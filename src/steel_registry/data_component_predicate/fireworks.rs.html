10/// Fields matched within one firework explosion.
11#[derive(Debug, Clone, PartialEq, Eq)]
12pub struct FireworkPredicate {
13    shape: Option<crate::data_components::components::FireworkExplosionShape>,
14    has_twinkle: Option<bool>,
15    has_trail: Option<bool>,
16}
17
18impl FireworkPredicate {
19    #[must_use]
20    pub fn matches(&self, explosion: &vanilla_components::FireworkExplosion) -> bool {
21        self.shape().is_none_or(|shape| shape == explosion.shape())
22            && self
23                .has_twinkle()
24                .is_none_or(|twinkle| twinkle == explosion.has_twinkle())
25            && self
26                .has_trail()
27                .is_none_or(|trail| trail == explosion.has_trail())
28    }
29
30    #[must_use]
31    pub const fn new(
32        shape: Option<crate::data_components::components::FireworkExplosionShape>,
33        has_twinkle: Option<bool>,
34        has_trail: Option<bool>,
35    ) -> Self {
36        Self {
37            shape,
38            has_twinkle,
39            has_trail,
40        }
41    }
42
43    #[must_use]
44    pub const fn shape(
45        &self,
46    ) -> Option<crate::data_components::components::FireworkExplosionShape> {
47        self.shape
48    }
49
50    #[must_use]
51    pub const fn has_twinkle(&self) -> Option<bool> {
52        self.has_twinkle
53    }
54
55    #[must_use]
56    pub const fn has_trail(&self) -> Option<bool> {
57        self.has_trail
58    }
59
60    fn from_nbt_value(tag: &NbtTag) -> Option<Self> {
61        let compound = tag.compound()?;
62        Some(Self {
63            shape: decode_optional(compound, "shape", |tag| {
64                match tag.string()?.to_owned().try_into_string().ok()?.as_str() {
65                    "small_ball" => {
66                        Some(crate::data_components::components::FireworkExplosionShape::SmallBall)
67                    }
68                    "large_ball" => {
69                        Some(crate::data_components::components::FireworkExplosionShape::LargeBall)
70                    }
71                    "star" => {
72                        Some(crate::data_components::components::FireworkExplosionShape::Star)
73                    }
74                    "creeper" => {
75                        Some(crate::data_components::components::FireworkExplosionShape::Creeper)
76                    }
77                    "burst" => {
78                        Some(crate::data_components::components::FireworkExplosionShape::Burst)
79                    }
80                    _ => None,
81                }
82            })?,
83            has_twinkle: decode_optional(compound, "has_twinkle", NbtNumeric::codec_bool)?,
84            has_trail: decode_optional(compound, "has_trail", NbtNumeric::codec_bool)?,
85        })
86    }
87
88    pub(super) fn to_nbt_value(&self) -> NbtTag {
89        let mut compound = NbtCompound::new();
90        if let Some(shape) = self.shape {
91            compound.insert("shape", shape.serialized_name());
92        }
93        if let Some(has_twinkle) = self.has_twinkle {
94            compound.insert("has_twinkle", has_twinkle);
95        }
96        if let Some(has_trail) = self.has_trail {
97            compound.insert("has_trail", has_trail);
98        }
99        NbtTag::Compound(compound)
100    }
101}

103impl HashComponent for FireworkPredicate {

105        let mut entries = Vec::new();
106        if let Some(shape) = self.shape {
107            push_hash_entry(&mut entries, "shape", shape.serialized_name());
108        }
109        if let Some(has_twinkle) = self.has_twinkle {

113            push_hash_entry(&mut entries, "has_trail", &has_trail);

116    }

164/// Predicate over firework explosions and flight duration.
165#[derive(Debug, Clone, PartialEq, Eq)]
166pub struct FireworksPredicate {
167    explosions: Option<CollectionPredicate<FireworkPredicate>>,
168    flight_duration: IntBounds,
169}
170
171impl FireworksPredicate {
172    #[must_use]
173    pub const fn new(
174        explosions: Option<CollectionPredicate<FireworkPredicate>>,
175        flight_duration: IntBounds,
176    ) -> Self {
177        Self {
178            explosions,
179            flight_duration,
180        }
181    }
182
183    #[must_use]
184    pub const fn explosions(&self) -> Option<&CollectionPredicate<FireworkPredicate>> {
185        self.explosions.as_ref()
186    }
187
188    #[must_use]
189    pub const fn flight_duration(&self) -> IntBounds {
190        self.flight_duration
191    }
192}
193
194impl DataComponentPredicate for FireworksPredicate {
195    fn matches(&self, components: &dyn DataComponentGetter) -> bool {
196        components
197            .component(vanilla_components::FIREWORKS)
198            .is_some_and(|fireworks| {
199                self.flight_duration().matches(fireworks.flight_duration())
200                    && self.explosions().is_none_or(|explosions| {
201                        explosions
202                            .matches(fireworks.explosions().iter(), FireworkPredicate::matches)
203                    })
204            })
205    }
206}
207
208impl DataComponentPredicateCodec for FireworksPredicate {
209    fn from_nbt_value(tag: &NbtTag) -> Option<Self> {
210        let compound = tag.compound()?;
211        Some(Self {
212            explosions: decode_optional(compound, "explosions", |tag| {
213                CollectionPredicate::from_nbt_with(tag, FireworkPredicate::from_nbt_value)
214            })?,
215            flight_duration: compound
216                .get("flight_duration")
217                .map_or(Some(IntBounds::ANY), IntBounds::from_owned_nbt)?,
218        })
219    }
220
221    fn to_nbt_value(&self) -> NbtTag {
222        let mut compound = NbtCompound::new();
223        if let Some(explosions) = &self.explosions {
224            compound.insert(
225                "explosions",
226                explosions.to_nbt_with(FireworkPredicate::to_nbt_value),
227            );
228        }
229        if !self.flight_duration.is_any() {
230            compound.insert("flight_duration", self.flight_duration.as_nbt_tag());
231        }
232        NbtTag::Compound(compound)
233    }
234}

236impl HashComponent for FireworksPredicate {
237    fn hash_component(&self, hasher: &mut ComponentHasher) {
238        let mut entries = Vec::new();
239        if let Some(explosions) = &self.explosions {
240            let mut value_hasher = ComponentHasher::new();
241            explosions.hash_with(&mut value_hasher, HashComponent::compute_hash);
242            crate::item_predicate::push_prehashed_entry(&mut entries, "explosions", value_hasher);
243        }
244        if !self.flight_duration.is_any() {
245            push_hash_entry(&mut entries, "flight_duration", &self.flight_duration);
246        }
247        hash_entries(hasher, &mut entries);
248    }
249}
