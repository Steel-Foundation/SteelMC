20/// A shaped recipe pattern.
21#[derive(Debug)]
22pub struct ShapedRecipe {
23    pub properties: RecipeProperties,
24    pub width: usize,
25    pub height: usize,
26    pub pattern: Box<[Ingredient]>,
27    pub result: ItemStackTemplate,
28    /// Whether the pattern is horizontally symmetric.
29    pub symmetrical: bool,
30}
31
32impl ShapedRecipe {
33    #[must_use]
34    pub fn new(
35        mut properties: RecipeProperties,
36        width: usize,
37        height: usize,
38        pattern: Box<[Ingredient]>,
39        result: ItemStackTemplate,
40    ) -> Self {
41        let symmetrical = Self::compute_symmetrical(width, &pattern);
42        if properties.placement.is_none() {
43            properties.placement = Some(super::PlacementInfo::from_optional_ingredients(&pattern));
44        }
45        Self {
46            properties,
47            width,
48            height,
49            pattern,
50            result,
51            symmetrical,
52        }
53    }
54
55    #[must_use]
56    pub fn with_placement(mut self, placement: super::PlacementInfo) -> Self {
57        self.properties.placement = Some(placement);
58        self
59    }
60
61    fn compute_symmetrical(width: usize, pattern: &[Ingredient]) -> bool {
62        if width == 0 {
63            return true;
64        }
65        let height = pattern.len() / width;
66        for y in 0..height {
67            for x in 0..width / 2 {
68                if !pattern[y * width + x].eq_ingredient(&pattern[y * width + (width - 1 - x)]) {
69                    return false;
70                }
71            }
72        }
73        true
74    }
75
76    #[must_use]
77    pub const fn fits_in(&self, width: usize, height: usize) -> bool {
78        self.width <= width && self.height <= height
79    }
80}

241impl CraftingRecipe {
242    #[must_use]
243    pub const fn properties(&self) -> &RecipeProperties {
244        match self {
245            Self::Shaped(recipe) => &recipe.properties,
246            Self::Shapeless(recipe) => &recipe.properties,
247            Self::Transmute(recipe) => &recipe.properties,
248            Self::Dye(recipe) => &recipe.properties,
249            Self::DecoratedPot(recipe) => &recipe.properties,
250            Self::Imbue(recipe) => &recipe.properties,
251            Self::BannerDuplicate(recipe) => &recipe.properties,
252            Self::BookCloning(recipe) => &recipe.properties,
253            Self::FireworkRocket(recipe) => &recipe.properties,
254            Self::FireworkStar(recipe) => &recipe.properties,
255            Self::FireworkStarFade(recipe) => &recipe.properties,
256            Self::MapExtending(recipe) => &recipe.properties,
257            Self::RepairItem(recipe) => &recipe.properties,
258            Self::ShieldDecoration(recipe) => &recipe.properties,
259        }
260    }
261
262    #[must_use]
263    pub const fn result(&self) -> Option<&ItemStackTemplate> {
264        match self {
265            Self::Shaped(recipe) => Some(&recipe.result),
266            Self::Shapeless(recipe) => Some(&recipe.result),
267            Self::Transmute(recipe) => recipe.result.as_ref(),
268            Self::Dye(recipe) => Some(&recipe.result),
269            Self::DecoratedPot(recipe) => Some(&recipe.result),
270            Self::Imbue(recipe) => Some(&recipe.result),
271            Self::BannerDuplicate(recipe) => Some(&recipe.result),
272            Self::BookCloning(recipe) => Some(&recipe.result),
273            Self::FireworkRocket(recipe) => Some(&recipe.result),
274            Self::FireworkStar(recipe) => Some(&recipe.result),
275            Self::FireworkStarFade(recipe) => Some(&recipe.result),
276            Self::MapExtending(recipe) => recipe.result.as_ref(),
277            Self::RepairItem(_) => None,
278            Self::ShieldDecoration(recipe) => Some(&recipe.result),
279        }
280    }
281
282    /// Interprets this passive recipe data for a concrete input snapshot.
283    #[must_use]
284    pub fn assemble(&self, input: &CraftingInput) -> ItemStack {
285        match self {
286            Self::Shaped(recipe) => recipe.result.create(),
287            Self::Shapeless(recipe) => recipe.result.create(),
288            Self::Transmute(recipe) => {
289                let material_count = input
290                    .items
291                    .iter()
292                    .filter(|stack| !stack.is_empty() && recipe.material.test(stack))
293                    .count();
294                let extra_count = if recipe.add_material_count_to_result {
295                    i32::try_from(material_count).unwrap_or(i32::MAX)
296                } else {
297                    0
298                };
299                input
300                    .items
301                    .iter()
302                    .find(|stack| !stack.is_empty() && recipe.input.test(stack))
303                    .map_or_else(ItemStack::empty, |source| match &recipe.result {
304                        Some(result) => result.apply(
305                            result.count().saturating_add(extra_count),
306                            source.components_patch(),
307                        ),
308                        None => source.copy_with_count(1i32.saturating_add(extra_count)),
309                    })
310            }
311            Self::Dye(recipe) => assemble_dye(recipe, input),
312            Self::DecoratedPot(recipe) => {
313                // Vanilla stores one decoration item per side, never the slot's count.
314                let decoration = |x: usize, y: usize| {
315                    ItemStackTemplate::from_stack(&input.get(x, y).copy_with_count(1)).ok()
316                };
317                let decorations = PotDecorations::new(
318                    decoration(1, 0),
319                    decoration(0, 1),
320                    decoration(2, 1),
321                    decoration(1, 2),
322                );
323                let mut result = recipe.result.create();
324                result.set(POT_DECORATIONS, decorations);
325                result
326            }
327            Self::Imbue(recipe) => {
328                let mut result = recipe.result.create();
329                if let Some(contents) = input.get(1, 1).get(POTION_CONTENTS) {
330                    result.set(POTION_CONTENTS, contents.clone());
331                }
332                result
333            }
334            Self::BannerDuplicate(recipe) => input
335                .items
336                .iter()
337                .find(|stack| {
338                    recipe.banner.test(stack)
339                        && stack
340                            .get(BANNER_PATTERNS)
341                            .is_some_and(|patterns| !patterns.layers().is_empty())
342                })
343                .map_or_else(ItemStack::empty, |source| {
344                    recipe
345                        .result
346                        .apply(recipe.result.count(), source.components_patch())
347                }),
348            Self::BookCloning(recipe) => assemble_book_cloning(recipe, input),
349            Self::FireworkRocket(recipe) => assemble_firework_rocket(recipe, input),
350            Self::FireworkStar(recipe) => assemble_firework_star(recipe, input),
351            Self::FireworkStarFade(recipe) => assemble_firework_fade(recipe, input),
352            Self::MapExtending(recipe) => {
353                let source = input.get(1, 1);
354                let mut result = match &recipe.result {
355                    Some(result) => result.apply(result.count(), source.components_patch()),
356                    None => source.copy_with_count(1),
357                };
358                result.set(MAP_POST_PROCESSING, MapPostProcessing::Scale);
359                result
360            }
361            Self::RepairItem(_) => assemble_repair(input),
362            Self::ShieldDecoration(recipe) => assemble_shield(recipe, input),
363        }
364    }
365}

367fn assemble_dye(recipe: &DyeRecipe, input: &CraftingInput) -> ItemStack {
368    let mut target = None;
369    let mut dyes = Vec::new();
370    for stack in input.items.iter().filter(|stack| !stack.is_empty()) {
371        if recipe.target.test(stack) {
372            target = Some(stack);
373        } else if recipe.dye.test(stack) {
374            dyes.push(stack.get(DYE).copied().unwrap_or(DyeColor::White));
375        }
376    }
377    let Some(target) = target else {
378        return ItemStack::empty();
379    };
380    if dyes.is_empty() {
381        return ItemStack::empty();
382    }
383    let color = DyedItemColor::apply_dyes(target.get(DYED_COLOR).copied(), &dyes);
384    let mut result = recipe
385        .result
386        .apply(recipe.result.count(), target.components_patch());
387    result.set(DYED_COLOR, color);
388    result
389}

391fn assemble_book_cloning(recipe: &BookCloningRecipe, input: &CraftingInput) -> ItemStack {
392    let Some(source) = input
393        .items
394        .iter()
395        .find(|stack| !stack.is_empty() && recipe.source.test(stack))
396    else {
397        return ItemStack::empty();
398    };
399    let Some(content) = source.get(WRITTEN_BOOK_CONTENT) else {
400        return ItemStack::empty();
401    };
402    let material_count = input
403        .items
404        .iter()
405        .filter(|stack| !stack.is_empty() && recipe.material.test(stack))
406        .count();
407    let Ok(extra_count) = i32::try_from(material_count.saturating_sub(1)) else {
408        return ItemStack::empty();
409    };
410    let mut result = recipe.result.apply(
411        recipe.result.count().saturating_add(extra_count),
412        source.components_patch(),
413    );
414    result.set(WRITTEN_BOOK_CONTENT, content.craft_copy());
415    result
416}

418fn assemble_firework_rocket(recipe: &FireworkRocketRecipe, input: &CraftingInput) -> ItemStack {
419    let mut flight_duration = 0;
420    let mut explosions = Vec::new();
421    for stack in input.items.iter().filter(|stack| !stack.is_empty()) {
422        if recipe.fuel.test(stack) {
423            flight_duration += 1;
424        } else if recipe.star.test(stack)
425            && let Some(explosion) = stack.get(FIREWORK_EXPLOSION)
426        {
427            explosions.push(explosion.clone());
428        }
429    }
430    let Ok(fireworks) = Fireworks::new(flight_duration, explosions) else {
431        return ItemStack::empty();
432    };
433    let mut result = recipe.result.create();
434    result.set(FIREWORKS, fireworks);
435    result
436}

438fn assemble_firework_star(recipe: &FireworkStarRecipe, input: &CraftingInput) -> ItemStack {
439    let mut shape = FireworkExplosionShape::SmallBall;
440    let mut has_twinkle = false;
441    let mut has_trail = false;
442    let mut colors = Vec::new();
443    for stack in input.items.iter().filter(|stack| !stack.is_empty()) {
444        if let Some((found_shape, _)) = recipe
445            .shapes
446            .iter()
447            .find(|(_, ingredient)| ingredient.test(stack))
448        {
449            shape = *found_shape;
450        } else if recipe.twinkle.test(stack) {
451            has_twinkle = true;
452        } else if recipe.trail.test(stack) {
453            has_trail = true;
454        } else if recipe.dye.test(stack) {
455            colors.push(
456                stack
457                    .get(DYE)
458                    .copied()
459                    .unwrap_or(DyeColor::White)
460                    .firework_color(),
461            );
462        }
463    }
464    let mut result = recipe.result.create();
465    result.set(
466        FIREWORK_EXPLOSION,
467        FireworkExplosion::new(shape, colors, Vec::new(), has_trail, has_twinkle),
468    );
469    result
470}

472fn assemble_firework_fade(recipe: &FireworkStarFadeRecipe, input: &CraftingInput) -> ItemStack {
473    let Some(target) = input
474        .items
475        .iter()
476        .find(|stack| !stack.is_empty() && recipe.target.test(stack))
477    else {
478        return ItemStack::empty();
479    };
480    let fade_colors: Vec<_> = input
481        .items
482        .iter()
483        .filter(|stack| !stack.is_empty() && recipe.dye.test(stack))
484        .map(|stack| {
485            stack
486                .get(DYE)
487                .copied()
488                .unwrap_or(DyeColor::White)
489                .firework_color()
490        })
491        .collect();
492    let mut result = recipe
493        .result
494        .apply(recipe.result.count(), target.components_patch());
495    let explosion = target
496        .get(FIREWORK_EXPLOSION)
497        .cloned()
498        .unwrap_or_default()
499        .with_fade_colors(fade_colors);
500    result.set(FIREWORK_EXPLOSION, explosion);
501    result
502}

504fn assemble_repair(input: &CraftingInput) -> ItemStack {
505    let mut inputs = input.items.iter().filter(|stack| !stack.is_empty());
506    let Some(first) = inputs.next() else {
507        return ItemStack::empty();
508    };
509    let Some(second) = inputs.next() else {
510        return ItemStack::empty();
511    };
512    let durability = first.get_max_damage().max(second.get_max_damage());
513    let remaining = (first.get_max_damage() - first.get_damage_value())
514        + (second.get_max_damage() - second.get_damage_value())
515        + durability * 5 / 100;
516    let mut result = ItemStack::new(first.item());
517    result.set(MAX_DAMAGE, durability);
518    result.set_damage_value((durability - remaining).max(0));
519
520    let mut curses = ItemEnchantments::empty();
521    for enchantments in [
522        first.get_enchantments_for_crafting(),
523        second.get_enchantments_for_crafting(),
524    ]
525    .into_iter()
526    .flatten()
527    {
528        for (key, level) in enchantments.iter() {
529            let Some(enchantment) = REGISTRY.enchantments.by_key(key) else {
530                continue;
531            };
532            if REGISTRY.enchantments.is_in_tag(
533                enchantment,
534                &vanilla_enchantment_tags::EnchantmentTag::CURSE,
535            ) {
536                curses.upgrade(key.clone(), *level);
537            }
538        }
539    }
540    result.set(ENCHANTMENTS, curses);
541    result
542}

612/// Positioned, immutable crafting-grid snapshot.
613#[derive(Debug, Clone)]
614pub struct CraftingInput {
615    pub width: usize,
616    pub height: usize,
617    pub items: Vec<ItemStack>,
618    ingredient_count: usize,
619    map_data: Vec<Option<MapRecipeData>>,

621
622impl CraftingInput {
623    pub const EMPTY: Self = Self {
624        width: 0,
625        height: 0,
626        items: Vec::new(),
627        ingredient_count: 0,
628        map_data: Vec::new(),
629    };
630
631    #[must_use]
632    pub fn new(width: usize, height: usize, items: Vec<ItemStack>) -> Self {
633        debug_assert_eq!(items.len(), width * height);
634        let ingredient_count = items.iter().filter(|stack| !stack.is_empty()).count();
635        let map_data = vec![None; items.len()];
636        Self {
637            width,
638            height,
639            items,
640            ingredient_count,
641            map_data,
642        }
643    }
644
645    #[must_use]
646    pub fn with_map_data(mut self, slot: usize, data: MapRecipeData) -> Self {
647        if let Some(map_data) = self.map_data.get_mut(slot) {
648            *map_data = Some(data);
649        }
650        self
651    }
652
653    #[must_use]
654    pub fn map_data(&self, slot: usize) -> Option<MapRecipeData> {
655        self.map_data.get(slot).copied().flatten()
656    }
657
658    #[must_use]
659    pub fn positioned(
660        width: usize,
661        height: usize,
662        items: Vec<ItemStack>,
663    ) -> PositionedCraftingInput {
664        if width == 0 || height == 0 {
665            return PositionedCraftingInput::EMPTY;
666        }
667
668        let mut left = width;
669        let mut right = 0;
670        let mut top = height;
671        let mut bottom = 0;
672        for y in 0..height {
673            for x in 0..width {
674                if !items[y * width + x].is_empty() {
675                    left = left.min(x);
676                    right = right.max(x);
677                    top = top.min(y);
678                    bottom = bottom.max(y);
679                }
680            }
681        }
682        if left > right || top > bottom {
683            return PositionedCraftingInput::EMPTY;
684        }
685
686        let new_width = right - left + 1;
687        let new_height = bottom - top + 1;
688        if new_width == width && new_height == height {
689            return PositionedCraftingInput {
690                input: Self::new(width, height, items),
691                left,
692                top,
693            };
694        }
695
696        let mut positioned = Vec::with_capacity(new_width * new_height);
697        for y in 0..new_height {
698            for x in 0..new_width {
699                positioned.push(items[(x + left) + (y + top) * width].clone());
700            }
701        }
702        PositionedCraftingInput {
703            input: Self::new(new_width, new_height, positioned),
704            left,
705            top,
706        }
707    }
708
709    #[must_use]
710    pub fn get(&self, x: usize, y: usize) -> &ItemStack {
711        &self.items[y * self.width + x]
712    }
713
714    #[must_use]
715    pub const fn ingredient_count(&self) -> usize {
716        self.ingredient_count
717    }
718}
