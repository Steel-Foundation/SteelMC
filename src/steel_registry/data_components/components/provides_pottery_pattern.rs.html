15/// Decorated pot pattern supplied by a pottery sherd item.
16#[derive(Debug, Clone, PartialEq, Eq)]
17pub struct ProvidesPotteryPattern {
18    pub pattern: DecoratedPotPatternRef,
19}
20
21impl WriteTo for ProvidesPotteryPattern {
22    fn write(&self, writer: &mut impl Write) -> Result<()> {
23        let id = self.pattern.try_id().ok_or_else(|| {
24            Error::other(format!(
25                "Unknown decorated pot pattern: {}",
26                self.pattern.key
27            ))
28        })?;
29        let id = i32::try_from(id)
30            .map_err(|_| Error::other(format!("Decorated pot pattern id out of range: {id}")))?;
31        VarInt(id).write(writer)
32    }
33}
34
35impl ReadFrom for ProvidesPotteryPattern {
36    fn read(data: &mut Cursor<&[u8]>) -> Result<Self> {
37        let id = VarInt::read(data)?.0;
38        let id = usize::try_from(id)
39            .map_err(|_| Error::other(format!("Negative decorated pot pattern id: {id}")))?;
40        let pattern = REGISTRY
41            .decorated_pot_patterns
42            .by_id(id)
43            .ok_or_else(|| Error::other(format!("Unknown decorated pot pattern id: {id}")))?;
44        Ok(Self { pattern })
45    }
46}
47
48impl ToNbtTag for ProvidesPotteryPattern {
49    fn to_nbt_tag(self) -> simdnbt::owned::NbtTag {
50        self.pattern.key.to_string().to_nbt_tag()
51    }
52}
53
54impl FromNbtTag for ProvidesPotteryPattern {
55    fn from_nbt_tag(tag: simdnbt::borrow::NbtTag) -> Option<Self> {
56        let key = Identifier::from_str(&tag.string()?.to_str()).ok()?;
57        REGISTRY
58            .decorated_pot_patterns
59            .by_key(&key)
60            .map(|pattern| Self { pattern })
61    }
62}
63
64impl HashComponent for ProvidesPotteryPattern {
65    fn hash_component(&self, hasher: &mut ComponentHasher) {
66        hasher.put_string(&self.pattern.key.to_string());
67    }
68}

84    fn parse(tag: simdnbt::owned::NbtTag) -> Option<ProvidesPotteryPattern> {
85        let mut bytes = Vec::new();
86        tag.write(&mut bytes);
87        let borrowed = read_tag(&mut Cursor::new(bytes.as_slice())).ok()?;
88        ProvidesPotteryPattern::from_nbt_tag(borrowed.as_tag())
89    }

91    #[test]
92    fn round_trips_both_codecs_and_hashes_by_key() {
93        init_vanilla_registry();
94        let component = ProvidesPotteryPattern {
95            pattern: &vanilla_decorated_pot_patterns::ANGLER,
96        };
97
98        let mut network = Vec::new();
99        component
100            .write(&mut network)
101            .expect("pottery pattern should encode");
102        assert_eq!(
103            ProvidesPotteryPattern::read(&mut Cursor::new(network.as_slice()))
104                .expect("pottery pattern should decode"),
105            component
106        );
107
108        let nbt = component.clone().to_nbt_tag();
109        assert_eq!(
110            nbt,
111            simdnbt::owned::NbtTag::String("minecraft:angler".into())
112        );
113        assert_eq!(parse(nbt), Some(component));
114    }
