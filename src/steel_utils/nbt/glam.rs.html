10fn numeric_f32_list(tag: BorrowedNbtTag, len: usize) -> Option<Vec<f32>> {
11    let values = nbt_collection_values(&tag.to_owned())?;
12    let values: Vec<_> = values
13        .iter()
14        .filter_map(NbtNumeric::codec_f32)
15        .take(len)
16        .collect();
17
18    (values.len() == len).then_some(values)
19}

33/// A rotation storing an angle and axis (in 3 components).
34///
35/// This is only used to provide a more accurate codec implementation
36/// for quaternions (`Quat`s).
37struct AxisAngle4f {
38    angle: f32,
39    axis: Vec3,
40}
41
42impl FromNbtTag for AxisAngle4f {
43    fn from_nbt_tag(tag: BorrowedNbtTag) -> Option<Self> {
44        let compound = tag.compound()?;
45        Some(Self {
46            angle: compound.get("angle")?.codec_f32()?,
47            axis: vec3_from_nbt_tag(compound.get("axis")?)?,
48        })
49    }
50}

52impl From<AxisAngle4f> for Quat {
53    fn from(value: AxisAngle4f) -> Self {
54        let angle = if value.angle < 0.0 {
55            (TAU + value.angle % TAU) % TAU
56        } else {
57            value.angle % TAU
58        };
59
60        let half_angle = angle * 0.5;
61        let sin = f64::from(half_angle).sin() as f32;
62
63        let mut cos = f64::from(1.0 - sin * sin).sqrt() as f32;
64        let shift = half_angle + FRAC_PI_2;
65        let mut quadrant = shift - (shift / TAU) as i32 as f32 * TAU;
66
67        if quadrant < 0.0 {
68            quadrant += TAU;
69        }
70
71        if quadrant >= PI {
72            cos = -cos;
73        }
74
75        Self::from_xyzw(
76            value.axis.x * sin,
77            value.axis.y * sin,
78            value.axis.z * sin,
79            cos,
80        )
81    }
82}

90/// Tries to convert a [`Quat`] from its NBT representation (recreates `ExtraCodecs.QUATERNIONF`).
91#[must_use]
92pub fn quat_from_nbt_tag(tag: BorrowedNbtTag) -> Option<Quat> {
93    // One of the two: 4 floats or AxisAngle4f
94    if let Some(l) = numeric_f32_list(tag, 4) {
95        let (x, y, z, w) = (l[0], l[1], l[2], l[3]);
96        let length_squared = x * x + (y * y + (z * z + w * w));
97        let inv_sqrt = 1.0 / (f64::from(length_squared).sqrt() as f32);
98
99        return Some(Quat::from_xyzw(
100            x * inv_sqrt,
101            y * inv_sqrt,
102            z * inv_sqrt,
103            w * inv_sqrt,
104        ));
105    }
106    Some(AxisAngle4f::from_nbt_tag(tag)?.into())
107}
