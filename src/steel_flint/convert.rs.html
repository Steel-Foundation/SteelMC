12/// Convert a Flint block specification to a `SteelMC` `BlockStateId`.
13///
14/// Returns `None` if the block ID is unknown or if any property is invalid.
15pub fn flint_block_to_state_id(block: &Block) -> Option<BlockStateId> {
16    let block_id = block.id.strip_prefix("minecraft:").unwrap_or(&block.id);
17    let identifier = Identifier::vanilla(block_id.to_string());
18
19    let block_ref = REGISTRY.blocks.by_key(&identifier)?;
20
21    REGISTRY.blocks.state_id_from_block_defaulted_properties(
22        block_ref,
23        block
24            .properties
25            .iter()
26            .map(|(k, v)| (k.as_str(), v.as_str())),
27    )
28}

30/// Convert a `SteelMC` `BlockStateId` to Flint `Block`.
31pub fn state_id_to_block(state_id: BlockStateId) -> Block {
32    let Some(block) = REGISTRY.blocks.by_state_id(state_id) else {
33        return Block::new("minecraft:air");
34    };
35
36    let id = if block.key.path == "void_air" || block.key.path == "cave_air" {
37        "minecraft:air".to_owned()
38    } else {
39        format!("minecraft:{}", block.key.path)
40    };
41
42    #[allow(clippy::disallowed_types)]
43    let properties: FxHashMap<String, String> = REGISTRY
44        .blocks
45        .properties(state_id)
46        .map(|(k, v)| (k.to_string(), v.to_string()))
47        .collect();
48
49    Block::with_properties(id, properties)
50}

52/// Convert Flint `BlockPos` to `SteelMC` `BlockPos`.
53pub const fn flint_pos_to_steel(pos: flint_core::BlockPos) -> SteelBlockPos {
54    SteelBlockPos::new(pos[0], pos[1], pos[2])
55}

57/// Convert a Flint gamerule value into the type the Steel rule expects.
58pub fn flint_gamerule_value(
59    rule: ErasedGameRuleRef,
60    value: GameruleValue,
61) -> anyhow::Result<GameRuleValue> {
62    match (rule.value_type(), value) {
63        (GameRuleType::Bool, GameruleValue::Bool(value)) => Ok(GameRuleValue::new(value)),
64        (GameRuleType::Int, GameruleValue::Integer(value)) => {
65            i32::try_from(value).map(GameRuleValue::new).map_err(|_| {
66                anyhow::anyhow!("Gamerule '{}' value {value} out of i32 range", rule.key())
67            })
68        }
69        (expected, value) => Err(anyhow::anyhow!(
70            "Gamerule '{}' expects {expected:?}, got '{value}'",
71            rule.key()
72        )),
73    }
74}

76/// Convert a `SteelMC` [`ClipHitResult`] to a [`BlockHitResult`].
77///
78/// The two types carry identical data; `World::clip` returns the former while
79/// the interaction API (`use_item_on`) takes the latter.
80pub const fn clip_to_block_hit(clip: ClipHitResult) -> BlockHitResult {
81    BlockHitResult {
82        location: clip.location,
83        direction: clip.direction,
84        block_pos: clip.block_pos,
85        miss: clip.miss,
86        inside: clip.inside,
87        world_border_hit: clip.world_border_hit,
88    }
89}

97        crate::init();
98        for id in ["minecraft:stone", "stone"] {
99            let state_id = flint_block_to_state_id(&Block::new(id)).expect("known block");
100            assert_eq!(state_id_to_block(state_id).id, "minecraft:stone");
