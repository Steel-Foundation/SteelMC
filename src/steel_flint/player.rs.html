32/// Test player implementation that wraps a real `Player`.

37impl SteelTestPlayer {
38    /// Creates a new test player in the given world.
39    ///
40    /// # Panics
41    /// Never in practice: the fixed test player name always passes validation.
42    pub fn new(world: Arc<World>, server: &Arc<Server>) -> Self {
43        // Same identity as the FlintCLI bot, so specs asserting an `Owner` UUID
44        // stay portable between runners.
45        let game_profile = GameProfile::new(
46            offline_uuid(TEST_PLAYER_NAME),
47            TEST_PLAYER_NAME.to_owned(),
48            vec![],
49        )
50        .expect("test player name is valid");
51        let player_connection = Arc::new(PlayerConnection::Other(Box::new(
52            FlintConnection::default(),
53        )));
54        let player = Player::new(
55            game_profile,
56            player_connection,
57            Arc::new(PlayerSession::new(10, 10)),
58            world,
59            Arc::downgrade(server),
60            server.config.clone(),
61            -1, // Negative entity ID for test players
62            ClientInformation::default(),
63        );
64        // Mark as loaded so interactions work
65        player.set_client_loaded(true);
66        Self {
67            player: Arc::new(player),
68        }
69    }
70}

72/// Converts a Flint [`PlayerSlot`] to a Steel inventory slot index.
73///
74/// Flint uses semantic slot names (e.g., `Hotbar1`, `OffHand`, `Helmet`),
75/// while Steel uses numeric indices. This function maps between the two:
76/// - Hotbar slots 1-9 → indices 0-8
77/// - `OffHand` → `PlayerInventory::SLOT_OFFHAND`
78/// - Armor slots → indices 36-39 (boots to helmet)
79const fn player_slot_to_index(slot: PlayerSlot) -> usize {
80    match slot {
81        PlayerSlot::Hotbar1 => 0,
82        PlayerSlot::Hotbar2 => 1,
83        PlayerSlot::Hotbar3 => 2,
84        PlayerSlot::Hotbar4 => 3,
85        PlayerSlot::Hotbar5 => 4,
86        PlayerSlot::Hotbar6 => 5,
87        PlayerSlot::Hotbar7 => 6,
88        PlayerSlot::Hotbar8 => 7,
89        PlayerSlot::Hotbar9 => 8,
90        PlayerSlot::OffHand => PlayerInventory::SLOT_OFFHAND,
91        PlayerSlot::Boots => 36,
92        PlayerSlot::Leggings => 37,
93        PlayerSlot::Chestplate => 38,
94        PlayerSlot::Helmet => 39,
95    }
96}

98/// Converts a Flint [`Item`] to a Steel [`ItemStack`].
99///
100/// Handles the `minecraft:` namespace prefix (strips it if present) and
101/// looks up the item in the registry. Returns an empty stack if the item
102/// is not found.
103fn flint_item_to_stack(item: &Item) -> ItemStack {
104    let item_id = item.id.strip_prefix("minecraft:").unwrap_or(&item.id);
105    let Some(item_ref) = REGISTRY
106        .items
107        .by_key(&Identifier::vanilla(item_id.to_string()))
108    else {
109        tracing::warn!("Unknown item: {} - returning empty stack", item.id);
110        return ItemStack::empty();
111    };
112    ItemStack::with_count(item_ref, i32::from(item.count))
113}
114
115/// Converts a Steel [`ItemStack`] to a Flint [`Item`].
116///
117/// Returns `None` for empty stacks. Adds the `minecraft:` namespace prefix
118/// to the item ID for consistency with Flint's expected format.
119fn stack_to_flint_item(stack: &ItemStack, requested_data: Vec<String>) -> Option<Item> {
120    if stack.is_empty() {
121        return None;
122    }
123
124    let id = format!("minecraft:{}", stack.item.key.path);
125    let mut map: FxHashMap<String, String> = FxHashMap::default();
126    for key in requested_data {
127        if let Some(data) = stack.get_effective_value_raw(&Identifier::vanilla(key.clone())) {
128            if data.downcast_ref::<()>().is_some() {
129                map.insert(key, String::new());
130            } else if let Some(b) = data.downcast_ref::<bool>() {
131                map.insert(key, b.to_string());
132            } else if let Some(i) = data.downcast_ref::<i32>() {
133                map.insert(key, i.to_string());
134            } else if let Some(f) = data.downcast_ref::<f32>() {
135                map.insert(key, f.to_string());
136            }
137            // TODO: handle other data types then needed
138        }
139    }
140    Some(Item {
141        id,
142        count: stack.count.try_into().unwrap_or(1),
143        data: map,
144    })
145}

147impl FlintPlayer for SteelTestPlayer {
148    fn as_any_mut(&mut self) -> &mut dyn Any {
149        self
150    }
151
152    fn set_slot(&mut self, slot: PlayerSlot, item: Option<&Item>) -> Result<(), anyhow::Error> {
153        let index = player_slot_to_index(slot);
154        let stack = item.map_or_else(ItemStack::empty, flint_item_to_stack);
155
156        let mut inv = self.player.inventory.lock();
157        inv.set_item(index, stack);
158        Ok(())
159    }
160
161    fn get_slot(
162        &mut self,
163        slot: PlayerSlot,
164        requested_data: Vec<String>,
165    ) -> Result<Option<Item>, anyhow::Error> {
166        let index = player_slot_to_index(slot);
167
168        let inv = self.player.inventory.lock();
169        let stack = inv.get_item(index);
170        Ok(stack_to_flint_item(stack, requested_data))
171    }
172
173    fn select_hotbar(&mut self, slot: u8) -> Result<(), anyhow::Error> {
174        if (1..=9).contains(&slot) {
175            // Flint uses 1-9, Steel uses 0-8
176            self.player.inventory.lock().set_selected_slot(slot - 1);
177        }
178        Ok(())
179    }
180
181    fn selected_hotbar(&self) -> u8 {
182        // Steel uses 0-8, Flint uses 1-9
183        self.player.inventory.lock().get_selected_slot() + 1
184    }
185
186    fn teleport(&mut self, pos: [f64; 3], rot: Option<[f32; 2]>) -> Result<(), anyhow::Error> {
187        self.player
188            .try_set_position(DVec3::new(pos[0], pos[1], pos[2]))?;
189        self.player.set_rotation((rot.unwrap_or([0.0, 0.0])).into());
190        Ok(())
191    }
192
193    fn interact(&mut self) -> Result<(), anyhow::Error> {
194        let world = self.player.get_world();
195        let (start, end) = self.player.get_ray_endpoints();
196        let clip = world.clip(start, end, ClipBlockShape::Outline, ClipFluid::None);
197
198        let hand = InteractionHand::MainHand;
199        let animation = self.player.interact_animation(hand);
200        let result = if clip.is_miss() {
201            game_mode::use_item(&self.player, &world, hand)
202        } else {
203            game_mode::use_item_on(&self.player, &world, hand, &clip_to_block_hit(clip))
204        };
205
206        if result.should_swing() {
207            self.player.swing(hand, animation, true);
208        }
209        self.player.broadcast_inventory_changes();
210
211        tracing::debug!("interact({start:?} -> {end:?}) -> {result:?}");
212        Ok(())
213    }
214
215    fn set_game_mode(&mut self, mode: GameMode) -> Result<(), anyhow::Error> {
216        self.player.set_game_mode(match mode {
217            GameMode::Survival => GameType::Survival,
218            GameMode::Creative => GameType::Creative,
219            GameMode::Adventure => GameType::Adventure,
220            GameMode::Spectator => GameType::Spectator,
221        });
222        Ok(())
223    }
224}
