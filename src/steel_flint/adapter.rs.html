42/// Returns IDs of blocks with a non-default behavior or a plain vanilla `Block` class.
43pub fn get_implemented_blocks() -> Vec<String> {
44    BLOCK_BEHAVIORS
45        .get_behaviors()
46        .iter()
47        .enumerate()
48        .filter(|(_, behavior)| !behavior.type_name().ends_with("DefaultBlockBehavior"))
49        .filter_map(|(id, _)| REGISTRY.blocks.by_id(id))
50        .chain(PLAIN_BLOCKS.iter().copied())
51        .map(|block| block.key.to_string())
52        .collect()
53}
54
55/// Returns IDs of items with a non-default, non-block-item behavior.
56pub fn get_implemented_items() -> Vec<String> {
57    ITEM_BEHAVIORS
58        .get_behaviors()
59        .iter()
60        .enumerate()
61        .filter(|(_, behavior)| {
62            let name = behavior.type_name();
63            !name.ends_with("DefaultItemBehavior") && !name.ends_with("BlockItem")
64        })
65        .filter_map(|(id, _)| REGISTRY.items.by_id(id))
66        .map(|item| item.key.to_string())
67        .collect()
68}
69
70/// Returns IDs of entities with an implementation.
71pub fn get_implemented_entities() -> Vec<String> {
72    ENTITIES
73        .get_implemented()
74        .map(|entity| entity.key.to_string())
75        .collect()
76}

78/// Flint run settings loaded from `flint.toml`.
79#[derive(Deserialize, Default)]
80pub struct FlintConfig {
81    filter: Option<FilterConfig>,
82    /// Base URL of the flint-viz instance to embed in failure URLs.
83    /// Priority: `FLINT_VIZ_URL` env var > toml `viz_url` > default.
84    #[serde(alias = "flint_viz_url", alias = "FLINT_VIZ_URL")]
85    viz_url: Option<String>,
86    /// Directory containing the flint tests.
87    /// Priority: `TEST_PATH` env var > toml `test_path` > `./test`.
88    test_path: Option<String>,
89}
90
91impl FlintConfig {
92    /// Resolves the test directory.
93    #[must_use]
94    pub fn test_path(&self) -> PathBuf {
95        var("TEST_PATH")
96            .ok()
97            .or_else(|| self.test_path.clone())
98            .map_or_else(|| PathBuf::from("./test"), PathBuf::from)
99    }
100}

102#[derive(Deserialize, Default)]
103struct FilterConfig {
104    tags: Option<rustc_hash::FxHashMap<String, bool>>,
105    implemented_only: Option<bool>,
106    test: Option<String>,
107    pattern: Option<String>,
108}

110/// Loads `flint.toml` from the working directory, falling back to defaults if it is missing or invalid.
111#[must_use]
112pub fn load_config() -> FlintConfig {
113    fs::read_to_string("flint.toml")
114        .ok()
115        .and_then(|s| from_str(&s).ok())
116        .unwrap_or_default()
117}

124/// Collects test file paths based on environment variables and `flint.toml`.
125/// Priority: `FLINT_TEST` > `FLINT_TAGS` > `implemented_only` (`FLINT_IMPLEMENTED_ONLY` env > toml) > `filter.tags` > `FLINT_PATTERN` > all
126#[must_use]
127pub fn collect_filtered_paths(loader: &TestLoader, cfg: &FlintConfig) -> Vec<PathBuf> {
128    let filter = cfg.filter.as_ref();
129
130    // Single test by name: env > toml
131    let test_name = var("FLINT_TEST")
132        .ok()
133        .or_else(|| filter.and_then(|f| f.test.clone()));
134    if let Some(name) = test_name {
135        println!("Running single test: {name}");
136        return loader
137            .collect_all_test_files()
138            .unwrap_or_default()
139            .into_iter()
140            .filter(|p| {
141                p.file_stem()
142                    .and_then(|s| s.to_str())
143                    .is_some_and(|n| n == name)
144            })
145            .collect();
146    }
147
148    // Tag filtering: env > implemented_only > toml tags
149    let env_tags = var("FLINT_TAGS").ok().map(|s| {
150        s.split(',')
151            .map(|t| t.trim().to_string())
152            .collect::<Vec<_>>()
153    });
154    let implemented_only = var("FLINT_IMPLEMENTED_ONLY").map_or_else(
155        |_| filter.and_then(|f| f.implemented_only).unwrap_or(false),
156        |v| matches!(v.as_str(), "1" | "true"),
157    );
158    if env_tags.is_none() && implemented_only {
159        let mut ids = get_implemented_items();
160        ids.extend(get_implemented_blocks());
161        ids.extend(get_implemented_entities());
162        let implemented: FxHashSet<String> = ids.iter().cloned().collect();
163        let paths: Vec<PathBuf> = loader
164            .collect_by_tags(&ids)
165            .into_iter()
166            .filter(|path| all_ids_implemented(path, &implemented))
167            .collect();
168        println!(
169            "Running {} test(s) whose minecraftIds are all implemented",
170            paths.len()
171        );
172        return paths;
173    }
174    let tags = env_tags.or_else(|| {
175        filter.and_then(|f| {
176            f.tags.as_ref().map(|map| {
177                map.iter()
178                    .filter(|&(_, &enabled)| enabled)
179                    .map(|(tag, _)| tag.clone())
180                    .collect()
181            })
182        })
183    });
184    if let Some(tags) = tags {
185        println!("Running tests with tags: {}", tags.join(", "));
186        return loader.collect_by_tags(&tags);
187    }
188
189    // Pattern matching: env > toml
190    let pattern = var("FLINT_PATTERN")
191        .ok()
192        .or_else(|| filter.and_then(|f| f.pattern.clone()));
193    if let Some(p) = pattern {
194        println!("Running tests matching pattern: {p}");
195        return loader
196            .collect_all_test_files()
197            .unwrap_or_default()
198            .into_iter()
199            .filter(|path| {
200                path.file_stem()
201                    .and_then(|s| s.to_str())
202                    .is_some_and(|name| matches_pattern(name, &p))
203            })
204            .collect();
205    }
206
207    // Default: all tests
208    println!("Running all flint tests");
209    loader.collect_all_test_files().unwrap_or_default()
210}

212/// Whether every `minecraftIds` entry of the test at `path` is implemented.
213/// Unreadable specs are kept so `load_specs` reports the real error.
214fn all_ids_implemented(path: &Path, implemented: &FxHashSet<String>) -> bool {
215    let Some(spec) = fs::read_to_string(path)
216        .ok()
217        .and_then(|json| serde_json::from_str::<MinimalTestSpec>(&json).ok())
218    else {
219        return true;
220    };
221    !spec.minecraft_ids.is_empty() && spec.minecraft_ids.iter().all(|id| implemented.contains(id))
222}

224/// Simple glob pattern matching (supports * wildcard)
225fn matches_pattern(name: &str, pattern: &str) -> bool {
226    if pattern == "*" {
227        return true;
228    }
229    if let Some(prefix) = pattern.strip_suffix('*') {
230        return name.starts_with(prefix);
231    }
232    if let Some(suffix) = pattern.strip_prefix('*') {
233        return name.ends_with(suffix);
234    }
235    name == pattern
236}

238/// Writes the CI summary to `log/flint_summary.json`.
239pub fn save_summary(summary: &TestSummary) -> io::Result<()> {
240    fs::create_dir_all("log")?;
241    fs::write("log/flint_summary.json", summary.create_ci_output(true))?;
242    println!("Summary saved to log/flint_summary.json");
243    Ok(())
244}

246fn resolve_viz_base_url(cfg: &FlintConfig) -> String {
247    var("FLINT_VIZ_URL")
248        .ok()
249        .or_else(|| cfg.viz_url.clone())
250        .unwrap_or_else(|| "https://flinttestmc.github.io/FlintViz".to_string())
251}

253/// For each failing test in `summary`, print a clickable flint-viz URL with
254/// the failing assertion(s) + inline `TestSpec` baked in. Designed to land
255/// directly below the failure tree so terminal emulators (and PR-comment
256/// renderers) auto-link it.
257pub fn print_failure_urls(
258    specs: &[flint_core::TestSpecLoadResult],
259    paths: &[PathBuf],
260    summary: &TestSummary,
261    cfg: &FlintConfig,
262) {
263    use flint_core::results::AssertionResult;
264    use flint_core::viz_link::{FailurePayload, failure_url};
265
266    let base = resolve_viz_base_url(cfg);
267    let mut emitted = false;
268    // specs/paths/results run in lockstep — `run_tests` preserves order.
269    for ((load_result, path), result) in specs.iter().zip(paths.iter()).zip(summary.results.iter())
270    {
271        if result.success || result.skipped {
272            continue;
273        }
274        let flint_core::TestSpecLoadResult::Loaded(spec) = load_result else {
275            continue;
276        };
277        let failures: Vec<_> = result
278            .assertions
279            .iter()
280            .filter_map(|a| match a {
281                AssertionResult::Failure(f) => Some(f.clone()),
282                AssertionResult::Success(_) => None,
283            })
284            .collect();
285        if failures.is_empty() {
286            continue;
287        }
288        let payload = FailurePayload::new(
289            spec.clone(),
290            Some(path.clone()),
291            failures,
292            result.total_ticks,
293        );
294        match failure_url(&payload, &base) {
295            Ok(url) => {
296                if !emitted {
297                    println!();
298                    emitted = true;
299                }
300                let name = spec.name.clone();
301                println!("[{name}]Open in flint-viz: {url}");
302            }
303            Err(err) => {
304                eprintln!(
305                    "warning: could not encode flint-viz URL for `{}`: {err}",
306                    result.test_name
307                );
308            }
309        }
310    }
311}

316    use crate::{TestLoader, TestRunner};

319    #[test]
320    #[ignore = "needs FlintBenchmark tests (TEST_PATH); run with --ignored"]
321    fn test_run_flint_selected() {
322        crate::init();
323        init_env();
324
325        let cfg = load_config();
326        let loader = TestLoader::new(&cfg.test_path(), true)
327            .unwrap_or_else(|e| panic!("error while loading test files: {e}"));
328        let paths = collect_filtered_paths(&loader, &cfg);
329        let specs = loader
330            .load_specs(&paths, false)
331            .unwrap_or_else(|e| panic!("error while loading test specs: {e}"));
332
333        let runner = TestRunner::new(Arc::new(SteelAdapter));
334        let summary = runner.run_tests(&specs);
335        summary.print_concise_summary();
336        save_summary(&summary).unwrap_or_else(|e| panic!("failed to save flint summary: {e}"));
337        print_failure_urls(&specs, &paths, &summary, &cfg);
338        assert_eq!(summary.failed_tests, 0, "Not all flint tests passed!");
339    }
340
341    #[test]
342    #[ignore = "needs FlintBenchmark tests (TEST_PATH); run with --ignored"]
343    fn test_run_all_flint_benchmarks() {
344        crate::init();
345        init_env();
346
347        let cfg = load_config();
348        let test_dir = cfg.test_path();
349        assert!(
350            test_dir.exists(),
351            "FlintBenchmark tests directory not found, skipping"
352        );
353
354        let loader = TestLoader::new(&test_dir, true)
355            .unwrap_or_else(|e| panic!("error while loading test files: {e}"));
356        let paths = loader
357            .collect_all_test_files()
358            .unwrap_or_else(|e| panic!("error while loading test files: {e}"));
359        assert!(
360            !paths.is_empty(),
361            "No test files matched the filter criteria in {}",
362            test_dir.to_str().unwrap_or("NONE")
363        );
364
365        println!("Found {} test(s) to run", paths.len());
366
367        let specs = loader
368            .load_specs(&paths, false)
369            .unwrap_or_else(|e| panic!("error while loading test specs: {e}"));
370
371        let runner = TestRunner::new(Arc::new(SteelAdapter));
372        let summary = runner.run_tests(&specs);
373        summary.print_concise_summary();
374        save_summary(&summary).unwrap_or_else(|e| panic!("failed to save flint summary: {e}"));
375        print_failure_urls(&specs, &paths, &summary, &cfg);
376        assert_eq!(summary.failed_tests, 0, "No tests were run");
377    }
