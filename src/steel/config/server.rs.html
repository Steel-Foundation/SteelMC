25/// The full server configuration as deserialized from TOML.
26///
27/// Contains both creation-time values (seed, world generator, storage)
28/// and runtime values that get moved into `RuntimeConfig`.
29#[derive(Debug, Clone, Deserialize)]
30#[serde(deny_unknown_fields)]
31pub struct ServerConfig {
32    /// The port the server will listen on.
33    pub server_port: u16,
34    /// The maximum number of players that can be on the server at once.
35    pub max_players: u32,
36    /// Allow `view_distance` above vanilla's 32-chunk cap.
37    #[serde(default)]
38    pub allow_extended_view_distance: bool,
39    /// The view distance of the server.
40    pub view_distance: u8,
41    /// The simulation distance of the server.
42    pub simulation_distance: u8,
43    /// Maximum queued neighbor-update tasks in one chained run; negative means unlimited.
44    #[serde(default = "default_max_chained_neighbor_updates")]
45    pub max_chained_neighbor_updates: i32,
46    /// Whether the server is in online mode.
47    pub online_mode: bool,
48    /// Optional authentication endpoint for online-mode `hasJoined` checks.
49    pub auth_server: Option<String>,
50    /// Optional endpoint for online-mode player name-to-profile lookups.
51    pub profile_server: Option<String>,
52    /// Optional endpoint for Mojang-compatible service public keys.
53    pub services_server: Option<String>,
54    /// Whether the server should use encryption. Required in online mode.
55    pub encryption: bool,
56    /// Whether vanilla floating/flying movement checks permit unauthorized flight.
57    #[serde(default)]
58    pub allow_flight: bool,
59    /// The message of the day.
60    pub motd: String,
61    /// Whether to use a favicon.
62    pub use_favicon: bool,
63    /// The path to the favicon.
64    pub favicon: String,
65    /// Whether to enforce secure chat.
66    pub enforce_secure_chat: bool,
67    /// Vanilla chat spam threshold window in seconds
68    #[serde(default = "default_spam_threshold_seconds")]
69    pub chat_spam_threshold_seconds: i32,
70    /// Vanilla command spam threshold window in seconds
71    #[serde(default = "default_spam_threshold_seconds")]
72    pub command_spam_threshold_seconds: i32,
73    /// The compression settings for the server.
74    pub compression: Option<CompressionInfo>,
75    /// All settings and configurations for server links.
76    pub server_links: Option<ServerLinks>,
77    /// Thread counts for server thread pools.
78    #[serde(default)]
79    pub threads: ThreadConfig,
80    /// The number of minutes required for a player to be idle for them to be kicked (timed out) from the server.
81    /// If this is equal to 0, no kicking will happen.
82    #[serde(default)]
83    pub player_idle_timeout: i32,
84}

134/// Validates the server configuration.
135///
136/// # Errors
137/// This function will return an error if the configuration is invalid.
138pub(super) fn validate(config: &ServerConfig) -> Result<(), &'static str> {
139    validate_login_security(config.online_mode, config.encryption)?;
140    if !config.allow_extended_view_distance && !(1..=32).contains(&config.view_distance) {
141        return Err("View distance must in range 1..32");
142    }
143    if config.allow_extended_view_distance
144        && !(1..=MAX_SUPPORTED_VIEW_DISTANCE).contains(&config.view_distance)
145    {
146        return Err("View distance must in range 1..128");
147    }
148    if let Some(auth_server) = &config.auth_server {
149        let Ok(url) = Url::parse(auth_server) else {
150            return Err("auth_server must be an absolute URL");
151        };
152        if !matches!(url.scheme(), "http" | "https") {
153            return Err("auth_server must use http or https");
154        }
155    }
156    if let Some(profile_server) = &config.profile_server {
157        let Ok(url) = Url::parse(profile_server) else {
158            return Err("profile_server must be an absolute URL");
159        };
160        if !matches!(url.scheme(), "http" | "https") {
161            return Err("profile_server must use http or https");
162        }
163    }
164    if let Some(services_server) = &config.services_server {
165        let Ok(url) = Url::parse(services_server) else {
166            return Err("services_server must be an absolute URL");
167        };
168        if !matches!(url.scheme(), "http" | "https") {
169            return Err("services_server must use http or https");
170        }
171    }
172    if config.simulation_distance > config.view_distance {
173        return Err("Simulation distance must be less than or equal to view distance");
174    }
175    if let Some(compression) = config.compression {
176        if compression.threshold.get() < MIN_COMPRESSION_THRESHOLD {
177            return Err("Compression threshold must be greater than or equal to 256");
178        }
179        if !(MIN_COMPRESSION_LEVEL..=MAX_COMPRESSION_LEVEL).contains(&compression.level) {
180            return Err("Compression level must be between 1 and 9");
181        }
182    }
183    if config.enforce_secure_chat {
184        if !config.online_mode {
185            return Err("online_mode must be true when enforce_secure_chat is enabled");
186        }
187        if !config.encryption {
188            return Err("encryption must be true when enforce_secure_chat is enabled");
189        }
190    }
191    Ok(())
192}
