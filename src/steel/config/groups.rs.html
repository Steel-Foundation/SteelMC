36impl PermissionGroupStore for FilePermissionGroupStore {
37    fn save_groups(
38        &self,
39        config: PermissionGroupsConfig,
40    ) -> BoxFuture<'static, Result<(), PermissionGroupStoreError>> {
41        let path = self.path.clone();
42        Box::pin(async move {
43            let serialized = serialize_groups_config(&config).map_err(|error| {
44                PermissionGroupStoreError::new(format!(
45                    "failed to serialize groups config: {error}"
46                ))
47            })?;
48            write_atomic_config(&path, serialized)
49                .await
50                .map_err(|error| {
51                    PermissionGroupStoreError::new(format!(
52                        "failed to write groups config {}: {error}",
53                        path.display()
54                    ))
55                })
56        })
57    }
58}

60fn serialize_groups_config(config: &PermissionGroupsConfig) -> Result<String, TomlSerializeError> {
61    let mut output = String::from(GROUPS_CONFIG_HEADER);
62    push_toml_field(&mut output, "default_groups", &config.default_groups)?;
63
64    for (name, group) in &config.groups {
65        output.push('\n');
66        output.push_str("[groups.");
67        output.push_str(name);
68        output.push_str("]\n");
69        push_group_config(&mut output, group)?;
70    }
71
72    Ok(output)
73}

75fn push_group_config(
76    output: &mut String,
77    group: &PermissionGroupConfig,
78) -> Result<(), TomlSerializeError> {
79    push_toml_field(output, "priority", &group.priority)?;
80    push_string_array_field(output, "inherits", &group.inherits)?;
81    push_string_array_field(output, "allow", &group.allow)?;
82    push_string_array_field(output, "deny", &group.deny)?;
83    push_metadata_rules(output, &group.metadata)
84}

86fn push_string_array_field(
87    output: &mut String,
88    key: &str,
89    values: &[String],
90) -> Result<(), TomlSerializeError> {
91    if values.is_empty() {
92        output.push_str(key);
93        output.push_str(" = []\n");
94        return Ok(());
95    }
96
97    output.push_str(key);
98    output.push_str(" = [\n");
99    for value in values {
100        output.push_str("    ");
101        output.push_str(&toml_value(value)?);
102        output.push_str(",\n");
103    }
104    output.push_str("]\n");
105    Ok(())
106}

108fn push_metadata_rules(
109    output: &mut String,
110    metadata: &[PermissionMetadataRuleConfig],
111) -> Result<(), TomlSerializeError> {
112    if metadata.is_empty() {
113        output.push_str("metadata = []\n");
114        return Ok(());
115    }
116
117    output.push_str("metadata = [\n");
118    for entry in metadata {
119        output.push_str("    { key = ");
120        output.push_str(&toml_value(&entry.key)?);
121        output.push_str(", value = ");
122        output.push_str(&permission_metadata_value_toml(&entry.value)?);
123        output.push_str(" },\n");
124    }
125    output.push_str("]\n");
126    Ok(())
127}

129fn push_toml_field<T: Serialize + ?Sized>(
130    output: &mut String,
131    key: &str,
132    value: &T,
133) -> Result<(), TomlSerializeError> {
134    output.push_str(key);
135    output.push_str(" = ");
136    output.push_str(&toml_value(value)?);
137    output.push('\n');
138    Ok(())
139}

141fn toml_value<T: Serialize + ?Sized>(value: &T) -> Result<String, TomlSerializeError> {
142    #[derive(Serialize)]
143    struct Field<'a, T: Serialize + ?Sized> {
144        value: &'a T,
145    }
146
147    let serialized = toml::to_string(&Field { value })?;
148    let serialized = serialized.trim_end();
149    Ok(serialized
150        .strip_prefix("value = ")
151        .unwrap_or(serialized)
152        .to_owned())
153}

155fn permission_metadata_value_toml(
156    value: &PermissionMetadataValue,
157) -> Result<String, TomlSerializeError> {
158    match value {
159        PermissionMetadataValue::Bool(value) => toml_value(value),
160        PermissionMetadataValue::Integer(value) => toml_value(value),
161        PermissionMetadataValue::String(value) => toml_value(value),
162    }
163}

165async fn write_atomic_config(path: &Path, contents: String) -> io::Result<()> {
166    let Some(parent) = path.parent() else {
167        return Err(io::Error::new(
168            io::ErrorKind::InvalidInput,
169            "config path has no parent",
170        ));
171    };
172    async_fs::create_dir_all(parent).await?;
173    let temp_path = path.with_extension("toml.tmp");
174    async_fs::write(&temp_path, contents).await?;
175    async_fs::rename(temp_path, path).await
176}

178pub(super) fn load_or_create_groups(path: &Path) -> Result<PermissionGroupsConfig, String> {
179    let config: PermissionGroupsConfig = if path.exists() {
180        let contents = fs::read_to_string(path)
181            .map_err(|error| format!("failed to read groups config {}: {error}", path.display()))?;
182        toml::from_str(&contents)
183            .map_err(|error| format!("failed to parse groups config {}: {error}", path.display()))?
184    } else {
185        fs::write(path, DEFAULT_GROUPS).map_err(|error| {
186            format!("failed to write groups config {}: {error}", path.display())
187        })?;
188        toml::from_str(DEFAULT_GROUPS)
189            .map_err(|error| format!("failed to parse default groups config: {error}"))?
190    };
191    PermissionGroups::from_config(config.clone()).map_err(|error| {
192        format!(
193            "failed to validate groups config {}: {error}",
194            path.display()
195        )
196    })?;
197    Ok(config)
198}
