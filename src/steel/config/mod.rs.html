36/// Top-level TOML deserialization target — used once at startup, not stored globally.
37#[derive(Debug, Clone, Deserialize)]
38#[serde(deny_unknown_fields)]
39pub struct SteelConfig {
40    /// The full server configuration (`[server]` section)
41    pub server: ServerConfig,
42    /// Logging configuration (`[log]` section)
43    pub log: Option<LogConfig>,
44    /// World and domain configuration from `worlds.toml`.
45    #[serde(skip, default = "empty_worlds_config")]
46    pub worlds: WorldsConfig,
47    /// Permission group configuration from `groups.toml`.
48    #[serde(skip, default)]
49    pub groups: PermissionGroupsConfig,
50    /// Path to the loaded `groups.toml`.
51    #[serde(skip, default)]
52    pub groups_path: Option<PathBuf>,
53}
54
55impl SteelConfig {
56    /// Builds the store used for persistence-first permission group updates.
57    #[must_use]
58    pub fn permission_group_store(&self) -> Option<Arc<dyn PermissionGroupStore>> {
59        self.groups_path.as_ref().map(|path| {
60            Arc::new(FilePermissionGroupStore::new(path.clone())) as Arc<dyn PermissionGroupStore>
61        })
62    }
63}

77/// Loads the server configuration from the given path, or creates it if it doesn't exist.
78pub fn load_or_create(path: &Path) -> Result<SteelConfig, String> {
79    let mut config = if path.exists() {
80        let config_str = fs::read_to_string(path)
81            .map_err(|e| format!("failed to read config file {}: {e}", path.display()))?;
82        let config: SteelConfig = toml::from_str(config_str.as_str())
83            .map_err(|e| format!("failed to parse config: {e}"))?;
84        validate(&config.server).map_err(|e| format!("failed to validate config: {e}"))?;
85        config
86    } else {
87        let parent = path
88            .parent()
89            .ok_or_else(|| format!("failed to get config directory for {}", path.display()))?;
90        fs::create_dir_all(parent).map_err(|e| {
91            format!(
92                "failed to create config directory {}: {e}",
93                parent.display()
94            )
95        })?;
96        fs::write(path, DEFAULT_CONFIG)
97            .map_err(|e| format!("failed to write config file {}: {e}", path.display()))?;
98        let config: SteelConfig = toml::from_str(DEFAULT_CONFIG)
99            .map_err(|e| format!("failed to parse default config: {e}"))?;
100        validate(&config.server).map_err(|e| format!("failed to validate default config: {e}"))?;
101        config
102    };
103
104    let worlds_path = path
105        .parent()
106        .ok_or_else(|| format!("failed to get config directory for {}", path.display()))?
107        .join("worlds.toml");
108    config.worlds = load_or_create_worlds(&worlds_path)?;
109    let groups_path = path
110        .parent()
111        .ok_or_else(|| format!("failed to get config directory for {}", path.display()))?
112        .join("groups.toml");
113    config.groups = load_or_create_groups(&groups_path)?;
114    config.groups_path = Some(groups_path);
115
116    // If icon file doesnt exist, write it
117    #[cfg(feature = "stand-alone")]
118    if config.server.use_favicon && !Path::new(&config.server.favicon).exists() {
119        fs::write(Path::new(&config.server.favicon), DEFAULT_FAVICON).map_err(|e| {
120            format!(
121                "failed to write favicon file {}: {e}",
122                config.server.favicon
123            )
124        })?;
125    }
126
127    Ok(config)
128}
129
130fn load_or_create_worlds(path: &Path) -> Result<WorldsConfig, String> {
131    if path.exists() {
132        let worlds_str = fs::read_to_string(path)
133            .map_err(|e| format!("failed to read worlds config file {}: {e}", path.display()))?;
134        toml::from_str(worlds_str.as_str())
135            .map_err(|e| format!("failed to parse worlds config {}: {e}", path.display()))
136    } else {
137        fs::write(path, DEFAULT_WORLDS)
138            .map_err(|e| format!("failed to write worlds config file {}: {e}", path.display()))?;
139        toml::from_str(DEFAULT_WORLDS)
140            .map_err(|e| format!("failed to parse default worlds config: {e}"))
141    }
142}
