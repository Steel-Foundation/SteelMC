28/// The main server struct.
29pub struct SteelServer {
30    /// The TCP listener for incoming connections.
31    pub tcp_listener: TcpListener,
32    /// The cancellation token for graceful shutdown.
33    pub cancel_token: CancellationToken,
34    /// The next client ID to be assigned.
35    pub client_id: u64,
36    /// The shared server state.
37    pub server: Arc<Server>,
38    /// Session id UUID state
39    pub connection_session: Arc<ServerConnectionSession>,
40}

42/// Startup error for expected operational failures.
43#[derive(Debug)]
44pub enum SteelServerError {
45    /// Core server startup failed.
46    Core(String),
47    /// TCP listener could not bind.
48    Bind {
49        /// Server port that failed to bind.
50        port: u16,
51        /// Underlying IO error.
52        source: io::Error,
53    },
54}
55
56impl fmt::Display for SteelServerError {
57    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
58        match self {
59            Self::Core(error) => f.write_str(error),
60            Self::Bind { port, source } => {
61                write!(f, "failed to bind to server port {port}: {source}")
62            }
63        }
64    }
65}
66
67impl Error for SteelServerError {}

69impl SteelServer {
70    /// Creates a new Steel server.
71    ///
72    pub async fn new(
73        chunk_runtime: Arc<Runtime>,
74        cancel_token: CancellationToken,
75        steel_config: config::SteelConfig,
76    ) -> Result<Self, SteelServerError> {
77        Self::new_with_commands(
78            chunk_runtime,
79            cancel_token,
80            steel_config,
81            CommandRegistry::new(),
82        )
83        .await
84    }
85
86    /// Creates a new Steel server with additional commands registered atomically at startup.
87    pub async fn new_with_commands(
88        chunk_runtime: Arc<Runtime>,
89        cancel_token: CancellationToken,
90        steel_config: config::SteelConfig,
91        command_registry: CommandRegistry,
92    ) -> Result<Self, SteelServerError> {
93        log::info!("Starting Steel Server ({GIT_HASH_SHORT})");
94
95        let permission_group_store = steel_config.permission_group_store();
96        let server_port = steel_config.server.server_port;
97        let worlds_config = steel_config.worlds;
98        let permission_groups =
99            PermissionGroupManager::new(steel_config.groups, permission_group_store).map_err(
100                |error| {
101                    SteelServerError::Core(format!("failed to validate groups config: {error}"))
102                },
103            )?;
104        let player_idle_timeout = steel_config.server.player_idle_timeout;
105        let runtime_config = steel_config.server.into_runtime_config();
106
107        let server = Server::new_with_commands(
108            chunk_runtime,
109            cancel_token.clone(),
110            runtime_config,
111            worlds_config,
112            permission_groups,
113            command_registry,
114        )
115        .await
116        .map_err(SteelServerError::Core)?;
117
118        server
119            .player_idle_timeout
120            .store(player_idle_timeout, Ordering::Relaxed);
121
122        let tcp_listener = TcpListener::bind(SocketAddrV4::new(Ipv4Addr::UNSPECIFIED, server_port))
123            .await
124            .map_err(|source| SteelServerError::Bind {
125                port: server_port,
126                source,
127            })?;
128
129        Ok(Self {
130            tcp_listener,
131            cancel_token,
132            client_id: 0,
133            server: Arc::new(server),
134            connection_session: Arc::new(ServerConnectionSession::default()),
135        })
136    }
137
138    /// Starts the server and begins accepting connections.
139    pub async fn start(&mut self, task_tracker: TaskTracker) {
140        log::info!("Started Steel Server");
141
142        let server = self.server.clone();
143        let token = self.cancel_token.clone();
144        let server_handle = tokio::spawn(async move {
145            server.run(token).await;
146        });
147
148        loop {
149            select! {
150                () = self.cancel_token.cancelled() => {
151                    break;
152                }
153                accept_result = self.tcp_listener.accept() => {
154                    let Ok((connection, address)) = accept_result else {
155                        continue;
156                    };
157                    if let Err(e) = connection.set_nodelay(true) {
158                        log::warn!("Failed to set TCP_NODELAY: {e}");
159                    }
160                    let (java_client, sender_recv, net_reader) = JavaTcpClient::new(
161                        connection,
162                        address,
163                        self.client_id,
164                        self.cancel_token.child_token(),
165                        self.server.clone(),
166                        self.connection_session.clone(),
167                        task_tracker.clone(),
168                    );
169                    self.client_id = self.client_id.wrapping_add(1);
170                    log::info!("Accepted connection from Java Edition: {address} (id {})", self.client_id);
171
172                    let java_client = Arc::new(java_client);
173                    java_client.start_outgoing_packet_task(sender_recv);
174                    java_client.start_incoming_packet_task(net_reader);
175                    // Java_client won't drop until the incoming and outcoming task close
176                    // So we dont need to care about them here anymore
177                }
178            }
179        }
180        let _ = server_handle.await;
181    }
