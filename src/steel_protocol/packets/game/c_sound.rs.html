6/// Sound source categories (matches vanilla `SoundSource` enum order).
7#[derive(Debug, Clone, Copy, PartialEq, Eq)]
8#[repr(u8)]
9pub enum SoundSource {
10    Master = 0,
11    Music = 1,
12    Records = 2,
13    Weather = 3,
14    Blocks = 4,
15    Hostile = 5,
16    Neutral = 6,
17    Players = 7,
18    Ambient = 8,
19    Voice = 9,
20    Ui = 10,
21}
22
23impl SoundSource {
24    /// Returns the vanilla command literal for this category.
25    #[must_use]
26    pub const fn name(self) -> &'static str {
27        match self {
28            Self::Master => "master",
29            Self::Music => "music",
30            Self::Records => "record",
31            Self::Weather => "weather",
32            Self::Blocks => "block",
33            Self::Hostile => "hostile",
34            Self::Neutral => "neutral",
35            Self::Players => "player",
36            Self::Ambient => "ambient",
37            Self::Voice => "voice",
38            Self::Ui => "ui",
39        }
40    }
41
42    /// Returns the `VarInt` value for the enum.
43    #[must_use]
44    pub const fn as_varint(self) -> i32 {
45        self as i32
46    }
47}

49/// Sent to play a sound effect at a specific position.
50///
51/// The position is encoded at 8x precision (divide by 8 to get actual block coordinates).
52/// This allows sub-block positioning for more accurate sound placement.
53#[derive(WriteTo, ClientPacket, Clone, Debug)]
54#[packet_id(Play = C_SOUND)]
55pub struct CSound {
56    /// The holder-encoded sound event.
57    pub sound: SoundEventHolder,
58    /// The sound source category (`VarInt`).
59    #[write(as = VarInt)]
60    pub source: i32,
61    /// X position multiplied by 8 (fixed-point).
62    pub pos: IVec3,
63    /// Volume (1.0 = normal).
64    pub volume: f32,
65    /// Pitch (1.0 = normal).
66    pub pitch: f32,
67    /// Random seed for sound variations.
68    pub seed: i64,
69}
70
71impl CSound {
72    /// Creates a new sound packet.
73    ///
74    /// # Arguments
75    /// * `sound` - Sound event to play
76    /// * `source` - Sound source category
77    /// * `x`, `y`, `z` - Position in block coordinates (will be scaled by 8)
78    /// * `volume` - Volume multiplier (1.0 = normal)
79    /// * `pitch` - Pitch multiplier (1.0 = normal)
80    /// * `seed` - Random seed for sound variations
81    #[must_use]
82    pub fn new(
83        sound: SoundEventRef,
84        source: SoundSource,
85        pos: DVec3,
86        volume: f32,
87        pitch: f32,
88        seed: i64,
89    ) -> Self {
90        Self::new_holder(
91            SoundEventHolder::registry(sound),
92            source,
93            pos,
94            volume,
95            pitch,
96            seed,
97        )
98    }
99
100    /// Creates a sound packet from a registered or direct sound holder.
101    #[must_use]
102    pub fn new_holder(
103        sound: SoundEventHolder,
104        source: SoundSource,
105        pos: DVec3,
106        volume: f32,
107        pitch: f32,
108        seed: i64,
109    ) -> Self {
110        Self {
111            sound,
112            source: source.as_varint(),
113            pos: IVec3::new(
114                (pos.x * 8.0) as i32,
115                (pos.y * 8.0) as i32,
116                (pos.z * 8.0) as i32,
117            ),
118            volume,
119            pitch,
120            seed,
121        }
122    }
123
124    /// Creates a block sound packet at the center of a block position.
125    ///
126    /// # Arguments
127    /// * `sound` - Sound event to play
128    /// * `pos` - Block position (will be centered at +0.5)
129    /// * `volume` - Volume multiplier
130    /// * `pitch` - Pitch multiplier
131    /// * `seed` - Random seed
132    #[must_use]
133    pub fn block_sound(
134        sound: SoundEventRef,
135        pos: steel_utils::BlockPos,
136        volume: f32,
137        pitch: f32,
138        seed: i64,
139    ) -> Self {
140        Self::new(
141            sound,
142            SoundSource::Blocks,
143            pos.0.as_dvec3().map(|v| v + 0.5),
144            volume,
145            pitch,
146            seed,
147        )
148    }
149}

151#[cfg(test)]
152mod tests {
153
154    use steel_registry::init_vanilla_registry;
155    use steel_registry::{RegistryEntry, sound_events};
156    use steel_utils::BlockPos;
157
158    use super::CSound;
159
160    #[test]
161    fn registered_sound_packet_uses_holder_id() {
162        init_vanilla_registry();
163
164        let packet = CSound::block_sound(
165            &sound_events::BLOCK_WOODEN_BUTTON_CLICK_ON,
166            BlockPos::ZERO,
167            1.0,
168            1.0,
169            0,
170        );
171
172        let expected_holder_id = sound_events::BLOCK_WOODEN_BUTTON_CLICK_ON.id() as i32 + 1;
173        assert_eq!(
174            sound_events::BLOCK_WOODEN_BUTTON_CLICK_ON.packet_holder_id(),
175            expected_holder_id
176        );
177        assert!(matches!(
178            packet.sound,
179            steel_registry::sound_event::SoundEventHolder::Registry(sound)
180                if sound.packet_holder_id() == expected_holder_id
181        ));
182    }
