28fn command() -> CommandNodeBuilder<CommandSource, SteelCommandRuntime> {
29    literal("teleport")
30        .then(
31            argument("location", SteelArgumentType::vec3(true))
32                .executes(teleport_source_to_position),
33        )
34        .then(
35            argument("destination", SteelArgumentType::entity())
36                .executes(teleport_source_to_entity),
37        )
38        .then(
39            argument("targets", SteelArgumentType::entities())
40                .then(target_position_branch())
41                .then(
42                    argument("destination", SteelArgumentType::entity())
43                        .executes(teleport_targets_to_entity),
44                ),
45        )
46}
47
48fn target_position_branch() -> CommandNodeBuilder<CommandSource, SteelCommandRuntime> {
49    argument("location", SteelArgumentType::vec3(true))
50        .executes(teleport_targets_to_position)
51        .then(
52            argument("rotation", SteelArgumentType::rotation())
53                .executes(teleport_targets_to_position_with_rotation),
54        )
55        .then(
56            literal("facing")
57                .then(
58                    literal("entity").then(
59                        argument("facingEntity", SteelArgumentType::entity())
60                            .executes(teleport_targets_facing_entity_feet)
61                            .then(
62                                argument("facingAnchor", SteelArgumentType::entity_anchor())
63                                    .executes(teleport_targets_facing_entity_anchor),
64                            ),
65                    ),
66                )
67                .then(
68                    argument("facingLocation", SteelArgumentType::vec3(true))
69                        .executes(teleport_targets_facing_position),
70                ),
71        )
72}

179fn teleport_to_entity(
180    context: &SteelCommandContext<CommandSource>,
181    targets: &[SharedEntity],
182    destination: &SharedEntity,
183) -> Result<i32, CommandSyntaxError> {
184    let Some(target_world) = destination.level() else {
185        return Err(CommandSyntaxError::dynamic(
186            "Teleport destination is not in a live world",
187        ));
188    };
189    let position = destination.position();
190    ensure_spawnable_position(position)?;
191    let rotation = destination.rotation();
192    ensure_same_domain_targets(targets, &target_world)?;
193
194    for target in targets {
195        let transition = TeleportTransition {
196            target_world: Arc::clone(&target_world),
197            position,
198            rotation: wrap_rotation(rotation),
199            velocity: DVec3::ZERO,
200            relatives: RelativeMovement::NONE,
201            portal_cooldown: 0,
202            as_passenger: false,
203            post_transition: TeleportPostTransition::do_nothing(),
204        };
205        perform_teleport(context.source(), target, transition, None);
206    }
207
208    send_entity_success(context.source(), targets, destination.as_ref());
209    target_count(targets)
210}

212fn teleport_to_position(
213    context: &SteelCommandContext<CommandSource>,
214    targets: &[SharedEntity],
215    destination: Coordinates,
216    rotation: Option<Coordinates>,
217    facing: Option<TeleportFacing>,
218) -> Result<i32, CommandSyntaxError> {
219    let source = context.source();
220    let position = destination.position(source);
221    ensure_spawnable_position(position)?;
222    let resolved_rotation = rotation.map(|rotation| rotation.rotation(source));
223    ensure_same_domain_targets(targets, source.world())?;
224
225    for target in targets {
226        let same_world = target
227            .level()
228            .is_some_and(|world| Arc::ptr_eq(&world, source.world()));
229        let relatives = teleport_relatives(
230            (
231                destination.is_x_relative(),
232                destination.is_y_relative(),
233                destination.is_z_relative(),
234            ),
235            rotation.map(|rotation| (rotation.is_y_relative(), rotation.is_x_relative())),
236            same_world,
237        );
238        let target_rotation = target.rotation();
239        let desired_rotation = resolved_rotation.unwrap_or(target_rotation);
240        let transition = TeleportTransition {
241            target_world: Arc::clone(source.world()),
242            position: packet_position(position, target.position(), relatives),
243            rotation: packet_rotation(desired_rotation, target_rotation, relatives),
244            velocity: DVec3::ZERO,
245            relatives,
246            portal_cooldown: 0,
247            as_passenger: false,
248            post_transition: TeleportPostTransition::do_nothing(),
249        };
250        perform_teleport(source, target, transition, facing.as_ref());
251    }
252
253    send_position_success(source, targets, position);
254    target_count(targets)
255}

265fn perform_teleport(
266    source: &CommandSource,
267    target: &SharedEntity,
268    transition: TeleportTransition,
269    facing: Option<&TeleportFacing>,
270) {
271    if let Some(player) = target.as_player() {
272        if player.is_sleeping() {
273            player.stop_sleeping();
274        }
275        player.send_packet(CSetCamera {
276            camera_id: player.id(),
277        });
278    }
279    if change_entity_world(Arc::clone(target), &transition).is_none() {
280        return;
281    }
282
283    // Vanilla applies command post-effects through the original target reference.
284    if let Some(facing) = facing {
285        match facing {
286            TeleportFacing::Position(position) => target.look_at(source.anchor(), *position),
287            TeleportFacing::Entity {
288                target: facing_target,
289                anchor,
290            } => target.look_at_entity(source.anchor(), facing_target.as_ref(), *anchor),
291        }
292    }
293
294    if target
295        .as_living_entity()
296        .is_none_or(|living| !living.is_fall_flying())
297    {
298        let velocity = target.velocity();
299        target.set_velocity(DVec3::new(velocity.x, 0.0, velocity.z));
300        target.set_on_ground(true);
301    }
302    if let Some(pathfinder) = target.as_pathfinder_mob() {
303        pathfinder.mob_base().navigation().lock().stop();
304    }
305}

307fn ensure_spawnable_position(position: DVec3) -> Result<(), CommandSyntaxError> {
308    if World::is_in_spawnable_bounds(BlockPos::from(position)) {
309        return Ok(());
310    }
311    Err(CommandSyntaxError::dynamic(TextComponent::from(
312        &translations::COMMANDS_TELEPORT_INVALID_POSITION,
313    )))
314}

316fn ensure_same_domain_targets(
317    targets: &[SharedEntity],
318    target_world: &World,
319) -> Result<(), CommandSyntaxError> {
320    for target in targets {
321        let Some(source_world) = target.level() else {
322            continue;
323        };
324        ensure_same_domain(source_world.domain(), target_world.domain())?;
325    }
326    Ok(())
327}
328
329fn ensure_same_domain(source_domain: &str, target_domain: &str) -> Result<(), CommandSyntaxError> {
330    if source_domain == target_domain {
331        return Ok(());
332    }
333    Err(CommandSyntaxError::dynamic(
334        "Entities cannot be teleported across Steel domains",
335    ))
336}

338fn teleport_relatives(
339    position_relative: (bool, bool, bool),
340    rotation_relative: Option<(bool, bool)>,
341    same_world: bool,
342) -> RelativeMovement {
343    let mut flags = 0;
344    let (relative_x, relative_y, relative_z) = position_relative;
345    if relative_x {
346        flags |= RelativeMovement::DELTA_X;
347        if same_world {
348            flags |= RelativeMovement::X;
349        }
350    }
351    if relative_y {
352        flags |= RelativeMovement::DELTA_Y;
353        if same_world {
354            flags |= RelativeMovement::Y;
355        }
356    }
357    if relative_z {
358        flags |= RelativeMovement::DELTA_Z;
359        if same_world {
360            flags |= RelativeMovement::Z;
361        }
362    }
363
364    let (relative_yaw, relative_pitch) = rotation_relative.unwrap_or((true, true));
365    if relative_yaw {
366        flags |= RelativeMovement::Y_ROT;
367    }
368    if relative_pitch {
369        flags |= RelativeMovement::X_ROT;
370    }
371    RelativeMovement::new(flags)
372}

374fn packet_position(destination: DVec3, current: DVec3, relatives: RelativeMovement) -> DVec3 {
375    DVec3::new(
376        if relatives.is_x_relative() {
377            destination.x - current.x
378        } else {
379            destination.x
380        },
381        if relatives.is_y_relative() {
382            destination.y - current.y
383        } else {
384            destination.y
385        },
386        if relatives.is_z_relative() {
387            destination.z - current.z
388        } else {
389            destination.z
390        },
391    )
392}

394fn packet_rotation(
395    desired: (f32, f32),
396    current: (f32, f32),
397    relatives: RelativeMovement,
398) -> (f32, f32) {
399    wrap_rotation((
400        if relatives.is_y_rot_relative() {
401            desired.0 - current.0
402        } else {
403            desired.0
404        },
405        if relatives.is_x_rot_relative() {
406            desired.1 - current.1
407        } else {
408            desired.1
409        },
410    ))
411}
412
413fn wrap_rotation((yaw, pitch): (f32, f32)) -> (f32, f32) {
414    (wrap_degrees(yaw), wrap_degrees(pitch))
415}
416
417fn wrap_degrees(value: f32) -> f32 {
418    let wrapped = value.rem_euclid(360.0);
419    if wrapped >= 180.0 {
420        wrapped - 360.0
421    } else {
422        wrapped
423    }
424}

426fn send_entity_success(source: &CommandSource, targets: &[SharedEntity], destination: &dyn Entity) {
427    let message = if let [target] = targets {
428        translations::COMMANDS_TELEPORT_SUCCESS_ENTITY_SINGLE
429            .message([
430                TextComponent::plain(target.plain_text_name()),
431                TextComponent::plain(destination.plain_text_name()),
432            ])
433            .component()
434    } else {
435        translations::COMMANDS_TELEPORT_SUCCESS_ENTITY_MULTIPLE
436            .message([
437                TextComponent::plain(targets.len().to_string()),
438                TextComponent::plain(destination.plain_text_name()),
439            ])
440            .component()
441    };
442    source.send_success(&message, true);
443}

445fn send_position_success(source: &CommandSource, targets: &[SharedEntity], position: DVec3) {
446    let [x, y, z] = [
447        format!("{:.6}", position.x),
448        format!("{:.6}", position.y),
449        format!("{:.6}", position.z),
450    ];
451    let message = if let [target] = targets {
452        translations::COMMANDS_TELEPORT_SUCCESS_LOCATION_SINGLE
453            .message([
454                TextComponent::plain(target.plain_text_name()),
455                TextComponent::plain(x),
456                TextComponent::plain(y),
457                TextComponent::plain(z),
458            ])
459            .component()
460    } else {
461        translations::COMMANDS_TELEPORT_SUCCESS_LOCATION_MULTIPLE
462            .message([
463                TextComponent::plain(targets.len().to_string()),
464                TextComponent::plain(x),
465                TextComponent::plain(y),
466                TextComponent::plain(z),
467            ])
468            .component()
469    };
470    source.send_success(&message, true);
471}

492    fn child(dispatcher: &Dispatcher, parent: NodeId, name: &str) -> NodeId {
493        let Some(children) = dispatcher.children(parent) else {
494            panic!("parent node should exist");
495        };
496        let Some(child) = children.iter().copied().find(|child| {
497            dispatcher
498                .node(*child)
499                .is_some_and(|node| node.name() == name)
500        }) else {
501            panic!("child {name} should exist");
502        };
503        child
504    }

506    #[test]
507    fn teleport_graph_matches_vanilla_target_and_facing_shapes() {
508        init_vanilla_registry();
509        let Ok(dispatcher) = create_dispatcher() else {
510            panic!("built-in commands should register");
511        };
512        for root_name in ["teleport", "tp"] {
513            let root = child(&dispatcher, dispatcher.root(), root_name);
514            let location = child(&dispatcher, root, "location");
515            assert!(matches!(
516                dispatcher.node(location),
517                Some(node)
518                    if node.is_executable()
519                        && node.argument_type() == Some(&SteelArgumentType::vec3(true))
520            ));
521
522            let destination = child(&dispatcher, root, "destination");
523            assert!(matches!(
524                dispatcher.node(destination),
525                Some(node)
526                    if node.is_executable()
527                        && node.argument_type() == Some(&SteelArgumentType::entity())
528            ));
529
530            let targets = child(&dispatcher, root, "targets");
531            let target_location = child(&dispatcher, targets, "location");
532            let rotation = child(&dispatcher, target_location, "rotation");
533            assert!(matches!(
534                dispatcher.node(rotation),
535                Some(node)
536                    if node.is_executable()
537                        && node.argument_type() == Some(&SteelArgumentType::rotation())
538            ));
539
540            let facing = child(&dispatcher, target_location, "facing");
541            let entity = child(&dispatcher, facing, "entity");
542            let facing_entity = child(&dispatcher, entity, "facingEntity");
543            let anchor = child(&dispatcher, facing_entity, "facingAnchor");
544            assert!(matches!(
545                dispatcher.node(anchor),
546                Some(node)
547                    if node.is_executable()
548                        && node.argument_type() == Some(&SteelArgumentType::entity_anchor())
549            ));

551            assert!(matches!(
