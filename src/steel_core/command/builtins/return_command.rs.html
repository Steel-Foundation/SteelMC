1//! Vanilla command-frame returns.
2
3use std::sync::Arc;
4
5use steel_utils::Identifier;
6
7use super::super::{
8    brigadier::{ArgumentType, CommandNodeBuilder, NodeId},
9    execution::{
10        ChainModifiers, CommandSource, CustomCommandExecutor, CustomModifierExecutor,
11        ExecutionCommandSource, ExecutionControl, SteelCommandRuntime, SteelContextChain, argument,
12        literal,
13    },
14    registration::CommandRegistration,
15};
16
17pub(super) fn registration() -> CommandRegistration<CommandSource> {
18    CommandRegistration::new(Identifier::vanilla_static("return"), command)
19}
20
21fn command(dispatcher_root: NodeId) -> CommandNodeBuilder<CommandSource, SteelCommandRuntime> {
22    literal("return")
23        .then(
24            argument("value", ArgumentType::integer(i32::MIN, i32::MAX))
25                .executes_custom(ReturnValue),
26        )
27        .then(literal("fail").executes_custom(ReturnFail))
28        .then(literal("run").redirects_custom(dispatcher_root, ReturnRun, false))
29}
30
31struct ReturnValue;
32
33impl CustomCommandExecutor<CommandSource> for ReturnValue {
34    fn run(
35        &self,
36        source: Arc<CommandSource>,
37        chain: &SteelContextChain<CommandSource>,
38        _modifiers: ChainModifiers,
39        control: &mut ExecutionControl<'_, CommandSource>,
40    ) {
41        let Ok(value) = chain.top_context().integer("value") else {
42            unreachable!("the return value executor always follows its integer argument")
43        };
44        source.callback().on_result(true, value);
45        control.return_success(value);
46    }
47}
48
49struct ReturnFail;
50
51impl CustomCommandExecutor<CommandSource> for ReturnFail {
52    fn run(
53        &self,
54        source: Arc<CommandSource>,
55        _chain: &SteelContextChain<CommandSource>,
56        _modifiers: ChainModifiers,
57        control: &mut ExecutionControl<'_, CommandSource>,
58    ) {
59        source.callback().on_result(false, 0);
60        control.return_failure();
61    }
62}
63
64struct ReturnRun;
65
66impl CustomModifierExecutor<CommandSource> for ReturnRun {
67    fn apply(
68        &self,
69        original_source: Arc<CommandSource>,
70        sources: Vec<Arc<CommandSource>>,
71        chain: &SteelContextChain<CommandSource>,
72        modifiers: ChainModifiers,
73        control: &mut ExecutionControl<'_, CommandSource>,
74    ) {
75        if sources.is_empty() {
76            if modifiers.is_return() {
77                control.queue_fallthrough();
78            }
79            return;
80        }
81
82        let Some(next_stage) = chain.next_stage() else {
83            unreachable!("return run redirects to a following command-root stage")
84        };
85        control.discard_frame();
86        control.queue_contexts(
87            next_stage,
88            original_source,
89            sources,
90            modifiers.with_return(),
91        );
92    }
93}

95#[cfg(test)]
96mod tests {
97    use steel_registry::init_vanilla_registry;
98
99    use super::super::create_dispatcher;
100    use crate::command::{brigadier::ArgumentType, execution::SteelArgumentType};
101
102    #[test]
103    fn return_graph_matches_vanillas_three_forms() {
104        init_vanilla_registry();
105        let Ok(dispatcher) = create_dispatcher() else {
106            panic!("built-in dispatcher should build");
107        };
108        let Some(root) = dispatcher.children(dispatcher.root()).and_then(|children| {
109            children.iter().copied().find(|child| {
110                dispatcher
111                    .node(*child)
112                    .is_some_and(|node| node.name() == "return")
113            })
114        }) else {
115            panic!("return root should exist");
116        };
117        let Some(children) = dispatcher.children(root) else {
118            panic!("return root should have children");
119        };
120        assert_eq!(children.len(), 3);
121
122        let value = children.iter().copied().find(|child| {
123            dispatcher
124                .node(*child)
125                .is_some_and(|node| node.name() == "value")
126        });
127        let Some(value) = value.and_then(|value| dispatcher.node(value)) else {
128            panic!("return value argument should exist");
129        };
130        assert_eq!(
131            value.argument_type(),
132            Some(&SteelArgumentType::from(ArgumentType::integer(
133                i32::MIN,
134                i32::MAX,
135            )))
136        );
137        assert!(value.is_executable());
138
139        let fail = children.iter().copied().find(|child| {
140            dispatcher
141                .node(*child)
142                .is_some_and(|node| node.name() == "fail")
143        });
144        let Some(fail) = fail.and_then(|fail| dispatcher.node(fail)) else {
145            panic!("return fail should exist");
146        };
147        assert!(fail.is_executable());
148
149        let run = children.iter().copied().find(|child| {
150            dispatcher
151                .node(*child)
152                .is_some_and(|node| node.name() == "run")
153        });
154        let Some(run) = run.and_then(|run| dispatcher.node(run)) else {
155            panic!("return run should exist");
156        };
157        assert_eq!(run.redirect(), Some(dispatcher.root()));
158        assert!(!run.is_forked_redirect());
159        assert!(run.has_redirect_modifier());
160        assert!(!run.is_executable());
161    }
