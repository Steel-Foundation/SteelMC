16pub(super) fn registration() -> CommandRegistration<CommandSource> {
17    CommandRegistration::new(Identifier::vanilla_static("kick"), |_| command())
18}
19
20fn command() -> CommandNodeBuilder<CommandSource, SteelCommandRuntime> {
21    literal("kick").then(
22        argument("targets", SteelArgumentType::players())
23            .executes(kick_players)
24            .then(argument("reason", SteelArgumentType::message()).executes(kick_players)),
25    )
26}
27
28fn kick_players(ctx: &SteelCommandContext<CommandSource>) -> Result<i32, CommandSyntaxError> {
29    let mut count: i32 = 0;
30
31    let targets = ctx.players("targets")?;
32    let message = match ctx.message("reason") {
33        Ok(message) => {
34            let parsed = message.resolve_component(ctx.source())?;
35
36            if let Content::Text { text } = &parsed.content {
37                TextComponent::from_snbt(text).unwrap_or(parsed)
38            } else {
39                parsed
40            }
41        }
42        Err(_) => translations::MULTIPLAYER_DISCONNECT_KICKED
43            .msg()
44            .component(),
45    };
46
47    for players in targets {
48        players.disconnect(message.clone());
49        ctx.source().send_success(
50            &translations::COMMANDS_KICK_SUCCESS
51                .message([players.display_name(), message.clone()])
52                .component(),
53            true,
54        );
55        count += 1;
56    }
57
58    Ok(count)
59}

69    type Dispatcher = CommandDispatcher<CommandSource, SteelCommandRuntime>;
70
71    fn child(dispatcher: &Dispatcher, parent: NodeId, name: &str) -> NodeId {
72        let Some(children) = dispatcher.children(parent) else {
73            panic!("parent node should exist");
74        };
75        let Some(child) = children.iter().copied().find(|child| {
76            dispatcher
77                .node(*child)
78                .is_some_and(|node| node.name() == name)
79        }) else {
80            panic!("child {name} should exist");
81        };
82        child
83    }

85    #[test]
86    fn clear_graph_matches_vanilla_argument_shape() {
87        init_vanilla_registry();
88        let Ok(dispatcher) = create_dispatcher() else {
89            panic!("built-in commands should register");
90        };
91        let kick = child(&dispatcher, dispatcher.root(), "kick");
92        let Some(clear_node) = dispatcher.node(kick) else {
93            panic!("kick node should exist");
94        };
95        assert!(clear_node.is_restricted());
96
97        let targets = child(&dispatcher, kick, "targets");
98        assert_eq!(
99            dispatcher
100                .node(targets)
101                .and_then(|node| node.argument_type()),
102            Some(&SteelArgumentType::players())
103        );
104        assert!(matches!(
105            dispatcher.node(targets),
106            Some(node) if node.is_executable()
107        ));
108
109        let reason = child(&dispatcher, targets, "reason");
110        assert_eq!(
111            dispatcher
112                .node(reason)
113                .and_then(|node| node.argument_type()),
114            Some(&SteelArgumentType::message())
115        );
