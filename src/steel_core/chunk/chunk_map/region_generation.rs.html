24/// Why [`ChunkMap::generate_region`] could not generate a region.
25#[derive(Debug, Clone, Error, PartialEq, Eq)]
26pub enum RegionGenerationError {
27    /// The map has active or unloading holders, or its generation refill loop is running.
28    #[error("region generation needs an empty chunk map with its refill loop stopped")]
29    MapInUse,
30    /// `min` exceeds `max` on an axis, or the region's dependency halo leaves the valid chunk range.
31    #[error("invalid region bounds {min:?}..={max:?}")]
32    InvalidBounds {
33        /// Requested minimum corner.
34        min: ChunkPos,
35        /// Requested maximum corner.
36        max: ChunkPos,
37    },
38    /// A step was disallowed, cancelled or panicked.
39    #[error("{status:?} generation failed at {pos:?}")]
40    StepFailed {
41        /// Status the failed step targeted.
42        status: ChunkStatus,
43        /// Chunk the step ran on.
44        pos: ChunkPos,
45    },
46}

48impl ChunkMap {
49    /// Generates every chunk in `min..=max` to Full and returns their holders in X-then-Z order.
50    ///
51    /// Each status is finished across the region and its dependency halo before the next
52    /// starts, so shared dependencies run once rather than once per requested chunk. Chunks
53    /// whose steps may touch the same data run in a fixed order and the rest in parallel, so the
54    /// result equals generating one chunk at a time and does not depend on thread scheduling.
55    ///
56    /// The map must be empty with its refill loop stopped, and nothing else may drive it until
57    /// this returns. The holders have no tickets, so the map never unloads them.
58    pub async fn generate_region(
59        self: &Arc<Self>,
60        min: ChunkPos,
61        max: ChunkPos,
62    ) -> Result<Vec<Arc<ChunkHolder>>, RegionGenerationError> {
63        let halo = FULL_STEP.get_accumulated_radius_of(ChunkStatus::Empty) as i32;
64        let limit = ChunkPos::MAX_COORDINATE_VALUE - halo;
65        let in_range = |low: i32, high: i32| low <= high && low >= -limit && high <= limit;
66        if !in_range(min.0.x, max.0.x) || !in_range(min.0.y, max.0.y) {
67            return Err(RegionGenerationError::InvalidBounds { min, max });
68        }
69        self.insert_region_holders(min, max, halo)?;
70
71        let mut status = Some(ChunkStatus::Empty);
72        while let Some(current) = status {
73            let radius = FULL_STEP.get_accumulated_radius_of(current) as i32;
74            self.generate_region_stage(
75                GENERATION_PYRAMID.get_step_to(current),
76                region_positions(min, max, radius),
77            )
78            .await?;
79            status = current.next();
80        }
81
82        Ok(region_positions(min, max, 0)
83            .into_iter()
84            .map(|pos| self.region_holder(pos))
85            .collect())
86    }
87
88    /// Installs a holder for every region and halo chunk, allowing only the status it needs.
89    fn insert_region_holders(
90        &self,
91        min: ChunkPos,
92        max: ChunkPos,
93        halo: i32,
94    ) -> Result<(), RegionGenerationError> {
95        let _source_phase_guard = self.source_phase_guard.lock();
96        if !self.generation_refill_stopped.load(Ordering::Acquire)
97            || !self.chunks.is_empty()
98            || !self.unloading_chunks.is_empty()
99        {
100            return Err(RegionGenerationError::MapInUse);
101        }
102
103        for pos in region_positions(min, max, halo) {
104            let dx = (min.0.x - pos.0.x).max(pos.0.x - max.0.x).max(0);
105            let dz = (min.0.y - pos.0.y).max(pos.0.y - max.0.y).max(0);
106            let distance = dx.max(dz) as usize;
107            let needed_status = if distance == 0 {
108                ChunkStatus::Full
109            } else {
110                let Some(status) = FULL_STEP.accumulated_dependencies.get(distance) else {
111                    panic!("Full dependencies must cover their own radius");
112                };
113                status
114            };
115            let holder = Arc::new(ChunkHolder::new_with_full_publications(
116                pos,
117                ticket_level_for_status(needed_status),
118                None,
119                self.world_gen_context.min_y(),
120                self.world_gen_context.height(),
121                Arc::downgrade(&self.full_publications),
122            ));
123            if self.chunks.insert_sync(pos, holder).is_err() {
124                return Err(RegionGenerationError::MapInUse);
125            }
126        }
127        Ok(())
128    }

130    /// Applies `step` to `positions` as if one chunk at a time in [`region_step_order`].
131    async fn generate_region_stage(
132        self: &Arc<Self>,
133        step: &'static ChunkStep,
134        positions: Vec<ChunkPos>,
135    ) -> Result<(), RegionGenerationError> {
136        let (read_radius, _) = region_step_radii(step);
137        let order = region_step_order(step, positions);
138        let starts = order
139            .iter()
140            .map(|&pos| {
141                if step.target_status == ChunkStatus::Features {
142                    self.referenced_structure_starts(pos)
143                } else {
144                    Vec::new()
145                }
146            })
147            .collect::<Vec<_>>();
148        let dependencies =
149            region_step_dependencies(region_conflict_distance(step), &order, &starts);
150
151        let mut tasks = Vec::<RegionTask>::with_capacity(order.len());
152        for (&pos, dependencies) in order.iter().zip(dependencies) {
153            let earlier = dependencies
154                .into_iter()
155                .map(|index| tasks[index].clone())
156                .collect::<Vec<_>>();
157            let chunk_map = Arc::clone(self);
158            let task = async move {
159                for task in earlier {
160                    task.await?;
161                }
162                chunk_map.apply_region_step(step, pos, read_radius).await
163            };
164            tasks.push(task.boxed().shared());
165        }
166        join_all(tasks).await.into_iter().collect()
167    }

169    /// Runs one step on one chunk through the same path as ticket-driven generation.
170    async fn apply_region_step(
171        self: Arc<Self>,
172        step: &'static ChunkStep,
173        pos: ChunkPos,
174        cache_radius: i32,
175    ) -> Result<(), RegionGenerationError> {
176        let chunk_map = Arc::clone(&self);
177        let cache = Arc::new(StaticCache2D::create(
178            pos.0.x,
179            pos.0.y,
180            cache_radius,
181            move |x, z| chunk_map.region_holder(ChunkPos::new(x, z)),
182        ));
183        let holder = cache.get(pos.0.x, pos.0.y);
184        let ready = holder.apply_step(step, &self, &cache, Arc::clone(&self.generation_pool));
185        let failed = RegionGenerationError::StepFailed {
186            status: step.target_status,
187            pos,
188        };
189        match ready {
190            Some(ready) => ready.await.ok_or(failed),
191            None => Err(failed),
192        }
193    }

195    /// Returns the chunks holding structure starts that Features at `pos` places pieces from.
196    fn referenced_structure_starts(&self, pos: ChunkPos) -> Vec<ChunkPos> {
197        let holder = self.region_holder(pos);
198        let Some(chunk) = holder.try_chunk(ChunkStatus::StructureReferences) else {
199            panic!("Features dependencies must have structure references at {pos:?}");
200        };
201        let references = chunk.structure_references();
202        references
203            .values()
204            .flat_map(|sources| sources.iter().copied())
205            .collect()
206    }
207
208    /// Returns a holder installed by [`Self::insert_region_holders`].
209    fn region_holder(&self, pos: ChunkPos) -> Arc<ChunkHolder> {
210        let Some(holder) = self.chunks.read_sync(&pos, |_, holder| Arc::clone(holder)) else {
211            panic!("region generation installs a holder for every dependency, missing {pos:?}");
212        };
213        holder
214    }

217/// Returns how far around its center `step` reads and writes chunks.
218///
219/// Light worksets span a cache window wider than the Light step's declared dependencies.
220const fn region_step_radii(step: &ChunkStep) -> (i32, i32) {
221    let read = step.direct_dependencies.get_radius() as i32;
222    match step.target_status {
223        ChunkStatus::Light if read < LIGHT_CACHE_RADIUS => (LIGHT_CACHE_RADIUS, LIGHT_CACHE_RADIUS),
224        ChunkStatus::Light => (read, LIGHT_CACHE_RADIUS),
225        _ => (read, step.block_state_write_radius),
226    }
227}

229/// Returns the distance within which one chunk's step may write what another's reads or writes.
230///
231/// Steps that only write their own chunk read neighbors at lower statuses, so they never conflict.
232const fn region_conflict_distance(step: &ChunkStep) -> i32 {
233    let (read, write) = region_step_radii(step);
234    if write <= 0 {
235        0
236    } else if read > write {
237        write + read
238    } else {
239        write * 2
240    }
241}

243/// Orders positions so that chunks sharing a phase are beyond the step's conflict distance.
244///
245/// Phases group positions by coordinates modulo the conflict distance plus one, then sort by X
246/// and Z. Any fixed order gives the same determinism; phases keep more chunks independent.
247pub(super) fn region_step_order(step: &ChunkStep, mut positions: Vec<ChunkPos>) -> Vec<ChunkPos> {
248    let period = region_conflict_distance(step) + 1;
249    positions.sort_by_key(|pos| {
250        (
251            pos.0.x.rem_euclid(period),
252            pos.0.y.rem_euclid(period),
253            pos.0.x,
254            pos.0.y,
255        )
256    });
257    positions
258}

260/// Returns, for each position in `order`, the earlier positions it must wait for.
261///
262/// A position waits for earlier ones within `conflict` and for the previous user of each
263/// structure-start chunk in its `starts` entry, which orders all users of that start.
264fn region_step_dependencies(
265    conflict: i32,
266    order: &[ChunkPos],
267    starts: &[Vec<ChunkPos>],
268) -> Vec<Vec<usize>> {
269    let indices = order
270        .iter()
271        .enumerate()
272        .map(|(index, &pos)| (pos, index))
273        .collect::<FxHashMap<_, _>>();
274    let mut start_users = FxHashMap::<ChunkPos, usize>::default();
275    order
276        .iter()
277        .zip(starts)
278        .enumerate()
279        .map(|(index, (pos, starts))| {
280            let mut earlier = (pos.0.x - conflict..=pos.0.x + conflict)
281                .flat_map(|x| {
282                    (pos.0.y - conflict..=pos.0.y + conflict).map(move |z| ChunkPos::new(x, z))
283                })
284                .filter_map(|other| indices.get(&other).copied())
285                .filter(|&other| other < index)
286                .collect::<Vec<_>>();
287            earlier.extend(
288                starts
289                    .iter()
290                    .filter_map(|&start| start_users.insert(start, index))
291                    .filter(|&user| user < index),
292            );
293            earlier
294        })
295        .collect()
296}

298/// Lists the rectangle expanded by `radius` in ascending X-then-Z order.
299fn region_positions(min: ChunkPos, max: ChunkPos, radius: i32) -> Vec<ChunkPos> {
300    (min.0.x - radius..=max.0.x + radius)
301        .flat_map(|x| (min.0.y - radius..=max.0.y + radius).map(move |z| ChunkPos::new(x, z)))
302        .collect()
303}

309    /// Returns the positions `pos` waits for when the given positions run `status` in order.
310    fn waits_for(
311        status: ChunkStatus,
312        positions: &[ChunkPos],
313        starts: &[Vec<ChunkPos>],
314        pos: ChunkPos,
315    ) -> Vec<ChunkPos> {
316        let step = GENERATION_PYRAMID.get_step_to(status);
317        let order = positions.to_vec();
318        let dependencies = region_step_dependencies(region_conflict_distance(step), &order, starts);
319        let Some(index) = order.iter().position(|&other| other == pos) else {
320            panic!("{pos:?} should be listed");
321        };
322        dependencies[index]
323            .iter()
324            .map(|&other| order[other])
325            .collect()
326    }
