24/// Lock-free index of sections containing randomly-ticking blocks or fluids.
25///
26/// Section writers update their bit while still holding the section lock. Readers
27/// use relaxed loads because this metadata only decides whether to attempt work;
28/// section contents remain protected by the section lock and brief staleness is
29/// acceptable in the same way as Vanilla's unsynchronized derived counters.
30#[derive(Debug)]
31pub(crate) struct RandomTickSectionBits {
32    words: Box<[AtomicU64]>,
33    section_count: usize,
34}
35
36impl RandomTickSectionBits {
37    fn new(section_count: usize) -> Self {
38        let word_count = section_count.div_ceil(u64::BITS as usize);
39        let words = (0..word_count)
40            .map(|_| AtomicU64::new(0))
41            .collect::<Vec<_>>()
42            .into_boxed_slice();
43        Self {
44            words,
45            section_count,
46        }
47    }
48
49    fn set(&self, section_index: usize, randomly_ticking: bool) {
50        debug_assert!(section_index < self.section_count);
51        let word_index = section_index / u64::BITS as usize;
52        let mask = 1_u64 << (section_index % u64::BITS as usize);
53        if randomly_ticking {
54            self.words[word_index].fetch_or(mask, Ordering::Relaxed);
55        } else {
56            self.words[word_index].fetch_and(!mask, Ordering::Relaxed);
57        }
58    }
59
60    fn contains(&self, section_index: usize) -> bool {
61        debug_assert!(section_index < self.section_count);
62        let word_index = section_index / u64::BITS as usize;
63        let mask = 1_u64 << (section_index % u64::BITS as usize);
64        self.words[word_index].load(Ordering::Relaxed) & mask != 0
65    }
66
67    /// Returns the next eligible section at or above `start`.
68    ///
69    /// Each call reloads the current word, so a random-tick callback changing a
70    /// later section can affect that later section in the same chunk pass.
71    #[must_use]
72    pub(crate) fn next(&self, start: usize) -> Option<usize> {
73        if start >= self.section_count {
74            return None;
75        }
76
77        let mut word_index = start / u64::BITS as usize;
78        let bit_index = start % u64::BITS as usize;
79        let mut bits = self.words[word_index].load(Ordering::Relaxed) & (u64::MAX << bit_index);
80        loop {
81            if bits != 0 {
82                let section_index =
83                    word_index * u64::BITS as usize + bits.trailing_zeros() as usize;
84                return (section_index < self.section_count).then_some(section_index);
85            }
86            word_index += 1;
87            let word = self.words.get(word_index)?;
88            bits = word.load(Ordering::Relaxed);
89        }
90    }
91
92    #[inline]
93    #[must_use]
94    pub(crate) fn is_empty(&self) -> bool {
95        self.next(0).is_none()
96    }
97}

101pub struct SectionHolder {
102    /// The chunk section data (requires lock to access).
103    section: SyncRwLock<ChunkSection>,
104    /// Shared lock-free random-tick section index.
105    randomly_ticking_sections: Arc<RandomTickSectionBits>,

107}

109impl SectionHolder {
110    /// Creates a new section holder.
111    #[must_use]
112    pub fn new(section: ChunkSection) -> Self {
113        let randomly_ticking_sections = Arc::new(RandomTickSectionBits::new(1));
114        Self::with_random_tick_index(section, randomly_ticking_sections, 0)
115    }
116
117    fn with_random_tick_index(
118        section: ChunkSection,
119        randomly_ticking_sections: Arc<RandomTickSectionBits>,
120        section_index: usize,
121    ) -> Self {
122        let randomly_ticking = section.is_randomly_ticking();
123        let result = Self {
124            section: SyncRwLock::new(section),
125            randomly_ticking_sections,
126            section_index,
127        };
128        if randomly_ticking {
129            result.randomly_ticking_sections.set(section_index, true);
130        }
131        result
132    }

135    ///
136    /// The mirror may briefly be stale relative to a concurrent section writer.
137    /// Section contents and the authoritative counter remain protected by the
138    /// section lock.
139    #[inline]
140    #[must_use]
141    pub fn is_randomly_ticking(&self) -> bool {
142        self.randomly_ticking_sections.contains(self.section_index)
143    }

259        }

280const BLOCKS_PER_SECTION: u16 = BlockPalette::VOLUME as u16;
281const STABLE_AIR_OCCUPANCY_ROW_COUNT: usize = BlockPalette::SIZE * BlockPalette::SIZE;
282
283/// Section-local occupancy of every state except Vanilla's three immutable air blocks.
284///
285/// `Unknown` is deliberately treated as occupied. It is used after arbitrary mutable palette
286/// access so an optimization can never hide plugin or future mutation paths that bypass the
287/// tracked setters.
288#[derive(Debug)]
289enum StableAirOccupancy {
290    Unknown,
291    Empty,
292    Full,
293    Mixed {
294        rows: Box<[u16; STABLE_AIR_OCCUPANCY_ROW_COUNT]>,
295        occupied_count: u16,
296    },
297}
298
299impl StableAirOccupancy {
300    fn from_palette(states: &BlockPalette) -> Self {
301        let data = match states {
302            BlockPalette::Homogeneous(state) if is_stable_vanilla_air(*state) => {
303                return Self::Empty;
304            }
305            BlockPalette::Homogeneous(_) => return Self::Full,
306            BlockPalette::Heterogeneous(data) => data,
307            BlockPalette::Building(_) => return Self::Unknown,
308        };
309        // Classify each palette entry once instead of every block position.
310        let mut air_states = SmallVec::<[BlockStateId; 3]>::new();
311        let mut occupied_count = 0_u16;
312        for &(state, count) in data.palette() {
313            if is_stable_vanilla_air(state) {
314                air_states.push(state);
315            } else {
316                occupied_count += count;
317            }
318        }
319        if air_states.is_empty() {
320            return Self::Full;
321        }
322        if occupied_count == 0 {
323            return Self::Empty;
324        }
325
326        let mut rows = Box::new([0; STABLE_AIR_OCCUPANCY_ROW_COUNT]);
327        // Palette cubes iterate in Y, Z, X order, so each run of 16 values is one row.
328        for (index, state) in data.iter_values().enumerate() {
329            if !air_states.contains(state) {
330                rows[index / BlockPalette::SIZE] |= 1_u16 << (index % BlockPalette::SIZE);
331            }
332        }
333        debug_assert!(occupied_count < BLOCKS_PER_SECTION);
334        Self::Mixed {
335            rows,
336            occupied_count,
337        }
338    }
339
340    fn record_change(
341        &mut self,
342        x: usize,
343        y: usize,
344        z: usize,
345        old_state: BlockStateId,
346        new_state: BlockStateId,
347    ) {
348        let old_occupied = !is_stable_vanilla_air(old_state);
349        let new_occupied = !is_stable_vanilla_air(new_state);
350        if old_occupied == new_occupied {
351            return;
352        }
353
354        let row_index = y * BlockPalette::SIZE + z;
355        let bit = 1_u16 << x;
356        match self {
357            Self::Unknown => {}
358            Self::Empty => {
359                debug_assert!(new_occupied);
360                let mut rows = Box::new([0; STABLE_AIR_OCCUPANCY_ROW_COUNT]);
361                rows[row_index] = bit;
362                *self = Self::Mixed {
363                    rows,
364                    occupied_count: 1,
365                };
366            }
367            Self::Full => {
368                debug_assert!(old_occupied);
369                let mut rows = Box::new([u16::MAX; STABLE_AIR_OCCUPANCY_ROW_COUNT]);
370                rows[row_index] &= !bit;
371                *self = Self::Mixed {
372                    rows,
373                    occupied_count: BLOCKS_PER_SECTION - 1,
374                };
375            }
376            Self::Mixed {
377                rows,
378                occupied_count,
379            } => {
380                if new_occupied {
381                    rows[row_index] |= bit;
382                    *occupied_count += 1;
383                    if *occupied_count == BLOCKS_PER_SECTION {
384                        *self = Self::Full;
385                    }
386                } else {
387                    rows[row_index] &= !bit;
388                    *occupied_count -= 1;
389                    if *occupied_count == 0 {
390                        *self = Self::Empty;
391                    }
392                }
393            }
394        }
395    }

397    fn box_is_clear(
398        &self,
399        min_x: usize,
400        max_x: usize,
401        min_y: usize,
402        max_y: usize,
403        min_z: usize,
404        max_z: usize,
405    ) -> bool {
406        debug_assert!(min_x <= max_x && max_x < BlockPalette::SIZE);
407        debug_assert!(min_y <= max_y && max_y < BlockPalette::SIZE);
408        debug_assert!(min_z <= max_z && max_z < BlockPalette::SIZE);
409
410        let Self::Mixed { rows, .. } = self else {
411            return matches!(self, Self::Empty);
412        };
413        let below_max = if max_x == u16::BITS as usize - 1 {
414            u16::MAX
415        } else {
416            (1_u16 << (max_x + 1)) - 1
417        };
418        let x_mask = below_max & (u16::MAX << min_x);
419        for y in min_y..=max_y {
420            for z in min_z..=max_z {
421                if rows[y * BlockPalette::SIZE + z] & x_mask != 0 {
422                    return false;
423                }
424            }
425        }
426        true
427    }

430fn is_stable_vanilla_air(state: BlockStateId) -> bool {
431    // The cached air flag avoids three registry lookups for the common non-air state.
432    state.is_air()
433        && (state == vanilla_blocks::AIR.default_state()
434            || state == vanilla_blocks::CAVE_AIR.default_state()
435            || state == vanilla_blocks::VOID_AIR.default_state())
436}

488    /// Reads an entire column at `(x, z)` across all sections into a caller-owned buffer.
489    ///
490    /// Holds each section's read lock once for 16 Y reads instead of acquiring
491    /// a lock per block. Indexed by `relative_y` (0 = chunk min-y).
492    /// The buffer is resized if needed and reused across calls to avoid allocation.
493    pub fn read_column_into(&self, x: usize, z: usize, buf: &mut Vec<BlockStateId>) {
494        debug_assert!(x < BlockPalette::SIZE);
495        debug_assert!(z < BlockPalette::SIZE);
496
497        let total = self.sections.len() * 16;
498        if buf.len() != total {
499            buf.resize(total, BlockStateId::default());
500        }
501        for (i, holder) in self.sections.iter().enumerate() {
502            let guard = holder.read();
503            let base = i * 16;
504            guard
505                .states
506                .copy_column_into(x, z, &mut buf[base..base + 16]);
507        }
508    }

576    /// Writes multiple blocks in one column, holding each section's write guard
577    /// across all writes to that section. Most efficient when blocks are grouped
578    /// by section (e.g. descending `relative_y` from a top-to-bottom scan).
579    pub fn write_column_blocks(&self, x: usize, z: usize, blocks: &[(usize, BlockStateId)]) {
580        const DIM: usize = BlockPalette::SIZE;
581        debug_assert!(x < DIM);
582        debug_assert!(z < DIM);
583
584        let mut i = 0;
585        while i < blocks.len() {
586            let section_idx = blocks[i].0 / DIM;
587            let mut guard = self.sections[section_idx].write();
588            guard.states.enter_building_mode();
589            let Some(cube) = guard.states.as_building_slice_mut() else {
590                unreachable!("just entered building mode")
591            };
592            let xz_base = z * DIM + x;
593            while i < blocks.len() && blocks[i].0 / DIM == section_idx {
594                let (rel_y, value) = blocks[i];
595                let local_y = rel_y % DIM;
596                cube[local_y * DIM * DIM + xz_base] = value;
597                i += 1;
598            }
599        }
600    }

602    /// Writes a batch of blocks at arbitrary positions, holding each section's
603    /// write guard across consecutive entries in the same section. Blocks should
604    /// be roughly grouped by section index for best performance.
605    ///
606    /// Each touched section enters worldgen Building mode (raw cube, no palette
607    /// tracking) so writes are O(1) stores. Per-write goes through a flat
608    /// `&mut [V]` view of the cube — bypasses the 3-arm `set` match and the
609    /// unused old-value load. `recalculate_counts` finalizes.
610    pub fn write_block_batch(&self, blocks: &[(usize, usize, usize, BlockStateId)]) {
611        const DIM: usize = BlockPalette::SIZE;
612        let mut i = 0;
613        while i < blocks.len() {
614            let section_idx = blocks[i].1 / DIM;
615            let mut guard = self.sections[section_idx].write();
616            guard.states.enter_building_mode();
617            let Some(cube) = guard.states.as_building_slice_mut() else {
618                // enter_building_mode just transitioned to Building.
619                unreachable!("just entered building mode")
620            };
621            while i < blocks.len() && blocks[i].1 / DIM == section_idx {
622                let (x, rel_y, z, value) = blocks[i];
623                let local_y = rel_y % DIM;
624                cube[local_y * DIM * DIM + z * DIM + x] = value;
625                i += 1;
626            }
627        }
628    }

630    /// Writes a batch of blocks while maintaining section counters and palette state.
631    ///
632    /// Blocks should be grouped by section index so each touched section only needs
633    /// one write guard.
634    pub(crate) fn write_tracked_block_batch(&self, blocks: &[(usize, usize, usize, BlockStateId)]) {
635        const DIM: usize = BlockPalette::SIZE;
636        let mut i = 0;
637        while i < blocks.len() {
638            let section_idx = blocks[i].1 / DIM;
639            let mut guard = self.sections[section_idx].write();
640            while i < blocks.len() && blocks[i].1 / DIM == section_idx {
641                let (x, relative_y, z, value) = blocks[i];
642                guard.set_block_state(x, relative_y % DIM, z, value);
643                i += 1;
644            }
645        }
646    }

662        let mut guard = self.sections[idx].write();
663        guard.set_block_state(relative_x, relative_y, relative_z, value);
664    }

667    ///
668    /// Callers must finalize by recounting touched sections before save,
669    /// promotion, or packet serialization.

671        &self,
672        relative_x: usize,
673        relative_y: usize,
674        relative_z: usize,
675        value: BlockStateId,

688///
689/// Contains a 16x16x16 cube of block states and biomes, along with cached
690/// counts for optimization (similar to vanilla's `LevelChunkSection`).

698    /// Used to quickly check if a section is empty.

824    /// Appends block-light source positions in `ScalableLux` local-index order.
825    pub fn append_block_light_sources(
826        &self,
827        chunk_min_x: i32,
828        section_min_y: i32,
829        chunk_min_z: i32,
830        sources: &mut Vec<BlockPos>,
831    ) {
832        if !self.maybe_has_block_light_sources() {
833            return;
834        }
835
836        for local_index in 0..BlockPalette::VOLUME {
837            let state = self.states.get_at_index(local_index);
838            if state.get_light_emission() == 0 {
839                continue;
840            }

842            sources.push(BlockPos::new(
843                chunk_min_x + (local_index & 15) as i32,
844                section_min_y + (local_index >> 8) as i32,
845                chunk_min_z + ((local_index >> 4) & 15) as i32,
846            ));
847        }
848    }

880    /// Recalculates cached counters from extracted per-state metadata.
881    ///
882    /// Iterates the palette (`O(palette_size)`) rather than every cube cell
883    /// (`O(4096)`): each block-state appears at most once in the palette and
884    /// carries its own occurrence count, so we just classify each unique state
885    /// and multiply by its count. Mirrors Moonrise's `BlockCountingBitStorage`.
886    /// For a `Homogeneous` section that's a single classify; for typical
887    /// `Heterogeneous` sections palette is well under 16 entries.
888    pub fn recalculate_counts(&mut self) {
889        self.recalculate_counts_from_palette(Self::block_state_section_counts);
890    }
891
892    fn recalculate_counts_from_palette(
893        &mut self,
894        mut counts_for_state: impl FnMut(BlockStateId) -> BlockStateSectionCounts,
895    ) {
896        self.states.finalize_building();
897        let stable_air_occupancy = StableAirOccupancy::from_palette(&self.states);
898
899        let mut non_empty: u16 = 0;
900        let mut fluid: u16 = 0;
901        let mut ticking_blocks: u16 = 0;
902        let mut ticking_fluids: u16 = 0;
903
904        match &self.states {
905            BlockPalette::Homogeneous(state) => {
906                let counts = counts_for_state(*state);
907                Self::accumulate_counter_traits(
908                    &mut non_empty,
909                    &mut fluid,
910                    &mut ticking_blocks,
911                    &mut ticking_fluids,
912                    counts,
913                    BLOCKS_PER_SECTION,
914                );
915            }
916            BlockPalette::Heterogeneous(data) => {
917                for &(state, count) in data.palette() {
918                    let counts = counts_for_state(state);
919                    Self::accumulate_counter_traits(
920                        &mut non_empty,
921                        &mut fluid,
922                        &mut ticking_blocks,
923                        &mut ticking_fluids,
924                        counts,
925                        count,
926                    );
927                }
928            }
929            BlockPalette::Building(_) => unreachable!("finalize_building was just called"),
930        }
931
932        self.non_empty_block_count = non_empty;
933        self.fluid_count = fluid;
934        self.ticking_block_count = ticking_blocks;
935        self.ticking_fluid_count = ticking_fluids;
936        self.stable_air_occupancy = stable_air_occupancy;
937    }
938
939    const fn accumulate_counter_traits(
940        non_empty: &mut u16,
941        fluid: &mut u16,
942        ticking_blocks: &mut u16,
943        ticking_fluids: &mut u16,
944        counts: BlockStateSectionCounts,
945        block_count: u16,
946    ) {
947        if !counts.is_air {
948            *non_empty += block_count;
949        }
950        if counts.has_fluid {
951            *fluid += block_count;
952        }
953        if counts.randomly_ticking_block {
954            *ticking_blocks += block_count;
955        }
956        if counts.randomly_ticking_fluid {
957            *ticking_fluids += block_count;
958        }
959    }

961    /// Whether this section's palette contains any POI-type block state.
962    ///
963    /// Lets the Full-stage POI populate skip the full 4096-block
964    /// `scan_and_populate` for the overwhelming majority of sections
965    /// (stone/dirt/air) that hold no POI blocks — a palette scan of `O(≤16)`
966    /// instead of `O(4096)`. Mirrors vanilla's `LevelChunkSection.maybeHas`.
967    #[must_use]
968    pub fn contains_poi(&self) -> bool {
969        let poi = &REGISTRY.poi_types;
970        match &self.states {
971            BlockPalette::Homogeneous(state) => poi.is_poi_state(*state),
972            BlockPalette::Heterogeneous(data) => data
973                .palette()
974                .iter()
975                .any(|(state, _)| poi.is_poi_state(*state)),
976            // Not yet finalized (only happens mid-worldgen, not at promotion);
977            // fall back to scanning rather than risk missing a POI.
978            BlockPalette::Building(_) => true,
979        }
980    }

982    /// Sets a block state and updates the cached counters.
983    ///
984    /// Returns the old block state.
985    ///
986    pub fn set_block_state(
987        &mut self,
988        x: usize,
989        y: usize,
990        z: usize,
991        new_state: BlockStateId,
992    ) -> BlockStateId {
993        self.ensure_counter_ready_for_delta();
994        let old_state = self.states.set(x, y, z, new_state);
995
996        if old_state != new_state {
997            let old_counts = Self::block_state_section_counts(old_state);
998            let new_counts = Self::block_state_section_counts(new_state);
999            self.apply_count_change(old_counts, new_counts);
1000            self.stable_air_occupancy
1001                .record_change(x, y, z, old_state, new_state);
1002        }
1003
1004        old_state
1005    }

1009    /// Returns the old block state. Cached counters are intentionally not
1010    /// updated; callers must recount before light, promotion, save, or packet
1011    /// serialization.

1013        &mut self,
1014        x: usize,
1015        y: usize,
1016        z: usize,
1017        new_state: BlockStateId,
1018    ) -> BlockStateId {
1019        self.invalidate_stable_air_occupancy();
1020        self.states.enter_building_mode();
1021        self.states.set(x, y, z, new_state)
1022    }

1026        let metadata = state.get_ticking_metadata();
1027        BlockStateSectionCounts {
1028            is_air: metadata.is_air(),
1029            has_fluid: metadata.has_fluid(),

1031            randomly_ticking_fluid: metadata.randomly_ticking_fluid(),

1038        }

1041    fn ensure_counter_ready_for_delta(&mut self) {
1042        if matches!(&self.states, BlockPalette::Building(_)) {
1043            log::debug!(
1044                "finalizing worldgen Building palette before applying a counter-aware \
1045                 block-state delta"
1046            );
1047            self.recalculate_counts();
1048        }
1049    }

1051    const fn apply_count_change(
1052        &mut self,
1053        old_counts: BlockStateSectionCounts,
1054        new_counts: BlockStateSectionCounts,
1055    ) {
1056        if !old_counts.is_air && new_counts.is_air {
1057            self.non_empty_block_count -= 1;
1058        } else if old_counts.is_air && !new_counts.is_air {
1059            self.non_empty_block_count += 1;
1060        }
1061
1062        if old_counts.has_fluid && !new_counts.has_fluid {
1063            self.fluid_count -= 1;
1064        } else if !old_counts.has_fluid && new_counts.has_fluid {
1065            self.fluid_count += 1;
1066        }
1067
1068        if old_counts.randomly_ticking_block && !new_counts.randomly_ticking_block {
1069            self.ticking_block_count -= 1;
1070        } else if !old_counts.randomly_ticking_block && new_counts.randomly_ticking_block {
1071            self.ticking_block_count += 1;
1072        }
1073
1074        if old_counts.randomly_ticking_fluid && !new_counts.randomly_ticking_fluid {
1075            self.ticking_fluid_count -= 1;
1076        } else if !old_counts.randomly_ticking_fluid && new_counts.randomly_ticking_fluid {
1077            self.ticking_fluid_count += 1;
1078        }
1079    }

1081    /// Writes the chunk section to a writer.
1082    ///
1083    /// # Panics
1084    /// - If the writer fails to write.
1085    pub fn write(&self, writer: &mut Cursor<Vec<u8>>) {
1086        self.non_empty_block_count
1087            .write(writer)
1088            .expect("Failed to write block count");
1089        self.fluid_count
1090            .write(writer)
1091            .expect("Failed to write fluid count");
1092
1093        self.states
1094            .write(writer)
1095            .expect("Failed to write block states");
1096        self.biomes.write(writer).expect("Failed to write biomes");
1097    }
