1//! Reloadable server resources.
2//!
3//! A resource is prepared off the tick thread and then published from a safe point through the
4//! server job queue. Registrations made outside reload events stay registered for the current
5//! pass; every resource is reset to its built-in data once the new values are published, so the
6//! next pass starts from a clean baseline. Steel has no event system or datapack loader yet, so
7//! only the built-in data is rebuilt.
8

25/// Every resource that `/reload` refreshes.
26/// the complete list of Minecraft reloadable resources are:
27/// - TODO `function`
28/// - DONE `advancement`
29/// - TODO `structure`
30/// - TODO `tags`
31/// - TODO `loot_table`
32/// - TODO `predicate`
33/// - TODO `recipe`
34/// - TODO `slot_source`
35/// - TODO `number_provider`
36static RELOADABLE_RESOURCES: &[&dyn ReloadableResource] = &[&AdvancementsResource];
37
38/// A server resource that can be rebuilt from its registration sources.
39pub trait ReloadableResource: Send + Sync {
40    /// Stable identifier used in diagnostics.
41    fn name(&self) -> &'static str;
42
43    /// Discards plugin and datapack data from the previous reload pass.
44    ///
45    /// Runs after the freshly prepared data is published. Registrations made outside reload
46    /// events therefore stay valid for the pass that is currently loading, and are discarded
47    /// only once it has been applied.
48    fn reset(&self) {}
49
50    /// Builds the replacement value. This runs on the blocking pool.
51    fn prepare(&self) -> Result<Box<dyn PreparedResource>, String>;
52}
53
54/// A replacement value produced by [`ReloadableResource::prepare`].
55pub trait PreparedResource: Send {
56    /// Publishes the replacement. This runs on the server tick thread.
57    fn apply(self: Box<Self>, server: &Server) -> Result<(), String>;
58}
59
60/// A prepared resource waiting to be published with its resource name.
61struct PreparedReload {
62    name: &'static str,
63    resource: Box<dyn PreparedResource>,
64}
65
66/// Errors that prevent `/reload` from publishing a full set of resources.
67#[derive(Debug, Error)]
68pub enum ResourceReloadError {
69    /// A resource could not be built.
70    #[error("failed to prepare resource '{resource}': {message}")]
71    Prepare {
72        /// Resource identifier.
73        resource: &'static str,
74        /// Failure reported by the resource.
75        message: String,
76    },
77    /// A prepared resource could not be published.
78    #[error("failed to apply resource '{resource}': {message}")]
79    Apply {
80        /// Resource identifier.
81        resource: &'static str,
82        /// Failure reported by the resource.
83        message: String,
84    },
85    /// The blocking preparation task failed.
86    #[error("resource preparation task failed: {0}")]
87    Task(#[from] JoinError),
88    /// The server stopped before the prepared resources could be published.
89    #[error("the server stopped before the reload could be applied")]
90    Cancelled,
91}

93impl Server {
94    /// Reloads every reloadable resource.
95    ///
96    /// Preparation runs on the blocking pool. The prepared values are published by a server job,
97    /// so live world and player state is never mutated off the tick thread.
98    pub async fn reload_resources(self: &Arc<Self>) -> Result<(), ResourceReloadError> {
99        // TODO: Fire the reload event here so plugins and datapacks can register their resources
100        // for the new pass. Registrations made outside the event are already present and remain
101        // valid until this pass has been published.
102        let prepared = prepare_reloadable_resources().await?;
103        let (sender, receiver) = oneshot::channel();
104        self.jobs.spawn(ApplyResourceReloadJob {
105            prepared: Some(prepared),
106            sender: Some(sender),
107        });
108        receiver
109            .await
110            .map_err(|_| ResourceReloadError::Cancelled)??;
111
112        // Reset after publishing so registrations made outside reload events were part of this
113        // pass. The next pass starts again from the built-in data. A failed reload keeps them for
114        // the next attempt instead of dropping registrations that never reached a published tree.
115        reset_reloadable_resources();
116        Ok(())
117    }
118}
119
120fn reset_reloadable_resources() {
121    for resource in RELOADABLE_RESOURCES {
122        resource.reset();
123    }
124}
125
126async fn prepare_reloadable_resources() -> Result<Vec<PreparedReload>, ResourceReloadError> {
127    let mut tasks = Vec::with_capacity(RELOADABLE_RESOURCES.len());
128    for resource in RELOADABLE_RESOURCES {
129        let name = resource.name();
130        tasks.push((name, spawn_blocking(move || resource.prepare())));
131    }
132
133    let mut prepared = Vec::with_capacity(tasks.len());
134    for (name, task) in tasks {
135        let resource = task
136            .await
137            .map_err(ResourceReloadError::Task)?
138            .map_err(|message| ResourceReloadError::Prepare {
139                resource: name,
140                message,
141            })?;
142        prepared.push(PreparedReload { name, resource });
143    }
144    Ok(prepared)
145}
146
147fn apply_reloads(
148    prepared: Vec<PreparedReload>,
149    server: &Server,
150) -> Result<(), ResourceReloadError> {
151    for PreparedReload { name, resource } in prepared {
152        resource
153            .apply(server)
154            .map_err(|message| ResourceReloadError::Apply {
155                resource: name,
156                message,
157            })?;
158    }
159    Ok(())
160}
161
162struct ApplyResourceReloadJob {
163    prepared: Option<Vec<PreparedReload>>,
164    sender: Option<oneshot::Sender<Result<(), ResourceReloadError>>>,
165}
166
167impl ServerJob for ApplyResourceReloadJob {
168    fn poll(&mut self, context: &mut ServerJobContext) -> JobPoll {
169        let Some(prepared) = self.prepared.take() else {
170            return JobPoll::Finished;
171        };
172
173        let result = match context.server() {
174            Some(server) => apply_reloads(prepared, &server),
175            None => Err(ResourceReloadError::Cancelled),
176        };
177        if let Some(sender) = self.sender.take() {
178            let _ = sender.send(result);
179        }
180        JobPoll::Finished
181    }
182}
