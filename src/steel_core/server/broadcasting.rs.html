13    pub fn broadcast_system_chat(&self, message: &TextComponent, excluded_player: Option<Uuid>) {
14        log::info!("{}", message.to_plain(&DisplayResolutor));
15        self.online_players.iter_players(|uuid, player| {
16            if Some(*uuid) != excluded_player {
17                player.send_packet(CSystemChat::new(message, false, player));
18            }
19            true
20        });
21    }

34    fn log_chat_message(outgoing: &OutgoingChatMessage, chat_type: &ChatTypeBound) {
35        let tag = if outgoing.is_signed() {
36            ""
37        } else {
38            "[Not Secure] "
39        };
40        let sender_name = chat_type.sender_name.to_plain(&DisplayResolutor);
41        let content = outgoing.plain_content();
42
43        let formatted = match chat_type.registry_id {
44            id if id == vanilla_chat_types::SAY_COMMAND.id() as i32 => {
45                format!("[{sender_name}] {content}")
46            }
47            id if id == vanilla_chat_types::EMOTE_COMMAND.id() as i32 => {
48                format!("* {sender_name} {content}")
49            }
50            // Standard chat and fallback
51            _ => {
52                return steel_utils::chat!(sender_name, "{tag}{}", content);
53            }
54        };
55
56        steel_utils::console!("{tag}{formatted}");
57    }

59    /// Builds the tab list header/footer with recent and five-second tick statistics.
60    pub(super) fn tab_list_components(
61        tick_stats: TabListTickStats,
62    ) -> (TextComponent, TextComponent) {
63        // Color TPS based on value
64        let tps_color = if tick_stats.tps >= 19.5 {
65            Color::Green
66        } else if tick_stats.tps >= 15.0 {
67            Color::Yellow
68        } else {
69            Color::Red
70        };
71
72        let mspt_color = |mspt: f32| {
73            if mspt <= 50.0 {
74                Color::Aqua
75            } else {
76                Color::Red
77            }
78        };
79
80        let header = TextComponent::plain("\n").add_children(vec![
81            TextComponent::plain("Steel Dev Build").color(Color::Yellow),
82            TextComponent::plain("\n"),
83        ]);
84        let footer = TextComponent::plain("\n").add_children(vec![
85            TextComponent::plain("TPS: ").color(Color::Gray),
86            TextComponent::plain(format!("{:.1}", tick_stats.tps)).color(tps_color),
87            TextComponent::plain(" | ").color(Color::DarkGray),
88            TextComponent::plain("MSPT: ").color(Color::Gray),
89            TextComponent::plain(format!("{:.2}", tick_stats.recent_mspt))
90                .color(mspt_color(tick_stats.recent_mspt)),
91            TextComponent::plain(" recent | ").color(Color::Gray),
92            TextComponent::plain(format!("{:.2}", tick_stats.average_mspt))
93                .color(mspt_color(tick_stats.average_mspt)),
94            TextComponent::plain(" avg (5s) | ").color(Color::Gray),
95            TextComponent::plain(format!("{:.2}", tick_stats.p95_mspt))
96                .color(mspt_color(tick_stats.p95_mspt)),
97            TextComponent::plain(" p95").color(Color::Gray),
98            TextComponent::plain("\n"),
99        ]);
100
101        (header, footer)
102    }

123    pub(super) fn broadcast_player_join_message(
124        &self,
125        player: &Player,
126        previous_name: Option<&str>,
127    ) {
128        let display_name = player.display_name();
129        // Fallback to the current name when the cache has no prior entry.
130        let old_name = previous_name.unwrap_or(player.gameprofile.name());
131        let message: TextComponent = if player.gameprofile.name().eq_ignore_ascii_case(old_name) {
132            translations::MULTIPLAYER_PLAYER_JOINED
133                .message([display_name])
134                .into()
135        } else {
136            translations::MULTIPLAYER_PLAYER_JOINED_RENAMED
137                .message([display_name, TextComponent::plain(old_name.to_owned())])
138                .into()
139        };
140        let message = message.color(Color::Yellow);
141        self.broadcast_system_chat(&message, Some(player.gameprofile.id()));
142    }

152    /// Broadcasts the current tick rate and frozen state to all clients.
153    /// This should be called whenever the tick rate or frozen state changes.
154    pub fn broadcast_ticking_state(&self) {
155        let tick_manager = self.tick_rate_manager.read();
156        let packet = CTickingState::new(tick_manager.tick_rate(), tick_manager.is_frozen());
157        drop(tick_manager);

159        self.broadcast_to_online(packet);

172    /// Sends the current ticking state and step packets to a joining player.
173    /// This should be called when a player joins the server.
174    pub fn send_ticking_state_to_player(&self, player: &Player) {
175        let tick_manager = self.tick_rate_manager.read();
176        let state_packet = CTickingState::new(tick_manager.tick_rate(), tick_manager.is_frozen());
177        let step_packet = CTickingStep::new(tick_manager.frozen_ticks_to_run());
178        drop(tick_manager);
179
180        player.send_packet(state_packet);
181        player.send_packet(step_packet);
182    }

194    /// Resends the command tree and vanilla client permission-level projection.
195    pub fn resend_player_permission_context(self: &Arc<Self>, player: &Arc<Player>) {
196        let world = player.get_world();
197        player.send_packet(CEntityEvent {
198            entity_id: player.id(),
199            event: client_permission_event(player, &world),
200        });
201
202        let server = player.server();
203        if !Arc::ptr_eq(&server, self) {
204            tracing::error!(
205                player = %player.gameprofile.name(),
206                "cannot project commands from a different server"
207            );
208            return;
209        }
210        let Some(shared_player) = self.online_players.get_by_uuid(&player.gameprofile.id()) else {
211            tracing::error!(
212                player = %player.gameprofile.name(),
213                "cannot project commands for a player outside the online player map"
214            );
215            return;
216        };
217        if !Arc::ptr_eq(&shared_player, player) {
218            tracing::error!(
219                player = %player.gameprofile.name(),
220                "cannot project commands for a stale player handle"
221            );
222            return;
223        }
224        let source = CommandSource::new(CommandSender::Player(shared_player), server);
225        let commands = {
226            let dispatcher = self.command_dispatcher.read();
227            command_tree_packet(&dispatcher, &source)
228        };
229        match commands {
230            Ok(commands) => player.send_packet(commands),
231            Err(error) => tracing::error!(
232                player = %player.gameprofile.name(),
233                %error,
234                "failed to project the player's command tree"
235            ),
236        }
237    }
