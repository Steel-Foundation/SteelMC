172fn cap_positive_thread_count(
173    configured_threads: Option<usize>,
174    available_threads: usize,
175) -> Option<usize> {
176    let configured_threads = configured_threads.filter(|&threads| threads > 0)?;
177    Some(configured_threads.min(available_threads.max(1)))

217fn can_teleport_between_worlds(
218    entity: &dyn Entity,
219    source_world: &World,
220    target_world: &World,
221    projectile_owner_seen_credits: impl Fn(&uuid::Uuid) -> Option<bool>,
222) -> bool {
223    if is_end_return_transition(source_world.dimension_type, target_world.dimension_type) {
224        return can_entity_return_from_end_to_overworld(entity, projectile_owner_seen_credits);
225    }

230fn is_end_return_transition(
231    source_dimension_type: DimensionTypeRef,
232    target_dimension_type: DimensionTypeRef,
233) -> bool {
234    source_dimension_type == &vanilla_dimension_types::THE_END
235        && target_dimension_type == &vanilla_dimension_types::OVERWORLD
236}

238fn is_nether_dimension_type(world: &World) -> bool {
239    world.dimension_type == &vanilla_dimension_types::THE_NETHER

242fn can_entity_return_from_end_to_overworld(
243    entity: &dyn Entity,
244    projectile_owner_seen_credits: impl Fn(&uuid::Uuid) -> Option<bool>,
245) -> bool {
246    if entity.entity_type() == &vanilla_entities::ENDER_PEARL
247        && entity
248            .projectile_owner_uuid()
249            .and_then(|uuid| projectile_owner_seen_credits(&uuid))
250            == Some(false)
251    {
252        return false;
253    }
254
255    direct_passengers_allow_end_return(entity)
256}
257
258fn direct_passengers_allow_end_return(entity: &dyn Entity) -> bool {
259    for passenger in entity.passengers() {
260        if passenger
261            .as_player()
262            .is_some_and(|player| !player.has_seen_credits())
263        {
264            return false;
265        }
266    }
267
268    true
269}

336/// Failure while atomically editing one player's persisted permission state.
337#[derive(Debug, thiserror::Error)]
338pub enum PlayerPermissionUpdateError<E> {
339    /// The caller rejected the proposed edit.
340    #[error("{0}")]
341    Edit(E),
342    /// The edit assigns a group that is not configured.
343    #[error("unknown permission group '{0}'")]
344    UnknownGroup(String),
345    /// The permission snapshot could not be persisted.
346    #[error("failed to update player permissions: {0}")]
347    Storage(io::Error),
348}
349
350impl<E> From<io::Error> for PlayerPermissionUpdateError<E> {
351    fn from(value: io::Error) -> Self {
352        Self::Storage(value)
353    }
354}

470    /// Waits until the server reaches `target_tick`.
471    pub async fn wait_until_tick(&self, target_tick: u64) {
472        loop {
473            let tick_changed = self.server_tick_changed.notified();
474            tokio::pin!(tick_changed);
475            tick_changed.as_mut().enable();
476
477            if self.current_tick() >= target_tick {
478                return;
479            }
480
481            tick_changed.await;
482        }
483    }

485    pub(crate) fn permission_rule_suggestions(&self) -> Vec<String> {
486        let mut suggestions = self
487            .command_permission_keys
488            .iter()
489            .cloned()
490            .collect::<BTreeSet<_>>();
491        let config = self.permission_groups.config_snapshot();
492        for group in config.groups.values() {
493            suggestions.extend(group.allow.iter().cloned());
494            suggestions.extend(group.deny.iter().cloned());
495        }
496        for (_, state) in self.player_permission_states.read().entries() {
497            suggestions.extend(state.overrides().entries().iter().map(|entry| {
498                PermissionRuleExpression::new(entry.key().clone(), entry.context().clone())
499                    .to_string()
500            }));
501        }
502        suggestions.into_iter().collect()
503    }

539    /// Creates a new server and atomically merges startup command extensions after built-ins.
540    #[expect(
541        clippy::too_many_lines,
542        reason = "server initialization is a single cohesive flow"
543    )]
544    pub async fn new_with_commands(
545        chunk_runtime: Arc<Runtime>,
546        cancel_token: CancellationToken,
547        config: RuntimeConfig,
548        worlds_config: WorldsConfig,
549        permission_groups: PermissionGroupManager,
550        command_registry: CommandRegistry,
551    ) -> Result<Self, String> {
552        validate_login_security(config.online_mode, config.encryption).map_err(str::to_owned)?;
553        let config = Arc::new(config);
554        init_globals();
555        log::info!(
556            "SteelMC is not affiliated with Mojang or Microsoft. Use is subject to the Minecraft EULA: https://aka.ms/MinecraftEULA"
557        );
558
559        // Authlib starts this fetch alongside server initialization and waits on first use.
560        // It runs whatever the login mode, because `handle_chat_session_update` reads these
561        // keys with no online-mode gate, as vanilla does.
562        let service_keys = Arc::new(
563            ServiceKeyStore::new(config.services_server.as_deref())
564                .map_err(|error| format!("failed to configure Minecraft services keys: {error}"))?,
565        );
566        let service_keys_ready = service_keys.start(cancel_token.clone());
567
568        let registry_cache = RegistryCache::new(config.compression);
569
570        let (generator_registry, storage_registry) = world_config_registries()?;
571        let resolved_worlds = worlds_config
572            .validate_and_resolve(&generator_registry, &storage_registry)
573            .map_err(|e| format!("failed to validate worlds.toml: {e}"))?;
574
575        let generation_pool: Arc<ThreadPool> = Arc::new({
576            let mut builder = ThreadPoolBuilder::new().thread_name(|i| format!("rayon-gen-{i}"));
577            if let Some(chunk_generation_threads) =
578                configured_chunk_generation_threads(config.chunk_generation_threads)
579            {
580                builder = builder.num_threads(chunk_generation_threads);
581            }
582            // Debug builds have deep call chains in density functions that overflow the default 2 MB stack
583            if cfg!(debug_assertions) {
584                builder = builder.stack_size(DEBUG_STACK_SIZE);
585            }
586            builder
587                .build()
588                .map_err(|e| format!("failed to create generation thread pool: {e}"))?
589        });
590        let chunk_encoding_pool = Arc::new({
591            let mut builder =
592                ThreadPoolBuilder::new().thread_name(|i| format!("rayon-chunk-encode-{i}"));
593            if let Some(chunk_encoding_threads) =
594                configured_chunk_encoding_threads(config.chunk_encoding_threads)
595            {
596                builder = builder.num_threads(chunk_encoding_threads);
597            }
598            builder
599                .build()
600                .map_err(|e| format!("failed to create chunk encoding thread pool: {e}"))?
601        });
602
603        let player_data_storage = PlayerDataStorage::new(
604            resolved_worlds.save_path.clone(),
605            resolved_worlds.player_storage.clone(),
606        )
607        .await
608        .map_err(|e| format!("failed to create player data storage: {e}"))?;
609        let player_permission_states = player_data_storage
610            .load_permission_subjects()
611            .await
612            .map_err(|error| format!("failed to load player permissions: {error}"))?;
613        let known_players = player_data_storage
614            .load_known_players()
615            .await
616            .map_err(|error| format!("failed to load known players: {error}"))?;
617        let mut worlds = WorldMap::new(
618            resolved_worlds.default_domain.clone(),
619            &resolved_worlds.domains,
620            &resolved_worlds.worlds,
621        );
622
623        for world_entry in &resolved_worlds.worlds {
624            let default_world_path = resolved_worlds
625                .save_path
626                .join(&world_entry.domain)
627                .join("worlds")
628                .join(&world_entry.name);
629            let storage_output = storage_registry
630                .create(
631                    &world_entry.storage,
632                    &resolved_worlds.save_path,
633                    Path::new(&default_world_path),
634                )
635                .map_err(|e| format!("failed to create storage for {}: {e}", world_entry.key))?;
636            let world_seed = LevelDataManager::load_seed_or_default(
637                storage_output.level_data_path.as_deref(),
638                world_entry.seed,
639            )
640            .await
641            .map_err(|e| {
642                format!(
643                    "failed to load level data seed for {}: {e}",
644                    world_entry.key
645                )
646            })?;
647            let generator_output = generator_registry
648                .create(
649                    storage_output.level_data_path.as_deref(),
650                    &world_entry.generator_config,
651                    world_seed,
652                    generation_pool.clone(),
653                )
654                .map_err(|e| format!("failed to create generator for {}: {e}", world_entry.key))?;
655            let generation_settings = generation_settings_for_world(world_entry, &generator_output);
656            let world = World::new_with_config_and_encoding_pool(
657                chunk_runtime.clone(),
658                world_entry.key.clone(),
659                generator_output.dimension_type,
660                world_seed,
661                WorldConfig {
662                    storage: storage_output.storage,
663                    level_data_path: storage_output
664                        .level_data_path
665                        .map(|path| path.to_string_lossy().into_owned()),
666                    generator: Arc::new(generator_output.generator),
667                    generation_settings,
668                    view_distance: config.view_distance,
669                    simulation_distance: config.simulation_distance,
670                    max_chained_neighbor_updates: config.max_chained_neighbor_updates,
671                    compression: config.compression,
672                    is_flat: generator_output.is_flat,
673                    sea_level: generator_output.sea_level,
674                    default_gamemode: world_entry.default_gamemode,
675                    difficulty: world_entry.difficulty,
676                },
677                generation_pool.clone(),
678                Arc::clone(&chunk_encoding_pool),
679            )
680            .await
681            .map_err(|e| format!("failed to create world {}: {e}", world_entry.key))?;
682            world
683                .initialize_spawn_if_needed()
684                .await
685                .map_err(|e| format!("failed to initialize spawn for {}: {e}", world_entry.key))?;
686            worlds.insert(world_entry.key.clone(), world);
687        }
688
689        let scoreboards = DomainScoreboards::load(&worlds)
690            .await
691            .map_err(|error| format!("failed to load domain scoreboards: {error}"))?;
692        let command_storage = DomainCommandStorage::load(&worlds)
693            .await
694            .map_err(|error| format!("failed to load domain command storage: {error}"))?;
695        let registered_commands = create_registered_dispatcher(command_registry)
696            .map_err(|error| format!("failed to register commands: {error}"))?;
697        let command_permission_keys = registered_commands
698            .permissions
699            .into_iter()
700            .map(|permission| permission.as_str().to_owned())
701            .collect();
702
703        // Steel finishes the initial attempt before opening its listener, except offline,
704        // where `enforces_secure_chat` needs online mode so nothing acts on the result.
705        if config.online_mode && service_keys_ready.await.is_err() {
706            log::error!("Minecraft services key fetch task stopped before its initial attempt");
707        }
708
709        Ok(Server {
710            config,
711            permission_groups,
712            cancel_token,
713            key_store: KeyStore::create(),
714            worlds,
715            online_players: PlayerMap::new(),
716            player_admissions: SyncMutex::new(FxHashMap::default()),
717            player_admission_changed: Notify::new(),
718            server_tick_changed: Notify::new(),
719            registry_cache,
720            tick_rate_manager: SyncRwLock::new(TickRateManager::new()),
721            player_idle_timeout: AtomicI32::new(0),
722            scoreboards,
723            command_storage,
724            command_dispatcher: SyncRwLock::new(registered_commands.dispatcher),
725            command_permission_keys,
726            command_requests: CommandRequestQueue::new(),
727            packet_processor: PacketProcessor::new(),
728            chunk_encoding_pool,
729            jobs: ServerJobQueue::new(),
730            player_data_storage,
731            player_permission_states: SyncRwLock::new(player_permission_states),
732            player_permission_updates: AsyncMutex::new(()),
733            known_players: SyncMutex::new(KnownPlayerCacheState::new(known_players)),
734            known_player_save_idle: Notify::new(),
735            profile_lookup_client: reqwest::Client::new(),
736            service_keys,
737            pending_player_joins: PlayerJoinQueue::new(),
738            pending_player_disconnects: PlayerDisconnectQueue::new(),
739            pending_world_changes: SyncMutex::new(vec![]),
740            pending_domain_switches: SyncMutex::new(vec![]),
741        })
742    }

806    /// Pauses later packets while `player` is replaced by a new incarnation.
807    pub(crate) fn begin_player_packet_transition(
808        &self,
809        player: &Arc<Player>,
810    ) -> Option<PlayerPacketTransition> {
811        if player.connection.closed() || !player.session.is_current_player(player) {
812            return None;
813        }
814        self.packet_processor.pause_player_session(&player.session)
815    }

817    /// Resumes packets retained by an exact player-replacement transition.
818    pub(crate) fn finish_player_packet_transition(
819        &self,
820        transition: PlayerPacketTransition,
821    ) -> bool {
822        self.packet_processor.resume_player_session(transition)
823    }

831    /// Returns Brigadier completions visible to a command sender.
832    pub fn command_completions(
833        self: &Arc<Self>,
834        sender: CommandSender,
835        input: &str,
836    ) -> Vec<CommandCompletion> {
837        if !CommandExecutionOwner::capture(sender.clone(), self).is_current(self) {
838            return Vec::new();
839        }
840        match self.build_command_suggestions(sender, input) {
841            Ok(suggestions) => {
842                let range = suggestions.range();
843                suggestions
844                    .list()
845                    .iter()
846                    .map(|suggestion| {
847                        CommandCompletion::new(
848                            range.start(),
849                            range.len(),
850                            suggestion.text().to_owned(),
851                        )
852                    })
853                    .collect()
854            }
855            Err(error) => {
856                tracing::warn!(%error, "failed to build command suggestions");
857                Vec::new()
858            }
859        }
860    }
