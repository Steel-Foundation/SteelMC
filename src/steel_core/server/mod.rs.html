183fn cap_positive_thread_count(
184    configured_threads: Option<usize>,
185    available_threads: usize,
186) -> Option<usize> {
187    let configured_threads = configured_threads.filter(|&threads| threads > 0)?;
188    Some(configured_threads.min(available_threads.max(1)))

198pub async fn test_server_with_worlds_and_config(
199    default_domain: String,
200    domains: &[ResolvedDomainConfig],
201    loaded_worlds: &[Arc<World>],
202    player_permission_states: PermissionSubjectIndex,
203    config: Arc<RuntimeConfig>,
204) -> Result<Arc<Server>, String> {
205    let mut worlds = WorldMap::new(default_domain, domains, &[]);
206    for world in loaded_worlds {
207        worlds.insert(world.key.clone(), Arc::clone(world));
208    }
209    worlds.validate_game_times()?;
210    let scoreboards = DomainScoreboards::load(&worlds)
211        .await
212        .map_err(|error| format!("test scoreboards should load: {error}"))?;
213    let command_storage = DomainCommandStorage::load(&worlds)
214        .await
215        .map_err(|error| format!("test command storage should load: {error}"))?;
216    let random_sequences = DomainRandomSequences::load(&worlds)
217        .await
218        .map_err(|error| format!("test random sequences should load: {error}"))?;
219    let player_data_storage = PlayerDataStorage::in_memory();
220    let registered_commands = create_registered_dispatcher(CommandRegistry::new())
221        .map_err(|error| format!("test commands should register: {error}"))?;
222    let command_permission_keys = registered_commands
223        .permissions
224        .iter()
225        .map(|permission| permission.as_str().to_owned())
226        .collect();
227    let permission_groups = PermissionGroupManager::transient(PermissionGroupsConfig::default())
228        .map_err(|error| format!("test permission groups should resolve: {error}"))?;
229    let registry_cache = RegistryCache::new(config.compression);
230    let chunk_encoding_pool = ThreadPoolBuilder::new()
231        .num_threads(1)
232        .build()
233        .map_err(|error| format!("test chunk encoding pool should initialize: {error}"))?;
234    let service_keys = ServiceKeyStore::new(None)
235        .map_err(|error| format!("test services key store should initialize: {error}"))?;
236    let damage_history = loaded_worlds
237        .first()
238        .and_then(|world| world.damage_history.upgrade())
239        .ok_or("test worlds must have a live damage history owner")?;
240    let shares_history = loaded_worlds.iter().all(|world| {
241        world
242            .damage_history
243            .upgrade()
244            .is_some_and(|history| Arc::ptr_eq(&history, &damage_history))
245    });
246    if !shares_history {
247        return Err("test worlds must share the server's history owner".to_owned());
248    }
249
250    Ok(Arc::new(Server {
251        damage_history,
252        config,
253        permission_groups,
254        cancel_token: CancellationToken::new(),
255        key_store: KeyStore::create(),
256        registry_cache,
257        worlds,
258        online_players: PlayerMap::new(),
259        player_admissions: SyncMutex::new(FxHashMap::default()),
260        player_admission_changed: Notify::new(),
261        server_tick_changed: Notify::new(),
262        tick_rate_manager: SyncRwLock::new(TickRateManager::new()),
263        perf: PerfRecorder::new(),
264        scoreboards,
265        command_storage,
266        random_sequences,
267        command_dispatcher: SyncRwLock::new(registered_commands.dispatcher),
268        command_permission_keys,
269        command_requests: CommandRequestQueue::new(),
270        packet_processor: PacketProcessor::new(),
271        chunk_encoding_pool: Arc::new(chunk_encoding_pool),
272        jobs: ServerJobQueue::new(),
273        player_data_storage,
274        player_permission_states: SyncRwLock::new(player_permission_states),
275        player_permission_updates: AsyncMutex::new(()),
276        known_players: SyncMutex::new(KnownPlayerCacheState::new(KnownPlayers::new())),
277        known_player_save_idle: Notify::new(),
278        profile_lookup_client: reqwest::Client::new(),
279        service_keys: Arc::new(service_keys),
280        pending_player_joins: PlayerJoinQueue::new(),
281        pending_player_disconnects: PlayerDisconnectQueue::new(),
282        pending_world_changes: SyncMutex::new(Vec::new()),
283        pending_domain_switches: SyncMutex::new(Vec::new()),
284        player_idle_timeout: AtomicI32::new(0),
285    }))

316fn can_teleport_between_worlds(
317    entity: &dyn Entity,
318    source_world: &World,
319    target_world: &World,
320    projectile_owner_seen_credits: impl Fn(&uuid::Uuid) -> Option<bool>,
321) -> bool {
322    if is_end_return_transition(source_world.dimension_type, target_world.dimension_type) {
323        return can_entity_return_from_end_to_overworld(entity, projectile_owner_seen_credits);
324    }

329fn is_end_return_transition(
330    source_dimension_type: DimensionTypeRef,
331    target_dimension_type: DimensionTypeRef,
332) -> bool {
333    source_dimension_type == &vanilla_dimension_types::THE_END
334        && target_dimension_type == &vanilla_dimension_types::OVERWORLD
335}

337fn is_nether_dimension_type(world: &World) -> bool {
338    world.dimension_type == &vanilla_dimension_types::THE_NETHER
339}
340
341fn can_entity_return_from_end_to_overworld(
342    entity: &dyn Entity,
343    projectile_owner_seen_credits: impl Fn(&uuid::Uuid) -> Option<bool>,
344) -> bool {
345    if entity.entity_type() == &vanilla_entities::ENDER_PEARL
346        && entity
347            .projectile_owner_uuid()
348            .and_then(|uuid| projectile_owner_seen_credits(&uuid))
349            == Some(false)
350    {
351        return false;
352    }
353
354    direct_passengers_allow_end_return(entity)
355}
356
357fn direct_passengers_allow_end_return(entity: &dyn Entity) -> bool {
358    for passenger in entity.passengers() {
359        if passenger
360            .as_player()
361            .is_some_and(|player| !player.has_seen_credits())
362        {
363            return false;
364        }
365    }
366
367    true
368}

436/// Failure while atomically editing one player's persisted permission state.
437#[derive(Debug, thiserror::Error)]
438pub enum PlayerPermissionUpdateError<E> {
439    /// The caller rejected the proposed edit.
440    #[error("{0}")]
441    Edit(E),
442    /// The edit assigns a group that is not configured.
443    #[error("unknown permission group '{0}'")]
444    UnknownGroup(String),
445    /// The permission snapshot could not be persisted.
446    #[error("failed to update player permissions: {0}")]
447    Storage(io::Error),
448}
449
450impl<E> From<io::Error> for PlayerPermissionUpdateError<E> {
451    fn from(value: io::Error) -> Self {
452        Self::Storage(value)
453    }

575    /// Waits until the server reaches `target_tick`.
576    pub async fn wait_until_tick(&self, target_tick: u64) {
577        loop {
578            let tick_changed = self.server_tick_changed.notified();
579            tokio::pin!(tick_changed);
580            tick_changed.as_mut().enable();
581
582            if self.current_tick() >= target_tick {
583                return;
584            }
585
586            tick_changed.await;
587        }
588    }

590    pub(crate) fn permission_rule_suggestions(&self) -> Vec<String> {
591        let mut suggestions = self
592            .command_permission_keys
593            .iter()
594            .cloned()
595            .collect::<BTreeSet<_>>();
596        let config = self.permission_groups.config_snapshot();
597        for group in config.groups.values() {
598            suggestions.extend(group.allow.iter().cloned());
599            suggestions.extend(group.deny.iter().cloned());
600        }
601        for (_, state) in self.player_permission_states.read().entries() {
602            suggestions.extend(state.overrides().entries().iter().map(|entry| {
603                PermissionRuleExpression::new(entry.key().clone(), entry.context().clone())
604                    .to_string()
605            }));
606        }
607        suggestions.into_iter().collect()
608    }

610    pub(crate) fn permission_metadata_suggestions(&self) -> Vec<String> {
611        let mut suggestions = BTreeSet::new();
612        let config = self.permission_groups.config_snapshot();
613        for group in config.groups.values() {
614            suggestions.extend(group.metadata.iter().map(|rule| rule.key.clone()));
615        }
616        for (_, state) in self.player_permission_states.read().entries() {
617            suggestions.extend(state.metadata_overrides().entries().iter().map(|entry| {
618                PermissionMetadataExpression::new(entry.key().clone(), entry.context().clone())
619                    .to_string()
620            }));
621        }
622        suggestions.into_iter().collect()
623    }

644    /// Creates a new server and atomically merges startup command extensions after built-ins.
645    #[expect(
646        clippy::too_many_lines,
647        reason = "server initialization is a single cohesive flow"
648    )]
649    pub async fn new_with_commands(
650        chunk_runtime: Arc<Runtime>,
651        cancel_token: CancellationToken,
652        config: RuntimeConfig,
653        worlds_config: WorldsConfig,
654        permission_groups: PermissionGroupManager,
655        command_registry: CommandRegistry,
656    ) -> Result<Self, String> {
657        validate_login_security(config.online_mode, config.encryption).map_err(str::to_owned)?;
658        let config = Arc::new(config);
659        init_globals();
660        log::info!(
661            "SteelMC is not affiliated with Mojang or Microsoft. Use is subject to the Minecraft EULA: https://aka.ms/MinecraftEULA"
662        );
663
664        // Authlib starts this fetch alongside server initialization and waits on first use.
665        // It runs whatever the login mode, because `handle_chat_session_update` reads these
666        // keys with no online-mode gate, as vanilla does.
667        let service_keys = Arc::new(
668            ServiceKeyStore::new(config.services_server.as_deref())
669                .map_err(|error| format!("failed to configure Minecraft services keys: {error}"))?,
670        );
671        let service_keys_ready = service_keys.start(cancel_token.clone());
672
673        let registry_cache = RegistryCache::new(config.compression);
674
675        let (generator_registry, storage_registry) = world_config_registries()?;
676        let resolved_worlds = worlds_config
677            .validate_and_resolve(&generator_registry, &storage_registry)
678            .map_err(|e| format!("failed to validate worlds.toml: {e}"))?;
679
680        let mut world_storage = storage_registry.resolve_worlds(&resolved_worlds)?;
681
682        let generation_pool: Arc<ThreadPool> = Arc::new({
683            let mut builder = ThreadPoolBuilder::new().thread_name(|i| format!("rayon-gen-{i}"));
684            if let Some(chunk_generation_threads) =
685                configured_chunk_generation_threads(config.chunk_generation_threads)
686            {
687                builder = builder.num_threads(chunk_generation_threads);
688            }
689            // Debug builds have deep call chains in density functions that overflow the default 2 MB stack
690            if cfg!(debug_assertions) {
691                builder = builder.stack_size(DEBUG_STACK_SIZE);
692            }
693            builder
694                .build()
695                .map_err(|e| format!("failed to create generation thread pool: {e}"))?
696        });
697        let chunk_encoding_pool = Arc::new({
698            let mut builder =
699                ThreadPoolBuilder::new().thread_name(|i| format!("rayon-chunk-encode-{i}"));
700            if let Some(chunk_encoding_threads) =
701                configured_chunk_encoding_threads(config.chunk_encoding_threads)
702            {
703                builder = builder.num_threads(chunk_encoding_threads);
704            }
705            builder
706                .build()
707                .map_err(|e| format!("failed to create chunk encoding thread pool: {e}"))?
708        });
709
710        let player_data_storage = PlayerDataStorage::from_selection(
711            resolved_worlds.save_path.clone(),
712            &resolved_worlds.player_storage,
713        )
714        .await
715        .map_err(|e| format!("failed to create player data storage: {e}"))?;
716        let player_permission_states = player_data_storage
717            .load_permission_subjects()
718            .await
719            .map_err(|error| format!("failed to load player permissions: {error}"))?;
720        let known_players = player_data_storage
721            .load_known_players()
722            .await
723            .map_err(|error| format!("failed to load known players: {error}"))?;
724        let mut worlds = WorldMap::new(
725            resolved_worlds.default_domain.clone(),
726            &resolved_worlds.domains,
727            &resolved_worlds.worlds,
728        );
729
730        let damage_history = Arc::new(DamageHistory::default());
731        let mut construct_world = async |world_entry: &ResolvedWorldConfig,
732                                         game_time_source: GameTimeSource|
733               -> Result<Arc<World>, String> {
734            let storage_output = world_storage
735                .remove(&world_entry.key)
736                .ok_or_else(|| format!("world {} has no resolved storage", world_entry.key))?;
737            let world_seed = LevelDataManager::load_seed_or_default(
738                storage_output.level_data_path.as_deref(),
739                world_entry.seed,
740            )
741            .await
742            .map_err(|e| {
743                format!(
744                    "failed to load level data seed for {}: {e}",
745                    world_entry.key
746                )
747            })?;
748            let generator_output = generator_registry
749                .create(
750                    storage_output.level_data_path.as_deref(),
751                    &world_entry.generator_config,
752                    world_seed,
753                    Arc::clone(&generation_pool),
754                )
755                .map_err(|e| format!("failed to create generator for {}: {e}", world_entry.key))?;
756            let generation_settings = generation_settings_for_world(world_entry, &generator_output);
757            let world = World::new_with_config_and_encoding_pool(
758                Arc::clone(&chunk_runtime),
759                world_entry.key.clone(),
760                generator_output.dimension_type,
761                world_seed,
762                WorldConfig {
763                    damage_history: Arc::clone(&damage_history),
764                    game_time_source,
765                    storage: storage_output.storage,
766                    level_data_path: storage_output
767                        .level_data_path
768                        .map(|path| path.to_string_lossy().into_owned()),
769                    generator: Arc::new(generator_output.generator),
770                    generation_settings,
771                    view_distance: config.view_distance,
772                    simulation_distance: config.simulation_distance,
773                    max_chained_neighbor_updates: config.max_chained_neighbor_updates,
774                    compression: config.compression,
775                    is_flat: generator_output.is_flat,
776                    sea_level: generator_output.sea_level,
777                    default_gamemode: world_entry.default_gamemode,
778                    difficulty: world_entry.difficulty,
779                },
780                Arc::clone(&generation_pool),
781                Arc::clone(&chunk_encoding_pool),
782            )
783            .await
784            .map_err(|e| format!("failed to create world {}: {e}", world_entry.key))?;
785            Ok(world)
786        };
787        let mut spawn_worlds = Vec::with_capacity(resolved_worlds.worlds.len());
788        for domain in &resolved_worlds.domains {
789            let primary_config = resolved_worlds
790                .worlds
791                .iter()
792                .find(|world| world.key == domain.default_world && world.domain == domain.name)
793                .ok_or_else(|| {
794                    format!(
795                        "domain {} has no configured primary {}",
796                        domain.name, domain.default_world
797                    )
798                })?;
799            let primary = construct_world(primary_config, GameTimeSource::Primary).await?;
800            let clock = Arc::clone(&primary.game_time);
801            spawn_worlds.push(Arc::clone(&primary));
802            worlds.insert(primary_config.key.clone(), primary);
803            for world_entry in resolved_worlds
804                .worlds
805                .iter()
806                .filter(|world| world.domain == domain.name && world.key != domain.default_world)
807            {
808                let world =
809                    construct_world(world_entry, GameTimeSource::Derived(Arc::clone(&clock)))
810                        .await?;
811                spawn_worlds.push(Arc::clone(&world));
812                worlds.insert(world_entry.key.clone(), world);
813            }
814        }
815        worlds.validate_game_times()?;
816
817        let scoreboards = DomainScoreboards::load(&worlds)
818            .await
819            .map_err(|error| format!("failed to load domain scoreboards: {error}"))?;
820        let command_storage = DomainCommandStorage::load(&worlds)
821            .await
822            .map_err(|error| format!("failed to load domain command storage: {error}"))?;
823        let random_sequences = DomainRandomSequences::load(&worlds)
824            .await
825            .map_err(|error| format!("failed to load domain random sequences: {error}"))?;
826        // Spawn chunk postprocessing can evaluate loot. Bind domain services first,
827        // then initialize spawns in primary-first construction order.
828        for world in spawn_worlds {
829            world.initialize_spawn_if_needed().await.map_err(|error| {
830                format!("failed to initialize spawn for {}: {error}", world.key)
831            })?;
832        }
833        let registered_commands = create_registered_dispatcher(command_registry)
834            .map_err(|error| format!("failed to register commands: {error}"))?;
835        let command_permission_keys = registered_commands
836            .permissions
837            .into_iter()
838            .map(|permission| permission.as_str().to_owned())
839            .collect();
840
841        // Steel finishes the initial attempt before opening its listener, except offline,
842        // where `enforces_secure_chat` needs online mode so nothing acts on the result.
843        if config.online_mode && service_keys_ready.await.is_err() {
844            log::error!("Minecraft services key fetch task stopped before its initial attempt");
845        }
846
847        Ok(Server {
848            damage_history,
849            config,
850            permission_groups,
851            cancel_token,
852            key_store: KeyStore::create(),
853            worlds,
854            online_players: PlayerMap::new(),
855            player_admissions: SyncMutex::new(FxHashMap::default()),
856            player_admission_changed: Notify::new(),
857            server_tick_changed: Notify::new(),
858            registry_cache,
859            tick_rate_manager: SyncRwLock::new(TickRateManager::new()),
860            perf: PerfRecorder::new(),
861            player_idle_timeout: AtomicI32::new(0),
862            scoreboards,
863            command_storage,
864            random_sequences,
865            command_dispatcher: SyncRwLock::new(registered_commands.dispatcher),
866            command_permission_keys,
867            command_requests: CommandRequestQueue::new(),
868            packet_processor: PacketProcessor::new(),
869            chunk_encoding_pool,
870            jobs: ServerJobQueue::new(),
871            player_data_storage,
872            player_permission_states: SyncRwLock::new(player_permission_states),
873            player_permission_updates: AsyncMutex::new(()),
874            known_players: SyncMutex::new(KnownPlayerCacheState::new(known_players)),
875            known_player_save_idle: Notify::new(),
876            profile_lookup_client: reqwest::Client::new(),
877            service_keys,
878            pending_player_joins: PlayerJoinQueue::new(),
879            pending_player_disconnects: PlayerDisconnectQueue::new(),
880            pending_world_changes: SyncMutex::new(vec![]),
881            pending_domain_switches: SyncMutex::new(vec![]),
882        })
883    }

914        &self,
915        sender: CommandSender,
916        mut command: String,
917        signing_context: Option<CommandSigningContext>,
918    ) -> Result<(), CommandQueueFull> {
919        if command.starts_with('/') {

921        }
922
923        self.queue_command(sender, command, signing_context)

951    pub(crate) fn queue_chat_session_update(
952        &self,
953        player: &Arc<Player>,
954        session: RemoteChatSession,
955    ) {
956        let Some(transition) = self.begin_player_packet_transition(player) else {
957            return;
958        };
959
960        let owner = CommandExecutionOwner::capture(CommandSender::Player(Arc::clone(player)), self);
961        if self
962            .command_requests
963            .submit(CommandRequest::ChatSessionUpdate {
964                owner,
965                session: Box::new(session),
966                transition,
967            })
968            .is_err()
969        {
970            self.finish_player_packet_transition(transition);
971            tracing::error!("failed to queue chat session update");
972        }
973    }

999    /// pauses later packets until the current playersession transition is complete
1000    pub(crate) fn begin_player_packet_transition(
1001        &self,
1002        player: &Arc<Player>,
1003    ) -> Option<PlayerPacketTransition> {
1004        if player.connection.closed() || !player.session.is_current_player(player) {
1005            return None;
1006        }
1007        self.packet_processor.pause_player_session(&player.session)
1008    }

1012        &self,
1013        transition: PlayerPacketTransition,
1014    ) -> bool {
1015        self.packet_processor.resume_player_session(transition)

1024    /// Returns Brigadier completions visible to a command sender.
1025    pub fn command_completions(
1026        self: &Arc<Self>,
1027        sender: CommandSender,
1028        input: &str,
1029    ) -> Vec<CommandCompletion> {
1030        if !CommandExecutionOwner::capture(sender.clone(), self).is_current(self) {
1031            return Vec::new();
1032        }
1033        match self.build_command_suggestions(sender, input) {
1034            Ok(suggestions) => {
1035                let range = suggestions.range();
1036                suggestions
1037                    .list()
1038                    .iter()
1039                    .map(|suggestion| {
1040                        CommandCompletion::new(
1041                            range.start(),
1042                            range.len(),
1043                            suggestion.text().to_owned(),
1044                        )
1045                    })
1046                    .collect()
1047            }
1048            Err(error) => {
1049                tracing::warn!(%error, "failed to build command suggestions");
1050                Vec::new()
1051            }
1052        }
1053    }

1055    /// Returns whether a command requires signed arguments when parsed for the given sender.
1056    pub fn command_requires_signed_arguments(
1057        self: &Arc<Self>,
1058        command: &str,
1059        sender: CommandSender,
1060    ) -> bool {
1061        let source = CommandSource::new(sender, Arc::clone(self));
1062        let dispatcher = self.command_dispatcher.read();
1063        let parse = dispatcher.parse(command, source);
1064        dispatcher.has_signed_arguments(&parse)
1065    }
1066
1067    /// Inspects the parsed command nodes and collects the names and text slices
1068    /// of all arguments requiring cryptographic signatures (e.g. `minecraft:message`).
1069    pub fn collect_signable_arguments<'a>(
1070        self: &Arc<Self>,
1071        command: &'a str,
1072        sender: CommandSender,
1073    ) -> Vec<(String, &'a str)> {
1074        let source = CommandSource::new(sender, Arc::clone(self));
1075        let dispatcher = self.command_dispatcher.read();
1076        let parse = dispatcher.parse(command, source);
1077
1078        dispatcher.collect_signable_arguments(&parse, command)
1079    }
