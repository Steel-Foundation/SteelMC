68// Radius-four TNT exposure repeatedly visits a few hundred nearby positions. A 256-entry direct
69// cache retains almost all of that working set without heap allocation or cross-explosion lifetime.
70// Its size must remain a power of two because lookups wrap with a mask.
71const EXPLOSION_EXPOSURE_CACHE_SIZE: usize = 256;
72const EXPLOSION_EXPOSURE_CACHE_MASK: usize = EXPLOSION_EXPOSURE_CACHE_SIZE - 1;
73const EXPLOSION_EXPOSURE_FIBONACCI_HASH_MULTIPLIER: u64 = 0x9e37_79b9_7f4a_7c15;
74const EMPTY_EXPOSURE_CACHE_GENERATION: u32 = 0;
75const FIRST_EXPOSURE_CACHE_GENERATION: u32 = 1;
76
77// A normal radius-four explosion queries at most 19 positions on each axis. Two bits per position
78// fit inline, avoiding one heap allocation while unusual radii use the compatibility path.
79const EXPLOSION_EXPOSURE_CLEAR_GRID_AXIS_SIZE: usize = 19;
80const EXPLOSION_EXPOSURE_CLEAR_GRID_STATE_BITS: usize = 2;
81const EXPLOSION_EXPOSURE_CLEAR_GRID_STATES_PER_WORD: usize =
82    u64::BITS as usize / EXPLOSION_EXPOSURE_CLEAR_GRID_STATE_BITS;
83const EXPLOSION_EXPOSURE_CLEAR_GRID_STATE_MASK: u64 =
84    (1 << EXPLOSION_EXPOSURE_CLEAR_GRID_STATE_BITS) - 1;
85const EXPLOSION_EXPOSURE_CLEAR_GRID_POSITIONS: usize = EXPLOSION_EXPOSURE_CLEAR_GRID_AXIS_SIZE
86    * EXPLOSION_EXPOSURE_CLEAR_GRID_AXIS_SIZE
87    * EXPLOSION_EXPOSURE_CLEAR_GRID_AXIS_SIZE;
88const EXPLOSION_EXPOSURE_CLEAR_GRID_WORDS: usize =
89    EXPLOSION_EXPOSURE_CLEAR_GRID_POSITIONS.div_ceil(EXPLOSION_EXPOSURE_CLEAR_GRID_STATES_PER_WORD);

99const UNRESOLVED_EXPOSURE_GRID_WORD: u64 = ExplosionExposureClearGridState::Unresolved as u64;

101impl ExplosionExposureClearGridState {
102    const fn for_collision(is_empty: bool) -> Self {
103        if is_empty {
104            Self::StaticEmpty
105        } else {
106            Self::RequiresExactCollision
107        }
108    }
109
110    const fn from_encoded(encoded: u64) -> Self {
111        if encoded == Self::Unresolved as u64 {
112            Self::Unresolved
113        } else if encoded == Self::StaticEmpty as u64 {
114            Self::StaticEmpty
115        } else {
116            // The unused fourth encoding conservatively takes the exact path too.
117            Self::RequiresExactCollision
118        }
119    }

130struct ExplosionExposureClearGrid {
131    bounds: Option<ExplosionExposureClearGridBounds>,
132    // Each lookup reads one packed word instead of two bitsets.
133    states: [u64; EXPLOSION_EXPOSURE_CLEAR_GRID_WORDS],
134}
135
136impl ExplosionExposureClearGrid {
137    const fn new() -> Self {
138        Self {
139            bounds: None,
140            states: [UNRESOLVED_EXPOSURE_GRID_WORD; EXPLOSION_EXPOSURE_CLEAR_GRID_WORDS],
141        }
142    }
143
144    fn configure(&mut self, min: BlockPos, max: BlockPos) {
145        let Some(size_x) = inclusive_axis_size(min.x(), max.x()) else {
146            self.bounds = None;
147            return;
148        };
149        let Some(size_y) = inclusive_axis_size(min.y(), max.y()) else {
150            self.bounds = None;
151            return;
152        };
153        let Some(size_z) = inclusive_axis_size(min.z(), max.z()) else {
154            self.bounds = None;
155            return;
156        };
157        let Some(volume) = size_x
158            .checked_mul(size_y)
159            .and_then(|plane| plane.checked_mul(size_z))
160        else {
161            self.bounds = None;
162            return;
163        };
164        if volume > EXPLOSION_EXPOSURE_CLEAR_GRID_POSITIONS {
165            self.bounds = None;
166            return;
167        }
168
169        self.bounds = Some(ExplosionExposureClearGridBounds {
170            min,
171            size_x,
172            size_y,
173            size_z,
174        });
175        self.states.fill(UNRESOLVED_EXPOSURE_GRID_WORD);
176    }
177
178    fn clear(&mut self) {
179        self.states.fill(UNRESOLVED_EXPOSURE_GRID_WORD);
180    }
181
182    fn index(&self, pos: BlockPos) -> Option<usize> {
183        let bounds = self.bounds?;
184        // Wrapping subtraction maps every below-min coordinate to a large unsigned value. Since
185        // the configured grid is tiny, the same range checks safely reject both underflow and
186        // coordinates above the maximum without three checked-subtraction branches.
187        let x = pos.x().wrapping_sub(bounds.min.x()) as u32 as usize;
188        let y = pos.y().wrapping_sub(bounds.min.y()) as u32 as usize;
189        let z = pos.z().wrapping_sub(bounds.min.z()) as u32 as usize;
190        if x >= bounds.size_x || y >= bounds.size_y || z >= bounds.size_z {
191            return None;
192        }
193        Some((y * bounds.size_z + z) * bounds.size_x + x)
194    }
195
196    const fn state(&self, index: usize) -> ExplosionExposureClearGridState {
197        let word = index / EXPLOSION_EXPOSURE_CLEAR_GRID_STATES_PER_WORD;
198        let shift = (index % EXPLOSION_EXPOSURE_CLEAR_GRID_STATES_PER_WORD)
199            * EXPLOSION_EXPOSURE_CLEAR_GRID_STATE_BITS;
200        let encoded = (self.states[word] >> shift) & EXPLOSION_EXPOSURE_CLEAR_GRID_STATE_MASK;
201        ExplosionExposureClearGridState::from_encoded(encoded)
202    }
203
204    fn record(&mut self, index: usize, state: ExplosionExposureClearGridState) {
205        debug_assert_ne!(state, ExplosionExposureClearGridState::Unresolved);
206        let word = index / EXPLOSION_EXPOSURE_CLEAR_GRID_STATES_PER_WORD;
207        let shift = (index % EXPLOSION_EXPOSURE_CLEAR_GRID_STATES_PER_WORD)
208            * EXPLOSION_EXPOSURE_CLEAR_GRID_STATE_BITS;
209        let mask = EXPLOSION_EXPOSURE_CLEAR_GRID_STATE_MASK << shift;
210        let value = (state as u64) << shift;
211        self.states[word] = (self.states[word] & !mask) | value;
212    }
213}

364        #[cfg(test)]
365        {
366            self.stats.block_visits += 1;
367        }
368
369        let mut unresolved_grid_index = None;
370        if let Some(index) = self.clear_grid.index(pos) {
371            match self.clear_grid.state(index) {
372                ExplosionExposureClearGridState::StaticEmpty => {

378                }

380                ExplosionExposureClearGridState::Unresolved => {

382                }
383            }

389    fn block_intersects_ray_exact(
390        &mut self,
391        pos: BlockPos,
392        from: DVec3,
393        to: DVec3,
394        unresolved_grid_index: Option<usize>,
395    ) -> bool {
396        let cache_index = explosion_exposure_cache_index(pos);
397        let entry = self.entries[cache_index];
398        if entry.generation == self.generation && entry.pos == pos {
399            #[cfg(test)]
400            {
401                self.stats.cache_hits += 1;
402            }
403            if let Some(index) = unresolved_grid_index {
404                let grid_state =
405                    ExplosionExposureClearGridState::for_collision(entry.collision.is_empty());
406                self.clear_grid.record(index, grid_state);
407                #[cfg(test)]
408                {
409                    self.stats.clear_grid_resolutions += 1;
410                }
411            }
412            return Self::static_collision_intersects(entry.collision, pos, from, to);
413        }
414
415        #[cfg(test)]
416        {
417            self.stats.state_lookups += 1;
418            self.stats.collision_lookups += 1;
419        }
420        let block_read = self
421            .world
422            .explosion_exposure_block_state(pos, &mut self.full_chunks);
423        let state = block_read.state;
424        let block = state.get_block();
425        let behavior = BLOCK_BEHAVIORS.get_behavior(block);
426        if block.config.dynamic_shape {
427            let boxes =
428                behavior.get_collision_boxes(state, self.world, pos, self.collision_context);
429            self.retain_cache_after_collision_query(block);
430            return boxes
431                .into_iter()
432                .any(|aabb| World::clip_local_aabb(pos, from, to, aabb).is_some());
433        }
434
435        let shape = behavior.get_collision_shape(state, self.world, pos, self.collision_context);
436        let offset = if shape.is_empty() {
437            DVec3::ZERO
438        } else {
439            behavior.get_collision_shape_offset(state, self.world, pos, self.collision_context)
440        };
441        let collision = OffsetVoxelShape::new(shape, offset);
442        let retain_cache = self.retain_cache_after_collision_query(block);
443        let intersects = Self::static_collision_intersects(collision, pos, from, to);
444        if block_read.cacheable && retain_cache {
445            let collision_is_empty = collision.is_empty();
446            let mut recorded_in_grid = false;
447            if let Some(index) = unresolved_grid_index {
448                let grid_state = ExplosionExposureClearGridState::for_collision(collision_is_empty);
449                self.clear_grid.record(index, grid_state);
450                recorded_in_grid = true;
451                #[cfg(test)]
452                {
453                    self.stats.clear_grid_resolutions += 1;
454                }
455            }
456            // Dense clear entries no longer need a large direct-cache slot. Retaining only
457            // potentially colliding shapes reduces destructive conflicts on the exact path.
458            if !recorded_in_grid || !collision_is_empty {
459                self.entries[cache_index] = ExplosionExposureCacheEntry {
460                    pos,
461                    collision,
462                    generation: self.generation,
463                };
464            }
465        }
466        intersects
467    }

486#[inline]
487fn explosion_exposure_cache_index(pos: BlockPos) -> usize {
488    let tag = PackedBlockPos::from(pos).as_raw() as u64;
489    // Fibonacci multiplication followed by xor folding disperses nearby packed positions into
490    // the low bits used by the power-of-two cache.
491    let mut mixed = tag.wrapping_mul(EXPLOSION_EXPOSURE_FIBONACCI_HASH_MULTIPLIER);
492    mixed ^= mixed >> (u64::BITS / 2);
493    mixed ^= mixed >> (u64::BITS / 4);
494    (mixed as usize) & EXPLOSION_EXPOSURE_CACHE_MASK
495}

497#[inline]
498fn is_collision_path_clear(
499    start_pos: DVec3,
500    end_pos: DVec3,
501    mut blocks_ray: impl FnMut(BlockPos) -> bool,
502) -> bool {
503    if start_pos == end_pos {
504        return true;
505    }
506
507    let to = vanilla_lerp_vec3(VANILLA_RAY_ENDPOINT_ADJUSTMENT, end_pos, start_pos);
508    let from = vanilla_lerp_vec3(VANILLA_RAY_ENDPOINT_ADJUSTMENT, start_pos, end_pos);
509    let mut block = BlockPos::from(from);
510    if blocks_ray(block) {
511        return false;
512    }
513
514    let difference = to - from;
515    let step = glam::IVec3::new(
516        minecraft_sign(difference.x),
517        minecraft_sign(difference.y),
518        minecraft_sign(difference.z),
519    );
520    let delta = DVec3::new(
521        if step.x == 0 {
522            f64::MAX
523        } else {
524            f64::from(step.x) / difference.x
525        },
526        if step.y == 0 {
527            f64::MAX
528        } else {
529            f64::from(step.y) / difference.y
530        },
531        if step.z == 0 {
532            f64::MAX
533        } else {
534            f64::from(step.z) / difference.z
535        },
536    );
537    let mut next = DVec3::new(
538        delta.x
539            * if step.x > 0 {
540                1.0 - (from.x - from.x.floor())
541            } else {
542                from.x - from.x.floor()
543            },
544        delta.y
545            * if step.y > 0 {
546                1.0 - (from.y - from.y.floor())
547            } else {
548                from.y - from.y.floor()
549            },
550        delta.z
551            * if step.z > 0 {
552                1.0 - (from.z - from.z.floor())
553            } else {
554                from.z - from.z.floor()
555            },
556    );
557
558    while next.x <= 1.0 || next.y <= 1.0 || next.z <= 1.0 {
559        if next.x < next.y && next.x < next.z {
560            block.0.x += step.x;
561            next.x += delta.x;
562        } else if next.y < next.z {
563            block.0.y += step.y;
564            next.y += delta.y;
565        } else {
566            block.0.z += step.z;
567            next.z += delta.z;
568        }
569        if blocks_ray(block) {
570            return false;
571        }
572    }
573
574    true
575}

577/// Returns the inclusive block-coordinate envelope visited by one axis of Vanilla's block DDA.
578///
579/// Axis crossing counts are independent of inter-axis tie order: the traversal cannot finish
580/// until every finite `next` value is greater than one. Replaying the scalar recurrence also
581/// retains the rare terminal step caused by floating-point rounding at `next == 1`.
582pub(crate) fn collision_path_axis_block_bounds(
583    start: f64,
584    end: f64,
585    max_steps: usize,
586) -> Option<(i32, i32)> {
587    if !start.is_finite() || !end.is_finite() {
588        return None;
589    }
590
591    let to = vanilla_lerp(VANILLA_RAY_ENDPOINT_ADJUSTMENT, end, start);
592    let from = vanilla_lerp(VANILLA_RAY_ENDPOINT_ADJUSTMENT, start, end);
593    let can_initialize_block_traversal = to.is_finite()
594        && from.is_finite()
595        && from >= f64::from(i32::MIN)
596        && from < f64::from(i32::MAX) + 1.0;
597    if !can_initialize_block_traversal {
598        return None;
599    }
600
601    let mut block = from.floor() as i32;
602    let mut min = block;
603    let mut max = block;
604    let difference = to - from;
605    let step = minecraft_sign(difference);
606    let delta = if step == 0 {
607        f64::MAX
608    } else {
609        f64::from(step) / difference
610    };
611    let from_fraction = from - from.floor();
612    let mut next = delta
613        * if step > 0 {
614            1.0 - from_fraction
615        } else {
616            from_fraction
617        };
618    if !delta.is_finite() || !next.is_finite() {
619        return None;
620    }
621
622    let mut steps = 0;
623    while next <= 1.0 {
624        if steps == max_steps {
625            return None;
626        }
627        block = block.checked_add(step)?;
628        min = min.min(block);
629        max = max.max(block);
630        next += delta;
631        steps += 1;
632    }
633    Some((min, max))
634}

637const fn minecraft_sign(number: f64) -> i32 {

714    /// Performs a vanilla-style block/fluid clip in the world.
715    #[must_use]
716    pub fn clip(
717        &self,
718        start_pos: DVec3,
719        end_pos: DVec3,
720        block_shape: ClipBlockShape,
721        fluid: ClipFluid,
722    ) -> ClipHitResult {
723        if start_pos == end_pos {
724            return Self::clip_miss(start_pos, end_pos);
725        }
726
727        let adjust = -1.0e-7f64;
728        let to = end_pos.lerp(start_pos, adjust);
729        let from = start_pos.lerp(end_pos, adjust);
730
731        let mut block = BlockPos::new(
732            from.x.floor() as i32,
733            from.y.floor() as i32,
734            from.z.floor() as i32,
735        );
736
737        if let Some(hit) = self.clip_block_and_fluid(block, start_pos, end_pos, block_shape, fluid)
738        {
739            return hit;
740        }
741
742        let difference = to - from;
743
744        let step = difference.signum().as_ivec3();
745
746        let delta = DVec3::new(
747            if step.x == 0 {
748                f64::MAX
749            } else {
750                (f64::from(step.x)) / difference.x
751            },
752            if step.y == 0 {
753                f64::MAX
754            } else {
755                (f64::from(step.y)) / difference.y
756            },
757            if step.z == 0 {
758                f64::MAX
759            } else {
760                (f64::from(step.z)) / difference.z
761            },
762        );
763
764        let mut next = DVec3::new(
765            delta.x
766                * (if step.x > 0 {
767                    1.0 - (from.x - from.x.floor())
768                } else {
769                    from.x - from.x.floor()
770                }),
771            delta.y
772                * (if step.y > 0 {
773                    1.0 - (from.y - from.y.floor())
774                } else {
775                    from.y - from.y.floor()
776                }),
777            delta.z
778                * (if step.z > 0 {
779                    1.0 - (from.z - from.z.floor())
780                } else {
781                    from.z - from.z.floor()
782                }),
783        );
784
785        while next.x <= 1.0 || next.y <= 1.0 || next.z <= 1.0 {
786            if next.x < next.y && next.x < next.z {
787                block.0.x += step.x;
788                next.x += delta.x;
789            } else if next.y < next.z {
790                block.0.y += step.y;
791                next.y += delta.y;
792            } else {
793                block.0.z += step.z;
794                next.z += delta.z;
795            }
796
797            if let Some(hit) =
798                self.clip_block_and_fluid(block, start_pos, end_pos, block_shape, fluid)
799            {
800                return hit;
801            }
802        }
803
804        Self::clip_miss(start_pos, end_pos)
805    }

837        let border = self.world_border_snapshot();

848                inside: false,

855    pub(super) fn clip_block_and_fluid(
856        &self,
857        pos: BlockPos,
858        from: DVec3,
859        to: DVec3,
860        block_shape: ClipBlockShape,
861        fluid: ClipFluid,
862    ) -> Option<ClipHitResult> {
863        let state = self.get_block_state(pos);
864        let block_result = Self::clip_shape(
865            pos,
866            from,
867            to,
868            self.clip_block_shape(state, pos, block_shape),
869        )
870        .map(|hit| Self::clip_with_interaction_override(pos, from, to, state, hit));
871        let fluid_result = self.clip_fluid_shape(pos, from, to, state, fluid);
872
873        match (block_result, fluid_result) {
874            (Some(block_hit), Some(fluid_hit)) => {
875                let block_distance = from.distance_squared(block_hit.location);
876                let fluid_distance = from.distance_squared(fluid_hit.location);
877                if block_distance <= fluid_distance {
878                    Some(block_hit)
879                } else {
880                    Some(fluid_hit)
881                }
882            }
883            (Some(hit), None) | (None, Some(hit)) => Some(hit),
884            (None, None) => None,
885        }
886    }

890        from: DVec3,
891        to: DVec3,

1004    pub(super) fn clip_shape(
1005        block_pos: BlockPos,
1006        from: DVec3,
1007        to: DVec3,
1008        shape: OffsetVoxelShape,
1009    ) -> Option<ClipHitResult> {
1010        if shape.is_empty() {
1011            return None;
1012        }
1013
1014        if (to - from).length_squared() < 1.0e-7 {
1015            return None;
1016        }
1017
1018        let block_vec = DVec3::new(
1019            f64::from(block_pos.x()),
1020            f64::from(block_pos.y()),
1021            f64::from(block_pos.z()),
1022        );
1023        let inside_test_point = from + (to - from) * 0.001;
1024        if Self::shape_contains_world_point(shape, block_vec, inside_test_point) {
1025            return Some(ClipHitResult {
1026                location: inside_test_point,
1027                direction: Self::approximate_nearest_direction(to - from).opposite(),
1028                block_pos,
1029                miss: false,
1030                inside: true,
1031                world_border_hit: false,
1032            });
1033        }
1034
1035        let mut closest: Option<(f64, Direction)> = None;
1036
1037        for shape in shape.iter() {
1038            let world_min = DVec3::new(shape.min_x(), shape.min_y(), shape.min_z()) + block_vec;
1039            let world_max = DVec3::new(shape.max_x(), shape.max_y(), shape.max_z()) + block_vec;
1040
1041            if let Some(hit) = Self::intersects_aabb_with_t(from, to, world_min, world_max)
1042                && hit.0 > 0.0
1043                && hit.0 < 1.0
1044                && closest.is_none_or(|(best_t, _)| hit.0 < best_t)
1045            {
1046                closest = Some(hit);
1047            }
1048        }
1049
1050        closest.map(|(t, direction)| ClipHitResult {
1051            location: from + (to - from) * t,
1052            direction,
1053            block_pos,
1054            miss: false,
1055            inside: false,
1056            world_border_hit: false,
1057        })
1058    }

1145    pub(super) fn approximate_nearest_direction(vector: DVec3) -> Direction {
1146        let mut result = Direction::North;
1147        let mut highest_dot = 0.0;
1148        for direction in [
1149            Direction::Down,
1150            Direction::Up,
1151            Direction::North,
1152            Direction::South,
1153            Direction::West,
1154            Direction::East,
1155        ] {
1156            let dot = vector.dot(direction.offset_vec().as_dvec3());
1157            if dot > highest_dot {
1158                highest_dot = dot;
1159                result = direction;
1160            }
1161        }
1162        result
1163    }

1165    /// Mirrors Minecraft  `AABB.getDirection` for one block-local shape box.
1166    pub(super) fn intersects_aabb_with_t(
1167        start: DVec3,
1168        end: DVec3,
1169        min: DVec3,
1170        max: DVec3,
1171    ) -> Option<(f64, Direction)> {
1172        let delta = end - start;
1173        let mut scale = 1.0;
1174        let mut direction = None;
1175
1176        if delta.x > AABB_CLIP_EPSILON {
1177            Self::clip_aabb_face(
1178                &mut scale,
1179                &mut direction,
1180                delta,
1181                start,
1182                min.x,
1183                [min.y, max.y, min.z, max.z],
1184                Direction::West,
1185            );
1186        } else if delta.x < -AABB_CLIP_EPSILON {
1187            Self::clip_aabb_face(
1188                &mut scale,
1189                &mut direction,
1190                delta,
1191                start,
1192                max.x,
1193                [min.y, max.y, min.z, max.z],
1194                Direction::East,
1195            );
1196        }
1197
1198        let y_delta = DVec3::new(delta.y, delta.z, delta.x);
1199        let y_start = DVec3::new(start.y, start.z, start.x);
1200        if delta.y > AABB_CLIP_EPSILON {
1201            Self::clip_aabb_face(
1202                &mut scale,
1203                &mut direction,
1204                y_delta,
1205                y_start,
1206                min.y,
1207                [min.z, max.z, min.x, max.x],
1208                Direction::Down,
1209            );
1210        } else if delta.y < -AABB_CLIP_EPSILON {
1211            Self::clip_aabb_face(
1212                &mut scale,
1213                &mut direction,
1214                y_delta,
1215                y_start,
1216                max.y,
1217                [min.z, max.z, min.x, max.x],
1218                Direction::Up,
1219            );
1220        }
1221
1222        let z_delta = DVec3::new(delta.z, delta.x, delta.y);
1223        let z_start = DVec3::new(start.z, start.x, start.y);
1224        if delta.z > AABB_CLIP_EPSILON {
1225            Self::clip_aabb_face(
1226                &mut scale,
1227                &mut direction,
1228                z_delta,
1229                z_start,
1230                min.z,
1231                [min.x, max.x, min.y, max.y],
1232                Direction::North,
1233            );
1234        } else if delta.z < -AABB_CLIP_EPSILON {
1235            Self::clip_aabb_face(
1236                &mut scale,
1237                &mut direction,
1238                z_delta,
1239                z_start,
1240                max.z,
1241                [min.x, max.x, min.y, max.y],
1242                Direction::South,
1243            );
1244        }
1245
1246        direction.map(|direction| (scale, direction))
1247    }

1249    #[inline]
1250    fn clip_aabb_face(
1251        scale: &mut f64,
1252        direction: &mut Option<Direction>,
1253        delta: DVec3,
1254        start: DVec3,
1255        face: f64,
1256        side_bounds: [f64; 4],
1257        new_direction: Direction,
1258    ) {
1259        let candidate_scale = (face - start.x) / delta.x;
1260        let side_b = start.y + candidate_scale * delta.y;
1261        let side_c = start.z + candidate_scale * delta.z;
1262        if 0.0 < candidate_scale
1263            && candidate_scale < *scale
1264            && side_bounds[0] - AABB_CLIP_EPSILON < side_b
1265            && side_b < side_bounds[1] + AABB_CLIP_EPSILON
1266            && side_bounds[2] - AABB_CLIP_EPSILON < side_c
1267            && side_c < side_bounds[3] + AABB_CLIP_EPSILON
1268        {
1269            *scale = candidate_scale;
1270            *direction = Some(new_direction);
1271        }
1272    }

1274    /// Performs a raytrace in the world.
1275    ///
1276    /// Adapted from Pumpkin project.
1277    pub fn raytrace<F>(
1278        &self,
1279        start_pos: DVec3,
1280        end_pos: DVec3,
1281        hit_check: F,
1282    ) -> (Option<BlockPos>, Option<Direction>)
1283    where
1284        F: Fn(BlockPos, &Self) -> RaytraceAction,
1285    {
1286        if start_pos == end_pos {
1287            return (None, None);
1288        }
1289
1290        let adjust = -1.0e-7f64;
1291        let to = end_pos.lerp(start_pos, adjust);
1292        let from = start_pos.lerp(end_pos, adjust);
1293
1294        let mut block = BlockPos::new(
1295            from.x.floor() as i32,
1296            from.y.floor() as i32,
1297            from.z.floor() as i32,
1298        );
1299
1300        match hit_check(block, self) {
1301            RaytraceAction::ImmediateHit => return (Some(block), None),
1302            RaytraceAction::CheckShape => {
1303                let (hit, face) = self.ray_outline_check(block, start_pos, end_pos);
1304                if hit {
1305                    return (Some(block), face);
1306                }
1307            }
1308            RaytraceAction::Pass => {}
1309        }
1310
1311        let difference = to - from;
1312
1313        let step = difference.signum().as_ivec3();
1314
1315        let delta = DVec3::new(
1316            if step.x == 0 {
1317                f64::MAX
1318            } else {
1319                (f64::from(step.x)) / difference.x
1320            },
1321            if step.y == 0 {
1322                f64::MAX
1323            } else {
1324                (f64::from(step.y)) / difference.y
1325            },
1326            if step.z == 0 {
1327                f64::MAX
1328            } else {
1329                (f64::from(step.z)) / difference.z
1330            },
1331        );
1332
1333        let mut next = DVec3::new(
1334            delta.x
1335                * (if step.x > 0 {
1336                    1.0 - (from.x - from.x.floor())
1337                } else {
1338                    from.x - from.x.floor()
1339                }),
1340            delta.y
1341                * (if step.y > 0 {
1342                    1.0 - (from.y - from.y.floor())
1343                } else {
1344                    from.y - from.y.floor()
1345                }),
1346            delta.z
1347                * (if step.z > 0 {
1348                    1.0 - (from.z - from.z.floor())
1349                } else {
1350                    from.z - from.z.floor()
1351                }),
1352        );
1353
1354        while next.x <= 1.0 || next.y <= 1.0 || next.z <= 1.0 {
1355            // Vanilla parity: traverseBlocks tie-breaking — Z wins on any tie.
1356            // X wins only when strictly less than both Y and Z.
1357            // Y wins only when strictly less than both X and Z.
1358            // Everything else (including all ties) goes to Z.
1359            let block_direction = if next.x < next.y && next.x < next.z {
1360                block.0.x += step.x;
1361                next.x += delta.x;
1362                if step.x > 0 {
1363                    Direction::West
1364                } else {
1365                    Direction::East
1366                }
1367            } else if next.y < next.x && next.y < next.z {
1368                block.0.y += step.y;
1369                next.y += delta.y;
1370                if step.y > 0 {
1371                    Direction::Down
1372                } else {
1373                    Direction::Up
1374                }
1375            } else {
1376                block.0.z += step.z;
1377                next.z += delta.z;
1378                if step.z > 0 {
1379                    Direction::North
1380                } else {
1381                    Direction::South
1382                }
1383            };
1384
1385            match hit_check(block, self) {
1386                RaytraceAction::ImmediateHit => {
1387                    return (Some(block), Some(block_direction));
1388                }
1389                RaytraceAction::CheckShape => {
1390                    let (hit, face) = self.ray_outline_check(block, start_pos, end_pos);
1391                    if hit {
1392                        return (Some(block), face);
1393                    }
1394                }
1395                RaytraceAction::Pass => {}
1396            }
1397        }
1398
1399        (None, None)
1400    }
