10/// Time-adjusted difficulty used by mob spawning and spawn bonuses.
11#[derive(Clone, Copy, Debug)]
12pub struct DifficultyInstance {
13    base: Difficulty,
14    effective_difficulty: f32,
15}
16
17impl DifficultyInstance {
18    const GLOBAL_TIME_OFFSET: f32 = -72_000.0;
19    const MAX_GLOBAL_TIME: f32 = 1_440_000.0;
20    const MAX_LOCAL_TIME: f32 = 3_600_000.0;
21    const MOON_BRIGHTNESS_SCALE: f32 = 0.25;
22
23    /// Builds the instance from vanilla's four difficulty inputs.
24    #[must_use]
25    pub fn new(
26        base: Difficulty,
27        total_game_time: i64,
28        local_game_time: i64,
29        moon_brightness: f32,
30    ) -> Self {
31        let effective_difficulty =
32            Self::calculate(base, total_game_time, local_game_time, moon_brightness);
33        Self {
34            base,
35            effective_difficulty,
36        }
37    }
38
39    fn calculate(
40        base: Difficulty,
41        total_game_time: i64,
42        local_game_time: i64,
43        moon_brightness: f32,
44    ) -> f32 {
45        if base == Difficulty::Peaceful {
46            return 0.0;
47        }
48
49        let is_hard = base == Difficulty::Hard;
50        let mut scale = 0.75;
51        let global_scale = ((total_game_time as f32 + Self::GLOBAL_TIME_OFFSET)
52            / Self::MAX_GLOBAL_TIME)
53            .clamp(0.0, 1.0)
54            * 0.25;
55        scale += global_scale;
56
57        let mut local_scale = (local_game_time as f32 / Self::MAX_LOCAL_TIME).clamp(0.0, 1.0)
58            * if is_hard { 1.0 } else { 0.75 };
59        local_scale += (moon_brightness * Self::MOON_BRIGHTNESS_SCALE).clamp(0.0, global_scale);
60        if base == Difficulty::Easy {
61            local_scale *= 0.5;
62        }
63
64        scale += local_scale;
65        f32::from(u8::from(base)) * scale
66    }
67
68    /// Returns the configured world difficulty.
69    #[must_use]
70    pub const fn difficulty(self) -> Difficulty {
71        self.base
72    }
73
74    /// Returns the time-adjusted difficulty level.
75    #[must_use]
76    pub const fn effective_difficulty(self) -> f32 {
77        self.effective_difficulty
78    }
79
80    /// Returns whether the effective difficulty reached vanilla hard.
81    #[must_use]
82    pub fn is_hard(self) -> bool {
83        self.effective_difficulty >= f32::from(u8::from(Difficulty::Hard))
84    }
85
86    /// Returns vanilla's equipment and attribute spawn-roll multiplier.
87    #[must_use]
88    pub fn special_multiplier(self) -> f32 {
89        if self.effective_difficulty < 2.0 {
90            0.0
91        } else if self.effective_difficulty > 4.0 {
92            1.0
93        } else {
94            (self.effective_difficulty - 2.0) / 2.0
95        }
96    }
97}

99impl World {
100    /// Returns vanilla difficulty scaling for a block position.
101    ///
102    /// Vanilla only contributes moon brightness and chunk time when the chunk is
103    /// loaded at Full status.
104    #[must_use]
105    pub fn get_current_difficulty_at(&self, pos: BlockPos) -> DifficultyInstance {
106        let chunk_pos = ChunkPos::from_block_pos(pos);
107        let local_game_time = self
108            .chunk_map
109            .with_full_chunk(chunk_pos, |chunk| chunk.common().inhabited_time());
110
111        let (base, total_game_time, moon_brightness) = {
112            let level_data = self.level_data.read();
113            let base = level_data.data().difficulty;
114            let total_game_time = level_data
115                .world_clocks()
116                .total_ticks(&vanilla_world_clocks::OVERWORLD)
117                .unwrap_or(0);
118            let moon_brightness = local_game_time.map_or(0.0, |_| {
119                environment::moon_brightness(self.dimension_type, level_data.world_clocks())
120            });
121            (base, total_game_time, moon_brightness)
122        };
123
124        DifficultyInstance::new(
125            base,
126            total_game_time,
127            local_game_time.unwrap_or(0),
128            moon_brightness,
129        )
130    }
131}

133#[cfg(test)]
134mod tests {
135    use super::*;
136
137    const CLOSE_ENOUGH: f32 = 0.000_001;
138
139    fn assert_close(left: f32, right: f32) {
140        assert!(
141            (left - right).abs() < CLOSE_ENOUGH,
142            "left={left}, right={right}"
143        );
144    }
145
146    #[test]
147    fn peaceful_is_always_zero() {
148        let instance = DifficultyInstance::new(Difficulty::Peaceful, i64::MAX, i64::MAX, 1.0);
149        assert_close(instance.effective_difficulty(), 0.0);
150        assert_close(instance.special_multiplier(), 0.0);
151        assert!(!instance.is_hard());
152    }
153
154    #[test]
155    fn normal_difficulty_combines_global_local_and_moon_time() {
156        // Global time is capped, local time is capped, and moon brightness is
157        // limited to the global time bonus.
158        let instance = DifficultyInstance::new(Difficulty::Normal, 1_512_000, 3_600_000, 1.0);
159        assert_close(instance.effective_difficulty(), 4.0);
160        assert_close(instance.special_multiplier(), 1.0);
161        assert!(instance.is_hard());
162    }
163
164    #[test]
165    fn easy_scales_only_the_local_contribution() {
166        let instance = DifficultyInstance::new(Difficulty::Easy, 0, 3_600_000, 0.0);
167        // global = 0.0, local = 1.0 * 0.75 * 0.5 = 0.375, scale = 1.125.
168        assert_close(instance.effective_difficulty(), 1.125);
169        assert_close(instance.special_multiplier(), 0.0);
170    }
171
172    #[test]
173    fn hard_contributes_more_local_time_than_normal() {
174        let hard = DifficultyInstance::new(Difficulty::Hard, 0, 3_600_000, 0.0);
175        let normal = DifficultyInstance::new(Difficulty::Normal, 0, 3_600_000, 0.0);
176        // hard = 3 * (0.75 + 1.0), normal = 2 * (0.75 + 0.75).
177        assert_close(hard.effective_difficulty(), 5.25);
178        assert_close(normal.effective_difficulty(), 3.0);
179    }
180}
