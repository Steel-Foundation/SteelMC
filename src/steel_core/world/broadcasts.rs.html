109    /// Broadcasts an already-encoded packet to all players tracking the given chunk.
110    ///
111    /// Use this when you have a pre-encoded packet to avoid re-encoding.
112    pub fn broadcast_to_nearby_encoded(
113        &self,
114        chunk: ChunkPos,
115        packet: EncodedPacket,
116        exclude: Option<i32>,
117    ) {
118        let tracking_players = self.get_packet_tracking_players(chunk);
119        for entity_id in tracking_players {
120            if Some(entity_id) == exclude {
121                continue;
122            }
123            if let Some(player) = self.players.get_by_entity_id(entity_id) {
124                player.connection.send_encoded(packet.clone());
125            }
126        }
127    }

129    /// Returns players whose view includes the chunk and whose client has the base chunk packet.
130    pub fn get_packet_tracking_players(&self, chunk: ChunkPos) -> Vec<i32> {
131        self.player_area_map
132            .get_tracking_players(chunk)
133            .into_iter()
134            .filter(|entity_id| {
135                self.players
136                    .get_by_entity_id(*entity_id)
137                    .is_some_and(|player| player.chunk_sender().lock().is_chunk_sent(chunk))
138            })
139            .collect()

142    /// Returns players on the tracked border of a chunk whose client has its base chunk packet.
143    pub fn get_light_packet_tracking_players(&self, chunk: ChunkPos) -> Vec<i32> {
144        self.player_area_map
145            .get_tracking_players(chunk)
146            .into_iter()
147            .filter(|entity_id| {
148                let Some(player) = self.players.get_by_entity_id(*entity_id) else {
149                    return false;
150                };
151                let Some(view) = *player.last_tracking_view.lock() else {
152                    return false;
153                };
154                let chunk_sender = player.chunk_sender().lock();
155                let is_chunk_sent = |pos| chunk_sender.is_chunk_sent(pos);
156                Self::chunk_is_on_packet_tracked_border(view, chunk, &is_chunk_sent)
157            })
158            .collect()
159    }

161    pub(super) fn chunk_is_on_packet_tracked_border(
162        view: PlayerChunkView,
163        chunk: ChunkPos,
164        is_chunk_sent: &impl Fn(ChunkPos) -> bool,
165    ) -> bool {
166        if !Self::chunk_is_packet_tracked(view, chunk, is_chunk_sent) {
167            return false;
168        }
169
170        for dx in -1..=1 {
171            for dz in -1..=1 {
172                if dx == 0 && dz == 0 {
173                    continue;
174                }
175
176                let neighbor = ChunkPos::new(chunk.0.x + dx, chunk.0.y + dz);
177                if !Self::chunk_is_packet_tracked(view, neighbor, is_chunk_sent) {
178                    return true;
179                }
180            }
181        }
182
183        false
184    }

186    pub(super) fn chunk_is_packet_tracked(
187        view: PlayerChunkView,
188        chunk: ChunkPos,
189        is_chunk_sent: &impl Fn(ChunkPos) -> bool,
190    ) -> bool {
191        view.contains(chunk) && is_chunk_sent(chunk)
192    }

269    pub(super) fn encode_movement_sync_packet(
270        &self,
271        packet: EntityMovementSyncPacket,
272    ) -> Option<EncodedMovementSync> {
273        let encoded = match packet {
274            EntityMovementSyncPacket::Position(packet) => self.encode_play(packet),
275            EntityMovementSyncPacket::PositionRotation(packet) => self.encode_play(packet),
276            EntityMovementSyncPacket::Rotation(packet) => self.encode_play(packet),
277            EntityMovementSyncPacket::HeadRotation(packet) => self.encode_play(packet),
278            EntityMovementSyncPacket::PositionSync(packet) => self.encode_play(packet),
279            EntityMovementSyncPacket::Velocity(packet) => self.encode_play(packet),
280            EntityMovementSyncPacket::ProjectileVelocity(motion, power) => {
281                return Some(EncodedMovementSync::Bundle(vec![
282                    self.encode_play(motion)?,
283                    self.encode_play(power)?,
284                ]));
285            }
286        };
287        encoded.map(EncodedMovementSync::Single)
288    }
289
290    fn encode_play<P: ClientPacket>(&self, packet: P) -> Option<EncodedPacket> {
291        EncodedPacket::from_bare(packet, self.compression, ConnectionProtocol::Play).ok()
292    }
