40impl WorldBorderSnapshot {
41    #[must_use]
42    pub(crate) fn is_block_within_bounds(self, pos: BlockPos) -> bool {
43        self.is_within_bounds_with_margin(f64::from(pos.x()), f64::from(pos.z()), 0.0)
44    }
45
46    #[must_use]
47    pub(crate) fn is_within_bounds_with_margin(self, x: f64, z: f64, margin: f64) -> bool {
48        x >= self.min_x - margin
49            && x < self.max_x + margin
50            && z >= self.min_z - margin
51            && z < self.max_z + margin
52    }
53
54    #[must_use]
55    pub(crate) fn clamp_to_bounds(self, x: f64, y: f64, z: f64) -> BlockPos {
56        BlockPos::from(self.clamp_vec3_to_bound(DVec3::new(x, y, z)))
57    }
58
59    #[must_use]
60    pub(crate) fn clamp_vec3_to_bound(self, position: DVec3) -> DVec3 {
61        let epsilon = f64::from(1.0E-5_f32);
62        DVec3::new(
63            clamp_f64(position.x, self.min_x, self.max_x - epsilon),
64            position.y,
65            clamp_f64(position.z, self.min_z, self.max_z - epsilon),
66        )
67    }
68
69    #[must_use]
70    pub(crate) fn is_within_bounds(self, bounding_box: WorldAabb) -> bool {
71        let epsilon = f64::from(1.0E-5_f32);
72        self.is_within_bounds_with_margin(bounding_box.min_x(), bounding_box.min_z(), 0.0)
73            && self.is_within_bounds_with_margin(
74                bounding_box.max_x() - epsilon,
75                bounding_box.max_z() - epsilon,
76                0.0,
77            )
78    }
79
80    #[must_use]
81    pub(crate) fn distance_to_border(self, x: f64, z: f64) -> f64 {
82        let from_north = z - self.min_z;
83        let from_south = self.max_z - z;
84        let from_west = x - self.min_x;
85        let from_east = self.max_x - x;
86        from_west.min(from_east).min(from_north).min(from_south)
87    }
88
89    #[must_use]
90    pub(crate) fn is_inside_close_to_border(self, x: f64, z: f64, bounding_box: WorldAabb) -> bool {
91        let box_max = bounding_box
92            .width()
93            .abs()
94            .max(bounding_box.depth().abs())
95            .max(1.0);
96        self.distance_to_border(x, z) < box_max * 2.0
97            && self.is_within_bounds_with_margin(x, z, box_max)
98    }
99
100    #[must_use]
101    pub(crate) fn outside_damage_amount(
102        self,
103        x: f64,
104        z: f64,
105        bounding_box: WorldAabb,
106    ) -> Option<f32> {
107        if self.is_within_bounds(bounding_box) {
108            return None;
109        }
110
111        let distance = self.distance_to_border(x, z) + self.safe_zone;
112        if distance >= 0.0 || self.damage_per_block <= 0.0 {
113            return None;
114        }
115
116        Some((-distance * self.damage_per_block).floor().max(1.0) as f32)
117    }

119    #[must_use]
120    pub(crate) fn collision_shapes_for(self, aabb: WorldAabb) -> Vec<WorldAabb> {
121        let min_x = self.min_x.floor();
122        let min_z = self.min_z.floor();
123        let max_x = self.max_x.ceil();
124        let max_z = self.max_z.ceil();
125        let shapes = [
126            WorldAabb::new(
127                f64::NEG_INFINITY,
128                f64::NEG_INFINITY,
129                f64::NEG_INFINITY,
130                min_x,
131                f64::INFINITY,
132                f64::INFINITY,
133            ),
134            WorldAabb::new(
135                max_x,
136                f64::NEG_INFINITY,
137                f64::NEG_INFINITY,
138                f64::INFINITY,
139                f64::INFINITY,
140                f64::INFINITY,
141            ),
142            WorldAabb::new(
143                f64::NEG_INFINITY,
144                f64::NEG_INFINITY,
145                f64::NEG_INFINITY,
146                f64::INFINITY,
147                f64::INFINITY,
148                min_z,
149            ),
150            WorldAabb::new(
151                f64::NEG_INFINITY,
152                f64::NEG_INFINITY,
153                max_z,
154                f64::INFINITY,
155                f64::INFINITY,
156                f64::INFINITY,
157            ),
158        ];
159
160        shapes
161            .into_iter()
162            .filter(|shape| shape.intersects(aabb))
163            .collect()
164    }

167/// Invalid world border state or update.
168#[derive(Debug, Clone, PartialEq)]
169pub enum WorldBorderError {
170    /// A floating-point field was NaN or infinite.
171    NonFinite(&'static str, f64),
172    /// A center coordinate exceeded vanilla's allowed range.
173    CenterOutOfRange(&'static str, f64),
174    /// A size lerp was requested with a negative duration.
175    NegativeLerpTime(i64),
176}
177
178impl fmt::Display for WorldBorderError {
179    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
180        match *self {
181            Self::NonFinite(field, value) => {
182                write!(f, "world border {field} must be finite, got {value}")
183            }
184            Self::CenterOutOfRange(field, value) => write!(
185                f,
186                "world border {field} must be within +/-{MAX_CENTER_COORDINATE}, got {value}"
187            ),
188            Self::NegativeLerpTime(value) => {
189                write!(
190                    f,
191                    "world border lerp time must be non-negative, got {value}"
192                )
193            }
194        }
195    }
196}

200#[derive(Debug, Clone, Copy, PartialEq)]
201enum BorderExtent {
202    Static {
203        size: f64,
204    },
205    Moving {
206        from: f64,
207        to: f64,
208        lerp_duration: i64,
209        lerp_progress: i64,
210        size: f64,
211        previous_size: f64,
212    },
213}
214
215impl BorderExtent {
216    const fn static_size(size: f64) -> Self {
217        Self::Static { size }
218    }
219
220    #[expect(
221        clippy::float_cmp,
222        reason = "Vanilla collapses lerps when start and target sizes are exactly equal."
223    )]
224    fn moving(from: f64, to: f64, ticks: i64) -> Result<Self, WorldBorderError> {
225        if ticks < 0 {
226            return Err(WorldBorderError::NegativeLerpTime(ticks));
227        }
228        if ticks == 0 || from == to {
229            return Ok(Self::static_size(to));
230        }
231
232        Ok(Self::Moving {
233            from,
234            to,
235            lerp_duration: ticks,
236            lerp_progress: ticks,
237            size: from,
238            previous_size: from,
239        })
240    }
241
242    const fn size(self) -> f64 {
243        match self {
244            Self::Static { size } | Self::Moving { size, .. } => size,
245        }
246    }
247
248    const fn lerp_time(self) -> i64 {
249        match self {
250            Self::Static { .. } => 0,
251            Self::Moving { lerp_progress, .. } => lerp_progress,
252        }
253    }
254
255    const fn lerp_target(self) -> f64 {
256        match self {
257            Self::Static { size } => size,
258            Self::Moving { to, .. } => to,
259        }
260    }
261
262    fn status(self) -> BorderStatus {
263        match self {
264            Self::Static { .. } => BorderStatus::Stationary,
265            Self::Moving { from, to, .. } if to < from => BorderStatus::Shrinking,
266            Self::Moving { .. } => BorderStatus::Growing,
267        }
268    }
269
270    fn tick(self) -> Self {
271        let Self::Moving {
272            from,
273            to,
274            lerp_duration,
275            lerp_progress,
276            size,
277            ..
278        } = self
279        else {
280            return self;
281        };
282
283        let next_progress = lerp_progress - 1;
284        if next_progress <= 0 {
285            return Self::Static { size: to };
286        }
287
288        let progress = (lerp_duration - next_progress) as f64 / lerp_duration as f64;
289        Self::Moving {
290            from,
291            to,
292            lerp_duration,
293            lerp_progress: next_progress,
294            size: from + (to - from) * progress,
295            previous_size: size,
296        }
297    }
298}

312impl WorldBorder {
313    pub(crate) fn new(data: WorldBorderData) -> Result<Self, WorldBorderError> {
314        validate_center("center_x", data.center_x)?;
315        validate_center("center_z", data.center_z)?;
316        validate_finite("damage_per_block", data.damage_per_block)?;
317        validate_finite("safe_zone", data.safe_zone)?;
318        validate_finite("size", data.size)?;
319        validate_finite("lerp_target", data.lerp_target)?;
320
321        let extent = if data.lerp_time > 0 {
322            BorderExtent::moving(data.size, data.lerp_target, data.lerp_time)?
323        } else {
324            BorderExtent::static_size(data.size)
325        };
326
327        Ok(Self {
328            center_x: data.center_x,
329            center_z: data.center_z,
330            damage_per_block: data.damage_per_block,
331            safe_zone: data.safe_zone,
332            warning_time: data.warning_time,
333            warning_blocks: data.warning_blocks,
334            absolute_max_size: DEFAULT_ABSOLUTE_MAX_SIZE,
335            extent,
336        })
337    }

339    #[must_use]
340    pub(crate) fn snapshot(&self) -> WorldBorderSnapshot {
341        let old_size = self.extent.size();
342        let new_size = self.extent.lerp_target();
343        let half_size = old_size / 2.0;
344        let absolute_max = f64::from(self.absolute_max_size);
345        WorldBorderSnapshot {
346            center_x: self.center_x,
347            center_z: self.center_z,
348            old_size,
349            new_size,
350            lerp_time: self.extent.lerp_time(),
351            absolute_max_size: self.absolute_max_size,
352            warning_blocks: self.warning_blocks,
353            warning_time: self.warning_time,
354            damage_per_block: self.damage_per_block,
355            safe_zone: self.safe_zone,
356            min_x: clamp_f64(self.center_x - half_size, -absolute_max, absolute_max),
357            min_z: clamp_f64(self.center_z - half_size, -absolute_max, absolute_max),
358            max_x: clamp_f64(self.center_x + half_size, -absolute_max, absolute_max),
359            max_z: clamp_f64(self.center_z + half_size, -absolute_max, absolute_max),
360            status: self.extent.status(),
361        }
362    }

433fn validate_center(field: &'static str, value: f64) -> Result<(), WorldBorderError> {
434    validate_finite(field, value)?;
435    if !(-MAX_CENTER_COORDINATE..=MAX_CENTER_COORDINATE).contains(&value) {
436        return Err(WorldBorderError::CenterOutOfRange(field, value));
437    }
438    Ok(())
439}
440
441const fn validate_finite(field: &'static str, value: f64) -> Result<(), WorldBorderError> {
442    if value.is_finite() {
443        Ok(())
444    } else {
445        Err(WorldBorderError::NonFinite(field, value))
446    }
447}
448
449fn clamp_f64(value: f64, min: f64, max: f64) -> f64 {
450    if value < min {
451        min
452    } else if value > max {
453        max
454    } else {
455        value
456    }
457}
