1//! Behavior shared by player and breeze wind charges.
2//!
3//! Wind charges fly in a straight line (no acceleration, inertia 1), never burn,
4//! ignore other wind charges and end crystals, and burst into a block-triggering
5//! wind explosion on any impact.
6
7use std::sync::Arc;
8
9use glam::DVec3;
10use steel_registry::{vanilla_damage_types, vanilla_entities};
11use steel_utils::WorldAabb;
12
13use crate::enchantment_helper::{self, EnchantmentPostAttackContext};
14use crate::entity::damage::DamageSource;
15use crate::entity::projectile::{AbstractHurtingProjectile, ProjectileHit};
16use crate::entity::{Entity, RemovalReason, SharedEntity};
17use crate::world::{ClipHitResult, LevelReader as _};
18
19/// Damage dealt to a directly hit entity.
20const ENTITY_HIT_DAMAGE: f32 = 1.0;
21/// How far a block-hit burst is pushed out along the hit face normal.
22const BLOCK_HIT_EXPLOSION_OFFSET: f64 = 0.25;
23/// Wind charges burst on their own this many blocks above the build limit.
24const MAX_HEIGHT_ABOVE_BUILD_LIMIT: i32 = 30;
25/// How far the bounding box sits below the entity position.
26const BOUNDING_BOX_Y_OFFSET: f64 = 0.15_f32 as f64;
27
28/// Behavior shared by player and breeze wind charges.
29///
30/// Concrete wind charges route their `Entity`/`Projectile` overrides to the
31/// `wind_charge_*` methods, since Rust trait defaults cannot override supertraits.
32pub trait AbstractWindCharge: AbstractHurtingProjectile {
33    /// Bursts this wind charge at `position`.
34    fn explode(self: Arc<Self>, position: DVec3);
35
36    /// Bursts once far above the build limit, otherwise ticks as a hurting projectile.
37    fn wind_charge_tick(self: Arc<Self>) {
38        let Some(world) = self.level() else {
39            return;
40        };
41        let max_y = world.max_y_exclusive() - 1;
42        if self.block_position().y() > max_y + MAX_HEIGHT_ABOVE_BUILD_LIMIT {
43            let position = self.position();
44            Arc::clone(&self).explode(position);
45            self.set_removed(RemovalReason::Discarded);
46            return;
47        }
48        self.hurting_projectile_tick();
49    }
50
51    /// Entity-sized box lowered by `BOUNDING_BOX_Y_OFFSET`.
52    fn wind_charge_bounding_box(&self, position: DVec3) -> WorldAabb {
53        let dimensions = self.entity_type().dimensions;
54        let half_width = f64::from(dimensions.width / 2.0);
55        let min_y = position.y - BOUNDING_BOX_Y_OFFSET;
56        WorldAabb::new(
57            position.x - half_width,
58            min_y,
59            position.z - half_width,
60            position.x + half_width,
61            min_y + f64::from(dimensions.height),
62            position.z + half_width,
63        )
64    }
65
66    /// Never collides with other wind charges.
67    fn wind_charge_can_collide_with(&self, other: &dyn Entity) -> bool {
68        !other.is_abstract_wind_charge()
69            && other.can_be_collided_with(Some(self.as_entity_event_source()))
70            && !self.is_passenger_of_same_vehicle(other)
71    }
72
73    /// Never hits other wind charges or end crystals.
74    fn wind_charge_can_hit_entity(&self, entity: &dyn Entity) -> bool {
75        !entity.is_abstract_wind_charge()
76            && entity.entity_type() != &vanilla_entities::END_CRYSTAL
77            && self.hurting_can_hit_entity(entity)
78    }
79
80    /// Deals one point of wind charge damage credited to a living owner, then
81    /// bursts at this charge's position.
82    fn wind_charge_on_hit_entity(self: Arc<Self>, entity: &SharedEntity) {
83        let owner = self
84            .get_owner()
85            .filter(|owner| owner.as_living_entity().is_some());
86        if let Some(living_owner) = owner.as_ref().and_then(|owner| owner.as_living_entity()) {
87            living_owner.set_last_hurt_mob(Some(entity));
88        }
89
90        let direct = Arc::clone(&self).into_shared_entity();
91        let mut source = DamageSource::environment(&vanilla_damage_types::WIND_CHARGE)
92            .with_direct_entity(direct);
93        if let Some(owner) = owner {
94            source = source.with_causing_entity(owner);
95        }
96        if let Some(world) = entity.level()
97            && entity.hurt(&world, &source, ENTITY_HIT_DAMAGE)
98            && entity.as_living_entity().is_some()
99        {
100            let context = EnchantmentPostAttackContext::new(entity, &source);
101            enchantment_helper::do_post_attack_effects(&world, entity.as_ref(), &context);
102        }
103
104        let position = self.position();
105        self.explode(position);
106    }
107
108    /// Runs the block's projectile callback, then bursts just outside the hit face.
109    fn wind_charge_on_hit_block(self: Arc<Self>, hit: &ClipHitResult) {
110        self.projectile_on_hit_block(hit);
111        let normal = hit.direction.offset_vec().as_dvec3() * BLOCK_HIT_EXPLOSION_OFFSET;
112        Arc::clone(&self).explode(hit.location + normal);
113        self.set_removed(RemovalReason::Discarded);
114    }
115
116    /// Dispatches the hit, then discards.
117    fn wind_charge_on_hit(self: Arc<Self>, hit: &ProjectileHit) {
118        Arc::clone(&self).projectile_on_hit(hit);
119        self.set_removed(RemovalReason::Discarded);
120    }
121}
