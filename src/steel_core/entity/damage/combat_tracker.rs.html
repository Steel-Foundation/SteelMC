20/// Ticks after the last hit before damage status resets outside combat.
21const RESET_DAMAGE_STATUS_TIME: i32 = 100;
22/// Ticks after the last hit before combat status resets.
23const RESET_COMBAT_STATUS_TIME: i32 = 300;
24const INTENTIONAL_GAME_DESIGN_BUG: &str = "https://bugs.mojang.com/browse/MCPE-28723";
25const INTENTIONAL_GAME_DESIGN_BUG_ID: &str = "MCPE-28723";

27/// A single recorded hit.
28#[derive(Debug, Clone)]
29pub struct CombatEntry {
30    /// The damage source that produced this entry.
31    pub source: DamageSource,
32    /// Damage dealt after absorption.
33    pub damage: f32,
34    /// Where the victim was climbing or swimming when hit.
35    pub fall_location: Option<FallLocation>,
36    /// The victim's fall distance when hit.
37    pub fall_distance: f32,
38}
39
40/// Where a falling victim was climbing or swimming, used to pick the accident message.
41#[derive(Debug, Clone, Copy, PartialEq, Eq)]
42pub enum FallLocation {
43    /// Not climbing.
44    Generic,
45    /// Ladders and trapdoors.
46    Ladder,
47    /// Vines.
48    Vines,
49    /// Weeping vines and their plant segments.
50    WeepingVines,
51    /// Twisting vines and their plant segments.
52    TwistingVines,
53    /// Scaffolding.
54    Scaffolding,
55    /// Any other climbable block.
56    OtherClimbable,
57    /// Falling in water.
58    Water,
59}
60
61impl FallLocation {
62    /// Categorizes a block state the victim was climbing.
63    #[must_use]
64    pub fn from_block_state(state: BlockStateId) -> Self {
65        let block = state.get_block();
66        if block == &vanilla_blocks::LADDER || block.has_tag(&BlockTag::TRAPDOORS) {
67            Self::Ladder
68        } else if block == &vanilla_blocks::VINE {
69            Self::Vines
70        } else if block == &vanilla_blocks::WEEPING_VINES
71            || block == &vanilla_blocks::WEEPING_VINES_PLANT
72        {
73            Self::WeepingVines
74        } else if block == &vanilla_blocks::TWISTING_VINES
75            || block == &vanilla_blocks::TWISTING_VINES_PLANT
76        {
77            Self::TwistingVines
78        } else if block == &vanilla_blocks::SCAFFOLDING {
79            Self::Scaffolding
80        } else {
81            Self::OtherClimbable
82        }
83    }
84
85    /// Returns the `death.fell.accident.*` key for this location.
86    #[must_use]
87    pub const fn language_key(self) -> &'static str {
88        match self {
89            Self::Generic => "death.fell.accident.generic",
90            Self::Ladder => "death.fell.accident.ladder",
91            Self::Vines => "death.fell.accident.vines",
92            Self::WeepingVines => "death.fell.accident.weeping_vines",
93            Self::TwistingVines => "death.fell.accident.twisting_vines",
94            Self::Scaffolding => "death.fell.accident.scaffolding",
95            Self::OtherClimbable => "death.fell.accident.other_climbable",
96            Self::Water => "death.fell.accident.water",
97        }
98    }
99}

101/// Per-entity combat state.
102#[derive(Debug, Clone)]
103pub(crate) struct CombatTracker {
104    entries: Vec<CombatEntry>,
105    last_damage_time: i32,
106    combat_start_time: i32,
107    combat_end_time: i32,
108    in_combat: bool,
109    taking_damage: bool,
110}
111
112impl CombatTracker {
113    pub(crate) const fn new() -> Self {
114        Self {
115            entries: Vec::new(),
116            last_damage_time: 0,
117            combat_start_time: 0,
118            combat_end_time: 0,
119            in_combat: false,
120            taking_damage: false,
121        }
122    }
123
124    /// Records a hit and enters damage-taking state.
125    pub(crate) fn record_damage(
126        &mut self,
127        source: &DamageSource,
128        damage: f32,
129        fall_location: Option<FallLocation>,
130        fall_distance: f32,
131        tick_count: i32,
132    ) {
133        self.entries.push(CombatEntry {
134            source: source.clone(),
135            damage,
136            fall_location,
137            fall_distance,
138        });
139        self.last_damage_time = tick_count;
140        self.taking_damage = true;
141    }
142
143    /// Enters combat, returning whether the state changed.
144    pub(crate) const fn enter_combat(&mut self, tick_count: i32) -> bool {
145        if self.in_combat {
146            return false;
147        }
148        self.in_combat = true;
149        self.combat_start_time = tick_count;
150        self.combat_end_time = tick_count;
151        true
152    }
153
154    /// Resets expired combat state, returning whether combat ended.
155    pub(crate) fn recheck_status(&mut self, tick_count: i32, alive: bool) -> bool {
156        let reset = if self.in_combat {
157            RESET_COMBAT_STATUS_TIME
158        } else {
159            RESET_DAMAGE_STATUS_TIME
160        };
161        let expired = !alive || tick_count - self.last_damage_time > reset;
162        if !self.taking_damage || !expired {
163            return false;
164        }
165
166        let was_in_combat = self.in_combat;
167        self.taking_damage = false;
168        self.in_combat = false;
169        self.combat_end_time = tick_count;
170        self.entries.clear();
171        was_in_combat
172    }
173
174    /// Returns the combat duration in ticks.
175    #[must_use]
176    pub(crate) const fn duration(&self, tick_count: i32) -> i32 {
177        if self.in_combat {
178            tick_count - self.combat_start_time
179        } else {
180            self.combat_end_time - self.combat_start_time
181        }
182    }
183
184    pub(crate) fn entries(&self) -> &[CombatEntry] {
185        &self.entries
186    }
187
188    /// Drops recorded hits without changing combat status.
189    pub(crate) fn clear_entries(&mut self) {
190        self.entries.clear();
191    }
192}

194/// Builds the death message from the recorded combat entries.
195pub(crate) fn death_message<E: LivingEntity + ?Sized>(victim: &E) -> TextComponent {
196    let entries = victim.living_base().combat_entries();
197    let Some(killing_blow) = entries.last() else {
198        return translated_message(
199            "death.attack.generic".to_owned(),
200            vec![victim.display_name()],
201        );
202    };
203    let killing_source = &killing_blow.source;
204
205    match killing_source.damage_type.death_message_type {
206        DeathMessageType::FallVariants => {
207            if let Some(knock_off_entry) = most_significant_fall(&entries) {
208                return fall_message(victim, knock_off_entry, killing_source.causing_entity());
209            }
210        }
211        DeathMessageType::IntentionalGameDesign => {
212            let death_message_key =
213                format!("death.attack.{}", killing_source.damage_type.message_id);
214            let link =
215                TextComponent::translated(CHAT_SQUARE_BRACKETS.message([translated_message(
216                    format!("{death_message_key}.link"),
217                    Vec::new(),
218                )]))
219                .click_event(ClickEvent::open_url(INTENTIONAL_GAME_DESIGN_BUG))
220                .hover_event(HoverEvent::show_text(TextComponent::plain(
221                    INTENTIONAL_GAME_DESIGN_BUG_ID,
222                )));
223            return translated_message(
224                format!("{death_message_key}.message"),
225                vec![victim.display_name(), link],
226            );
227        }
228        DeathMessageType::Default => {}
229    }
230
231    killing_source.get_localized_death_message(victim)
232}

234/// Picks the entry describing the worst fall, if any.
235fn most_significant_fall(entries: &[CombatEntry]) -> Option<&CombatEntry> {
236    let mut result = None;
237    let mut alternative = None;
238    let mut alternative_damage = 0.0;
239    let mut best_fall = 0.0;
240
241    for (index, entry) in entries.iter().enumerate() {
242        let previous = index.checked_sub(1).map(|previous| &entries[previous]);
243        let is_fake_fall = entry
244            .source
245            .is(&vanilla_damage_type_tags::DamageTypeTag::ALWAYS_MOST_SIGNIFICANT_FALL);
246        let fall_distance = if is_fake_fall {
247            f32::MAX
248        } else {
249            entry.fall_distance
250        };
251
252        if (entry
253            .source
254            .is(&vanilla_damage_type_tags::DamageTypeTag::IS_FALL)
255            || is_fake_fall)
256            && fall_distance > 0.0
257            && (result.is_none() || fall_distance > best_fall)
258        {
259            result = if index > 0 { previous } else { Some(entry) };
260            best_fall = fall_distance;
261        }
262
263        if entry.fall_location.is_some()
264            && (alternative.is_none() || entry.damage > alternative_damage)
265        {
266            alternative = Some(entry);
267            alternative_damage = entry.damage;
268        }
269    }
270
271    if best_fall > 5.0 {
272        result
273    } else if alternative_damage > 5.0 {
274        alternative
275    } else {
276        None
277    }
278}

280/// Builds a fall death message from the entry that knocked the victim off.
281fn fall_message<E: LivingEntity + ?Sized>(
282    victim: &E,
283    knock_off_entry: &CombatEntry,
284    killing_entity: Option<&SharedEntity>,
285) -> TextComponent {
286    let knock_off_source = &knock_off_entry.source;
287    if !knock_off_source.is(&vanilla_damage_type_tags::DamageTypeTag::IS_FALL)
288        && !knock_off_source
289            .is(&vanilla_damage_type_tags::DamageTypeTag::ALWAYS_MOST_SIGNIFICANT_FALL)
290    {
291        let killer_name = killing_entity.map(|entity| entity.display_name());
292        let attacker_entity = knock_off_source.causing_entity();
293        let attacker_name = attacker_entity.map(|entity| entity.display_name());
294
295        if let Some(attacker_name) = attacker_name.as_ref()
296            && killer_name.as_ref() != Some(attacker_name)
297        {
298            return assisted_fall_message(
299                victim,
300                attacker_entity,
301                attacker_name.clone(),
302                "death.fell.assist.item",
303                "death.fell.assist",
304            );
305        }
306
307        return if let Some(killer_name) = killer_name {
308            assisted_fall_message(
309                victim,
310                killing_entity,
311                killer_name,
312                "death.fell.finish.item",
313                "death.fell.finish",
314            )
315        } else {
316            translated_message("death.fell.killer".to_owned(), vec![victim.display_name()])
317        };
318    }
319
320    let fall_location = knock_off_entry
321        .fall_location
322        .unwrap_or(FallLocation::Generic);
323    translated_message(
324        fall_location.language_key().to_owned(),
325        vec![victim.display_name()],
326    )
327}

329/// Builds a fall message naming `attacker_entity`, including their named item.
330fn assisted_fall_message<E: LivingEntity + ?Sized>(
331    victim: &E,
332    attacker_entity: Option<&SharedEntity>,
333    attacker_name: TextComponent,
334    message_with_item: &str,
335    message_without_item: &str,
336) -> TextComponent {
337    let attacker_item = attacker_entity
338        .and_then(|entity| entity.as_living_entity())
339        .map(|living| {
340            living
341                .living_base()
342                .equipment()
343                .lock()
344                .get_ref(EquipmentSlot::MainHand)
345                .clone()
346        })
347        .filter(|item| !item.is_empty());
348
349    match attacker_item {
350        Some(item) if item.has(CUSTOM_NAME) => translated_message(
351            message_with_item.to_owned(),
352            vec![
353                victim.display_name(),
354                attacker_name,
355                ITEM_BEHAVIORS.display_name(&item),
356            ],
357        ),
358        _ => translated_message(
359            message_without_item.to_owned(),
360            vec![victim.display_name(), attacker_name],
361        ),
362    }

385    #[test]
386    fn recheck_status_resets_damage_status_after_its_window() {
387        let mut tracker = CombatTracker::new();
388        tracker.record_damage(
389            &DamageSource::environment(&vanilla_damage_types::GENERIC),
390            1.0,
391            None,
392            0.0,
393            0,
394        );
395
396        assert!(!tracker.recheck_status(100, true));
397        assert_eq!(tracker.entries().len(), 1);
398
399        assert!(!tracker.recheck_status(101, true));
400        assert!(tracker.entries().is_empty());
401    }
402
403    #[test]
404    fn recheck_status_uses_the_longer_window_in_combat() {
405        let mut tracker = CombatTracker::new();
406        assert!(tracker.enter_combat(0));
407        tracker.record_damage(
408            &DamageSource::environment(&vanilla_damage_types::GENERIC),
409            1.0,
410            None,
411            0.0,
412            0,
413        );
414
415        assert!(!tracker.recheck_status(300, true));
416        assert!(tracker.recheck_status(301, true));
417    }

419    #[test]
420    fn most_significant_fall_attributes_a_lethal_fall_to_the_preceding_hit() {
421        init_vanilla_registry();
422        let entries = [
423            entry(&vanilla_damage_types::MOB_ATTACK, 1.0, 0.0, None),
424            entry(&vanilla_damage_types::FALL, 5.0, 10.0, None),
425        ];

427        let selected = most_significant_fall(&entries).expect("a 10 block fall is significant");
428
429        assert_eq!(selected.damage, 1.0);
430    }
