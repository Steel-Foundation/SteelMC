59/// Flap speed gained per tick while airborne (vanilla `Chicken.aiStep`).
60const FLAP_SPEED_AIR_GAIN: f32 = 4.0;
61/// Flap speed lost per tick while grounded (vanilla `Chicken.aiStep`).
62const FLAP_SPEED_GROUND_LOSS: f32 = 1.0;
63/// Scales the per-tick flap-speed adjustment (vanilla `Chicken.aiStep`).
64const FLAP_SPEED_ADJUST_SCALE: f32 = 0.3;
65const FLAP_SPEED_MIN: f32 = 0.0;
66const FLAP_SPEED_MAX: f32 = 1.0;
67/// Flapping strength restored whenever the chicken is airborne.
68const MIN_FLAPPING_STRENGTH: f32 = 1.0;
69/// Multiplicative flapping-strength decay per tick.
70const FLAPPING_STRENGTH_DECAY: f32 = 0.9;
71/// Scales accumulated flap rotation (vanilla `Chicken.aiStep`).
72const FLAP_ROTATION_SCALE: f32 = 2.0;
73/// Downward velocity multiplier while airborne, producing the chicken slow-fall.
74const FALL_DRAG_Y: f32 = 0.6;
75/// Divisor converting flap speed into the next flap distance threshold.
76const NEXT_FLAP_SPEED_DIVISOR: f32 = 2.0;

78/// Minimum ticks between egg lays (vanilla `nextInt(6000) + 6000`).
79const EGG_LAY_MIN_DELAY_TICKS: i32 = 6000;
80/// Range of the randomized egg-lay delay in ticks.

157    fn new_with_base(base: EntityBase, entity_type: EntityTypeRef) -> Self {
158        let living_base = LivingEntityBase::new(entity_type);
159        let mob_base = MobBase::new();
160        let ageable_base = AgeableMobBase::new();
161        let animal_base = AnimalBase::new();
162        AnimalBase::initialize_pathfinding_malus(&mob_base);
163        // Chickens do not avoid water (`setPathfindingMalus(PathType.WATER, 0.0F)`).
164        mob_base
165            .pathfinding_malus()
166            .lock()
167            .set(PathType::Water, 0.0);
168        let mut chicken_state = ChickenState::new();
169        chicken_state.egg_time = Self::roll_egg_lay_time();
170        let mut entity_data = ChickenEntityData::new();
171        living_base.initialize_synced_data(&mut entity_data);
172
173        {
174            // Keep vanilla Chicken goal priorities and speeds in the same order.
175            let mut goal_selector = mob_base.goal_selector().lock();
176            goal_selector.add_goal(0, FloatGoal::new(&mob_base));
177            goal_selector.add_goal(1, PanicGoal::new(1.4));
178            goal_selector.add_goal(2, BreedGoal::new(1.0));
179            goal_selector.add_goal(
180                3,
181                TemptGoal::new(
182                    1.0,
183                    |item_stack| {
184                        REGISTRY
185                            .items
186                            .is_in_tag(item_stack.item(), &ItemTag::CHICKEN_FOOD)
187                    },
188                    false,
189                ),
190            );
191            goal_selector.add_goal(4, FollowParentGoal::new(1.1));
192            goal_selector.add_goal(5, WaterAvoidingRandomStrollGoal::new(1.0));
193            goal_selector.add_goal(6, LookAtPlayerGoal::new(6.0));
194            goal_selector.add_goal(7, RandomLookAroundGoal::new());
195        }
196
197        Self {
198            base,
199            entity_type,
200            living_base,
201            mob_base,
202            ageable_base,
203            animal_base,
204            chicken_state: SyncMutex::new(chicken_state),
205            entity_data: SyncMutex::new(entity_data),
206        }

314    /// Runs vanilla `Chicken.aiStep` wing-flap and slow-fall side effects.
315    fn tick_flapping(&self) {
316        let on_ground = self.on_ground();
317        let velocity = self.velocity();
318
319        {
320            let mut state = self.chicken_state.lock();
321            state.o_flap = state.flap;
322            state.o_flap_speed = state.flap_speed;
323            let adjustment = if on_ground {
324                -FLAP_SPEED_GROUND_LOSS
325            } else {
326                FLAP_SPEED_AIR_GAIN
327            };
328            state.flap_speed = (state.flap_speed + adjustment * FLAP_SPEED_ADJUST_SCALE)
329                .clamp(FLAP_SPEED_MIN, FLAP_SPEED_MAX);
330            if !on_ground && state.flapping < MIN_FLAPPING_STRENGTH {
331                state.flapping = MIN_FLAPPING_STRENGTH;
332            }
333            state.flapping *= FLAPPING_STRENGTH_DECAY;
334            state.flap += state.flapping * FLAP_ROTATION_SCALE;
335        }
336
337        if !on_ground && velocity.y < 0.0 {
338            self.set_velocity(DVec3::new(
339                velocity.x,
340                velocity.y * f64::from(FALL_DRAG_Y),
341                velocity.z,
342            ));
343        }
344    }
345
346    /// Runs vanilla `Chicken.aiStep` egg-laying side effects.
347    fn tick_egg_laying(&self) {
348        if self.level().is_none() || !Entity::is_alive(self) || AgeableMob::is_baby(self) {
349            return;
350        }
351
352        let should_lay_egg = {
353            let mut state = self.chicken_state.lock();
354            // Vanilla gates egg laying on `!isChickenJockey()`.
355            if state.is_chicken_jockey {
356                return;
357            }
358            state.egg_time -= 1;
359            state.egg_time <= 0
360        };
361
362        if should_lay_egg {
363            if self.drop_gift_loot_table(&vanilla_loot_tables::GAMEPLAY_CHICKEN_LAY) {
364                let pitch = EGG_LAY_SOUND_BASE_PITCH
365                    + (rand::random::<f32>() - rand::random::<f32>()) * EGG_LAY_SOUND_PITCH_JITTER;
366                self.play_sound(
367                    &sound_events::ENTITY_CHICKEN_EGG,
368                    EGG_LAY_SOUND_VOLUME,
369                    pitch,
370                );
371                self.game_event(&vanilla_game_events::ENTITY_PLACE);
372            }
373            self.set_egg_time(Self::roll_egg_lay_time());
374        }
375    }
376
377    /// Rolls a gift loot table at this entity's position and drops each result.
378    ///
379    /// Mirrors vanilla `LivingEntity.dropFromGiftLootTable` for the
380    /// `gameplay/chicken_lay` table, returning whether any item was dropped.
381    fn drop_gift_loot_table(&self, loot_table: LootTableRef) -> bool {
382        let position = self.position();
383        let mut rng = rand::rng();
384        // Vanilla `LootContextParamSets.GIFT` carries only ORIGIN and THIS_ENTITY.
385        let mut context = LootContext::new(&mut rng)
386            .with_origin(position.x, position.y, position.z)
387            .with_this_entity(entity_loot_ref(self));
388
389        let items = loot_table.get_random_items(&mut context);
390        let dropped_any = !items.is_empty();
391        for item in items {
392            self.spawn_at_location(item, EGG_LAY_SPAWN_Y_OFFSET);
393        }
394        dropped_any
395    }

462    fn load_additional(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
463        self.load_mob(nbt);
464        self.load_ageable_mob(nbt);
465        self.load_animal(nbt);
466
467        self.set_chicken_jockey(nbt.byte("IsChickenJockey").is_some_and(|value| value != 0));
468        if let Some(egg_time) = nbt.int("EggLayTime") {
469            self.set_egg_time(egg_time);
470        }
471        if let Some(variant) = nbt.string("variant")
472            && let Ok(key) = Identifier::from_str(variant.to_str().as_ref())
473        {
474            self.set_variant_by_key(&key);
475        }
476        if let Some(sound_variant) = nbt.string("sound_variant")
477            && let Ok(key) = Identifier::from_str(sound_variant.to_str().as_ref())
478        {
479            self.set_sound_variant_by_key(&key);
480        }
481    }

625        &self,
626        world: &Arc<World>,
627        spawn_reason: EntitySpawnReason,
628        group_data: Option<SpawnGroupData>,
629    ) -> Option<SpawnGroupData> {
630        let biome = world.biome_at(self.block_position());
631        // Mirrors the cow/pig convention.
632        let (variant, sound_variant) = {
633            let mut random = LegacyRandom::from_seed(rand::random());
634            let variant = biome.and_then(|biome| {
635                REGISTRY
636                    .chicken_variants
637                    .select_spawn_variant(biome, &mut random)
638            });
639            let sound_variant = REGISTRY.chicken_sound_variants.pick_random(&mut random);
640            (variant, sound_variant)
641        };
642
643        if let Some(variant) = variant {
644            self.set_variant(variant);
645        }
646
647        if let Some(sound_variant) = sound_variant {
648            self.set_sound_variant(sound_variant);
649        }
650
651        self.finalize_spawn_ageable_mob(world, spawn_reason, group_data)
652    }
