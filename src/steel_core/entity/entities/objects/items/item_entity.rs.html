39/// Maximum age in ticks before despawn (5 minutes = 6000 ticks).
40const LIFETIME: i32 = 6000;
41
42/// Pickup delay set by `set_default_pickup_delay()` (0.5 seconds = 10 ticks).
43/// Note: Items spawn with 0 delay by default; this is only used when explicitly set.
44const DEFAULT_PICKUP_DELAY: i32 = 10;
45
46/// Pickup delay value meaning "never pickupable".
47const INFINITE_PICKUP_DELAY: i32 = 32767;
48
49/// Age value meaning "infinite lifetime" (never despawns).
50const INFINITE_LIFETIME: i32 = -32768;
51
52/// Default health (damage resistance).
53const DEFAULT_HEALTH: i32 = 5;
54
55/// Gravity applied per tick (blocks/tick^2). Vanilla: `ItemEntity.getDefaultGravity()`
56const DEFAULT_GRAVITY: f64 = 0.04;
57
58/// Air/vertical drag multiplier per tick.
59const AIR_DRAG: f64 = 0.98;
60const FLUID_VERTICAL_NUDGE: f64 = 5.0e-4;
61const ITEM_FLUID_HEIGHT_THRESHOLD: f64 = 0.1;
62const ITEM_WATER_DRAG: f64 = 0.99;
63const ITEM_LAVA_DRAG: f64 = 0.95;
64const MERGE_MAX_STACK_SIZE: i32 = 64;

66/// Mutable item-specific state that changes during item ticks, pickup, damage,
67/// merging, and save/load.
68struct ItemEntityState {
69    /// Age in ticks. Despawns at `LIFETIME` (6000). Special value -32768 = infinite.
70    age: i32,
71    /// Ticks until pickupable. 0 = can pickup, 32767 = never.
72    pickup_delay: i32,
73    /// Health (damage resistance). Item is destroyed when this reaches 0.
74    health: i32,
75    /// UUID of the entity that threw/dropped this item.
76    thrower: Option<Uuid>,
77    /// UUID of the only entity that can pick up this item.
78    /// If `None`, any player can pick it up. Vanilla calls this `target`.
79    owner: Option<Uuid>,
80}

84        Self {
85            age: 0,
86            pickup_delay: 0,
87            health: DEFAULT_HEALTH,
88            thrower: None,
89            owner: None,
90        }
91    }

304    /// Attempts to have a player pick up this item.
305    ///
306    /// Returns `true` if the item was fully picked up (and the entity should be removed),
307    /// `false` if pickup failed or was only partial.
308    ///
309    /// Mirrors vanilla's `ItemEntity.playerTouch(Player)`.
310    pub fn try_pickup(&self, player: &Arc<Player>) -> bool {
311        // Check pickup delay
312        if self.has_pickup_delay() {
313            return false;
314        }
315
316        // Check owner restriction
317        if let Some(owner_uuid) = self.get_owner()
318            && owner_uuid != player.gameprofile.id
319        {
320            return false;
321        }
322
323        // Get the item and try to add to inventory
324        let mut item = self.get_item();
325        let original_count = item.count();
326
327        // Try to add to player's inventory
328        let added = player.inventory.lock().add(&mut item);
329
330        // If nothing was added, bail out
331        if item.count() == original_count {
332            return false;
333        }
334
335        // Calculate how many items were picked up
336        let picked_up_count = original_count - item.count();
337
338        // Vanilla uses the stack's count to award the stat that many items picked up, which is
339        // wrong if not all the items get picked up by the player. This could be considered
340        // buggy, but it's possible that people depend on this behavior.
341        player.award_stat_with_count(
342            &vanilla_stat_types::ITEM_PICKED_UP,
343            item.item,
344            original_count,
345        );
346
347        if item.item.has_tag(&ItemTag::FISHES) {
348            player.award_custom_stat(&vanilla_custom_stats::FISH_CAUGHT);
349        }
350
351        // Send the take animation packet to nearby players
352        if let Some(world) = self.level() {
353            let pos = self.position();
354            let chunk_pos = steel_utils::ChunkPos::from_entity_pos(pos);
355
356            let take_packet = CTakeItemEntity::new(self.id(), player.id(), picked_up_count);
357            world.broadcast_to_nearby(chunk_pos, take_packet, None);
358        }
359
360        // Update or remove the item entity
361        if added {
362            // Fully picked up - mark for removal
363            self.set_removed(RemovalReason::Discarded);
364            true
365        } else {
366            // Partial pickup - update the remaining item
367            self.set_item(item);
368            false
369        }
370    }

372    /// Returns true if this item entity can be merged with others.
373    ///
374    /// Mirrors vanilla's `ItemEntity.isMergeable()`.
375    /// An item is mergeable if:
376    /// - It's not removed
377    /// - It doesn't have infinite pickup delay (32767)
378    /// - It doesn't have infinite lifetime (-32768)
379    /// - Its age is less than the despawn threshold (6000)
380    /// - Its count is less than max stack size
381    #[must_use]
382    pub fn is_mergeable(&self) -> bool {
383        let item = self.get_item();
384        let state = self.item_state.lock();
385        !self.is_removed()
386            && state.pickup_delay != INFINITE_PICKUP_DELAY
387            && state.age != INFINITE_LIFETIME
388            && state.age < LIFETIME
389            && item.count() < item.max_stack_size()
390    }

392    /// Checks if two item stacks can be merged together.
393    ///
394    /// Mirrors vanilla's `ItemEntity.areMergeable()`.
395    /// Returns true if the items are the same type with the same components,
396    /// and their combined count wouldn't exceed max stack size.
397    #[must_use]
398    pub fn are_mergeable(this_stack: &ItemStack, other_stack: &ItemStack) -> bool {
399        // Combined count must not exceed max stack size
400        if other_stack.count() + this_stack.count() > other_stack.max_stack_size() {
401            return false;
402        }
403        // Must be the same item with the same components
404        ItemStack::is_same_item_same_components(this_stack, other_stack)
405    }

407    /// Attempts to merge with another item entity.
408    ///
409    /// Mirrors vanilla's `ItemEntity.tryToMerge()`.
410    /// The item with fewer items is merged into the one with more.
411    fn try_to_merge(&self, other: &Self) {
412        let this_stack = self.get_item();
413        let other_stack = other.get_item();
414
415        // Both items must have the same owner (target)
416        if self.get_owner() != other.get_owner() {
417            return;
418        }
419
420        if !Self::are_mergeable(&this_stack, &other_stack) {
421            return;
422        }
423
424        // Merge smaller stack into larger stack
425        if other_stack.count() < this_stack.count() {
426            Self::merge_stacks(self, &this_stack, other, &other_stack);
427        } else {
428            Self::merge_stacks(other, &other_stack, self, &this_stack);
429        }
430    }

432    /// Merges the `from_item`'s stack into the `to_item`'s stack.
433    ///
434    /// Mirrors vanilla's `ItemEntity.merge(ItemEntity, ItemStack, ItemEntity, ItemStack)`.
435    fn merge_stacks(
436        to_item: &Self,
437        to_stack: &ItemStack,
438        from_item: &Self,
439        from_stack: &ItemStack,
440    ) {
441        // Calculate how many items to transfer
442        let max_count = to_stack.max_stack_size().min(MERGE_MAX_STACK_SIZE);
443        if to_stack.count() >= max_count {
444            return;
445        }
446        let space_available = max_count - to_stack.count();
447        let transfer_count = space_available.min(from_stack.count());
448
449        // Create new stacks
450        let new_to_stack = to_stack.copy_with_count(to_stack.count() + transfer_count);
451        let mut new_from_stack = from_stack.clone();
452        new_from_stack.shrink(transfer_count);
453
454        // Update the destination item
455        let (from_pickup_delay, from_age) = {
456            let state = from_item.item_state.lock();
457            (state.pickup_delay, state.age)
458        };
459        to_item.set_item(new_to_stack);
460
461        // Pickup delay is the max of both so merged items do not become instantly pickable.
462        // Age is the min of both so merged items do not despawn prematurely.
463        {
464            let mut state = to_item.item_state.lock();
465            state.pickup_delay = state.pickup_delay.max(from_pickup_delay);
466            state.age = state.age.min(from_age);
467        }
468
469        if new_from_stack.is_empty() {
470            from_item.set_removed(RemovalReason::Discarded);
471        } else {
472            from_item.set_item(new_from_stack);
473        }
474    }

476    /// Attempts to merge this item with nearby item entities.
477    ///
478    /// Mirrors vanilla's `ItemEntity.mergeWithNeighbors()`.
479    /// Searches for other mergeable item entities within 0.5 blocks horizontally
480    /// and attempts to merge with them.
481    pub fn merge_with_neighbors(&self, world: &Arc<World>) {
482        if !self.is_mergeable() {
483            return;
484        }
485
486        // Search area: 0.5 blocks horizontal, 0 vertical (vanilla uses inflate(0.5, 0.0, 0.5))
487        let search_box = self.bounding_box().inflate_xyz(0.5, 0.0, 0.5);
488
489        // Get all entities in the search area
490        for entity in world.get_entities_in_aabb(&search_box) {
491            // Skip self
492            if entity.id() == self.id() {
493                continue;
494            }
495
496            if let Some(other_item) = entity.downcast_ref::<Self>() {
497                // Double-check mergability (might have changed)
498                if other_item.is_mergeable() {
499                    self.try_to_merge(other_item);
500
501                    // If we've been removed (merged into other), stop
502                    if self.is_removed() {
503                        break;
504                    }
505                }
506            }
507        }
508    }
509
510    fn apply_fluid_movement_or_gravity(&self) {
511        let contact = self.fluid_contact();
512        if contact.water_height() > ITEM_FLUID_HEIGHT_THRESHOLD {
513            self.apply_fluid_movement(ITEM_WATER_DRAG);
514        } else if contact.lava_height() > ITEM_FLUID_HEIGHT_THRESHOLD {
515            self.apply_fluid_movement(ITEM_LAVA_DRAG);
516        } else {
517            self.apply_gravity();
518        }
519    }
520
521    fn apply_fluid_movement(&self, horizontal_drag: f64) {
522        let movement = self.velocity();
523        self.set_velocity(DVec3::new(
524            movement.x * horizontal_drag,
525            movement.y
526                + if movement.y < 0.06 {
527                    FLUID_VERTICAL_NUDGE
528                } else {
529                    0.0
530                },
531            movement.z * horizontal_drag,
532        ));
533    }

536impl Entity for ItemEntity {
537    fn base(&self) -> &EntityBase {
538        &self.base
539    }
540
541    fn entity_type(&self) -> EntityTypeRef {
542        self.entity_type
543    }
544
545    fn tick(&self) {
546        // Check if item is empty
547        if self.get_item().is_empty() {
548            self.set_removed(RemovalReason::Discarded);
549            return;
550        }
551
552        self.default_tick();
553
554        {
555            let mut state = self.item_state.lock();
556            if state.pickup_delay > 0 && state.pickup_delay != INFINITE_PICKUP_DELAY {
557                state.pickup_delay -= 1;
558            }
559        }
560
561        // Vanilla item tick stores previous position before applying movement.
562        self.set_old_position_to_current();
563        let old_pos = self.old_position();
564        // Store old movement for needsSync check (vanilla: ItemEntity.tick line 98)
565        let old_movement = self.velocity();
566        // Store old on_ground to detect changes (triggers immediate sync)
567        let old_on_ground = self.on_ground();
568
569        self.apply_fluid_movement_or_gravity();
570        self.update_no_physics_from_current_collision();
571
572        // Vanilla optimization: skip physics when at rest on ground.
573        // Only process physics if:
574        // 1. Not on ground, OR
575        // 2. Has significant horizontal movement, OR
576        // 3. Every 4th tick (for items that might need to fall through opened trapdoors, etc.)
577        // (vanilla: ItemEntity.tick line 121)
578        let vel = self.velocity();
579        let horizontal_movement_sq = vel.x * vel.x + vel.z * vel.z;
580        let should_move = !self.on_ground()
581            || horizontal_movement_sq > 1.0e-5
582            || (self.tick_count() + self.id()) % 4 == 0;
583
584        if should_move {
585            // Move with collision detection; movement handles velocity zeroing on collision.
586            if let Some(result) = self.move_entity(MoverType::SelfMovement, self.velocity()) {
587                self.apply_effects_from_blocks();
588                if self.is_removed() {
589                    return;
590                }
591
592                // Get world for block queries
593                if let Some(world) = self.level() {
594                    // Apply friction (vanilla: ItemEntity.tick line 125-128)
595                    let friction = if result.on_ground
596                        && let Some(block_pos) = self.block_pos_below_that_affects_movement()
597                    {
598                        let block_state = world.get_block_state(block_pos);
599                        f64::from(block_state.get_block().config.friction) * AIR_DRAG
600                    } else {
601                        AIR_DRAG
602                    };
603
604                    let mut velocity = self.velocity();
605                    velocity.x *= friction;
606                    velocity.z *= friction;
607                    velocity.y *= AIR_DRAG;
608
609                    // Bounce when landing on ground (vanilla: ItemEntity.tick lines 145-149)
610                    if result.on_ground && velocity.y < 0.0 {
611                        velocity.y *= -0.5;
612                    }
613
614                    self.set_velocity(velocity);
615                }
616            }
617        } else {
618            self.apply_effects_from_blocks_for_last_movements();
619            if self.is_removed() {
620                return;
621            }
622        }
623
624        // Item merging (vanilla: ItemEntity.tick lines 152-156)
625        // Merge rate depends on whether the item moved to a different block
626        let current_pos = self.position();
627        let moved_block = old_pos.x.floor() as i32 != current_pos.x.floor() as i32
628            || old_pos.y.floor() as i32 != current_pos.y.floor() as i32
629            || old_pos.z.floor() as i32 != current_pos.z.floor() as i32;
630        let merge_rate = if moved_block { 2 } else { 40 };
631
632        if self.tick_count() % merge_rate == 0
633            && self.is_mergeable()
634            && let Some(world) = self.level()
635        {
636            self.merge_with_neighbors(&world);
637        }
638
639        // Check if velocity changed significantly -> set needsSync (vanilla: ItemEntity.tick lines 160-164)
640        // Vanilla: if (getDeltaMovement().subtract(oldMovement).lengthSqr() > 0.01) needsSync = true
641        let new_movement = self.velocity();
642        let diff = DVec3::new(
643            new_movement.x - old_movement.x,
644            new_movement.y - old_movement.y,
645            new_movement.z - old_movement.z,
646        );
647        let diff_sq = diff.x * diff.x + diff.y * diff.y + diff.z * diff.z;
648        if diff_sq > 0.01 {
649            self.mark_velocity_sync();
650        }
651
652        // Also set needsSync when on_ground changes - this ensures immediate sync
653        // when the item lands or becomes airborne, preventing client desync
654        if self.on_ground() != old_on_ground {
655            self.mark_velocity_sync();
656        }
657
658        let should_despawn = {
659            let mut state = self.item_state.lock();
660            if state.age == INFINITE_LIFETIME {
661                false
662            } else {
663                state.age += 1;
664                state.age >= LIFETIME
665            }
666        };
667
668        if should_despawn {

704    fn hurt(&self, _world: &World, source: &DamageSource, amount: f32) -> bool {
705        // TODO: Check isInvulnerableToBase once the shared non-living entity hook is ported.
706        if !self.get_item().can_be_hurt_by(source.damage_type) {
707            return false;
708        }
709        let new_health = {
710            let mut state = self.item_state.lock();
711            state.health = (state.health as f32 - amount) as i32;
712            state.health
713        };
714        if new_health <= 0 {
715            // TODO: Call item.onDestroyed() when implemented
716            self.set_removed(RemovalReason::Killed);
717        }
718        true
719    }

721    fn save_additional(&self, nbt: &mut NbtCompound) {
722        // Match vanilla's ItemEntity.addAdditionalSaveData
723        let state = self.item_state.lock();
724        nbt.insert("Health", state.health as i16);
725        nbt.insert("Age", state.age as i16);
726        nbt.insert("PickupDelay", state.pickup_delay as i16);
727
728        if let Some(thrower) = state.thrower {
729            nbt.insert("Thrower", NbtTag::IntArray(thrower.to_int_array().to_vec()));
730        }
731        if let Some(owner) = state.owner {
732            nbt.insert("Owner", NbtTag::IntArray(owner.to_int_array().to_vec()));
733        }
734        drop(state);
735
736        let item = self.get_item();
737        if !item.is_empty() {
738            nbt.insert("Item", item.to_nbt_tag());
739        }
740    }
741
742    fn load_additional(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
743        // Match vanilla's ItemEntity.readAdditionalSaveData
744        let mut state = self.item_state.lock();
745        if let Some(health) = nbt.short("Health") {
746            state.health = i32::from(health);
747        }
748        if let Some(age) = nbt.short("Age") {
749            state.age = i32::from(age);
750        }
751        if let Some(pickup_delay) = nbt.short("PickupDelay") {
752            state.pickup_delay = i32::from(pickup_delay);
753        }
754
755        if let Some(thrower_arr) = nbt.int_array("Thrower")
756            && let Some(uuid) = Uuid::from_int_array(&thrower_arr)
757        {
758            state.thrower = Some(uuid);
759        }
760        if let Some(owner_arr) = nbt.int_array("Owner")
761            && let Some(uuid) = Uuid::from_int_array(&owner_arr)
762        {
763            state.owner = Some(uuid);
764        }
765        drop(state);
766
767        if let Some(item_tag) = nbt.compound("Item")
768            && let Some(item) = ItemStack::from_borrowed_compound(&item_tag)
769        {
770            self.entity_data.lock().item.set(item);
771        }
772
773        // Vanilla behavior: discard if item is empty after load
774        if self.get_item().is_empty() {
775            self.set_removed(RemovalReason::Discarded);
776        }
777    }
