23/// The default line width of text shown by a text display.
24pub const DEFAULT_LINE_WIDTH: i32 = 200;

26/// The default text opacity of text shown by a text display.
27///
28/// `-1` corresponds to full opacity.
29pub const DEFAULT_TEXT_OPACITY: i8 = -1;

31/// The default background color of text shown by a text display.
32pub const DEFAULT_BACKGROUND_COLOR: i32 = 0x4000_0000;

34/// A display entity that renders text with configurable text properties.
35///
36/// To **access** or **modify** the data of a text display, you will need to use [`Display::with_view`].
37/// This method takes a function with a [`TextDisplayView`] as a parameter, which can be used within the function.
38#[entity_behavior(class = "TextDisplay")]
39pub struct TextDisplayEntity {
40    base: EntityBase,
41    entity_type: EntityTypeRef,
42    entity_data: SyncMutex<TextDisplayEntityData>,
43}

100    fn piston_push_reaction(&self) -> PushReaction {
101        self.piston_push_reaction_display()
102    }
103
104    fn is_ignoring_block_triggers(&self) -> bool {
105        self.is_ignoring_block_triggers_display()
106    }
107
108    fn save_additional(&self, nbt: &mut NbtCompound) {
109        self.with_view(|view| {
110            <Self as Display>::save_display(&view, nbt);
111
112            nbt.insert("text", view.text_ref());
113            nbt.insert("line_width", view.line_width());
114            nbt.insert("background", view.background_color());
115            nbt.insert("text_opacity", view.text_opacity());
116
117            nbt.insert("shadow", view.shadow());
118            nbt.insert("see_through", view.see_through());
119            nbt.insert("default_background", view.default_background());
120
121            nbt.insert("alignment", view.alignment());
122        });
123    }
124
125    fn load_additional(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
126        self.with_view(|mut view| {
127            <Self as Display>::load_display(&mut view, nbt);
128
129            view.set_line_width(nbt.int_value("line_width").unwrap_or(DEFAULT_LINE_WIDTH));
130            view.set_text_opacity(
131                nbt.byte_value("text_opacity")
132                    .unwrap_or(DEFAULT_TEXT_OPACITY) as u8,
133            );
134            view.set_background_color(
135                nbt.int_value("background")
136                    .unwrap_or(DEFAULT_BACKGROUND_COLOR),
137            );
138
139            let mut flags = TextDisplayFlags::empty();
140            if nbt.byte_value("shadow").is_some_and(|v| v != 0) {
141                flags.insert(TextDisplayFlags::SHADOW);
142            }
143            if nbt.byte_value("see_through").is_some_and(|v| v != 0) {
144                flags.insert(TextDisplayFlags::SEE_THROUGH);
145            }
146            if nbt.byte_value("default_background").is_some_and(|v| v != 0) {
147                flags.insert(TextDisplayFlags::USE_DEFAULT_BACKGROUND);
148            }
149            view.set_flags(flags);
150
151            let alignment = nbt.get("alignment").and_then(Alignment::from_nbt_tag);
152            if let Some(alignment) = alignment {
153                view.set_alignment(alignment);
154            }
155
156            // TODO: Resolve the text component with this display entity's world.
157            let text = nbt.get("text").and_then(TextComponent::from_nbt_tag);
158            if let Some(text) = text {
159                view.set_text(text);
160            }
161        });
162    }
163}

329/// Flags that control some boolean properties of text displays.
330#[derive(Clone, Copy, Debug, PartialEq, Eq)]
331struct TextDisplayFlags(i8);
332
333bitflags! {
334    impl TextDisplayFlags: i8 {
335        const SHADOW = 1;
336        const SEE_THROUGH = 1 << 1;
337        const USE_DEFAULT_BACKGROUND = 1 << 2;
338        const ALIGN_LEFT = 1 << 3;
339        const ALIGN_RIGHT = 1 << 4;
340    }
341}
342
343/// The text alignment used by a text display.
344#[derive(Debug, Clone, Copy, PartialEq, Eq)]
345pub enum Alignment {
346    /// Center alignment.
347    Center,
348    /// Left alignment.
349    Left,
350    /// Right alignment.
351    Right,
352}
353
354impl From<TextDisplayFlags> for Alignment {
355    fn from(flags: TextDisplayFlags) -> Self {
356        if flags.contains(TextDisplayFlags::ALIGN_LEFT) {
357            Alignment::Left
358        } else if flags.contains(TextDisplayFlags::ALIGN_RIGHT) {
359            Alignment::Right
360        } else {
361            Alignment::Center
362        }
363    }
364}
365
366impl ToNbtTag for Alignment {
367    fn to_nbt_tag(self) -> NbtTag {
368        NbtTag::String(
369            match self {
370                Self::Center => "center",
371                Self::Left => "left",
372                Self::Right => "right",
373            }
374            .into(),
375        )
376    }
377}
378
379impl FromNbtTag for Alignment {
380    fn from_nbt_tag(tag: simdnbt::borrow::NbtTag) -> Option<Self> {
381        match tag.string()?.to_str().as_ref() {
382            "center" => Some(Self::Center),
383            "left" => Some(Self::Left),
384            "right" => Some(Self::Right),
385            _ => None,
386        }
387    }
