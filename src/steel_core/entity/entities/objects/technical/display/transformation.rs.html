1//! display transform NBT
2
3use simdnbt::borrow::NbtTag as BorrowedNbtTag;
4use simdnbt::owned::{NbtCompound, NbtTag};
5use simdnbt::{FromNbtTag, ToNbtTag};
6pub use steel_math::transformation::Transformation;
7use steel_utils::nbt::{
8    mat4_from_nbt_tag, quat_from_nbt_tag, quat_to_nbt_tag, vec3_from_nbt_tag, vec3_to_nbt_tag,
9};
10
11pub(super) struct TransformationNbt(pub(super) Transformation);
12
13impl TransformationNbt {
14    fn from_compound(tag: BorrowedNbtTag) -> Option<Transformation> {
15        let compound = tag.compound()?;
16        Some(Transformation {
17            translation: vec3_from_nbt_tag(compound.get("translation")?)?,
18            left_rotation: quat_from_nbt_tag(compound.get("left_rotation")?)?,
19            scale: vec3_from_nbt_tag(compound.get("scale")?)?,
20            right_rotation: quat_from_nbt_tag(compound.get("right_rotation")?)?,
21        })
22    }
23}
24
25impl FromNbtTag for TransformationNbt {
26    fn from_nbt_tag(tag: BorrowedNbtTag) -> Option<Self> {
27        let transformation = Self::from_compound(tag)
28            .or_else(|| mat4_from_nbt_tag(tag).map(Transformation::from))?;
29        Some(Self(transformation))
30    }
31}
32
33impl ToNbtTag for TransformationNbt {
34    fn to_nbt_tag(self) -> NbtTag {
35        let mut compound = NbtCompound::new();
36        compound.insert("translation", vec3_to_nbt_tag(self.0.translation));
37        compound.insert("left_rotation", quat_to_nbt_tag(self.0.left_rotation));
38        compound.insert("scale", vec3_to_nbt_tag(self.0.scale));
39        compound.insert("right_rotation", quat_to_nbt_tag(self.0.right_rotation));
40        NbtTag::Compound(compound)
41    }
42}
