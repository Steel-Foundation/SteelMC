32/// Ticks between two effect application passes.
33const TIME_BETWEEN_APPLICATIONS: i32 = 5;
34const MAX_RADIUS: f32 = 32.0;
35/// The cloud is discarded once its radius shrinks below this.
36const MINIMAL_RADIUS: f32 = 0.5;
37const DEFAULT_RADIUS: f32 = 3.0;
38const HEIGHT: f32 = 0.5;
39/// `Duration` value for a cloud that never expires.
40const INFINITE_DURATION: i32 = -1;
41const DEFAULT_WAIT_TIME: i32 = 20;
42const DEFAULT_REAPPLICATION_DELAY: i32 = 20;
43const DEFAULT_POTION_DURATION_SCALE: f32 = 1.0;
44/// Scale passed to instantaneous effects (instant health/damage) applied by the cloud.
45const INSTANTANEOUS_EFFECT_SCALE: f64 = 0.5;
46
47/// Server-side cloud state that is persisted but not synced.
48struct AreaEffectCloudState {
49    potion_contents: PotionContents,
50    potion_duration_scale: f32,
51    /// Entity id to the tick count at which it may be affected again.
52    victims: FxHashMap<i32, i32>,
53    duration: i32,
54    wait_time: i32,
55    reapplication_delay: i32,
56    duration_on_use: i32,
57    radius_on_use: f32,
58    radius_per_tick: f32,
59    owner: Option<EntityReference>,
60}
61
62impl AreaEffectCloudState {
63    fn new() -> Self {
64        Self {
65            potion_contents: PotionContents::empty(),
66            potion_duration_scale: DEFAULT_POTION_DURATION_SCALE,
67            victims: FxHashMap::default(),
68            duration: INFINITE_DURATION,
69            wait_time: DEFAULT_WAIT_TIME,
70            reapplication_delay: DEFAULT_REAPPLICATION_DELAY,
71            duration_on_use: 0,
72            radius_on_use: 0.0,
73            radius_per_tick: 0.0,
74            owner: None,
75        }
76    }
77}

207    fn server_tick(self: &Arc<Self>, world: &World) {
208        let tick_count = self.tick_count();
209        let (duration, wait_time, radius_per_tick) = {
210            let state = self.state.lock();
211            (state.duration, state.wait_time, state.radius_per_tick)
212        };
213        if duration != INFINITE_DURATION && tick_count.wrapping_sub(wait_time) >= duration {
214            self.set_removed(RemovalReason::Discarded);
215            return;
216        }
217
218        let should_wait = tick_count < wait_time;
219        if self.is_waiting() != should_wait {
220            self.set_waiting(should_wait);
221        }
222        if should_wait {
223            return;
224        }
225
226        let mut radius = self.radius();
227        if radius_per_tick != 0.0 {
228            radius += radius_per_tick;
229            if radius < MINIMAL_RADIUS {
230                self.set_removed(RemovalReason::Discarded);
231                return;
232            }
233            self.set_radius(radius);
234        }
235
236        if tick_count % TIME_BETWEEN_APPLICATIONS == 0 {
237            self.apply_effects(world, tick_count, radius);
238        }
239    }
240
241    /// Applies the cloud's effects to every eligible living entity inside it.
242    fn apply_effects(self: &Arc<Self>, world: &World, tick_count: i32, mut radius: f32) {
243        let (effects, reapplication_delay) = {
244            let mut state = self.state.lock();
245            state.victims.retain(|_, until| tick_count < *until);
246            if !state.potion_contents.has_effects() {
247                state.victims.clear();
248                return;
249            }
250            let scale = state.potion_duration_scale;
251            let effects: Vec<_> = state
252                .potion_contents
253                .all_effects()
254                .iter()
255                .map(|effect| effect.with_scaled_duration(scale))
256                .collect();
257            (effects, state.reapplication_delay)
258        };
259
260        let entities = world.get_entities_in_aabb_matching(&self.bounding_box(), |entity| {
261            !entity.is_spectator() && entity.as_living_entity().is_some()
262        });
263        if entities.is_empty() {
264            return;
265        }
266
267        let this: SharedEntity = Arc::<Self>::clone(self);
268        let owner = self.get_owner();
269        let position = self.position();
270
271        for entity in entities {
272            let Some(living) = entity.as_living_entity() else {
273                continue;
274            };
275            if self.state.lock().victims.contains_key(&entity.id())
276                || !living.is_affected_by_potions()
277                || !effects.iter().any(|effect| {
278                    living.can_be_affected(&to_runtime_instance(effect, effect.duration()))
279                })
280            {
281                continue;
282            }
283
284            let offset = entity.position() - position;
285            let dist = offset.x * offset.x + offset.z * offset.z;
286            if dist > f64::from(radius * radius) {
287                continue;
288            }
289
290            let (radius_on_use, duration_on_use) = {
291                let mut state = self.state.lock();
292                state
293                    .victims
294                    .insert(entity.id(), tick_count.wrapping_add(reapplication_delay));
295                (state.radius_on_use, state.duration_on_use)
296            };
297
298            for effect in &effects {
299                self.apply_effect(world, living, effect, &this, owner.as_ref());
300            }
301
302            if radius_on_use != 0.0 {
303                radius += radius_on_use;
304                if radius < MINIMAL_RADIUS {
305                    self.set_removed(RemovalReason::Discarded);
306                    return;
307                }
308                self.set_radius(radius);
309            }
310
311            if duration_on_use != 0 && self.add_duration(duration_on_use) {
312                self.set_removed(RemovalReason::Discarded);
313                return;
314            }
315        }
316    }
317
318    fn apply_effect(
319        &self,
320        world: &World,
321        living: &dyn LivingEntity,
322        effect: &RegistryMobEffectInstance,
323        this: &SharedEntity,
324        owner: Option<&SharedEntity>,
325    ) {
326        let behavior = MOB_EFFECT_BEHAVIORS.get_behavior(effect.effect());
327        if let Some(instantaneous) = behavior.as_instantaneous() {
328            instantaneous.apply_instantaneous(
329                world,
330                living,
331                effect.amplifier(),
332                Some(this),
333                owner,
334                INSTANTANEOUS_EFFECT_SCALE,
335            );
336        } else {
337            living.add_mob_effect_with_source(
338                to_runtime_instance(effect, effect.duration()),
339                Some(self as &dyn Entity),
340            );
341        }
342    }
343
344    /// Adds `delta` to a finite duration, returning whether the cloud has expired.
345    fn add_duration(&self, delta: i32) -> bool {
346        let mut state = self.state.lock();
347        if state.duration == INFINITE_DURATION {
348            return false;
349        }
350        state.duration = state.duration.wrapping_add(delta);
351        state.duration <= 0
352    }

355impl Entity for AreaEffectCloudEntity {
356    fn base(&self) -> &EntityBase {
357        &self.base
358    }
359
360    fn entity_type(&self) -> EntityTypeRef {
361        self.entity_type
362    }
363
364    fn tick(self: Arc<Self>) {
365        self.default_tick();
366        // Client-side particle rendering is driven by the synced radius, waiting
367        // flag and particle, so only the server tick runs here.
368        if let Some(world) = self.level() {
369            self.server_tick(&world);
370        }
371    }
372
373    fn no_physics(&self) -> bool {
374        true
375    }
376
377    fn piston_push_reaction(&self) -> PushReaction {
378        PushReaction::IgnoreEntity
379    }
380
381    fn dimensions_for_pose(&self, _pose: EntityPose) -> EntityDimensions {
382        EntityDimensions::with_default_eye_height(self.radius() * 2.0, HEIGHT)
383    }
384
385    fn hurt(&self, _world: &World, _source: &DamageSource, _amount: f32) -> bool {
386        false
387    }
388
389    fn restore_owner_reference(&self, owner: &SharedEntity) {
390        if let Some(reference) = &self.state.lock().owner {
391            reference.cache_entity(owner);
392        }
393    }
394
395    fn synced_data(&self) -> Option<&dyn EntitySyncedData> {
396        Some(&self.entity_data)
397    }
398
399    fn apply_implicit_item_components(&self, item_stack: &ItemStack) {
400        if let Some(contents) = item_stack.get(vanilla_components::POTION_CONTENTS) {
401            self.set_potion_contents(contents.clone());
402        }
403        if let Some(scale) = item_stack.get(vanilla_components::POTION_DURATION_SCALE) {
404            self.set_potion_duration_scale(*scale);
405        }
406    }
407
408    fn save_additional(&self, nbt: &mut NbtCompound) {
409        nbt.insert("Age", self.tick_count());
410        let state = self.state.lock();
411        nbt.insert("Duration", state.duration);
412        nbt.insert("WaitTime", state.wait_time);
413        nbt.insert("ReapplicationDelay", state.reapplication_delay);
414        nbt.insert("DurationOnUse", state.duration_on_use);
415        nbt.insert("RadiusOnUse", state.radius_on_use);
416        nbt.insert("RadiusPerTick", state.radius_per_tick);
417        nbt.insert("Radius", self.radius());
418        if let Some(owner) = &state.owner {
419            owner.store(nbt, "Owner");
420        }
421        if state.potion_contents != PotionContents::empty() {
422            nbt.insert(
423                "potion_contents",
424                state.potion_contents.clone().to_nbt_tag(),
425            );
426        }
427        if state.potion_duration_scale != DEFAULT_POTION_DURATION_SCALE {
428            nbt.insert("potion_duration_scale", state.potion_duration_scale);
429        }
430    }
431
432    fn load_additional(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
433        self.base.set_tick_count(nbt.int("Age").unwrap_or(0));
434        {
435            let mut state = self.state.lock();
436            state.duration = nbt.int("Duration").unwrap_or(INFINITE_DURATION);
437            state.wait_time = nbt.int("WaitTime").unwrap_or(DEFAULT_WAIT_TIME);
438            state.reapplication_delay = nbt
439                .int("ReapplicationDelay")
440                .unwrap_or(DEFAULT_REAPPLICATION_DELAY);
441            state.duration_on_use = nbt.int("DurationOnUse").unwrap_or(0);
442            state.radius_on_use = nbt.float("RadiusOnUse").unwrap_or(0.0);
443            state.radius_per_tick = nbt.float("RadiusPerTick").unwrap_or(0.0);
444            state.owner = EntityReference::read(&nbt, "Owner");
445            state.potion_duration_scale = nbt
446                .float("potion_duration_scale")
447                .unwrap_or(DEFAULT_POTION_DURATION_SCALE);
448        }
449        self.set_radius(nbt.float("Radius").unwrap_or(DEFAULT_RADIUS));
450        self.set_potion_contents(
451            nbt.get("potion_contents")
452                .and_then(PotionContents::from_nbt_tag)
453                .unwrap_or_else(PotionContents::empty),
454        );
455    }
