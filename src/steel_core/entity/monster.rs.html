25const BRIGHT_LIGHT_NO_ACTION_TIME_INCREMENT: i32 = 2;

28/// Marks hostile entities, including those outside the `Monster` hierarchy.
29/// Shared rules use this marker to reject leashing.
30pub trait Enemy {}
31
32impl<T: Monster + ?Sized> Enemy for T {}

34/// Shared hostile-mob behavior. Concrete mobs delegate to `_monster` hooks
35/// to avoid name collisions with base-trait methods.
36pub trait Monster: PathfinderMob + Enemy {
37    fn default_xp_reward_monster(&self) -> i32 {
38        DEFAULT_XP_REWARD
39    }
40
41    fn sound_source_monster(&self) -> SoundSource {
42        SoundSource::Hostile
43    }
44
45    fn hurt_sound_monster(&self, _source: &DamageSource) -> Option<SoundEventRef> {
46        Some(&sound_events::ENTITY_HOSTILE_HURT)
47    }
48
49    fn death_sound_monster(&self) -> Option<SoundEventRef> {
50        Some(&sound_events::ENTITY_HOSTILE_DEATH)
51    }
52
53    fn fall_sounds_monster(&self) -> (SoundEventRef, SoundEventRef) {
54        (
55            &sound_events::ENTITY_HOSTILE_SMALL_FALL,
56            &sound_events::ENTITY_HOSTILE_BIG_FALL,
57        )
58    }
59
60    /// Returns the hostile swim sound. Steel has no separate hook for
61    /// vanilla's `entity.hostile.splash` sound yet.
62    fn swim_sound_monster(&self) -> SoundEventRef {
63        &sound_events::ENTITY_HOSTILE_SWIM
64    }
65
66    /// Adds two idle ticks in bright light to hasten despawning.
67    fn update_no_action_time(&self) {
68        let Some(world) = self.level() else {
69            return;
70        };
71        let eye_pos = BlockPos::new(
72            fast_floor(self.position().x),
73            fast_floor(self.get_eye_y()),
74            fast_floor(self.position().z),
75        );
76        if world.light_level_dependent_magic_value(eye_pos) > LIGHT_MAGIC_VALUE_BRIGHTNESS_GATE {
77            self.set_no_action_time(self.no_action_time() + BRIGHT_LIGHT_NO_ACTION_TIME_INCREMENT);
78        }
79    }
80
81    fn monster_ai_step(&self, entity: &SharedEntity) -> Option<MoveResult> {
82        self.update_no_action_time();
83        self.mob_ai_step(entity)
84    }
85
86    /// Favors darker positions for pathfinding.
87    fn get_walk_target_value_monster(&self, pos: BlockPos) -> f32 {
88        let Some(world) = self.level() else {
89            return 0.0;
90        };
91        -world.pathfinding_cost_from_light_levels(pos)
92    }
93
94    fn is_preventing_player_rest(&self, _level: &World, _player: &Player) -> bool {
95        true
96    }
97
98    fn should_drop_experience_monster(&self) -> bool {
99        true
100    }
101
102    /// Allows loot when `mobDrops` is enabled, including for babies.
103    fn should_drop_loot_monster(&self, world: &World) -> bool {
104        world.get_game_rule(&MOB_DROPS)
105    }
106
107    /// Checks dimension light limits using the caller's seeded RNG.
108    fn is_dark_enough_to_spawn(level: &World, pos: BlockPos, random: &mut impl Random) -> bool
109    where
110        Self: Sized,
111    {
112        if i32::from(level.light_value_at(LightLayer::Sky, pos))
113            > random.next_i32_bounded(SKY_LIGHT_SPAWN_ROLL_BOUND)
114        {
115            return false;
116        }
117
118        let dimension_type = &level.dimension_type;
119        let block_light_limit = dimension_type.monster_spawn_block_light_limit;
120        if block_light_limit < i32::from(MAX_LIGHT_LEVEL)
121            && i32::from(level.light_value_at(LightLayer::Block, pos)) > block_light_limit
122        {
123            return false;
124        }
125
126        let sky_darkening = if level.is_thundering() {
127            THUNDERING_SKY_DARKENING
128        } else {
129            level.sky_darkening()
130        };
131        let brightness = level.max_local_raw_brightness(pos, sky_darkening);
132        i32::from(brightness)
133            <= sample_monster_spawn_light_test(&dimension_type.monster_spawn_light_level, random)
134    }
135
136    /// Checks difficulty, light, and the spawn surface.
137    fn check_monster_spawn_rules(
138        entity_type: EntityTypeRef,
139        level: &World,
140        spawn_reason: EntitySpawnReason,
141        pos: BlockPos,
142        random: &mut impl Random,
143    ) -> bool
144    where
145        Self: Sized,
146    {
147        is_spawn_allowed_on_difficulty(entity_type, level)
148            && (spawn_reason.ignores_light_requirements()
149                || Self::is_dark_enough_to_spawn(level, pos, random))
150            && <Self as Mob>::check_mob_spawn_rules(entity_type, level, spawn_reason, pos)
151    }
152
153    /// Checks difficulty and the spawn surface without a light requirement.
154    fn check_any_light_monster_spawn_rules(
155        entity_type: EntityTypeRef,
156        level: &World,
157        spawn_reason: EntitySpawnReason,
158        pos: BlockPos,
159    ) -> bool
160    where
161        Self: Sized,
162    {
163        is_spawn_allowed_on_difficulty(entity_type, level)
164            && <Self as Mob>::check_mob_spawn_rules(entity_type, level, spawn_reason, pos)
165    }
166
167    /// Adds sky visibility to monster spawn rules, except for spawners.
168    fn check_surface_monsters_spawn_rules(
169        entity_type: EntityTypeRef,
170        level: &World,
171        spawn_reason: EntitySpawnReason,
172        pos: BlockPos,
173        random: &mut impl Random,
174    ) -> bool
175    where
176        Self: Sized,
177    {
178        Self::check_monster_spawn_rules(entity_type, level, spawn_reason, pos, random)
179            && (spawn_reason.is_spawner() || level.can_see_sky(pos))
180    }
181}

183/// Applies vanilla's `EntityType.canSpawn` difficulty gate here until Steel
184/// has a creation/spawner gate that rejects hostile types on Peaceful.
185fn is_spawn_allowed_on_difficulty(entity_type: EntityTypeRef, level: &World) -> bool {
186    entity_type.allowed_in_peaceful || level.difficulty() != Difficulty::Peaceful
187}

189/// Samples the dimension's spawn-light provider. Vanilla uses uniform
190/// providers; unknown distributions currently return their minimum.
191fn sample_monster_spawn_light_test(
192    level: &MonsterSpawnLightLevel,
193    random: &mut impl Random,
194) -> i32 {
195    match level {
196        MonsterSpawnLightLevel::Simple(level) => *level,
197        MonsterSpawnLightLevel::Complex {
198            distribution_type,
199            min_inclusive,
200            max_inclusive,
201        } => {
202            if *distribution_type == "minecraft:uniform" {
203                random.next_i32_between(*min_inclusive, *max_inclusive)
204            } else {
205                *min_inclusive
206            }
207        }
208    }
209}
