1//! Multipart entities, shaped after `NeoForge`'s `PartEntity`.
2//!
3//! Vanilla has no shared base for `EnderDragonPart`, so engine code going through
4//! [`PartEntity`] never names a specific mob. Like vanilla's `ServerLevel.dragonParts`,
5//! parts stay out of the entity manager: they are never sectioned, saved or sent to
6//! clients, which synthesize them from the parent's ID (see
7//! [`crate::entity::EntityIdBlock`]).
8
9use std::sync::{Arc, OnceLock, Weak};
10
11use glam::DVec3;
12use steel_registry::entity_type::EntityDimensions;
13
14use super::{Entity, SharedEntity, WeakEntity};
15
16/// A sub-entity of a multipart parent, generalizing vanilla `EnderDragonPart`.
17pub trait PartEntity: Entity {
18    /// Returns this part's shared vanilla state.
19    fn part_base(&self) -> &PartEntityBase;
20
21    /// Returns the owning entity; `None` before it is bound or after it is dropped.
22    fn parent(&self) -> Option<SharedEntity> {
23        self.part_base().parent()
24    }
25
26    /// Returns this part's vanilla name, such as `"head"` or `"tail"`, shared by parts of one kind.
27    fn part_name(&self) -> &'static str {
28        self.part_base().name()
29    }
30
31    /// Moves this part. Parts are outside the entity manager, so this skips its movement commit.
32    fn set_part_position(&self, position: DVec3) {
33        self.base().set_position_local(position);
34    }
35}
36
37/// Shared state for a [`PartEntity`]: `EnderDragonPart`'s `parentMob`, `name` and `size`.
38#[derive(Debug)]
39pub struct PartEntityBase {
40    /// Bound by the world once the parent's `Arc` exists, since Rust cannot pass `this`
41    /// from the parent's constructor as vanilla does.
42    parent: OnceLock<WeakEntity>,
43    name: &'static str,
44    size: EntityDimensions,
45}
46
47impl PartEntityBase {
48    /// Creates part state with the vanilla name and hitbox size.
49    #[must_use]
50    pub const fn new(name: &'static str, size: EntityDimensions) -> Self {
51        Self {
52            parent: OnceLock::new(),
53            name,
54            size,
55        }
56    }
57
58    /// Returns the owning entity, if it is still alive and already bound.
59    #[must_use]
60    pub fn parent(&self) -> Option<SharedEntity> {
61        self.parent.get().and_then(Weak::upgrade)
62    }
63
64    /// Returns the part's vanilla name.
65    #[must_use]
66    pub const fn name(&self) -> &'static str {
67        self.name
68    }
69
70    /// Returns the part's hitbox size, which ignores the pose as in vanilla.
71    #[must_use]
72    pub const fn size(&self) -> EntityDimensions {
73        self.size
74    }
75
76    /// Binds the owning entity. Later calls are ignored, so a re-tracked parent cannot
77    /// rebind its parts.
78    pub fn bind_parent(&self, parent: &SharedEntity) {
79        let _ = self.parent.set(Arc::downgrade(parent));
80    }
81}
