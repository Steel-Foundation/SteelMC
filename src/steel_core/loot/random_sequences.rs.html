49/// Saved vanilla Xoroshiro streams for one domain, using its default world's seed.
50pub(crate) struct RandomSequences {
51    seed: i64,
52    salt: i32,
53    include_world_seed: bool,
54    include_sequence_id: bool,
55    sequences: SyncRwLock<FxHashMap<Identifier, Arc<SyncMutex<RandomSource>>>>,
56    revision: AtomicU64,
57    saved_revision: AtomicU64,
58}
59
60impl RandomSequences {
61    #[cfg(test)]
62    pub(crate) fn new_for_test(seed: i64) -> Self {
63        Self::from_persistent(seed, PersistentRandomSequences::default())
64            .expect("empty sequence data is valid")
65    }
66
67    fn from_persistent(seed: i64, persistent: PersistentRandomSequences) -> io::Result<Self> {
68        let mut sequences = FxHashMap::default();
69        for (key, sequence) in persistent.sequences {
70            let key = key
71                .parse::<Identifier>()
72                .map_err(|error| io::Error::new(io::ErrorKind::InvalidData, error))?;
73            let random = Xoroshiro::from_state(sequence.source.map(i64::cast_unsigned));
74            sequences.insert(
75                key,
76                Arc::new(SyncMutex::new(RandomSource::Xoroshiro(random))),
77            );
78        }
79        Ok(Self {
80            seed,
81            salt: persistent.salt,
82            include_world_seed: persistent.include_world_seed,
83            include_sequence_id: persistent.include_sequence_id,
84            sequences: SyncRwLock::new(sequences),
85            revision: AtomicU64::new(0),
86            saved_revision: AtomicU64::new(0),
87        })
88    }
89
90    fn create_sequence(&self, key: &Identifier) -> RandomSource {
91        let seed = (if self.include_world_seed {
92            self.seed
93        } else {
94            0
95        }) ^ i64::from(self.salt);
96        let random = if self.include_sequence_id {
97            let digest = md5::compute(key.to_string());
98            let mut lo = [0; 8];
99            let mut hi = [0; 8];
100            lo.copy_from_slice(&digest.0[..8]);
101            hi.copy_from_slice(&digest.0[8..]);
102            Xoroshiro::from_seed_with_hash(
103                seed as u64,
104                [u64::from_be_bytes(lo), u64::from_be_bytes(hi)],
105            )
106        } else {
107            Xoroshiro::from_seed(seed as u64)
108        };
109        RandomSource::Xoroshiro(random)
110    }
111
112    pub(crate) fn with_random<T>(
113        &self,
114        key: &Identifier,
115        use_random: impl FnOnce(&mut RandomSource) -> LootResult<T>,
116    ) -> LootResult<T> {
117        let existing = self.sequences.read().get(key).cloned();
118        let source = existing.unwrap_or_else(|| {
119            Arc::clone(
120                self.sequences
121                    .write()
122                    .entry(key.clone())
123                    .or_insert_with(|| Arc::new(SyncMutex::new(self.create_sequence(key)))),
124            )
125        });
126        // Handwritten tables may re-enter loot generation. Never wait on a stream
127        // already borrowed by that call (or another domain world worker).
128        // TODO: Coordinate shared-sequence access across world workers without
129        // turning contention into failed gameplay operations or zero signals.
130        let Some(mut random) = source.try_lock() else {
131            return Err(LootError::RandomUnavailable(
132                "named sequence is already in use",
133            ));
134        };
135        let result = use_random(&mut random);
136        self.revision.fetch_add(1, Ordering::Release);
137        result
138    }
139
140    fn pending_save(&self) -> io::Result<Option<(u64, PersistentRandomSequences)>> {
141        // Capture the revision before reading any stream. A concurrent draw must
142        // stay dirty even when it occurs between snapshotting different streams.
143        let revision = self.revision.load(Ordering::Acquire);
144        if revision == self.saved_revision.load(Ordering::Acquire) {
145            return Ok(None);
146        }
147        let sources: Vec<_> = self
148            .sequences
149            .read()
150            .iter()
151            .map(|(key, source)| (key.to_string(), Arc::clone(source)))
152            .collect();
153        let mut sequences = BTreeMap::new();
154        for (key, source) in sources {
155            let random = source.lock();
156            let RandomSource::Xoroshiro(random) = &*random else {
157                return Err(io::Error::new(
158                    io::ErrorKind::InvalidData,
159                    format!("named sequence {key} was replaced with a non-Xoroshiro source"),
160                ));
161            };
162            sequences.insert(
163                key,
164                PersistentSequence {
165                    source: random.state().map(u64::cast_signed),
166                },
167            );
168        }
169        Ok(Some((
170            revision,
171            PersistentRandomSequences {
172                salt: self.salt,
173                include_world_seed: self.include_world_seed,
174                include_sequence_id: self.include_sequence_id,
175                sequences,
176            },
177        )))
178    }
179}

181/// Loads and saves one shared sequence collection per domain.
182pub(crate) struct DomainRandomSequences {
183    domains: BTreeMap<String, Arc<RandomSequences>>,
184    save_lock: AsyncMutex<()>,
185}
186
187impl DomainRandomSequences {
188    pub(crate) async fn load(worlds: &WorldMap) -> io::Result<Self> {
189        let mut domains = BTreeMap::new();
190        for domain in worlds.domain_names() {
191            let world = worlds.default_world(domain).ok_or_else(|| {
192                io::Error::new(
193                    io::ErrorKind::NotFound,
194                    format!("domain {domain} has no default world"),
195                )
196            })?;
197            let persistent = world
198                .saved_data
199                .load_or_default(saved_data_names::RANDOM_SEQUENCES)
200                .await?;
201            let seed = world.level_data.read().seed();
202            let sequences = Arc::new(RandomSequences::from_persistent(seed, persistent)?);
203            for world in worlds.values().filter(|world| world.domain() == domain) {
204                *world.random_sequences.write() = Some(Arc::clone(&sequences));
205            }
206            domains.insert(domain.to_owned(), sequences);
207        }
208        Ok(Self {
209            domains,
210            save_lock: AsyncMutex::new(()),
211        })
212    }
213
214    pub(crate) async fn save(&self, worlds: &WorldMap) -> io::Result<usize> {
215        let _save_guard = self.save_lock.lock().await;
216        let mut saved = 0;
217        for (domain, sequences) in &self.domains {
218            let Some((revision, persistent)) = sequences.pending_save()? else {
219                continue;
220            };
221            let world = worlds.default_world(domain).ok_or_else(|| {
222                io::Error::new(
223                    io::ErrorKind::NotFound,
224                    format!("domain {domain} has no default world"),
225                )
226            })?;
227            world
228                .saved_data
229                .save(saved_data_names::RANDOM_SEQUENCES, &persistent)
230                .await?;
231            sequences
232                .saved_revision
233                .fetch_max(revision, Ordering::Release);
234            saved += 1;
235        }
236        Ok(saved)
237    }
238}
