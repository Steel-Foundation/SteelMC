49/// Saved vanilla Xoroshiro streams for one domain, using its default world's seed.
50pub(crate) struct RandomSequences {
51    seed: i64,
52    salt: i32,
53    include_world_seed: bool,
54    include_sequence_id: bool,
55    sequences: SyncRwLock<FxHashMap<Identifier, Arc<SyncMutex<RandomSource>>>>,
56    revision: AtomicU64,
57    saved_revision: AtomicU64,
58}
59
60impl RandomSequences {
61    fn from_persistent(seed: i64, persistent: PersistentRandomSequences) -> io::Result<Self> {
62        let mut sequences = FxHashMap::default();
63        for (key, sequence) in persistent.sequences {
64            let key = key
65                .parse::<Identifier>()
66                .map_err(|error| io::Error::new(io::ErrorKind::InvalidData, error))?;
67            let random = Xoroshiro::from_state(sequence.source.map(i64::cast_unsigned));
68            sequences.insert(
69                key,
70                Arc::new(SyncMutex::new(RandomSource::Xoroshiro(random))),
71            );
72        }
73        Ok(Self {
74            seed,
75            salt: persistent.salt,
76            include_world_seed: persistent.include_world_seed,
77            include_sequence_id: persistent.include_sequence_id,
78            sequences: SyncRwLock::new(sequences),
79            revision: AtomicU64::new(0),
80            saved_revision: AtomicU64::new(0),
81        })
82    }
83
84    fn create_sequence(&self, key: &Identifier) -> RandomSource {
85        let seed = (if self.include_world_seed {
86            self.seed
87        } else {
88            0
89        }) ^ i64::from(self.salt);
90        let random = if self.include_sequence_id {
91            let digest = md5::compute(key.to_string());
92            let mut lo = [0; 8];
93            let mut hi = [0; 8];
94            lo.copy_from_slice(&digest.0[..8]);
95            hi.copy_from_slice(&digest.0[8..]);
96            Xoroshiro::from_seed_with_hash(
97                seed as u64,
98                [u64::from_be_bytes(lo), u64::from_be_bytes(hi)],
99            )
100        } else {
101            Xoroshiro::from_seed(seed as u64)
102        };
103        RandomSource::Xoroshiro(random)
104    }
105
106    pub(crate) fn with_random<T>(
107        &self,
108        key: &Identifier,
109        use_random: impl FnOnce(&mut RandomSource) -> LootResult<T>,
110    ) -> LootResult<T> {
111        let existing = self.sequences.read().get(key).cloned();
112        let source = existing.unwrap_or_else(|| {
113            Arc::clone(
114                self.sequences
115                    .write()
116                    .entry(key.clone())
117                    .or_insert_with(|| Arc::new(SyncMutex::new(self.create_sequence(key)))),
118            )
119        });
120        // Handwritten tables may re-enter loot generation. Never wait on a stream
121        // already borrowed by that call (or another domain world worker).
122        // TODO: Coordinate shared-sequence access across world workers without
123        // turning contention into failed gameplay operations or zero signals.
124        let Some(mut random) = source.try_lock() else {
125            return Err(LootError::RandomUnavailable(
126                "named sequence is already in use",
127            ));
128        };
129        let result = use_random(&mut random);
130        self.revision.fetch_add(1, Ordering::Release);
131        result
132    }
133
134    fn pending_save(&self) -> io::Result<Option<(u64, PersistentRandomSequences)>> {
135        // Capture the revision before reading any stream. A concurrent draw must
136        // stay dirty even when it occurs between snapshotting different streams.
137        let revision = self.revision.load(Ordering::Acquire);
138        if revision == self.saved_revision.load(Ordering::Acquire) {
139            return Ok(None);
140        }
141        let sources: Vec<_> = self
142            .sequences
143            .read()
144            .iter()
145            .map(|(key, source)| (key.to_string(), Arc::clone(source)))
146            .collect();
147        let mut sequences = BTreeMap::new();
148        for (key, source) in sources {
149            let random = source.lock();
150            let RandomSource::Xoroshiro(random) = &*random else {
151                return Err(io::Error::new(
152                    io::ErrorKind::InvalidData,
153                    format!("named sequence {key} was replaced with a non-Xoroshiro source"),
154                ));
155            };
156            sequences.insert(
157                key,
158                PersistentSequence {
159                    source: random.state().map(u64::cast_signed),
160                },
161            );
162        }
163        Ok(Some((
164            revision,
165            PersistentRandomSequences {
166                salt: self.salt,
167                include_world_seed: self.include_world_seed,
168                include_sequence_id: self.include_sequence_id,
169                sequences,
170            },
171        )))
172    }
173}

175/// Loads and saves one shared sequence collection per domain.
176pub(crate) struct DomainRandomSequences {
177    domains: BTreeMap<String, Arc<RandomSequences>>,
178    save_lock: AsyncMutex<()>,
179}
180
181impl DomainRandomSequences {
182    pub(crate) async fn load(worlds: &WorldMap) -> io::Result<Self> {
183        let mut domains = BTreeMap::new();
184        for domain in worlds.domain_names() {
185            let world = worlds.default_world(domain).ok_or_else(|| {
186                io::Error::new(
187                    io::ErrorKind::NotFound,
188                    format!("domain {domain} has no default world"),
189                )
190            })?;
191            let persistent = world
192                .saved_data
193                .load_or_default(saved_data_names::RANDOM_SEQUENCES)
194                .await?;
195            let seed = world.level_data.read().seed();
196            let sequences = Arc::new(RandomSequences::from_persistent(seed, persistent)?);
197            for world in worlds.values().filter(|world| world.domain() == domain) {
198                *world.random_sequences.write() = Some(Arc::clone(&sequences));
199            }
200            domains.insert(domain.to_owned(), sequences);
201        }
202        Ok(Self {
203            domains,
204            save_lock: AsyncMutex::new(()),
205        })
206    }
207
208    pub(crate) async fn save(&self, worlds: &WorldMap) -> io::Result<usize> {
209        let _save_guard = self.save_lock.lock().await;
210        let mut saved = 0;
211        for (domain, sequences) in &self.domains {
212            let Some((revision, persistent)) = sequences.pending_save()? else {
213                continue;
214            };
215            let world = worlds.default_world(domain).ok_or_else(|| {
216                io::Error::new(
217                    io::ErrorKind::NotFound,
218                    format!("domain {domain} has no default world"),
219                )
220            })?;
221            world
222                .saved_data
223                .save(saved_data_names::RANDOM_SEQUENCES, &persistent)
224                .await?;
225            sequences
226                .saved_revision
227                .fetch_max(revision, Ordering::Release);
228            saved += 1;
229        }
230        Ok(saved)
231    }
232}
