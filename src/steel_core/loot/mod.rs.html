30/// A recoverable failure evaluating a compiled loot operation.
31#[derive(Debug, thiserror::Error)]
32pub enum LootError {
33    /// Container data changed during generation or another access is generating it.
34    #[error("container loot generation conflicted with another access")]
35    ContainerChanged,
36    /// The required world stream is unavailable or is already being evaluated.
37    #[error("loot random source unavailable: {0}")]
38    RandomUnavailable(&'static str),
39    #[error("loot table {table:?} requires unsupported operation {operation}")]
40    /// An operation is known but its gameplay implementation is unavailable.
41    Unsupported {
42        /// Table active when evaluation failed.
43        table: Option<Identifier>,
44        /// Unsupported operation or capability.
45        operation: &'static str,
46    },
47    #[error("loot table {0} has no registered implementation")]
48    /// No callable was registered for the requested table.
49    MissingTable(Identifier),
50}
51
52/// Result of evaluating a compiled table, condition, or modifier.
53pub type LootResult<T> = Result<T, LootError>;
