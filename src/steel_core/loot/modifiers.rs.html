12/// The supported custom-name projection of a block entity's collected components.
13pub(super) fn copy_block_entity_name(
14    ctx: &super::LootContext<'_>,
15    item: &mut ItemStack,
16) -> super::LootResult<()> {
17    let Some(block_entity) = ctx.block_entity else {
18        return Ok(());
19    };
20    // TODO: Use collected block-entity components for custom-name copying,
21    // including entities without Nameable and absent-name no-op behavior.
22    let Some(nameable) = block_entity.as_nameable() else {
23        return ctx.unsupported("copy_components/block_entity custom name");
24    };
25    if let Some(name) = nameable.custom_name() {
26        item.set(CUSTOM_NAME, name);
27    }
28    Ok(())
29}

31pub(super) fn set_enchantment(item: &mut ItemStack, key: Identifier, level: i32, add: bool) {
32    let Some(enchantments) = item.get_enchantments_for_crafting() else {
33        return;
34    };
35    let base = if add {
36        enchantments.get_level(&key) as i32
37    } else {
38        0
39    };
40    item.set_enchantments(
41        &[(
42            key,
43            base.wrapping_add(level)
44                .clamp(0, Enchantment::MAX_LEVEL as i32) as u32,
45        )],
46        false,
47    );
48}

50pub(super) fn smelt(item: &mut ItemStack, use_input_count: bool) {
51    if item.is_empty() {
52        return;
53    }
54    let input = SingleItemRecipeInput::new(item.copy_with_count(item.count()));
55    let Some(recipe) = REGISTRY
56        .recipes
57        .find_match(&vanilla_recipe_types::SMELTING, &input)
58    else {
59        log::warn!(
60            "Couldn't smelt {} because there is no smelting recipe",
61            item.item.key
62        );
63        return;
64    };
65    let mut result = recipe.data().result.create();
66    if result.is_empty() {
67        return;
68    }
69    let count = if use_input_count { item.count() } else { 1 };
70    result.set_count(
71        count
72            .wrapping_mul(result.count())
73            .min(result.max_stack_size()),
74    );
75    *item = result;
76}

78pub(super) fn uniform_bonus(
79    item: &mut ItemStack,
80    random: &mut RandomSource,
81    level: i32,
82    multiplier: i32,
83) {
84    let bonus = random.next_i32_bounded(multiplier * level + 1);
85    item.set_count(item.count().wrapping_add(bonus));
