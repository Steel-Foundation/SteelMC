1//! Canyon (ravine) carver.
2//!
3//! Carves a single long, narrow tunnel per chunk with per-height width
4//! variation; runs only from an overworld biome's carver list (`minecraft:canyon`, 1 % probability).
5
6use std::f32::consts::{PI, TAU};

19/// Vanilla `WorldCarver.getRange()` — 4 chunks each direction. Shared with
20/// the cave carver.
21const CARVER_RANGE: i32 = 4;
22const MAX_TUNNEL_DISTANCE: i32 = (CARVER_RANGE * 2 - 1) * 16;
23
24/// Position + rotation state that evolves along the canyon's length.
25#[derive(Debug, Clone, Copy)]
26struct CanyonState {
27    x: f64,
28    y: f64,
29    z: f64,
30    /// Yaw.
31    horizontal_rotation: f32,
32    /// Pitch.
33    vertical_rotation: f32,
34}
35
36/// Static per-tunnel parameters for the canyon's `do_carve` loop.
37#[derive(Debug, Clone, Copy)]
38struct CanyonTunnel {
39    tunnel_seed: i64,
40    thickness: f32,
41    distance: i32,
42    y_scale: f64,
43}
44
45impl<N, F> CarveRun<'_, '_, N, F>
46where
47    N: DimensionNoises,
48    F: FnMut(BlockPos) -> u16,
49{
50    /// Runs a canyon carver pass rooted in `source_pos`. `random` must have
51    /// been seeded by the caller via
52    /// `set_large_feature_seed(seed + carver_index, source.x, source.z)`
53    /// and the `isStartChunk` probability check must have already passed.
54    ///
55    /// Mirrors vanilla's `CanyonWorldCarver.carve` — one tunnel per chunk,
56    /// no splits, no rooms.
57    pub fn carve_canyon(
58        &mut self,
59        config: &CanyonCarverConfiguration,
60        source_pos: ChunkPos,
61        random: &mut LegacyRandom,
62    ) {
63        let source_min_x = source_pos.0.x * 16;
64        let source_min_z = source_pos.0.y * 16;
65
66        let lava_level_y = config
67            .base
68            .lava_level
69            .resolve_y(self.ctx.min_y, self.ctx.gen_depth);
70        let params = CarveParams {
71            replaceable_tag: &config.base.replaceable_tag,
72            replaceable_states: cached_replaceable_states(&config.base.replaceable_tag),
73            lava_level_y,
74            style: CarverStyle::Overworld,
75        };
76
77        let state = CanyonState {
78            x: f64::from(source_min_x + random.next_i32_bounded(16)),
79            y: f64::from(
80                config
81                    .base
82                    .y
83                    .sample(random, self.ctx.min_y, self.ctx.gen_depth),
84            ),
85            z: f64::from(source_min_z + random.next_i32_bounded(16)),
86            horizontal_rotation: random.next_f32() * TAU,
87            vertical_rotation: config.vertical_rotation.sample(random),
88        };
89
90        // Draw order: yScale, thickness, distance_factor→distance, tunnel_seed.
91        let y_scale = f64::from(config.base.y_scale.sample(random));
92        let thickness = config.shape.thickness.sample(random);
93        let distance =
94            (MAX_TUNNEL_DISTANCE as f32 * config.shape.distance_factor.sample(random)) as i32;
95        let tunnel_seed = random.next_i64();
96
97        let tunnel = CanyonTunnel {
98            tunnel_seed,
99            thickness,
100            distance,
101            y_scale,
102        };
103
104        self.do_carve_canyon(&params, config, state, tunnel);
105    }
106
107    fn do_carve_canyon(
108        &mut self,
109        params: &CarveParams<'_>,
110        config: &CanyonCarverConfiguration,
111        mut state: CanyonState,
112        tunnel: CanyonTunnel,
113    ) {
114        let mut random = LegacyRandom::from_seed(tunnel.tunnel_seed as u64);
115        let width_factors = config
116            .shape
117            .init_width_factors(self.ctx.gen_depth, &mut random);
118        let mut y_rota: f32 = 0.0;
119        let mut x_rota: f32 = 0.0;
120
121        for current_step in 0..tunnel.distance {
122            let progress = PI * current_step as f32 / tunnel.distance as f32;
123            let mut horizontal_radius = horizontal_tunnel_radius(progress, tunnel.thickness);
124            let mut vertical_radius = horizontal_radius * tunnel.y_scale;
125            horizontal_radius *=
126                f64::from(config.shape.horizontal_radius_factor.sample(&mut random));
127            vertical_radius = config.shape.update_vertical_radius(
128                &mut random,
129                vertical_radius,
130                tunnel.distance as f32,
131                current_step as f32,
132            );
133
134            let xc = trig::cos(f64::from(state.vertical_rotation));
135            let xs = trig::sin(f64::from(state.vertical_rotation));
136            state.x += f64::from(trig::cos(f64::from(state.horizontal_rotation)) * xc);
137            state.y += f64::from(xs);
138            state.z += f64::from(trig::sin(f64::from(state.horizontal_rotation)) * xc);
139            state.vertical_rotation *= 0.7;
140            state.vertical_rotation += x_rota * 0.05;
141            state.horizontal_rotation += y_rota * 0.05;
142            x_rota *= 0.8;
143            y_rota *= 0.5;
144            x_rota += (random.next_f32() - random.next_f32()) * random.next_f32() * 2.0;
145            y_rota += (random.next_f32() - random.next_f32()) * random.next_f32() * 4.0;
146
147            if random.next_i32_bounded(4) == 0 {
148                continue;
149            }
150
151            if !can_reach(
152                self.chunk_min_x,
153                self.chunk_min_z,
154                state.x,
155                state.z,
156                current_step,
157                tunnel.distance,
158                tunnel.thickness,
159            ) {
160                return;
161            }
162
163            let min_y = self.ctx.min_y;
164            let skip_checker = |xd: f64, yd: f64, zd: f64, world_y: i32| {
165                should_skip_canyon(&width_factors, min_y, xd, yd, zd, world_y)
166            };
167
168            self.carve_ellipsoid(
169                params,
170                state.x,
171                state.y,
172                state.z,
173                horizontal_radius,
174                vertical_radius,
175                skip_checker,
176            );
177        }
178    }
179}

181/// Vanilla `CanyonWorldCarver.shouldSkip`. Vanilla indexes
182/// `widthFactorPerHeight[yIndex - 1]` where `yIndex = world_y -
183/// context.getMinGenY()`; i.e. the previous Y's width factor applied to this
184/// block's radial test.
185fn should_skip_canyon(
186    width_factors: &[f32],
187    min_y: i32,
188    xd: f64,
189    yd: f64,
190    zd: f64,
191    world_y: i32,
192) -> bool {
193    let y_index = (world_y - min_y - 1) as usize;
194    let factor = width_factors[y_index];
195    (xd * xd + zd * zd) * f64::from(factor) + yd * yd / 6.0 >= 1.0
196}
