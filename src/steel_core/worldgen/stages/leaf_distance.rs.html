1//! Generation-only leaf-distance convergence.
2//!
3//! Vanilla lets tree edge-shape updates schedule leaf ticks, then converges the
4//! affected leaf components after promotion through ordinary gameplay updates.
5//! Steel resolves the current stable distances at generated Light instead. At
6//! that point every Features task that can write the center chunk is complete,
7//! the light work window provides a guarded 3x3 read cache, and only the center
8//! proto chunk is mutated. A later outer-ring Features task can exceptionally
9//! change the six-block halo inside a neighbor; generation accepts that rare
10//! boundary staleness to keep this work off the game tick. Loaded chunks and
11//! live leaf updates retain ordinary Vanilla tick behavior.

28const LEAF_DISTANCE_LIMIT: u8 = 7;
29const MAX_PROPAGATED_DISTANCE: u8 = LEAF_DISTANCE_LIMIT - 1;
30const LEAF_DISTANCES: [u8; LEAF_DISTANCE_LIMIT as usize] = [1, 2, 3, 4, 5, 6, 7];
31const DIRECTIONS: [Direction; 6] = [
32    Direction::Down,
33    Direction::Up,
34    Direction::North,
35    Direction::South,
36    Direction::West,
37    Direction::East,
38];

40#[derive(Clone, Copy)]
41enum LeafDistanceStateInfo {
42    Other,
43    Source,
44    Leaf {
45        states_by_distance: [BlockStateId; LEAF_DISTANCE_LIMIT as usize],
46    },
47}

49#[derive(Debug, Clone, Copy, PartialEq, Eq)]
50struct LeafDistanceUpdate {
51    pos: BlockPos,
52    state: BlockStateId,
53}

55#[derive(Clone, Copy)]
56struct LeafSolveBounds {
57    center: ChunkPos,
58    min_x: i32,
59    max_x: i32,
60    min_y: i32,
61    max_y_exclusive: i32,
62    min_z: i32,
63    max_z: i32,
64}
65
66impl LeafSolveBounds {
67    fn new(center: ChunkPos, min_y: i32, height: i32) -> Self {
68        let center_min_x = center.0.x * 16;
69        let center_min_z = center.0.y * 16;
70        Self {
71            center,
72            min_x: center_min_x - i32::from(MAX_PROPAGATED_DISTANCE),
73            max_x: center_min_x + 15 + i32::from(MAX_PROPAGATED_DISTANCE),
74            min_y,
75            max_y_exclusive: min_y + height,
76            min_z: center_min_z - i32::from(MAX_PROPAGATED_DISTANCE),
77            max_z: center_min_z + 15 + i32::from(MAX_PROPAGATED_DISTANCE),
78        }
79    }
80
81    const fn contains(self, pos: BlockPos) -> bool {
82        pos.x() >= self.min_x
83            && pos.x() <= self.max_x
84            && pos.y() >= self.min_y
85            && pos.y() < self.max_y_exclusive
86            && pos.z() >= self.min_z
87            && pos.z() <= self.max_z
88    }
89
90    const fn is_center(self, pos: BlockPos) -> bool {
91        ChunkPos::from_block_pos(pos).0.x == self.center.0.x
92            && ChunkPos::from_block_pos(pos).0.y == self.center.0.y
93    }
94}

96static LEAF_DISTANCE_STATE_INFO: LazyLock<Box<[LeafDistanceStateInfo]>> =
97    LazyLock::new(build_leaf_distance_state_info);
98
99pub(super) fn resolve_generated_leaf_distances(workset: &LightWorkset, holder: &ChunkHolder) {
100    let pending_leaf_ticks = pending_leaf_tick_positions(holder);
101    if pending_leaf_ticks.is_empty() {
102        return;
103    }
104
105    let center = holder.get_pos();
106    let min_y = holder.min_y();
107    let height = holder.height();
108    let updates = workset.with_chunk_read_cache(|chunk_cache| {
109        chunk_cache.with_section_read_cache(|section_cache| {
110            let layout = section_cache.layout();
111            plan_leaf_distance_updates(center, min_y, height, &pending_leaf_ticks, |pos| {
112                let Some(cached_block) = layout.cached_block(pos) else {
113                    panic!("generated leaf-distance solver read {pos:?} outside its light cache");
114                };
115                section_cache.get_block_state(cached_block)
116            })
117        })
118    });
119
120    let Some(chunk) = holder.try_chunk(ChunkStatus::InitializeLight) else {
121        panic!("center chunk disappeared during generated leaf-distance resolution");
122    };
123    let proto = chunk;
124
125    let mut writes = Vec::with_capacity(updates.len());
126    for update in updates {
127        let Ok(relative_x) = usize::try_from(update.pos.x().rem_euclid(16)) else {
128            panic!("leaf write X coordinate cannot be represented as usize");
129        };
130        let Ok(relative_y) = usize::try_from(update.pos.y() - min_y) else {
131            panic!("leaf write Y coordinate cannot be represented as usize");
132        };
133        let Ok(relative_z) = usize::try_from(update.pos.z().rem_euclid(16)) else {
134            panic!("leaf write Z coordinate cannot be represented as usize");
135        };
136        writes.push((relative_x, relative_y, relative_z, update.state));
137    }
138
139    // Distance-only leaf changes preserve heightmap and light-source predicates.
140    // The tracked batch still updates palette and random-tick section metadata,
141    // while intentionally omitting neighbor-shape and observer callbacks so the
142    // converged generation wave is not scheduled again.
143    chunk.write_block_batch_for_generation(ChunkStatus::InitializeLight, &writes);
144    let Some(removed_ticks) = proto
145        .scheduled_ticks
146        .remove_pending_blocks_matching(|tick| is_leaf_distance_block(tick.tick_type))
147    else {
148        panic!("generated leaf-distance resolution requires pending chunk ticks");
149    };
150    if !writes.is_empty() || removed_ticks != 0 {
151        chunk.mark_dirty();
152    }
153}

155fn pending_leaf_tick_positions(holder: &ChunkHolder) -> Vec<BlockPos> {
156    let Some(chunk) = holder.try_chunk(ChunkStatus::InitializeLight) else {
157        panic!("generated leaf-distance resolution requires InitializeLight");
158    };
159    let proto = chunk;
160    let Some(ticks) = proto.scheduled_ticks.pending_block_snapshot() else {
161        panic!("generated leaf-distance resolution requires pending chunk ticks");
162    };
163    ticks
164        .iter()
165        .filter(|tick| is_leaf_distance_block(tick.tick_type))
166        .map(|tick| tick.pos)
167        .collect()
168}

170fn plan_leaf_distance_updates(
171    center: ChunkPos,
172    min_y: i32,
173    height: i32,
174    seeds: &[BlockPos],
175    mut state_at: impl FnMut(BlockPos) -> BlockStateId,
176) -> Vec<LeafDistanceUpdate> {
177    // A center leaf can only receive a final value below 7 from a source at
178    // most six edges away, so the center chunk plus a six-block halo is enough.
179    let bounds = LeafSolveBounds::new(center, min_y, height);
180    let mut leaves = FxHashMap::default();
181    let mut source_adjacent = FxHashSet::default();
182    let mut component_queue = VecDeque::new();
183
184    for &seed in seeds {
185        if !bounds.is_center(seed) || !bounds.contains(seed) || leaves.contains_key(&seed) {
186            continue;
187        }
188
189        let state = state_at(seed);
190        if matches!(
191            leaf_distance_state_info(state),
192            LeafDistanceStateInfo::Leaf { .. }
193        ) {
194            leaves.insert(seed, state);
195            component_queue.push_back(seed);
196        }
197    }
198
199    while let Some(pos) = component_queue.pop_front() {
200        for direction in DIRECTIONS {
201            let neighbor = pos.relative(direction);
202            if !bounds.contains(neighbor) || leaves.contains_key(&neighbor) {
203                continue;
204            }
205
206            let state = state_at(neighbor);
207            match leaf_distance_state_info(state) {
208                LeafDistanceStateInfo::Source => {
209                    source_adjacent.insert(pos);
210                }
211                LeafDistanceStateInfo::Leaf { .. } => {
212                    leaves.insert(neighbor, state);
213                    component_queue.push_back(neighbor);
214                }
215                LeafDistanceStateInfo::Other => {}
216            }
217        }
218    }
219
220    let mut distances = FxHashMap::default();
221    let mut distance_queue = VecDeque::new();
222    for pos in source_adjacent {
223        distances.insert(pos, 1);
224        distance_queue.push_back(pos);
225    }
226
227    while let Some(pos) = distance_queue.pop_front() {
228        let Some(&distance) = distances.get(&pos) else {
229            panic!("queued leaf position has no computed distance");
230        };
231        if distance >= MAX_PROPAGATED_DISTANCE {
232            continue;
233        }
234
235        let next_distance = distance + 1;
236        for direction in DIRECTIONS {
237            let neighbor = pos.relative(direction);
238            if leaves.contains_key(&neighbor) && !distances.contains_key(&neighbor) {
239                distances.insert(neighbor, next_distance);
240                distance_queue.push_back(neighbor);
241            }
242        }
243    }
244
245    let mut updates = Vec::new();
246    for (pos, old_state) in leaves {
247        if !bounds.is_center(pos) {
248            continue;
249        }
250
251        let distance = distances.get(&pos).copied().unwrap_or(LEAF_DISTANCE_LIMIT);
252        let LeafDistanceStateInfo::Leaf { states_by_distance } =
253            leaf_distance_state_info(old_state)
254        else {
255            panic!("leaf component contains a non-leaf state");
256        };
257        let new_state = states_by_distance[usize::from(distance - 1)];
258        if new_state != old_state {
259            updates.push(LeafDistanceUpdate {
260                pos,
261                state: new_state,
262            });
263        }
264    }
265
266    updates.sort_unstable_by_key(|update| {
267        (
268            (update.pos.y() - min_y) / 16,
269            update.pos.y(),
270            update.pos.z(),
271            update.pos.x(),
272        )
273    });
274    updates
275}

277fn leaf_distance_state_info(state: BlockStateId) -> LeafDistanceStateInfo {
278    let Some(&info) = LEAF_DISTANCE_STATE_INFO.get(state.0 as usize) else {
279        panic!("invalid block state id {} in leaf-distance solver", state.0);
280    };
281    info
282}

284fn is_leaf_distance_block(block: BlockRef) -> bool {
285    matches!(
286        leaf_distance_state_info(block.default_state()),
287        LeafDistanceStateInfo::Leaf { .. }
288    )
289}

291fn build_leaf_distance_state_info() -> Box<[LeafDistanceStateInfo]> {
292    let mut state_info =
293        vec![LeafDistanceStateInfo::Other; REGISTRY.blocks.state_to_block_lookup.len()];
294
295    for (block_id, block) in REGISTRY.blocks.iter() {
296        let base_state = REGISTRY.blocks.block_to_base_state[block_id];
297        let state_count = block.state_count();
298        if block.has_tag(&BlockTag::PREVENTS_NEARBY_LEAF_DECAY) {
299            for offset in 0..state_count {
300                state_info[usize::from(base_state + offset)] = LeafDistanceStateInfo::Source;
301            }
302            continue;
303        }
304
305        if !block.has_tag(&BlockTag::LEAVES) || !has_leaf_distance_property(block) {
306            continue;
307        }
308
309        for offset in 0..state_count {
310            let state = BlockStateId(base_state + offset);
311            let states_by_distance = array::from_fn(|index| {
312                REGISTRY.blocks.set_property(
313                    state,
314                    &BlockStateProperties::DISTANCE,
315                    LEAF_DISTANCES[index],
316                )
317            });
318            state_info[usize::from(state.0)] = LeafDistanceStateInfo::Leaf { states_by_distance };
319        }
320    }
321
322    state_info.into_boxed_slice()
323}
324
325fn has_leaf_distance_property(block: BlockRef) -> bool {
326    let expected = &BlockStateProperties::DISTANCE;
327    // Match the complete 1..=7 descriptor, not just the serialized name:
328    // scaffolding owns a distinct 0..=7 property also named `distance`.
329    block.properties.iter().any(|property| {
330        property.get_name() == expected.name
331            && property.value_count() == expected.value_count()
332            && (0..expected.value_count()).all(|index| {
333                property.value_name_from_index(index) == expected.value_name_from_index(index)
334            })
335    })
336}

347        states: &FxHashMap<BlockPos, BlockStateId>,
348    ) -> impl FnMut(BlockPos) -> BlockStateId + '_ {
349        let air = vanilla_blocks::AIR.default_state();

353    #[test]
354    fn resolves_center_leaves_from_sources_across_chunk_boundaries() {
355        init_vanilla_registry();
356        let center = ChunkPos::new(0, 0);
357        let center_leaf = BlockPos::new(15, 8, 8);
358        let neighbor_leaf = BlockPos::new(16, 8, 8);
359        let source = BlockPos::new(17, 8, 8);
360        let distance_seven = vanilla_blocks::OAK_LEAVES
361            .default_state()
362            .set_value(&BlockStateProperties::DISTANCE, 7)
363            .set_value(&BlockStateProperties::PERSISTENT, true)
364            .set_value(&BlockStateProperties::WATERLOGGED, true);
365        let mut states = FxHashMap::default();
366        states.insert(center_leaf, distance_seven);
367        states.insert(neighbor_leaf, distance_seven);
368        states.insert(source, vanilla_blocks::OAK_LOG.default_state());
369
370        let updates =
371            plan_leaf_distance_updates(center, 0, 16, &[center_leaf], map_state_reader(&states));
372
373        assert_eq!(updates.len(), 1);
374        assert_eq!(updates[0].pos, center_leaf);
375        assert_eq!(
376            updates[0]
377                .state
378                .try_get_value(&BlockStateProperties::DISTANCE),
379            Some(2)
380        );
381        assert_eq!(
382            updates[0]
383                .state
384                .try_get_value(&BlockStateProperties::PERSISTENT),
385            Some(true)
386        );
387        assert_eq!(
388            updates[0]
389                .state
390                .try_get_value(&BlockStateProperties::WATERLOGGED),
391            Some(true)
392        );
393        assert!(updates.iter().all(|update| update.pos != neighbor_leaf));
394    }

426    #[test]
427    fn six_block_halo_preserves_the_distance_seven_boundary() {
428        init_vanilla_registry();
429        let seed = BlockPos::new(15, 8, 8);
430        let leaves = vanilla_blocks::OAK_LEAVES.default_state();
431        let mut distance_six_states = FxHashMap::default();
432        for x in 15..=20 {
433            let current_distance = if x == 15 { 1 } else { 7 };
434            distance_six_states.insert(
435                BlockPos::new(x, 8, 8),
436                leaves.set_value(&BlockStateProperties::DISTANCE, current_distance),
437            );
438        }
439        distance_six_states.insert(
440            BlockPos::new(21, 8, 8),
441            vanilla_blocks::OAK_LOG.default_state(),
442        );
443
444        let distance_six = plan_leaf_distance_updates(
445            ChunkPos::new(0, 0),
446            0,
447            16,
448            &[seed],
449            map_state_reader(&distance_six_states),
450        );
451        assert_eq!(
452            distance_six[0]
453                .state
454                .try_get_value(&BlockStateProperties::DISTANCE),
455            Some(6)
456        );
457
458        let mut distance_seven_states = distance_six_states;
459        distance_seven_states.insert(
460            BlockPos::new(21, 8, 8),
461            leaves.set_value(&BlockStateProperties::DISTANCE, 6),
462        );
463        distance_seven_states.insert(
464            BlockPos::new(22, 8, 8),
465            vanilla_blocks::OAK_LOG.default_state(),

467        let distance_seven = plan_leaf_distance_updates(

475            distance_seven[0]
