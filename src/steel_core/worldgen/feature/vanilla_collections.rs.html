7const INLINE_CAPACITY: usize = 32;
8const INITIAL_TABLE_CAPACITY: usize = 16;
9const LOAD_FACTOR_NUMERATOR: usize = 3;
10const LOAD_FACTOR_DENOMINATOR: usize = 4;
11const TREEIFY_THRESHOLD: usize = 8;
12const UNTREEIFY_THRESHOLD: usize = 6;
13const MIN_TREEIFY_CAPACITY: usize = 64;
14
15/// Block-position set for feature code that vanilla models as `HashSet<BlockPos>`.
16///
17/// This mirrors the target JDK 25 `HashMap` bucket, resize, and tree-bin behavior. Its iteration
18/// order affects feature RNG and leaf-distance propagation.
19#[derive(Default)]
20pub(super) struct JavaBlockPosSet {
21    entries: SmallVec<[BlockPos; INLINE_CAPACITY]>,
22    table_capacity: usize,
23    tree_bins: Vec<TreeBin>,
24}
25
26impl JavaBlockPosSet {
27    pub(super) fn insert(&mut self, pos: BlockPos) -> bool {
28        if self.entries.contains(&pos) {
29            return false;
30        }
31
32        if self.table_capacity == 0 {
33            self.table_capacity = INITIAL_TABLE_CAPACITY;
34        }
35
36        let bucket = Self::bucket(pos, self.table_capacity);
37        if let Some(tree_bin_index) = self.tree_bins.iter().position(|bin| bin.bucket == bucket) {
38            let (parent, root) = self.tree_bins[tree_bin_index].insert(pos);
39            let Some(parent_index) = self.entries.iter().position(|&entry| entry == parent) else {
40                unreachable!("tree-bin parent must be present in the iteration order");
41            };
42            self.entries.insert(parent_index + 1, pos);
43            self.move_position_to_bucket_front(root, bucket);
44        } else {
45            let insert_at = self
46                .entries
47                .iter()
48                .position(|&entry| Self::bucket(entry, self.table_capacity) > bucket)
49                .unwrap_or(self.entries.len());
50            self.entries.insert(insert_at, pos);
51
52            let bucket_len = self
53                .entries
54                .iter()
55                .filter(|&&entry| Self::bucket(entry, self.table_capacity) == bucket)
56                .count();
57            if bucket_len > TREEIFY_THRESHOLD {
58                if self.table_capacity < MIN_TREEIFY_CAPACITY {
59                    self.resize();
60                } else {
61                    self.treeify_bin(bucket);
62                }
63            }
64        }
65        if self.entries.len()
66            > self.table_capacity * LOAD_FACTOR_NUMERATOR / LOAD_FACTOR_DENOMINATOR
67        {
68            self.resize();
69        }
70        true
71    }
72
73    pub(super) fn contains(&self, pos: BlockPos) -> bool {
74        self.entries.contains(&pos)
75    }
76
77    pub(super) fn is_empty(&self) -> bool {
78        self.entries.is_empty()
79    }

81    pub(super) fn java_order(&self) -> impl Iterator<Item = &BlockPos> {
82        self.entries.iter()
83    }
84
85    pub(super) fn java_ordered_positions(&self) -> Vec<BlockPos> {
86        self.entries.to_vec()
87    }

89    pub(super) fn pop_java_ordered_position(&mut self) -> Option<BlockPos> {
90        if self.entries.is_empty() {
91            return None;
92        }
93
94        let pos = self.entries.remove(0);
95        let bucket = Self::bucket(pos, self.table_capacity);
96        if let Some(tree_bin_index) = self.tree_bins.iter().position(|bin| bin.bucket == bucket) {
97            if self
98                .entries
99                .iter()
100                .any(|&entry| Self::bucket(entry, self.table_capacity) == bucket)
101            {
102                self.tree_bins[tree_bin_index].remove(pos);
103            } else {
104                self.tree_bins.remove(tree_bin_index);
105            }
106        }
107        Some(pos)
108    }
109
110    const fn bucket(pos: BlockPos, table_capacity: usize) -> usize {
111        Self::spread_hash(pos) as usize & (table_capacity - 1)
112    }
113
114    const fn spread_hash(pos: BlockPos) -> u32 {
115        let hash = pos
116            .z()
117            .wrapping_mul(31)
118            .wrapping_add(pos.y())
119            .wrapping_mul(31)
120            .wrapping_add(pos.x()) as u32;
121        hash ^ (hash >> 16)
122    }
123
124    fn resize(&mut self) {
125        let old_capacity = self.table_capacity;
126        self.table_capacity *= 2;
127        let table_capacity = self.table_capacity;
128        self.entries
129            .sort_by_key(|&entry| Self::bucket(entry, table_capacity));
130
131        let old_tree_bins = mem::take(&mut self.tree_bins);
132        for mut tree_bin in old_tree_bins {
133            let low = self
134                .entries
135                .iter()
136                .copied()
137                .filter(|&entry| Self::bucket(entry, table_capacity) == tree_bin.bucket)
138                .collect::<SmallVec<[BlockPos; INLINE_CAPACITY]>>();
139            let high_bucket = tree_bin.bucket + old_capacity;
140            let high = self
141                .entries
142                .iter()
143                .copied()
144                .filter(|&entry| Self::bucket(entry, table_capacity) == high_bucket)
145                .collect::<SmallVec<[BlockPos; INLINE_CAPACITY]>>();
146
147            if low.is_empty() || high.is_empty() {
148                tree_bin.bucket = if low.is_empty() {
149                    high_bucket
150                } else {
151                    tree_bin.bucket
152                };
153                self.tree_bins.push(tree_bin);
154                continue;
155            }
156
157            self.retreeify_split_bin(tree_bin.bucket, &low);
158            self.retreeify_split_bin(high_bucket, &high);
159        }
160    }
161
162    fn treeify_bin(&mut self, bucket: usize) {
163        let positions = self
164            .entries
165            .iter()
166            .copied()
167            .filter(|&entry| Self::bucket(entry, self.table_capacity) == bucket)
168            .collect::<SmallVec<[BlockPos; INLINE_CAPACITY]>>();
169        let tree_bin = TreeBin::new(bucket, &positions);
170        let root = tree_bin.root_position();
171        self.tree_bins.push(tree_bin);
172        self.move_position_to_bucket_front(root, bucket);
173    }
174
175    fn retreeify_split_bin(&mut self, bucket: usize, positions: &[BlockPos]) {
176        if positions.len() <= UNTREEIFY_THRESHOLD {
177            return;
178        }
179        let tree_bin = TreeBin::new(bucket, positions);
180        let root = tree_bin.root_position();
181        self.tree_bins.push(tree_bin);
182        self.move_position_to_bucket_front(root, bucket);
183    }
184
185    fn move_position_to_bucket_front(&mut self, pos: BlockPos, bucket: usize) {
186        let Some(root_index) = self.entries.iter().position(|&entry| entry == pos) else {
187            return;
188        };
189        let Some(bucket_start) = self
190            .entries
191            .iter()
192            .position(|&entry| Self::bucket(entry, self.table_capacity) == bucket)
193        else {
194            return;
195        };
196        if root_index != bucket_start {
197            let pos = self.entries.remove(root_index);
198            self.entries.insert(bucket_start, pos);
199        }
200    }
201}

203struct TreeBin {
204    bucket: usize,
205    nodes: Vec<TreeNode>,
206    root: usize,
207}
208
209impl TreeBin {
210    fn new(bucket: usize, positions: &[BlockPos]) -> Self {
211        let mut tree = Self {
212            bucket,
213            nodes: Vec::new(),
214            root: 0,
215        };
216        let Some((&first, remaining)) = positions.split_first() else {
217            return tree;
218        };
219        tree.nodes.push(TreeNode::new(first));
220        for &pos in remaining {
221            tree.insert(pos);
222        }
223        tree
224    }
225
226    fn root_position(&self) -> BlockPos {
227        self.nodes[self.root].pos
228    }
229
230    fn insert(&mut self, pos: BlockPos) -> (BlockPos, BlockPos) {
231        let mut parent = self.root;
232        loop {
233            let ordering = tree_compare(pos, self.nodes[parent].pos);
234            let next = if ordering.is_lt() {
235                self.nodes[parent].left
236            } else {
237                self.nodes[parent].right
238            };
239            let Some(next) = next else {
240                let index = self.nodes.len();
241                self.nodes.push(TreeNode {
242                    pos,
243                    parent: Some(parent),
244                    left: None,
245                    right: None,
246                    red: true,
247                });
248                if ordering.is_lt() {
249                    self.nodes[parent].left = Some(index);
250                } else {
251                    self.nodes[parent].right = Some(index);
252                }
253                self.root = balance_insertion(&mut self.nodes, self.root, index);
254                return (self.nodes[parent].pos, self.root_position());
255            };
256            parent = next;
257        }
258    }
259
260    fn remove(&mut self, pos: BlockPos) {
261        let Some(node) = self.find(pos) else {
262            return;
263        };
264        let mut root = self.root;
265        let left = self.nodes[node].left;
266        let right = self.nodes[node].right;
267
268        let replacement = if let (Some(left), Some(right)) = (left, right) {
269            let mut successor = right;
270            while let Some(next) = self.nodes[successor].left {
271                successor = next;
272            }
273            let successor_was_red = self.nodes[successor].red;
274            self.nodes[successor].red = self.nodes[node].red;
275            self.nodes[node].red = successor_was_red;
276            let successor_right = self.nodes[successor].right;
277            let node_parent = self.nodes[node].parent;
278
279            if successor == right {
280                self.nodes[node].parent = Some(successor);
281                self.nodes[successor].right = Some(node);
282            } else {
283                let successor_parent = self.nodes[successor].parent;
284                self.nodes[node].parent = successor_parent;
285                if let Some(successor_parent) = successor_parent {
286                    if self.nodes[successor_parent].left == Some(successor) {
287                        self.nodes[successor_parent].left = Some(node);
288                    } else {
289                        self.nodes[successor_parent].right = Some(node);
290                    }
291                }
292                self.nodes[successor].right = Some(right);
293                self.nodes[right].parent = Some(successor);
294            }
295
296            self.nodes[node].left = None;
297            self.nodes[node].right = successor_right;
298            if let Some(successor_right) = successor_right {
299                self.nodes[successor_right].parent = Some(node);
300            }
301            self.nodes[successor].left = Some(left);
302            self.nodes[left].parent = Some(successor);
303            self.nodes[successor].parent = node_parent;
304            if let Some(node_parent) = node_parent {
305                if self.nodes[node_parent].left == Some(node) {
306                    self.nodes[node_parent].left = Some(successor);
307                } else {
308                    self.nodes[node_parent].right = Some(successor);
309                }
310            } else {
311                root = successor;
312            }
313
314            successor_right.unwrap_or(node)
315        } else {
316            left.or(right).unwrap_or(node)
317        };
318
319        if replacement != node {
320            let node_parent = self.nodes[node].parent;
321            self.nodes[replacement].parent = node_parent;
322            if let Some(node_parent) = node_parent {
323                if self.nodes[node_parent].left == Some(node) {
324                    self.nodes[node_parent].left = Some(replacement);
325                } else {
326                    self.nodes[node_parent].right = Some(replacement);
327                }
328            } else {
329                root = replacement;
330                self.nodes[replacement].red = false;
331            }
332            self.nodes[node].parent = None;
333            self.nodes[node].left = None;
334            self.nodes[node].right = None;
335        }
336
337        root = if self.nodes[node].red {
338            root
339        } else {
340            balance_deletion(&mut self.nodes, root, replacement)
341        };
342
343        if replacement == node
344            && let Some(parent) = self.nodes[node].parent
345        {
346            self.nodes[node].parent = None;
347            if self.nodes[parent].left == Some(node) {
348                self.nodes[parent].left = None;
349            } else if self.nodes[parent].right == Some(node) {
350                self.nodes[parent].right = None;
351            }
352        }
353        self.root = root;
354    }
355
356    fn find(&self, pos: BlockPos) -> Option<usize> {
357        let mut node = Some(self.root);
358        while let Some(index) = node {
359            if self.nodes[index].pos == pos {
360                return Some(index);
361            }
362            node = if tree_compare(pos, self.nodes[index].pos).is_lt() {
363                self.nodes[index].left
364            } else {
365                self.nodes[index].right
366            };
367        }
368        None
369    }
370}

393fn tree_compare(left: BlockPos, right: BlockPos) -> Ordering {
394    let left_hash = JavaBlockPosSet::spread_hash(left) as i32;
395    let right_hash = JavaBlockPosSet::spread_hash(right) as i32;
396    left_hash
397        .cmp(&right_hash)
398        .then_with(|| left.y().cmp(&right.y()))
399        .then_with(|| left.z().cmp(&right.z()))
400        .then_with(|| left.x().cmp(&right.x()))
401}

403fn balance_insertion(nodes: &mut [TreeNode], mut root: usize, mut node: usize) -> usize {
404    loop {
405        let Some(parent) = nodes[node].parent else {
406            nodes[node].red = false;
407            return node;
408        };
409        if !nodes[parent].red {
410            return root;
411        }
412        let Some(grandparent) = nodes[parent].parent else {
413            return root;
414        };
415
416        if nodes[grandparent].left == Some(parent) {
417            let uncle = nodes[grandparent].right;
418            if let Some(uncle) = uncle.filter(|&uncle| nodes[uncle].red) {
419                nodes[uncle].red = false;
420                nodes[parent].red = false;
421                nodes[grandparent].red = true;
422                node = grandparent;
423                continue;
424            }
425            if nodes[parent].right == Some(node) {
426                root = rotate_left(nodes, root, parent);
427                node = parent;
428            }
429            let Some(parent) = nodes[node].parent else {
430                return root;
431            };
432            let Some(grandparent) = nodes[parent].parent else {
433                return root;
434            };
435            nodes[parent].red = false;
436            nodes[grandparent].red = true;
437            return rotate_right(nodes, root, grandparent);
438        }
439
440        let uncle = nodes[grandparent].left;
441        if let Some(uncle) = uncle.filter(|&uncle| nodes[uncle].red) {
442            nodes[uncle].red = false;
443            nodes[parent].red = false;
444            nodes[grandparent].red = true;
445            node = grandparent;
446            continue;
447        }
448        if nodes[parent].left == Some(node) {
449            root = rotate_right(nodes, root, parent);
450            node = parent;
451        }
452        let Some(parent) = nodes[node].parent else {
453            return root;
454        };
455        let Some(grandparent) = nodes[parent].parent else {
456            return root;
457        };
458        nodes[parent].red = false;
459        nodes[grandparent].red = true;
460        return rotate_left(nodes, root, grandparent);
461    }
462}

464#[expect(
465    clippy::too_many_lines,
466    reason = "keeping both symmetric branches together makes the OpenJDK red-black deletion port auditable"
467)]
468fn balance_deletion(nodes: &mut [TreeNode], mut root: usize, mut node: usize) -> usize {
469    loop {
470        if node == root {
471            return root;
472        }
473        let Some(mut parent) = nodes[node].parent else {
474            nodes[node].red = false;
475            return node;
476        };
477        if nodes[node].red {
478            nodes[node].red = false;
479            return root;
480        }
481
482        if nodes[parent].left == Some(node) {
483            let mut sibling = nodes[parent].right;
484            if is_red(nodes, sibling) {
485                let sibling_index = sibling.unwrap_or(parent);
486                nodes[sibling_index].red = false;
487                nodes[parent].red = true;
488                root = rotate_left(nodes, root, parent);
489                let Some(new_parent) = nodes[node].parent else {
490                    return root;
491                };
492                parent = new_parent;
493                sibling = nodes[parent].right;
494            }
495            let Some(sibling_index) = sibling else {
496                node = parent;
497                continue;
498            };
499            let sibling_left = nodes[sibling_index].left;
500            let mut sibling_right = nodes[sibling_index].right;
501            if !is_red(nodes, sibling_right) && !is_red(nodes, sibling_left) {
502                nodes[sibling_index].red = true;
503                node = parent;
504                continue;
505            }
506            if !is_red(nodes, sibling_right) {
507                if let Some(sibling_left) = sibling_left {
508                    nodes[sibling_left].red = false;
509                }
510                nodes[sibling_index].red = true;
511                root = rotate_right(nodes, root, sibling_index);
512                let Some(new_parent) = nodes[node].parent else {
513                    return root;
514                };
515                parent = new_parent;
516                sibling = nodes[parent].right;
517            }
518            if let Some(sibling) = sibling {
519                nodes[sibling].red = nodes[parent].red;
520                sibling_right = nodes[sibling].right;
521                if let Some(sibling_right) = sibling_right {
522                    nodes[sibling_right].red = false;
523                }
524            }
525            nodes[parent].red = false;
526            root = rotate_left(nodes, root, parent);
527            node = root;
528            continue;
529        }
530
531        let mut sibling = nodes[parent].left;
532        if is_red(nodes, sibling) {
533            let sibling_index = sibling.unwrap_or(parent);
534            nodes[sibling_index].red = false;
535            nodes[parent].red = true;
536            root = rotate_right(nodes, root, parent);
537            let Some(new_parent) = nodes[node].parent else {
538                return root;
539            };
540            parent = new_parent;
541            sibling = nodes[parent].left;
542        }
543        let Some(sibling_index) = sibling else {
544            node = parent;
545            continue;
546        };
547        let mut sibling_left = nodes[sibling_index].left;
548        let sibling_right = nodes[sibling_index].right;
549        if !is_red(nodes, sibling_left) && !is_red(nodes, sibling_right) {
550            nodes[sibling_index].red = true;
551            node = parent;
552            continue;
553        }
554        if !is_red(nodes, sibling_left) {
555            if let Some(sibling_right) = sibling_right {
556                nodes[sibling_right].red = false;
557            }
558            nodes[sibling_index].red = true;
559            root = rotate_left(nodes, root, sibling_index);
560            let Some(new_parent) = nodes[node].parent else {
561                return root;
562            };
563            parent = new_parent;
564            sibling = nodes[parent].left;
565        }
566        if let Some(sibling) = sibling {
567            nodes[sibling].red = nodes[parent].red;
568            sibling_left = nodes[sibling].left;
569            if let Some(sibling_left) = sibling_left {
570                nodes[sibling_left].red = false;
571            }
572        }
573        nodes[parent].red = false;
574        root = rotate_right(nodes, root, parent);
575        node = root;
576    }
577}

579fn is_red(nodes: &[TreeNode], node: Option<usize>) -> bool {
580    node.is_some_and(|node| nodes[node].red)
581}

583fn rotate_left(nodes: &mut [TreeNode], mut root: usize, node: usize) -> usize {
584    let Some(right) = nodes[node].right else {
585        return root;
586    };
587    nodes[node].right = nodes[right].left;
588    if let Some(right_left) = nodes[right].left {
589        nodes[right_left].parent = Some(node);
590    }
591    nodes[right].parent = nodes[node].parent;
592    if let Some(parent) = nodes[node].parent {
593        if nodes[parent].left == Some(node) {
594            nodes[parent].left = Some(right);
595        } else {
596            nodes[parent].right = Some(right);
597        }
598    } else {
599        root = right;
600        nodes[root].red = false;
601    }
602    nodes[right].left = Some(node);
603    nodes[node].parent = Some(right);
604    root
605}
606
607fn rotate_right(nodes: &mut [TreeNode], mut root: usize, node: usize) -> usize {
608    let Some(left) = nodes[node].left else {
609        return root;
610    };
611    nodes[node].left = nodes[left].right;
612    if let Some(left_right) = nodes[left].right {
613        nodes[left_right].parent = Some(node);
614    }
615    nodes[left].parent = nodes[node].parent;
616    if let Some(parent) = nodes[node].parent {
617        if nodes[parent].right == Some(node) {
618            nodes[parent].right = Some(left);
619        } else {
620            nodes[parent].left = Some(left);
621        }
622    } else {
623        root = left;
624        nodes[root].red = false;
625    }
626    nodes[left].right = Some(node);
627    nodes[node].parent = Some(left);
628    root
629}
