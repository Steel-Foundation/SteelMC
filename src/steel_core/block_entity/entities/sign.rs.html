29/// Maximum distance (in blocks) a player can be from a sign while editing.
30/// If they move further away, the edit lock is released.
31const MAX_EDIT_DISTANCE: f64 = 4.0;
32
33/// Number of text lines on each side of a sign.
34pub const SIGN_LINES: usize = 4;
35
36/// Text and styling for one side of a sign.
37#[derive(Debug, Clone)]
38pub struct SignText {
39    /// The 4 lines of text (raw, unfiltered).
40    pub messages: [TextComponent; SIGN_LINES],
41    /// Text color (dye color applied to the sign).
42    pub color: DyeColor,
43    /// Whether the text has a glowing effect (from glow ink sac).
44    pub has_glowing_text: bool,
45}
46
47impl Default for SignText {
48    fn default() -> Self {
49        Self::new()
50    }
51}
52
53impl SignText {
54    /// Creates a new empty sign text with default color (black) and no glow.
55    #[must_use]
56    pub fn new() -> Self {
57        Self {
58            messages: array::from_fn(|_| TextComponent::new()),
59            color: DyeColor::Black,
60            has_glowing_text: false,
61        }
62    }
63
64    /// Gets a message line by index.
65    #[must_use]
66    pub fn get_message(&self, index: usize) -> Option<&TextComponent> {
67        self.messages.get(index)
68    }
69
70    /// Sets a message line by index.
71    pub fn set_message(&mut self, index: usize, message: TextComponent) {
72        if index < SIGN_LINES {
73            self.messages[index] = message;
74        }
75    }
76
77    /// Checks if any line has text content.
78    #[must_use]
79    pub fn has_message(&self) -> bool {
80        self.messages.iter().any(|msg| {
81            // Check if the text component has any actual content
82            match &msg.content {
83                Content::Text { text } => !text.is_empty(),
84                _ => true, // Translations, etc. count as having a message
85            }
86        })
87    }
88
89    /// Builds sign text from an item's `sign_text_front`/`sign_text_back`
90    /// component. Filtered messages are dropped: Steel has no text filtering.
91    fn from_component(component: Option<&SignTextComponent>) -> Self {
92        component.map_or_else(Self::new, |component| Self {
93            messages: component.messages.clone(),
94            color: component.color,
95            has_glowing_text: component.has_glowing_text,
96        })
97    }
98
99    /// Returns whether the sign editor can edit every line, i.e. all lines are
100    /// plain text. Styling is ignored; translations, scores, etc. are not editable.
101    #[must_use]
102    pub fn has_editable_text(&self) -> bool {
103        self.messages
104            .iter()
105            .all(|msg| matches!(msg.content, Content::Text { .. }))
106    }
107
108    /// Loads sign text from borrowed NBT.
109    pub fn load(&mut self, nbt: BorrowedNbtCompoundView<'_, '_>) {
110        if let Some(messages) = nbt.list("messages") {
111            let tags = messages.to_owned().as_nbt_tags();
112            let messages = tags
113                .iter()
114                .map(TextComponent::from_nbt)
115                .collect::<Option<Vec<_>>>();
116            if let Some(messages) = messages
117                && let Ok(messages) = <[TextComponent; SIGN_LINES]>::try_from(messages)
118            {
119                self.messages = messages;
120            }
121        }
122
123        // Load color
124        if let Some(color_str) = nbt.string("color") {
125            self.color =
126                DyeColor::from_serialized_name(&color_str.to_str()).unwrap_or(DyeColor::Black);
127        }
128
129        // Load glow
130        if let Some(glow) = nbt.byte("has_glowing_text") {
131            self.has_glowing_text = glow != 0;
132        }
133    }
134
135    /// Saves sign text to NBT.
136    pub fn save(&self, nbt: &mut NbtCompound) {
137        nbt.insert(
138            "messages",
139            NbtList::from(
140                self.messages
141                    .iter()
142                    .map(TextComponent::to_codec_nbt)
143                    .collect::<Vec<_>>(),
144            ),
145        );
146
147        // Save color
148        nbt.insert("color", self.color.serialized_name());
149
150        // Save glow
151        nbt.insert("has_glowing_text", i8::from(self.has_glowing_text));
152    }
153}

155/// Sign block entity.
156///
157/// Stores text on both front and back sides of the sign.
158pub struct SignBlockEntity {
159    base: BlockEntityBase,
160    sign: SyncMutex<SignState>,
161}
162
163struct SignState {
164    /// Text on the front side.
165    front_text: SignText,
166    /// Text on the back side.
167    back_text: SignText,
168    /// Whether the sign is waxed (prevents editing).
169    is_waxed: bool,
170    /// UUID of the player currently allowed to edit this sign.
171    /// Used to prevent multiple players from editing simultaneously.
172    player_who_may_edit: Option<Uuid>,
173}
174
175// SAFETY: This key identifies Steel's shared sign implementation for both sign
176// registry entries, rather than either registry entry itself.
177unsafe impl DowncastType for SignBlockEntity {
178    const TYPE_KEY: DowncastTypeKey = DowncastTypeKey::new("steel:block_entity/sign");
179}
180
181impl SignBlockEntity {
182    /// Creates a new sign block entity.
183    #[must_use]
184    pub fn new(level: Weak<World>, pos: BlockPos, state: BlockStateId) -> Self {
185        Self::with_type(level, &vanilla_block_entity_types::SIGN, pos, state)
186    }
187
188    /// Creates a new hanging sign block entity.
189    #[must_use]
190    pub fn new_hanging(level: Weak<World>, pos: BlockPos, state: BlockStateId) -> Self {
191        Self::with_type(level, &vanilla_block_entity_types::HANGING_SIGN, pos, state)
192    }
193
194    /// Creates a sign block entity with a specific type.
195    #[must_use]
196    pub fn with_type(
197        level: Weak<World>,
198        block_entity_type: BlockEntityTypeRef,
199        pos: BlockPos,
200        state: BlockStateId,
201    ) -> Self {
202        Self {
203            base: BlockEntityBase::new(block_entity_type, level, pos, state),
204            sign: SyncMutex::new(SignState {
205                front_text: SignText::new(),
206                back_text: SignText::new(),
207                is_waxed: false,
208                player_who_may_edit: None,
209            }),
210        }
211    }
212
213    /// Gets the UUID of the player currently allowed to edit this sign.
214    #[must_use]
215    pub fn get_player_who_may_edit(&self) -> Option<Uuid> {
216        self.sign.lock().player_who_may_edit
217    }
218
219    /// Sets the player allowed to edit this sign.
220    pub fn set_player_who_may_edit(&self, player: Option<Uuid>) {
221        self.sign.lock().player_who_may_edit = player;
222    }
223
224    /// Checks if another player (not the given one) is currently editing this sign.
225    #[must_use]
226    pub fn is_other_player_editing(&self, player_uuid: Uuid) -> bool {
227        self.sign
228            .lock()
229            .player_who_may_edit
230            .is_some_and(|editor| editor != player_uuid)
231    }
232
233    /// Gets the text for a side.
234    #[must_use]
235    pub fn get_text(&self, slot: SignTextSlot) -> SignText {
236        let sign = self.sign.lock();
237        if slot.is_front() {
238            sign.front_text.clone()
239        } else {
240            sign.back_text.clone()
241        }
242    }
243
244    /// Returns whether this sign is waxed.
245    #[must_use]
246    pub fn is_waxed(&self) -> bool {
247        self.sign.lock().is_waxed
248    }
249
250    /// Makes this sign waxed, returning whether its state changed.
251    pub fn wax(&self) -> bool {
252        let mut sign = self.sign.lock();
253        if sign.is_waxed {
254            return false;
255        }
256        sign.is_waxed = true;
257        true
258    }
259    /// Sets whether a side's text glows, returning whether its state changed or not :3.
260    ///
261    /// Mirrors vanilla `SignBlockEntity.updateText`: returns false when the side
262    /// already has the requested glow state, so callers can skip consuming the item.
263    pub fn set_glowing(&self, slot: SignTextSlot, glowing: bool) -> bool {
264        let mut sign = self.sign.lock();
265        let text = if slot.is_front() {
266            &mut sign.front_text
267        } else {
268            &mut sign.back_text
269        };
270        if text.has_glowing_text == glowing {
271            return false;
272        }
273        text.has_glowing_text = glowing;
274        true
275    }
276
277    /// Sets the text for a side.
278    pub fn set_text(&self, text: SignText, slot: SignTextSlot) {
279        let mut sign = self.sign.lock();
280        if slot.is_front() {
281            sign.front_text = text;
282        } else {
283            sign.back_text = text;
284        }
285    }
286}

288impl BlockEntity for SignBlockEntity {
289    fn base(&self) -> &BlockEntityBase {
290        &self.base
291    }
292
293    fn load_additional(&self, nbt: &BorrowedNbtCompound<'_>) {
294        // Convert to NbtCompound view for accessing methods
295        let nbt_view: BorrowedNbtCompoundView<'_, '_> = nbt.into();
296        let mut sign = self.sign.lock();
297
298        // Load front text
299        if let Some(front_nbt) = nbt_view.compound("front_text") {
300            sign.front_text.load(front_nbt);
301        }
302
303        // Load back text
304        if let Some(back_nbt) = nbt_view.compound("back_text") {
305            sign.back_text.load(back_nbt);
306        }
307
308        // Load waxed state
309        if let Some(waxed) = nbt_view.byte("is_waxed") {
310            sign.is_waxed = waxed != 0;
311        }
312    }
313
314    fn apply_components_from_item(&self, item: &ItemStack) {
315        let mut sign = self.sign.lock();
316        sign.front_text = SignText::from_component(item.get(SIGN_TEXT_FRONT));
317        sign.back_text = SignText::from_component(item.get(SIGN_TEXT_BACK));
318        sign.is_waxed = item.has(WAXED);
319    }
320
321    fn save_additional(&self, nbt: &mut NbtCompound) {
322        let sign = self.sign.lock();
323        // Save front text
324        let mut front_nbt = NbtCompound::new();
325        sign.front_text.save(&mut front_nbt);
326        nbt.insert("front_text", front_nbt);
327
328        // Save back text
329        let mut back_nbt = NbtCompound::new();
330        sign.back_text.save(&mut back_nbt);
331        nbt.insert("back_text", back_nbt);
332
333        // Save waxed state
334        nbt.insert("is_waxed", i8::from(sign.is_waxed));
335    }
336
337    fn get_update_tag(&self) -> Option<NbtCompound> {
338        // Send full sign data to client
339        let mut nbt = NbtCompound::new();
340        self.save_additional(&mut nbt);
341        Some(nbt)
342    }
343
344    fn tick(&self, world: &Arc<World>) {
345        // Clear the edit lock if the editing player is too far away or gone
346        let editor_uuid = self.sign.lock().player_who_may_edit;
347        let Some(editor_uuid) = editor_uuid else {
348            return;
349        };
350        let should_clear = world
351            .players
352            .get_by_uuid(&editor_uuid)
353            .is_none_or(|player| {
354                let pos = self.get_block_pos();
355                let player_pos = player.position();
356                let dx = player_pos.x - f64::from(pos.0.x) - 0.5;
357                let dy = player_pos.y - f64::from(pos.0.y) - 0.5;
358                let dz = player_pos.z - f64::from(pos.0.z) - 0.5;
359                let distance_sq = dx * dx + dy * dy + dz * dz;
360                distance_sq > MAX_EDIT_DISTANCE * MAX_EDIT_DISTANCE
361            });
362
363        if should_clear {
364            let mut sign = self.sign.lock();
365            if sign.player_who_may_edit == Some(editor_uuid) {
366                sign.player_who_may_edit = None;
367            }
368        }
369    }
370}

389    #[test]
390    fn plain_sign_lines_save_as_a_string_list() {
391        let mut text = SignText::new();
392        text.messages = array::from_fn(|index| TextComponent::plain(index.to_string()));
393
394        let mut nbt = NbtCompound::new();
395        text.save(&mut nbt);
396
397        assert_eq!(
398            nbt.get("messages"),
399            Some(&NbtTag::List(NbtList::String(vec![
400                "0".into(),
401                "1".into(),
402                "2".into(),
403                "3".into(),
404            ])))
405        );
406    }

408    #[test]
409    fn mixed_sign_lines_round_trip_through_the_component_codec() {
410        let mut expected = SignText::new();
411        expected.messages[0] = TextComponent::plain("plain");
412        expected.messages[1] = TextComponent::plain("styled").bold(true);
413
414        let mut nbt = NbtCompound::new();
415        expected.save(&mut nbt);
416        assert!(matches!(
417            nbt.get("messages"),
418            Some(NbtTag::List(NbtList::Compound(_)))
419        ));
420
421        let mut bytes = Vec::new();
422        NbtTag::Compound(nbt).write(&mut bytes);
423        let borrowed = read_tag(&mut Cursor::new(bytes.as_slice()))
424            .expect("saved sign text should be valid NBT");
425        let borrowed_tag = borrowed.as_tag();
426        let compound = borrowed_tag
427            .compound()
428            .expect("saved sign text should be a compound");
429
430        let mut decoded = SignText::new();
431        decoded.load(compound);
432
433        assert_eq!(decoded.messages, expected.messages);
434        assert_eq!(decoded.color, expected.color);
435        assert_eq!(decoded.has_glowing_text, expected.has_glowing_text);
436    }

448    #[test]
449    fn placed_sign_takes_text_and_wax_from_item_components() {
450        init_vanilla_registry();
451        let world_fixture = fresh_test_world("sign_item_components");
452        let world = &world_fixture.world;
453        let pos = BlockPos::new(0, 64, 0);
454        let sign = SignBlockEntity::new(
455            Arc::downgrade(world),
456            pos,
457            vanilla_blocks::OAK_SIGN.default_state(),
458        );
459
460        let mut messages = array::from_fn(|_| TextComponent::new());
461        messages[0] = TextComponent::plain("front");
462        let mut item = ItemStack::new(&vanilla_items::OAK_SIGN);
463        item.set(
464            SIGN_TEXT_FRONT,
465            SignTextComponent::new(messages.clone(), None, DyeColor::Red, true),
466        );
467        item.set(WAXED, ());
468
469        sign.apply_components_from_item(&item);
470
471        let front = sign.get_text(SignTextSlot::Front);
472        assert_eq!(front.messages, messages);
473        assert_eq!(front.color, DyeColor::Red);
474        assert!(front.has_glowing_text);
475        assert!(!sign.get_text(SignTextSlot::Back).has_message());
476        assert!(sign.is_waxed());
477    }
