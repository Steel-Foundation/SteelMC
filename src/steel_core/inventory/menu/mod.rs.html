45/// A menu opened by a player: the shared click machinery plus one
46/// [`MenuKind`].
47///
48/// The single concrete menu type. It owns the [`MenuBehavior`], the
49/// `MenuLayout`, and a boxed [`MenuKind`]. Click handlers are inherent methods.
50pub struct Menu {
51    behavior: MenuBehavior,
52    layout: MenuLayout,
53    kind: Box<dyn MenuKind>,
54    overrides_player_slots: bool,
55}

58    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
59        f.debug_struct("Menu")
60            .field("behavior", &self.behavior)
61            .field("kind", &self.kind.downcast_type_key())
62            .finish_non_exhaustive()
63    }
64}

151    /// Called when the menu is closed. Hands the carried item and input sections
152    /// back to the player, then runs the kind's cleanup. Items drop into the
153    /// world if the player can't take them (see
154    /// [`Player::returns_menu_items_to_inventory`]).
155    pub fn removed(&mut self, player: &Player) {
156        let return_to_inventory = player.returns_menu_items_to_inventory();
157
158        let carried = mem::take(self.behavior.carried_mut());
159        if !carried.is_empty() {
160            if return_to_inventory {
161                player.add_item_or_drop(carried, Prediction::ServerOnly);
162            } else {
163                let _ = player.drop_item(carried, false, Prediction::ServerOnly);
164            }
165        }
166        self.layout
167            .return_drained_items(&self.behavior, player, return_to_inventory);
168
169        let Self { behavior, kind, .. } = self;
170        kind.removed(behavior, player);
171    }

180    /// Clears or counts crafting-grid items in the base inventory menu,
181    /// returning the number cleared or counted. Returns 0 for any other menu.
182    pub(crate) fn clear_or_count_crafting_items(
183        &mut self,
184        predicate: &dyn Fn(&ItemStack) -> bool,
185        amount_to_remove: i32,
186        counting_only: bool,
187    ) -> i32 {
188        let Some(kind) = self.kind.downcast_ref::<InventoryKind>() else {
189            return 0;
190        };
191        let crafting_id = kind.crafting_id();
192        let mut guard = self.behavior.lock_all_containers();
193        let Some(crafting) = guard.get_typed_mut::<CraftingContainer>(crafting_id) else {
194            return 0;
195        };
196
197        crafting.clear_or_count_matching_items(predicate, amount_to_remove, counting_only)
198    }

239    /// Shift-click (quick move) for a slot: the kind's override if any, else the
240    /// declarative route table. Returns the item originally in the slot, or
241    /// empty if nothing moved.
242    fn quick_move_stack(
243        &mut self,
244        guard: &mut ContainerLockGuard,
245        slot_index: usize,
246        player: &Player,
247    ) -> ItemStack {
248        let Self {
249            behavior,
250            layout,
251            kind,
252            ..
253        } = self;
254        if let Some(result) = kind.quick_move(behavior, guard, slot_index, player) {
255            result
256        } else {
257            layout.quick_move(behavior, guard, slot_index, player)
258        }
259    }

261    /// Handles a click action in this menu. Packet clicks are validated via
262    /// [`Click::parse`]; invalid programmatically constructed clicks are ignored.
263    ///
264    /// TODO: Add `tryItemClickBehaviorOverride` for bundle item support.
265    pub fn clicked(&mut self, click: Click, player: &Player) {
266        if !click.is_valid_for(self.behavior().slot_count()) {
267            log::debug!(
268                "Ignoring programmatic container click that violates parsed-click invariants: \
269                 {click:?}"
270            );
271            return;
272        }
273
274        let armor_before = self.armor_items();
275        let has_infinite_materials = player.game_mode() == GameType::Creative;
276        if let Click::QuickCraft(action) = click {
277            let outcome = {
278                let mut guard = self.behavior().lock_all_containers();
279                let Self { behavior, kind, .. } = self;
280                kind.on_drag(behavior, &mut guard, action, player)
281            };
282            if outcome == ClickOutcome::Consume {
283                self.behavior_mut().reset_quick_craft();
284            } else {
285                let Self { behavior, kind, .. } = self;
286                behavior.do_quick_craft(action, has_infinite_materials, player, &|slot| {
287                    kind.can_drag_to(slot)
288                });
289            }
290        } else {
291            // Any non-quickcraft click resets an in-progress quickcraft.
292            if self.behavior().quickcraft().is_some() {
293                self.behavior_mut().reset_quick_craft();
294            }
295
296            // Menu-defined click hook. A consumed click skips default handling.
297            // The guard is dropped before the default arms re-lock the same containers.
298            let outcome = {
299                let mut guard = self.behavior().lock_all_containers();
300                let Self { behavior, kind, .. } = self;
301                kind.on_slot_clicked(behavior, &mut guard, click, player)
302            };
303
304            if outcome == ClickOutcome::Fallthrough {
305                match click {
306                    Click::Pickup { slot, button } => {
307                        self.behavior_mut().do_pickup(slot, button, player);
308                    }
309                    Click::DropCarried { button } => {
310                        self.behavior_mut().drop_carried(button, player);
311                    }
312                    Click::QuickMove { slot } => {
313                        self.do_quick_move(slot, player);
314                    }
315                    Click::Swap { slot, with } => {
316                        self.do_swap(slot, with, player);
317                    }
318                    Click::Clone { slot } => {
319                        self.behavior_mut().do_clone(slot, has_infinite_materials);
320                    }
321                    Click::Throw { slot, whole_stack } => {
322                        self.behavior_mut().do_throw(slot, whole_stack, player);
323                    }
324                    Click::PickupAll { slot, direction } => {
325                        self.do_pickup_all(slot, direction, player);
326                    }
327                    Click::QuickCraft(_) => unreachable!(),
328                }
329            }
330        }
331        // Recompute recipe-driven slots after the click. A QuickCraft drag has
332        // no slot on its end phase, so recompute on any non-empty menu.
333        let should_recompute = match click {
334            Click::DropCarried { .. } => false,
335            Click::QuickCraft(_) => !self.behavior().slots().is_empty(),
336            _ => true,
337        };
338        if should_recompute {
339            let mut guard = self.behavior().lock_all_containers();
340            self.slots_changed(&mut guard, player);
341        }
342        self.notify_armor_equipment_changes(player, armor_before);
343    }

345    /// Sets a slot on behalf of a direct player action, such as a creative mode
346    /// slot packet. Reports the resulting equipment changes like [`Self::clicked`].
347    pub(crate) fn set_slot_by_player(
348        &mut self,
349        slot_index: usize,
350        stack: ItemStack,
351        player: &Player,
352    ) {
353        let armor_before = self.armor_items();
354        {
355            let mut guard = self.behavior().lock_all_containers();
356            let Some(slot) = self.behavior().slots().get(slot_index) else {
357                return;
358            };
359            let previous = slot.get_item(&guard).clone();
360            slot.set_by_player(&mut guard, stack, &previous);
361        }
362        self.notify_armor_equipment_changes(player, armor_before);
363    }

365    fn armor_items(&self) -> Vec<(EquipmentSlot, ItemStack)> {
366        let guard = self.behavior().lock_all_containers();
367        self.behavior()
368            .slots()
369            .iter()
370            .filter_map(|slot| {
371                let armor_slot = slot.downcast_ref::<ArmorSlot>()?;
372                Some((
373                    armor_slot.equipment_slot(),
374                    armor_slot.get_item(&guard).clone(),
375                ))
376            })
377            .collect()
378    }

380    fn notify_armor_equipment_changes(
381        &self,
382        player: &Player,
383        armor_before: Vec<(EquipmentSlot, ItemStack)>,
384    ) {
385        for ((slot, previous), (_, current)) in armor_before.into_iter().zip(self.armor_items()) {
386            player.on_equip_item(slot, &previous, &current);
387        }
388    }

390    /// Handles quick move (shift-click).
391    fn do_quick_move(&mut self, slot_index: usize, player: &Player) {
392        let mut guard = self.behavior().lock_all_containers();
393
394        if !self.behavior().slots()[slot_index].may_pickup(&guard, player) {
395            return;
396        }
397
398        let initial_item = self.behavior().slots()[slot_index].get_item(&guard).clone();
399        if initial_item.is_empty() {
400            return;
401        }
402
403        // Loop while the slot still holds the same item type.
404        let mut result = self.quick_move_stack(&mut guard, slot_index, player);
405
406        while !result.is_empty() {
407            let current_item = self.behavior().slots()[slot_index].get_item(&guard).clone();
408            if !ItemStack::is_same_item(&current_item, &result) {
409                break;
410            }
411            result = self.quick_move_stack(&mut guard, slot_index, player);
412        }
413    }

415    /// Handles swap (number keys for a hotbar slot, or swap-hands for the
416    /// offhand).
417    fn do_swap(&mut self, slot_index: usize, with: SwapTarget, player: &Player) {
418        let player_inventory = ContainerRef::from(Arc::clone(&player.inventory));
419        let player_inv_id = ContainerId::from_arc(&player.inventory);
420        let mut guard = self.behavior().lock_all_containers_with(player_inventory);
421
422        let behavior = self.behavior();
423        let target_slot = &behavior.slots()[slot_index];
424        let inventory_slot = with.inventory_slot();
425
426        let target_item = target_slot.get_item(&guard).clone();
427        let Some(inventory) = guard.get(player_inv_id) else {
428            unreachable!("the explicitly locked player inventory must be present");
429        };
430        let source_item = inventory.get_item(inventory_slot).clone();
431
432        if source_item.is_empty() && target_item.is_empty() {
433            return;
434        }
435
436        if source_item.is_empty() {
437            // Move target -> inventory.
438            if target_slot.may_pickup(&guard, player) {
439                let Some(inventory) = guard.get_mut(player_inv_id) else {
440                    unreachable!("the explicitly locked player inventory must be present");
441                };
442                inventory.set_item(inventory_slot, target_item.clone());
443                target_slot.set_by_player(&mut guard, ItemStack::empty(), &target_item);
444                if let Some(remainder) = target_slot.on_take(&mut guard, &target_item, player) {
445                    player.add_item_or_drop_with_guard(
446                        &mut guard,
447                        remainder,
448                        Prediction::Predicted,
449                    );
450                }
451            }
452        } else if target_item.is_empty() {
453            // Move inventory -> target.
454            if target_slot.may_place(&source_item) {
455                let max_size = target_slot.get_max_stack_size_for_item(&guard, &source_item);
456                if source_item.count > max_size {
457                    let Some(inv) = guard.get_mut(player_inv_id) else {
458                        unreachable!("the explicitly locked player inventory must be present");
459                    };
460                    let to_place = inv.get_item_mut(inventory_slot).split(max_size);
461                    target_slot.set_by_player(&mut guard, to_place, &ItemStack::empty());
462                } else {
463                    let Some(inventory) = guard.get_mut(player_inv_id) else {
464                        unreachable!("the explicitly locked player inventory must be present");
465                    };
466                    inventory.set_item(inventory_slot, ItemStack::empty());
467                    target_slot.set_by_player(&mut guard, source_item, &ItemStack::empty());
468                }
469            }
470        } else {
471            // Swap target <-> inventory.
472            if target_slot.may_pickup(&guard, player) && target_slot.may_place(&source_item) {
473                let max_size = target_slot.get_max_stack_size_for_item(&guard, &source_item);
474                if source_item.count > max_size {
475                    // Source too big: place a partial stack, return target to inventory.
476                    let Some(inv) = guard.get_mut(player_inv_id) else {
477                        unreachable!("the explicitly locked player inventory must be present");
478                    };
479                    let to_place = inv.get_item_mut(inventory_slot).split(max_size);
480                    target_slot.set_by_player(&mut guard, to_place, &target_item);
481                    if let Some(remainder) = target_slot.on_take(&mut guard, &target_item, player) {
482                        player.add_item_or_drop_with_guard(
483                            &mut guard,
484                            remainder,
485                            Prediction::Predicted,
486                        );
487                    }
488                    let mut displaced = target_item;
489                    let Some(inventory) = guard.get_mut(player_inv_id) else {
490                        unreachable!("the explicitly locked player inventory must be present");
491                    };
492                    let added = inventory.add(&mut displaced);
493                    // Vanilla's Inventory::add consumes uninserted stacks in creative mode.
494                    if !added && !player.has_infinite_materials() {
495                        let _ = guard.run_unlocked(|| {
496                            player.drop_item(displaced, true, Prediction::Predicted)
497                        });
498                    }
499                } else {
500                    let Some(inventory) = guard.get_mut(player_inv_id) else {
501                        unreachable!("the explicitly locked player inventory must be present");
502                    };
503                    inventory.set_item(inventory_slot, target_item.clone());
504                    target_slot.set_by_player(&mut guard, source_item, &target_item);
505                    if let Some(remainder) = target_slot.on_take(&mut guard, &target_item, player) {
506                        player.add_item_or_drop_with_guard(
507                            &mut guard,
508                            remainder,
509                            Prediction::Predicted,
510                        );
511                    }
512                }
513            }
514        }
515    }

517    /// Handles pickup all (double-click): collects matching items from all slots
518    /// into the carried stack.
519    fn do_pickup_all(&mut self, slot_index: usize, direction: FillDirection, player: &Player) {
520        let mut guard = self.behavior().lock_all_containers();
521
522        let behavior = self.behavior();
523        let slot = &behavior.slots()[slot_index];
524        let slot_has_item = !slot.get_item(&guard).is_empty();
525        let slot_may_pickup = slot.may_pickup(&guard, player);
526
527        if behavior.carried().is_empty() || (slot_has_item && slot_may_pickup) {
528            return;
529        }
530
531        let max_stack = behavior.carried().max_stack_size();
532        let carried_item = behavior.carried().clone();
533        let slot_count = behavior.slots().len();
534
535        let (start, step): (i32, i32) = match direction {
536            FillDirection::Forward => (0, 1),
537            FillDirection::Backward => (slot_count as i32 - 1, -1),
538        };
539
540        // First pass collects non-full stacks, second pass the full ones.
541        for pass in 0..2 {
542            let mut i = start;
543            while i >= 0 && i < slot_count as i32 && self.behavior().carried().count < max_stack {
544                let target_slot = &self.behavior().slots()[i as usize];
545                let target_item = target_slot.get_item(&guard).clone();
546
547                if !target_item.is_empty()
548                    && can_item_quick_replace(&target_item, &carried_item, true)
549                    && target_slot.may_pickup(&guard, player)
550                    && self.can_take_item_for_pick_all(&carried_item, i as usize)
551                {
552                    // First pass skips full stacks, second pass includes them.
553                    if pass != 0 || target_item.count != target_item.max_stack_size() {
554                        let can_take = max_stack - self.behavior().carried().count;
555                        let to_take = target_item.count.min(can_take);
556                        let removed = target_slot.safe_take(&mut guard, to_take, can_take, player);
557                        self.behavior_mut()
558                            .carried_mut()
559                            .grow(removed.count.min(can_take));
560                    }
561                }
562
563                i += step;
564            }
565        }
566    }
