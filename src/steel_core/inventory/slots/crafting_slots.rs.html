17/// A Recipe Handler for Crafting Recipes
18#[derive(Clone)]
19pub struct CraftingHandler {
20    crafting_container: Shared<CraftingContainer>,
21    result_container: Shared<ResultContainer>,
22    grid_size: usize,
23}
24
25impl CraftingHandler {
26    /// Creates a new Crafting Recipe Handler
27    pub const fn new(
28        crafting_container: Shared<CraftingContainer>,
29        result_container: Shared<ResultContainer>,
30        grid_size: usize,
31    ) -> Self {
32        Self {
33            crafting_container,
34            result_container,
35            grid_size,
36        }
37    }
38
39    /// The `ContainerId` of the crafting container
40    #[must_use]
41    pub fn crafting_id(&self) -> ContainerId {
42        ContainerId::from_arc(&self.crafting_container)
43    }
44
45    /// A shared handle to the crafting container.
46    #[must_use]
47    pub fn crafting_container(&self) -> Shared<CraftingContainer> {
48        Arc::clone(&self.crafting_container)
49    }
50
51    /// The `ContainerId` of the result container
52    #[must_use]
53    pub fn result_id(&self) -> ContainerId {
54        ContainerId::from_arc(&self.result_container)
55    }
56}
57
58impl ResultHandler for CraftingHandler {
59    fn result_container(&self) -> ContainerRef {
60        ContainerRef::from(Arc::clone(&self.result_container))
61    }
62
63    fn dependencies(&self) -> Vec<ContainerRef> {
64        vec![ContainerRef::from(Arc::clone(&self.crafting_container))]
65    }
66
67    fn update_result(&self, guard: &mut ContainerLockGuard) {
68        // TODO: Enforce limited crafting and retain the recipe once player recipe books exist.
69        let crafting = guard
70            .get_typed::<CraftingContainer>(self.crafting_id())
71            .expect("crafting container not locked");
72
73        let result_stack =
74            recipe_manager::assemble_for_container(crafting).unwrap_or_else(ItemStack::empty);
75
76        let result_container = guard
77            .get_typed_mut::<ResultContainer>(self.result_id())
78            .expect("result container not locked");
79        result_container.set_item(0, result_stack);
80        result_container.set_changed();
81    }
82
83    fn on_result_taken(
84        &self,
85        guard: &mut ContainerLockGuard,
86        player: &Player,
87    ) -> Option<ItemStack> {
88        // TODO: Unlock the recipe and trigger RECIPE_CRAFTED once their foundations exist.
89        let mut remainder_overflow: Vec<ItemStack> = Vec::new();
90
91        let remainders_and_positioned = {
92            let crafting = guard
93                .get_typed::<CraftingContainer>(self.crafting_id())
94                .expect("crafting container not locked");
95            recipe_manager::get_remaining_items(crafting)
96        };
97
98        let Some((remainders, positioned)) = remainders_and_positioned else {
99            guard
100                .get_typed_mut::<ResultContainer>(self.result_id())
101                .expect("result container not locked")
102                .set_item(0, ItemStack::empty());
103            return None;
104        };
105
106        {
107            let crafting = guard
108                .get_typed_mut::<CraftingContainer>(self.crafting_id())
109                .expect("crafting container not locked");
110
111            let input = &positioned.input;
112
113            for y in 0..input.height {
114                for x in 0..input.width {
115                    let grid_slot = positioned.to_grid_slot(x, y, self.grid_size);
116                    let remainder_idx = x + y * input.width;
117                    let replacement = if remainder_idx < remainders.len() {
118                        remainders[remainder_idx].clone()
119                    } else {
120                        ItemStack::empty()
121                    };
122
123                    {
124                        let item = crafting.get_item_mut(grid_slot);
125                        if !item.is_empty() {
126                            item.shrink_one();
127                        }
128                    }
129
130                    if !replacement.is_empty() {
131                        let current_item = crafting.get_item(grid_slot).clone();
132
133                        if current_item.is_empty() {
134                            crafting.set_item(grid_slot, replacement);
135                        } else if ItemStack::is_same_item_same_components(
136                            &current_item,
137                            &replacement,
138                        ) {
139                            crafting.get_item_mut(grid_slot).grow(replacement.count());
140                        } else {
141                            remainder_overflow.push(replacement);
142                        }
143                    }
144                }
145            }
146
147            crafting.set_changed();
148        }
149
150        self.update_result(guard);
151
152        for remainder in remainder_overflow {
153            player.add_item_or_drop_with_guard(guard, remainder);
154        }
155
156        None
157    }
158
159    fn is_result_valid(&self, guard: &ContainerLockGuard, _player: &Player) -> bool {
160        let Some(result) = guard.get(self.result_id()) else {
161            return false;
162        };
163        let result_item = result.get_item(0);
164        if result_item.is_empty() {
165            return false;
166        }
167
168        let Some(crafting) = guard.get_typed::<CraftingContainer>(self.crafting_id()) else {
169            return false;
170        };
171
172        let Some(assembled) = recipe_manager::assemble_for_container(crafting) else {
173            return false;
174        };
175
176        ItemStack::matches(result_item, &assembled)
177    }
