36/// Normal single/double chest behavior. Sitting-cat blocking and piglin anger
37/// are explicitly deferred until their entity foundations exist.
38#[block_behavior]
39pub struct ChestBlock {

41    #[json_arg(sound_events)]

47/// A chest's inventories in vanilla slot order, retaining their real owners.
48pub struct ChestContainer {
49    pub(crate) owners: SmallVec<[SharedBlockEntity; 2]>,
50    /// Right half first for doubles; each half remains independently lockable.

73    /// Direction from a chest half to its partner.
74    #[must_use]
75    pub fn connected_direction(state: BlockStateId) -> Direction {
76        let facing = state.get_value(FACING);
77        if state.get_value(TYPE) == ChestType::Left {
78            facing.rotate_y_clockwise()
79        } else {
80            facing.rotate_y_counter_clockwise()
81        }
82    }
83
84    fn candidate_partner_facing(
85        &self,
86        world: &dyn LevelReader,
87        pos: BlockPos,
88        direction: Direction,
89    ) -> Option<Direction> {
90        let state = world.get_block_state(pos.relative(direction));
91        (state.get_block() == self.block && state.get_value(TYPE) == ChestType::Single)
92            .then(|| state.get_value(FACING))
93    }
94
95    fn is_blocked(world: &dyn LevelReader, pos: BlockPos) -> bool {
96        let above = pos.above();
97        is_redstone_conductor(world, world.get_block_state(above), above)
98    }

100    /// Combines compatible halves; automation can request ignoring lid obstruction.
101    #[must_use]
102    pub fn get_container(
103        state: BlockStateId,
104        world: &dyn LevelReader,
105        pos: BlockPos,
106        ignore_blocked: bool,
107    ) -> Option<ChestContainer> {
108        let first = world.get_block_entity(pos)?;
109        if first.get_type() != &vanilla_block_entity_types::CHEST
110            || (!ignore_blocked && Self::is_blocked(world, pos))
111        {
112            return None;
113        }
114        let chest_type = state.get_value(TYPE);
115        let mut owners: SmallVec<[SharedBlockEntity; 2]> = smallvec![first];
116        if chest_type != ChestType::Single {
117            let neighbor_pos = pos.relative(Self::connected_direction(state));
118            let neighbor = world.get_block_state(neighbor_pos);
119            if neighbor.get_block() == state.get_block()
120                && neighbor.get_value(TYPE) != ChestType::Single
121                && neighbor.get_value(TYPE) != chest_type
122                && neighbor.get_value(FACING) == state.get_value(FACING)
123            {
124                if !ignore_blocked && Self::is_blocked(world, neighbor_pos) {
125                    return None;
126                }
127                if let Some(other) = world.get_block_entity(neighbor_pos)
128                    && other.get_type() == &vanilla_block_entity_types::CHEST
129                {
130                    if chest_type == ChestType::Right {
131                        owners.push(other);
132                    } else {
133                        owners.insert(0, other);
134                    }
135                }
136            }
137        }
138        let containers = owners
139            .iter()
140            .map(|owner| owner.container_ref())
141            .collect::<Option<_>>()?;
142        Some(ChestContainer { owners, containers })
143    }
144
145    fn display_name(chest: &ChestContainer) -> TextComponent {
146        for owner in &chest.owners {
147            if let Some(name) = owner.as_nameable().and_then(Nameable::custom_name) {
148                return name;
149            }
150        }
151        if chest.owners.len() == 2 {
152            translations::CONTAINER_CHEST_DOUBLE.msg().component()
153        } else {
154            translations::CONTAINER_CHEST.msg().component()
155        }
156    }
157}

164    fn get_state_for_placement(&self, context: &BlockPlaceContext<'_>) -> Option<BlockStateId> {
165        let pos = context.place_pos();
166        let world = context.world.as_ref();
167        let mut facing = context.horizontal_direction().opposite();
168        let mut chest_type = ChestType::Single;
169        let clicked = context.clicked_face();
170        let secondary = context.is_secondary_use_active();
171        if clicked.get_axis().is_horizontal()
172            && secondary
173            && let Some(neighbor_facing) =
174                self.candidate_partner_facing(world, pos, clicked.opposite())
175            && neighbor_facing.get_axis() != clicked.get_axis()
176        {
177            facing = neighbor_facing;
178            chest_type = if facing.rotate_y_counter_clockwise() == clicked.opposite() {
179                ChestType::Right
180            } else {
181                ChestType::Left
182            };
183        }
184        if chest_type == ChestType::Single && !secondary {
185            if self.candidate_partner_facing(world, pos, facing.rotate_y_clockwise())
186                == Some(facing)
187            {
188                chest_type = ChestType::Left;
189            } else if self.candidate_partner_facing(world, pos, facing.rotate_y_counter_clockwise())
190                == Some(facing)
191            {
192                chest_type = ChestType::Right;
193            }
194        }
195        Some(
196            self.block
197                .default_state()
198                .set_value(FACING, facing)
199                .set_value(TYPE, chest_type)
200                .set_value(WATERLOGGED, context.is_water_source()),
201        )
202    }

204    fn update_shape(
205        &self,
206        state: BlockStateId,
207        world: &dyn ScheduledTickAccess,
208        pos: BlockPos,
209        direction: Direction,
210        _neighbor_pos: BlockPos,
211        neighbor: BlockStateId,
212    ) -> BlockStateId {
213        schedule_water_tick_if_waterlogged(state, world, pos);
214        if neighbor.get_block() == self.block && direction.get_axis().is_horizontal() {
215            let neighbor_type = neighbor.get_value(TYPE);
216            if state.get_value(TYPE) == ChestType::Single
217                && neighbor_type != ChestType::Single
218                && state.get_value(FACING) == neighbor.get_value(FACING)
219                && Self::connected_direction(neighbor) == direction.opposite()
220            {
221                return state.set_value(
222                    TYPE,
223                    if neighbor_type == ChestType::Left {
224                        ChestType::Right
225                    } else {
226                        ChestType::Left
227                    },
228                );
229            }
230        } else if Self::connected_direction(state) == direction {
231            return state.set_value(TYPE, ChestType::Single);
232        }
233        state
234    }

236    fn use_without_item(
237        &self,
238        state: BlockStateId,
239        world: &Arc<World>,
240        pos: BlockPos,
241        player: &Player,
242        _hit: &BlockHitResult,
243        _inv: &mut InventoryAccess,
244    ) -> InteractionResult {
245        let Some(chest) = Self::get_container(state, world.as_ref(), pos, false) else {
246            return InteractionResult::Success;
247        };
248        let title = Self::display_name(&chest);
249        player.award_custom_stat(&vanilla_custom_stats::OPEN_CHEST);
250        if !chest.owners.iter().all(|owner| {
251            owner
252                .downcast_ref::<ChestBlockEntity>()
253                .is_some_and(|chest| chest.can_open(player))
254        }) {
255            if !player.is_spectator() {
256                player.send_overlay_message(
257                    &translations::CONTAINER_IS_LOCKED
258                        .message([title])
259                        .component(),
260                );
261                let (x, y, z) = pos.get_center();
262                let mut center = DVec3::new(x, y, z);
263                if chest.owners.len() == 2 {
264                    center += DVec3::from(Self::connected_direction(state).offset_vec()) * 0.5;
265                }
266                world.play_sound_at(
267                    &sound_events::BLOCK_CHEST_LOCKED,
268                    SoundSource::Blocks,
269                    center,
270                    1.0,
271                    1.0,
272                    None,
273                );
274            }
275            return InteractionResult::Success;
276        }
277        for owner in &chest.owners {
278            if let Some(randomizable) = owner.as_randomizable_container()
279                && let Err(error) = randomizable.unpack_loot_table(Some(player))
280            {
281                log::error!(
282                    "Could not unpack chest loot at {:?}: {error}",
283                    owner.get_block_pos()
284                );
285                return InteractionResult::Success;
286            }
287        }
288        let inventory = Arc::clone(&player.inventory);
289        player.open_menu(title, move |context| {
290            block_chest(
291                inventory,
292                context.container_id,
293                chest.owners,
294                chest.containers,
295            )
296        });
297        InteractionResult::Success
298    }

318    fn get_analog_output_signal(
319        &self,
320        state: BlockStateId,
321        world: &World,
322        pos: BlockPos,
323        _direction: Direction,
324    ) -> i32 {
325        let Some(chest) = Self::get_container(state, world, pos, false) else {
326            return 0;
327        };
328        let Ok(guard) = ContainerLockGuard::lock_all(&chest.containers) else {
329            // TODO: Preserve vanilla access ordering so loot-generation contention
330            // cannot become a false zero comparator signal. Returning zero here
331            // also hides generation failures; distinguish them from empty contents.
332            return 0;
333        };
334        calculate_redstone_signal_from_containers(
335            chest
336                .containers
337                .iter()
338                .filter_map(|container| guard.get(container.container_id())),
339        )
340    }

352    fn tick(&self, _state: BlockStateId, world: &Arc<World>, pos: BlockPos) {
353        if let Some(entity) = world.get_block_entity(pos)
354            && let Some(chest) = entity.downcast_ref::<ChestBlockEntity>()
355        {
356            chest.recheck_open();
357        }
