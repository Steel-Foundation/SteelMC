45/// Behavior for vanilla door blocks.
46#[block_behavior]
47pub struct DoorBlock {
48    block: BlockRef,
49    #[json_arg(value, json = "type_can_open_by_hand")]
50    can_open_by_hand: bool,
51    #[json_arg(value, json = "type_can_open_by_wind_charge")]
52    can_open_by_wind_charge: bool,
53    #[json_arg(sound_events, json = "type_door_open")]
54    sound_open: SoundEventRef,
55    #[json_arg(sound_events, json = "type_door_close")]
56    sound_close: SoundEventRef,
57}

59const DOOR_HINGE: &EnumProperty<DoorHingeSide> = &BlockStateProperties::DOOR_HINGE;
60const DOUBLE_BLOCK_HALF: &EnumProperty<DoubleBlockHalf> = &BlockStateProperties::DOUBLE_BLOCK_HALF;
61const HORIZONTAL_FACING: &EnumProperty<Direction> = &BlockStateProperties::HORIZONTAL_FACING;
62const OPEN: &BoolProperty = &BlockStateProperties::OPEN;
63const POWERED: &BoolProperty = &BlockStateProperties::POWERED;
64
65impl DoorBlock {
66    const USE_UPDATE_FLAGS: UpdateFlags =
67        UpdateFlags::UPDATE_CLIENTS.union(UpdateFlags::UPDATE_IMMEDIATE);
68
69    /// Creates a new door block behavior.
70    #[must_use]
71    pub const fn new(
72        block: BlockRef,
73        can_open_by_hand: bool,
74        can_open_by_wind_charge: bool,
75        sound_open: SoundEventRef,
76        sound_close: SoundEventRef,
77    ) -> Self {
78        Self {
79            block,
80            can_open_by_hand,
81            can_open_by_wind_charge,
82            sound_open,
83            sound_close,
84        }
85    }
86
87    fn is_door(state: BlockStateId) -> bool {
88        state.try_get_value(DOOR_HINGE).is_some()
89            && state.try_get_value(DOUBLE_BLOCK_HALF).is_some()
90    }
91
92    fn is_lower_door(state: BlockStateId) -> bool {
93        Self::is_door(state) && state.get_value(DOUBLE_BLOCK_HALF) == DoubleBlockHalf::Lower
94    }
95
96    fn hinge_for_placement(context: &BlockPlaceContext<'_>) -> DoorHingeSide {
97        let pos = context.place_pos();
98        let above_pos = pos.above();
99        let place_direction = context.horizontal_direction();
100
101        let left_direction = place_direction.rotate_y_counter_clockwise();
102        let left_pos = left_direction.relative(pos);
103        let left_state = context.world.get_block_state(left_pos);
104        let left_above_pos = left_direction.relative(above_pos);
105        let left_above_state = context.world.get_block_state(left_above_pos);
106
107        let right_direction = place_direction.rotate_y_clockwise();
108        let right_pos = right_direction.relative(pos);
109        let right_state = context.world.get_block_state(right_pos);
110        let right_above_pos = right_direction.relative(above_pos);
111        let right_above_state = context.world.get_block_state(right_above_pos);
112
113        let solid_block_balance = i32::from(shapes::is_offset_shape_full_block(
114            right_state.get_collision_shape_at(right_pos),
115        )) + i32::from(shapes::is_offset_shape_full_block(
116            right_above_state.get_collision_shape_at(right_above_pos),
117        )) - i32::from(shapes::is_offset_shape_full_block(
118            left_state.get_collision_shape_at(left_pos),
119        )) - i32::from(shapes::is_offset_shape_full_block(
120            left_above_state.get_collision_shape_at(left_above_pos),
121        ));
122
123        let door_left = Self::is_lower_door(left_state);
124        let door_right = Self::is_lower_door(right_state);
125
126        if (!door_left || door_right) && solid_block_balance <= 0 {
127            if (!door_right || door_left) && solid_block_balance >= 0 {
128                let (step_x, step_z) = place_direction.offset_xz();
129                let click_x = context.click_location().x - f64::from(pos.x());
130                let click_z = context.click_location().z - f64::from(pos.z());
131
132                if (step_x >= 0 || click_z >= 0.5)
133                    && (step_x <= 0 || click_z <= 0.5)
134                    && (step_z >= 0 || click_x <= 0.5)
135                    && (step_z <= 0 || click_x >= 0.5)
136                {
137                    DoorHingeSide::Left
138                } else {
139                    DoorHingeSide::Right
140                }
141            } else {
142                DoorHingeSide::Left
143            }
144        } else {
145            DoorHingeSide::Right
146        }
147    }
148
149    fn has_correct_tool_for_drops(player: &Player, state: BlockStateId) -> bool {
150        let inv = player.inventory.lock();
151        let main_hand = inv.get_item_in_hand(InteractionHand::MainHand);
152        main_hand.is_correct_tool_for_drops(state)
153            || !state.get_block().config.requires_correct_tool_for_drops
154    }
155
156    fn prevent_drop_from_bottom_part(
157        world: &Arc<World>,
158        pos: BlockPos,
159        state: BlockStateId,
160        player: &Player,
161    ) {
162        if state.get_value(DOUBLE_BLOCK_HALF) != DoubleBlockHalf::Upper {
163            return;
164        }
165
166        let bottom_pos = pos.below();
167        let bottom_state = world.get_block_state(bottom_pos);
168        if bottom_state.get_block() != state.get_block()
169            || bottom_state.get_value(DOUBLE_BLOCK_HALF) != DoubleBlockHalf::Lower
170        {
171            return;
172        }
173
174        let replacement = fluid_state_to_block(bottom_state.get_fluid_state());
175        world.set_block(
176            bottom_pos,
177            replacement,
178            UpdateFlags::UPDATE_ALL | UpdateFlags::UPDATE_SUPPRESS_DROPS,
179        );
180        world.destroy_block_effect(bottom_pos, u32::from(bottom_state.0), Some(player.id()));
181    }
182
183    fn play_sound(&self, world: &Arc<World>, pos: BlockPos, open: bool, exclude: Option<i32>) {
184        let sound = if open {
185            self.sound_open
186        } else {
187            self.sound_close
188        };
189        world.play_block_sound(sound, pos, 1.0, 1.0, exclude);
190    }
191}

193impl BlockBehavior for DoorBlock {
194    fn get_state_for_placement(&self, context: &BlockPlaceContext<'_>) -> Option<BlockStateId> {
195        let pos = context.place_pos();
196        if pos.y() >= context.world.max_y_exclusive() - 1 {
197            return None;
198        }
199        if !context.world.get_block_state(pos.above()).is_replaceable() {
200            return None;
201        }
202
203        let powered = context.world.has_neighbor_signal(pos)
204            || context.world.has_neighbor_signal(pos.above());
205        Some(
206            self.block
207                .default_state()
208                .set_value(HORIZONTAL_FACING, context.horizontal_direction())
209                .set_value(DOOR_HINGE, Self::hinge_for_placement(context))
210                .set_value(POWERED, powered)
211                .set_value(OPEN, powered)
212                .set_value(DOUBLE_BLOCK_HALF, DoubleBlockHalf::Lower),
213        )
214    }
215
216    fn update_shape(
217        &self,
218        state: BlockStateId,
219        world: &dyn ScheduledTickAccess,
220        pos: BlockPos,
221        direction: Direction,
222        _neighbor_pos: BlockPos,
223        neighbor_state: BlockStateId,
224    ) -> BlockStateId {
225        let half = state.get_value(DOUBLE_BLOCK_HALF);
226        if direction.get_axis() == Axis::Y
227            && (half == DoubleBlockHalf::Lower) == (direction == Direction::Up)
228        {
229            if Self::is_door(neighbor_state) && neighbor_state.get_value(DOUBLE_BLOCK_HALF) != half
230            {
231                return neighbor_state.set_value(DOUBLE_BLOCK_HALF, half);
232            }
233            return vanilla_blocks::AIR.default_state();
234        }
235
236        if half == DoubleBlockHalf::Lower
237            && direction == Direction::Down
238            && !self.can_survive(state, world, pos)
239        {
240            return vanilla_blocks::AIR.default_state();
241        }
242
243        state
244    }
245
246    fn can_survive(&self, state: BlockStateId, world: &dyn LevelReader, pos: BlockPos) -> bool {
247        let below_pos = pos.below();
248        let below_state = world.get_block_state(below_pos);
249        if state.get_value(DOUBLE_BLOCK_HALF) == DoubleBlockHalf::Lower {
250            world.is_face_sturdy(below_state, below_pos, Direction::Up)
251        } else {
252            below_state.get_block() == self.block
253        }
254    }
255
256    fn is_pathfindable(&self, state: BlockStateId, computation_type: PathComputationType) -> bool {
257        match computation_type {
258            PathComputationType::Land | PathComputationType::Air => state.get_value(OPEN),
259            PathComputationType::Water => false,
260        }
261    }
262
263    fn is_wooden_door(&self, state: BlockStateId) -> bool {
264        self.can_open_by_hand && Self::is_door(state)
265    }
266
267    fn set_door_open(
268        &self,
269        state: BlockStateId,
270        world: &Arc<World>,
271        pos: BlockPos,
272        source_entity: Option<&dyn Entity>,
273        open: bool,
274    ) -> bool {
275        if !Self::is_door(state) || state.get_value(OPEN) == open {
276            return false;
277        }
278
279        let new_state = state.set_value(OPEN, open);
280        if !world.set_block(pos, new_state, Self::USE_UPDATE_FLAGS) {
281            return false;
282        }
283
284        self.play_sound(world, pos, open, source_entity.map(Entity::id));
285        let event = if open {
286            &vanilla_game_events::BLOCK_OPEN
287        } else {
288            &vanilla_game_events::BLOCK_CLOSE
289        };
290        world.game_event(event, pos, &GameEventContext::new(source_entity, None));
291        true
292    }
293
294    fn set_placed_by(
295        &self,
296        state: BlockStateId,
297        world: &Arc<World>,
298        pos: BlockPos,
299        _source: &PlacementSource<'_>,
300    ) {
301        world.set_block(
302            pos.above(),
303            state.set_value(DOUBLE_BLOCK_HALF, DoubleBlockHalf::Upper),
304            UpdateFlags::UPDATE_ALL,
305        );
306    }
307
308    fn player_will_destroy(
309        &self,
310        state: BlockStateId,
311        world: &Arc<World>,
312        pos: BlockPos,
313        player: &Player,
314    ) -> BlockStateId {
315        if player.has_infinite_materials() || !Self::has_correct_tool_for_drops(player, state) {
316            Self::prevent_drop_from_bottom_part(world, pos, state, player);
317        }
318        state
319    }
320
321    fn use_without_item(
322        &self,
323        state: BlockStateId,
324        world: &Arc<World>,
325        pos: BlockPos,
326        player: &Player,
327        _hit_result: &BlockHitResult,
328        _inv: &mut InventoryAccess,
329    ) -> InteractionResult {
330        if !self.can_open_by_hand {
331            return InteractionResult::Pass;
332        }
333
334        let open = !state.get_value(OPEN);
335        let new_state = state.set_value(OPEN, open);
336        world.set_block(pos, new_state, Self::USE_UPDATE_FLAGS);
337        self.play_sound(world, pos, open, Some(player.id()));
338        let event = if open {
339            &vanilla_game_events::BLOCK_OPEN
340        } else {
341            &vanilla_game_events::BLOCK_CLOSE
342        };
343        world.game_event(event, pos, &GameEventContext::new(Some(player), None));
344        InteractionResult::Success
345    }
346
347    fn on_explosion_hit(
348        &self,
349        state: BlockStateId,
350        world: &Arc<World>,
351        pos: BlockPos,
352        explosion: &dyn Explosion,
353        on_hit: &mut dyn FnMut(ItemStack, BlockPos),
354    ) {
355        if explosion.can_trigger_blocks()
356            && state.get_value(DOUBLE_BLOCK_HALF) == DoubleBlockHalf::Lower
357            && self.can_open_by_wind_charge
358            && !state.get_value(POWERED)
359        {
360            self.set_door_open(state, world, pos, None, !state.get_value(OPEN));
361        }
362
363        self.default_on_explosion_hit(state, world, pos, explosion, on_hit);
364    }
365
366    fn handle_neighbor_changed(
367        &self,
368        state: BlockStateId,
369        world: &Arc<World>,
370        pos: BlockPos,
371        source_block: BlockRef,
372        _moved_by_piston: bool,
373    ) {
374        if source_block == self.block {
375            return;
376        }
377
378        let half = state.get_value(DOUBLE_BLOCK_HALF);
379        let other_half_pos = if half == DoubleBlockHalf::Lower {
380            pos.above()
381        } else {
382            pos.below()
383        };
384        let signal = world.has_neighbor_signal(pos) || world.has_neighbor_signal(other_half_pos);
385        if signal == state.get_value(POWERED) {
386            return;
387        }
388
389        if signal != state.get_value(OPEN) {
390            self.play_sound(world, pos, signal, None);
391            let event = if signal {
392                &vanilla_game_events::BLOCK_OPEN
393            } else {
394                &vanilla_game_events::BLOCK_CLOSE
395            };
396            world.game_event(event, pos, &GameEventContext::default());
397        }
398
399        let new_state = state.set_value(POWERED, signal).set_value(OPEN, signal);
400        world.set_block(pos, new_state, UpdateFlags::UPDATE_CLIENTS);
401    }
402}

567    #[test]
568    fn lower_half_copies_transformed_upper_half_state() {
569        init_vanilla_registry();
570        let behavior = DoorBlock::new(
571            &vanilla_blocks::SPRUCE_DOOR,
572            true,
573            true,
574            &sound_events::BLOCK_WOODEN_DOOR_OPEN,
575            &sound_events::BLOCK_WOODEN_DOOR_CLOSE,
576        );
577        let lower = vanilla_blocks::SPRUCE_DOOR
578            .default_state()
579            .set_value(HORIZONTAL_FACING, Direction::West)
580            .set_value(DOOR_HINGE, DoorHingeSide::Right)
581            .set_value(DOUBLE_BLOCK_HALF, DoubleBlockHalf::Lower)
582            .set_value(OPEN, false)
583            .set_value(POWERED, false);
584        let upper = vanilla_blocks::SPRUCE_DOOR
585            .default_state()
586            .set_value(HORIZONTAL_FACING, Direction::South)
587            .set_value(DOOR_HINGE, DoorHingeSide::Left)
588            .set_value(DOUBLE_BLOCK_HALF, DoubleBlockHalf::Upper)
589            .set_value(OPEN, false)
590            .set_value(POWERED, false);
591        let level = TestLevel::default();
592
593        let updated = behavior.update_shape(
594            lower,
595            &level,
596            BlockPos::ZERO,
597            Direction::Up,
598            BlockPos::ZERO.above(),
599            upper,
600        );
601
602        assert_eq!(updated.get_value(DOUBLE_BLOCK_HALF), DoubleBlockHalf::Lower);
603        assert_eq!(updated.get_value(HORIZONTAL_FACING), Direction::South);
604        assert_eq!(updated.get_value(DOOR_HINGE), DoorHingeSide::Left);
605    }
