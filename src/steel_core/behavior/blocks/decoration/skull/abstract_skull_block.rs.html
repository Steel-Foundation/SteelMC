15const POWERED: &BoolProperty = &BlockStateProperties::POWERED;
16
17/// Shared server behavior for `AbstractSkullBlock`.
18pub(super) trait AbstractSkullBlock: BlockBehavior {
19    #[must_use]
20    fn state_for_placement(&self, context: &BlockPlaceContext<'_>) -> Option<BlockStateId>;
21
22    fn default_state_for_placement(&self, context: &BlockPlaceContext<'_>) -> Option<BlockStateId> {
23        self.state_for_placement(context).map(|state| {
24            state.set_value(
25                POWERED,
26                context.world.has_neighbor_signal(context.place_pos()),
27            )
28        })
29    }
30
31    fn default_handle_neighbor_changed(
32        &self,
33        state: BlockStateId,
34        world: &Arc<World>,
35        pos: BlockPos,
36        _source_block: BlockRef,
37        _moved_by_piston: bool,
38    ) {
39        let signal: bool = world.has_neighbor_signal(pos);
40        if signal != state.get_value(POWERED) {
41            world.set_block(
42                pos,
43                state.set_value(POWERED, signal),
44                UpdateFlags::UPDATE_CLIENTS,
45            );
46        }
47    }
48
49    fn default_is_pathfindable(
50        &self,
51        _state: BlockStateId,
52        _computation_type: PathComputationType,
53    ) -> bool {
54        false
55    }
56
57    fn new_skull_block_entity(
58        &self,
59        level: Weak<World>,
60        pos: BlockPos,
61        state: BlockStateId,
62    ) -> BlockEntityCreation {
63        BlockEntityCreation::from_registered_factory(BLOCK_ENTITIES.create(
64            &vanilla_block_entity_types::SKULL,
65            level,
66            pos,
67            state,
68        ))
69    }
70
71    fn default_get_clone_item_stack(
72        &self,
73        block: BlockRef,
74        state: BlockStateId,
75        block_entity: Option<SharedBlockEntity>,
76        include_data: bool,
77    ) -> Option<ItemStack> {
78        let item = ItemStack::new(REGISTRY.items.by_block(block));
79        if !include_data {
80            return Some(item);
81        }
82
83        let Some(skull_block_entity) = block_entity
84            .as_ref()
85            .and_then(|block_entity| block_entity.downcast_ref::<SkullBlockEntity>())
86        else {
87            return Some(item);
88        };
89
90        Some(skull_block_entity.skull_as_picked_item(state))
91    }
92
93    fn default_get_drops(
94        &self,
95        state: BlockStateId,
96        context: &BlockLootContext,
97    ) -> Option<Vec<ItemStack>> {
98        let block_entity = context.block_entity()?;
99        let skull_block_entity = block_entity.downcast_ref::<SkullBlockEntity>()?;
100
101        let item = skull_block_entity.skull_as_dropped_item(state);
102        Some(vec![item])
103    }
104}
