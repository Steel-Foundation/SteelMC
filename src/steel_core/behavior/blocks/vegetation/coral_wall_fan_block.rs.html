18/// Vanilla `CoralWallFanBlock` survival (live coral wall fans).
19///
20/// Inherits `canSurvive` from `BaseCoralWallFanBlock`.
21#[block_behavior]
22pub struct CoralWallFanBlock {
23    block: BlockRef,
24    #[json_arg(vanilla_blocks, json = "dead_block")]
25    dead_block: BlockRef,
26}
27
28const WATERLOGGED: &BoolProperty = &BlockStateProperties::WATERLOGGED;
29const HORIZONTAL_FACING: &EnumProperty<Direction> = &BlockStateProperties::HORIZONTAL_FACING;
30
31impl CoralWallFanBlock {
32    /// Creates a new live coral wall fan block behavior.
33    #[must_use]
34    pub const fn new(block: BlockRef, dead_block: BlockRef) -> Self {
35        Self { block, dead_block }
36    }
37
38    fn dead_state(&self, state: BlockStateId) -> BlockStateId {
39        self.dead_block
40            .default_state()
41            .set_value(WATERLOGGED, false)
42            .set_value(HORIZONTAL_FACING, state.get_value(HORIZONTAL_FACING))
43    }
44}
45
46impl BlockBehavior for CoralWallFanBlock {
47    fn can_survive(&self, state: BlockStateId, world: &dyn LevelReader, pos: BlockPos) -> bool {
48        let facing = state.get_value(HORIZONTAL_FACING);
49        BaseCoralWallFanBlock::can_survive(world, pos, facing)
50    }
51
52    fn on_place(
53        &self,
54        state: BlockStateId,
55        world: &Arc<World>,
56        pos: BlockPos,
57        _old_state: BlockStateId,
58        _moved_by_piston: bool,
59    ) {
60        CoralBlock::schedule_die_tick(state, world, pos, self.block);
61    }
62
63    fn update_shape(
64        &self,
65        state: BlockStateId,
66        world: &dyn ScheduledTickAccess,
67        pos: BlockPos,
68        direction: Direction,
69        _neighbor_pos: BlockPos,
70        _neighbor_state: BlockStateId,
71    ) -> BlockStateId {
72        if direction.opposite() == state.get_value(HORIZONTAL_FACING)
73            && !self.can_survive(state, world, pos)
74        {
75            return vanilla_blocks::AIR.default_state();
76        }
77
78        schedule_water_tick_if_waterlogged(state, world, pos);
79
80        CoralBlock::schedule_die_tick(state, world, pos, self.block);
81        state
82    }
83
84    fn get_state_for_placement(&self, context: &BlockPlaceContext<'_>) -> Option<BlockStateId> {
85        let state = self
86            .block
87            .default_state()
88            .set_value(WATERLOGGED, context.is_full_water());
89
90        context
91            .get_nearest_looking_directions()
92            .into_iter()
93            .filter(|direction| direction.is_horizontal())
94            .map(|direction| state.set_value(HORIZONTAL_FACING, direction.opposite()))
95            .find(|state| self.can_survive(*state, context.world, context.place_pos()))
96    }
97
98    fn tick(&self, state: BlockStateId, world: &Arc<World>, pos: BlockPos) {
99        if !CoralBlock::scan_for_water(state, world, pos) {
100            world.set_block(pos, self.dead_state(state), UpdateFlags::UPDATE_CLIENTS);
101        }
102    }
103}

105#[cfg(test)]
106mod tests {
107    use steel_registry::init_vanilla_registry;
108
109    use super::*;
110
111    #[test]
112    fn dead_wall_fan_state_preserves_facing() {
113        init_vanilla_registry();
114        let behavior = CoralWallFanBlock::new(
115            &vanilla_blocks::TUBE_CORAL_WALL_FAN,
116            &vanilla_blocks::DEAD_TUBE_CORAL_WALL_FAN,
117        );
118        let state = vanilla_blocks::TUBE_CORAL_WALL_FAN
119            .default_state()
120            .set_value(HORIZONTAL_FACING, Direction::West)
121            .set_value(WATERLOGGED, true);
122
123        let dead_state = behavior.dead_state(state);
124
125        assert_eq!(dead_state.get_value(HORIZONTAL_FACING), Direction::West);
126        assert!(!dead_state.get_value(WATERLOGGED));
127    }
