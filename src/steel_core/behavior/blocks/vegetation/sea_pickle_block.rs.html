27/// Vanilla `SeaPickleBlock` survival.
28#[block_behavior]
29pub struct SeaPickleBlock {
30    block: BlockRef,
31}
32
33const MAX_PICKLES: u8 = 4;
34
35const WATERLOGGED: &BoolProperty = &BlockStateProperties::WATERLOGGED;
36const PICKLES: &IntProperty = &BlockStateProperties::PICKLES;
37
38impl SeaPickleBlock {
39    /// Creates a new sea pickle block behavior.
40    #[must_use]
41    pub const fn new(block: BlockRef) -> Self {
42        Self { block }
43    }
44
45    fn may_place_on(world: &dyn LevelReader, state: BlockStateId, pos: BlockPos) -> bool {
46        BLOCK_BEHAVIORS
47            .get_behavior(state.get_block())
48            .get_collision_boxes(state, world, pos, BlockCollisionContext::empty())
49            .iter()
50            .any(|aabb| !aabb.is_empty() && aabb.max_y() >= 1.0)
51            || world.is_face_sturdy(state, pos, Direction::Up)
52    }
53    fn is_dead(state: BlockStateId) -> bool {
54        !state.get_value(WATERLOGGED)
55    }
56}
57
58impl BlockBehavior for SeaPickleBlock {
59    fn update_shape(
60        &self,
61        state: BlockStateId,
62        world: &dyn ScheduledTickAccess,
63        pos: BlockPos,
64        _direction: Direction,
65        _neighbor_pos: BlockPos,
66        _neighbor_state: BlockStateId,
67    ) -> BlockStateId {
68        if !self.can_survive(state, world, pos) {
69            return vanilla_blocks::AIR.default_state();
70        }
71
72        schedule_water_tick_if_waterlogged(state, world, pos);
73        state
74    }
75
76    fn can_survive(&self, _state: BlockStateId, world: &dyn LevelReader, pos: BlockPos) -> bool {
77        let below_pos = pos.below();
78        Self::may_place_on(world, world.get_block_state(below_pos), below_pos)
79    }
80
81    fn get_state_for_placement(&self, context: &BlockPlaceContext<'_>) -> Option<BlockStateId> {
82        let state = context.world.get_block_state(context.place_pos());
83        if state.get_block() == self.block {
84            return Some(state.set_value(PICKLES, MAX_PICKLES.min(state.get_value(PICKLES) + 1)));
85        }
86        let replaced_fluid_state = get_fluid_state_from_block(state);
87        let is_water_source = replaced_fluid_state.is_water() && replaced_fluid_state.is_source();
88        Some(
89            self.block
90                .default_state()
91                .set_value(WATERLOGGED, is_water_source),
92        )
93    }
94    fn can_be_replaced(&self, state: BlockStateId, context: &BlockPlaceContext<'_>) -> bool {
95        if !context.is_secondary_use_active()
96            && context.with_item(|item| item.item() == REGISTRY.items.by_block(state.get_block()))
97            && state.get_value(PICKLES) < MAX_PICKLES
98        {
99            return true;
100        }
101        default_can_be_replaced(state, context)
102    }
103    fn is_pathfindable(
104        &self,
105        _state: BlockStateId,
106        _computation_type: PathComputationType,
107    ) -> bool {
108        false
109    }
110    fn as_bonemealable(&self) -> Option<&dyn Bonemealable> {
111        Some(self)
112    }
113}

115impl Bonemealable for SeaPickleBlock {
116    fn is_valid_bonemeal_target(
117        &self,
118        state: BlockStateId,
119        world: &dyn LevelReader,
120        pos: BlockPos,
121    ) -> bool {
122        !Self::is_dead(state)
123            && world
124                .get_block_state(pos.below())
125                .get_block()
126                .has_tag(&BlockTag::CORAL_BLOCKS)
127    }
128    fn perform_bonemeal(
129        &self,
130        state: BlockStateId,
131        world: &Arc<World>,
132        rng: &mut dyn Rng,
133        pos: BlockPos,
134    ) {
135        let mut z_span = 1;
136        let x_start = pos.x() - 2;
137        let mut z_offset = 0;
138
139        for (count, x) in (0..5).enumerate() {
140            for z in 0..z_span {
141                let end_y = 2 + pos.y() - 1;
142
143                for start_y in (end_y - 2)..end_y {
144                    let position = BlockPos::new(x_start + x, start_y, pos.z() - z_offset + z);
145
146                    if position != pos
147                        && rng.random_range(0..6) == 0
148                        && world.get_block_state(position).get_block() == &vanilla_blocks::WATER
149                    {
150                        let below_state = world.get_block_state(position.below());
151
152                        if below_state.get_block().has_tag(&BlockTag::CORAL_BLOCKS) {
153                            let sea_pickle_state = vanilla_blocks::SEA_PICKLE
154                                .default_state()
155                                .set_value(PICKLES, rng.random_range(0..MAX_PICKLES) + 1);
156
157                            world.set_block(position, sea_pickle_state, UpdateFlags::UPDATE_ALL);
158                        }
159                    }
160                }
161            }
162
163            if count < 2 {
164                z_span += 2;
165                z_offset += 1;
166            } else {
167                z_span -= 2;
168                z_offset -= 1;
169            }
170        }
171
172        let final_state = state.set_value(PICKLES, MAX_PICKLES);
173
174        world.set_block(pos, final_state, UpdateFlags::UPDATE_CLIENTS);
175    }
176}

178#[cfg(test)]
179mod tests {
180    use steel_registry::{init_vanilla_registry, vanilla_fluids};
181
182    use super::*;
183    use crate::behavior::init_behaviors;
184    use crate::test_support::TestLevel;
185
186    #[test]
187    fn sea_pickle_checks_survival_before_scheduling_water() {
188        init_vanilla_registry();
189        init_behaviors();
190        let behavior = SeaPickleBlock::new(&vanilla_blocks::SEA_PICKLE);
191        let state = vanilla_blocks::SEA_PICKLE
192            .default_state()
193            .set_value(WATERLOGGED, true);
194        let pos = BlockPos::new(0, 64, 0);
195        let unsupported = TestLevel::default();
196
197        assert!(
198            behavior
199                .update_shape(
200                    state,
201                    &unsupported,
202                    pos,
203                    Direction::North,
204                    pos.north(),
205                    vanilla_blocks::AIR.default_state(),
206                )
207                .is_air()
208        );
209        assert!(unsupported.scheduled_fluid_ticks.borrow().is_empty());
210
211        let supported =
212            TestLevel::default().with_block(pos.below(), vanilla_blocks::STONE.default_state());
213        assert_eq!(
214            behavior.update_shape(
215                state,
216                &supported,
217                pos,
218                Direction::North,
219                pos.north(),
220                vanilla_blocks::AIR.default_state(),
221            ),
222            state
223        );
224        assert_eq!(
225            supported
226                .scheduled_fluid_ticks
227                .borrow()
228                .iter()
229                .map(|tick| (tick.fluid, tick.delay))
230                .collect::<Vec<_>>(),
231            vec![(&vanilla_fluids::WATER, 5)]
232        );
233    }
