21const THROW: ThrowParams = ThrowParams {
22    sound: &sound_events::ENTITY_SPLASH_POTION_THROW,
23    sound_source: SoundSource::Players,
24    sound_volume: 0.5,
25    y_offset: -20.0,
26    power: 0.5,
27    uncertainty: 1.0,

30/// Splash-potion behavior providing Vanilla's potion-content-dependent name
31/// and the water default instance inherited from `PotionItem`.
32// TODO: Implement the `ProjectileItem` dispenser dispatch.
33#[item_behavior]
34pub struct SplashPotionItem;
35
36impl ItemBehavior for SplashPotionItem {
37    fn get_name<'a>(&self, stack: &'a ItemStack) -> Cow<'a, TextComponent> {
38        potion_name(stack)
39    }
40
41    fn get_default_instance(&self, item: ItemRef) -> ItemStack {
42        potion_default_instance(item)
43    }
44
45    fn use_on(&self, context: &mut UseOnContext) -> InteractionResult {
46        potion_use_on(context)
47    }
48
49    fn use_item(&self, context: &mut UseItemContext) -> InteractionResult {
50        let world = context.world;
51        throw_item_projectile(context, &THROW, |spawn_pos| {
52            SplashPotionEntity::new(
53                &vanilla_entities::SPLASH_POTION,
54                next_entity_id(),
55                spawn_pos,
56                Arc::downgrade(world),
57            )
58        });
59        InteractionResult::Success
60    }
61}
