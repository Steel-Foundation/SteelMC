21const SPLASH_THROW: ThrowParams = ThrowParams {
22    sound: &sound_events::ENTITY_SPLASH_POTION_THROW,
23    sound_source: SoundSource::Players,
24    sound_volume: 0.5,
25    y_offset: -20.0,
26    power: 0.5,
27    uncertainty: 1.0,
28};
29
30const LINGERING_THROW: ThrowParams = ThrowParams {
31    sound: &sound_events::ENTITY_LINGERING_POTION_THROW,
32    sound_source: SoundSource::Neutral,
33    ..SPLASH_THROW
34};

36/// Splash-potion behavior providing Vanilla's potion-content-dependent name
37/// and the water default instance inherited from `PotionItem`.
38// TODO: Implement the `ProjectileItem` dispenser dispatch.
39#[item_behavior]
40pub struct SplashPotionItem;
41
42impl ItemBehavior for SplashPotionItem {
43    fn get_name<'a>(&self, stack: &'a ItemStack) -> Cow<'a, TextComponent> {
44        potion_name(stack)
45    }
46
47    fn get_default_instance(&self, item: ItemRef) -> ItemStack {
48        potion_default_instance(item)
49    }
50
51    fn use_on(&self, context: &mut UseOnContext) -> InteractionResult {
52        potion_use_on(context)
53    }
54
55    fn use_item(&self, context: &mut UseItemContext) -> InteractionResult {
56        let world = context.world;
57        throw_item_projectile(context, &SPLASH_THROW, |spawn_pos| {
58            SplashPotionEntity::new(
59                &vanilla_entities::SPLASH_POTION,
60                next_entity_id(),
61                spawn_pos,
62                Arc::downgrade(world),
63            )
64        });
65        InteractionResult::Success
66    }
67}
