35/// Handles all bucket variants (empty, water, lava).
36#[item_behavior]
37pub struct BucketItem {
38    #[json_arg(vanilla_blocks, json = "content", optional = "empty")]
39    fluid_block: Option<BlockRef>,
40}
41
42impl BucketItem {
43    /// Creates a new bucket behavior. `None` = empty bucket, `Some(block)` = filled.
44    #[must_use]
45    pub const fn new(fluid_block: Option<BlockRef>) -> Self {
46        Self { fluid_block }
47    }
48}

50impl ItemBehavior for BucketItem {
51    fn use_item(&self, context: &mut UseItemContext) -> InteractionResult {
52        match self.fluid_block {
53            None => use_empty_bucket(context),
54            Some(fluid_block) => use_filled_bucket(fluid_block, context),
55        }
56    }
57}

59fn filled_bucket_success_stack(context: &UseItemContext) -> ItemStack {
60    if context.player.has_infinite_materials() {
61        context
62            .inv
63            .with_item(|item| item.copy_with_count(item.count()))
64    } else {
65        ItemStack::new(&vanilla_items::BUCKET)
66    }
67}

69fn use_empty_bucket(context: &mut UseItemContext) -> InteractionResult {
70    let (start, end) = context.player.get_ray_endpoints();
71
72    // Raytrace: stop on source fluids
73    let (hit_block, hit_direction) = context.world.raytrace(start, end, |pos, world| {
74        let state = world.get_block_state(pos);
75        let block = state.get_block();
76
77        if block == &vanilla_blocks::AIR {
78            return RaytraceAction::Pass;
79        }
80
81        let fluid_state = state.get_fluid_state();
82        if fluid_state.is_source() {
83            return RaytraceAction::ImmediateHit;
84        }
85        // Vanilla parity: ClipContext.Fluid.SOURCE_ONLY — flowing fluid is transparent.
86        if !fluid_state.is_empty() {
87            return RaytraceAction::Pass;
88        }
89
90        RaytraceAction::CheckShape
91    });
92
93    // Vanilla returns PASS when raytrace misses (allows other handlers to try)
94    let (Some(hit_pos), Some(direction)) = (hit_block, hit_direction) else {
95        return InteractionResult::Pass;
96    };
97
98    let item = context.inv.with_item(|item| item.clone());
99    if !context
100        .player
101        .may_use_item_at(direction.relative(hit_pos), direction, &item)
102    {
103        return InteractionResult::Fail;
104    }
105
106    let hit_state = context.world.get_block_state(hit_pos);
107    let block_behavior = BLOCK_BEHAVIORS.get_behavior(hit_state.get_block());
108
109    if let Some(result) =
110        block_behavior.pickup_block(context.world, hit_pos, hit_state, Some(context.player))
111    {
112        context.inv.with_item(|item| {
113            context
114                .player
115                .award_stat(&vanilla_stat_types::ITEM_USED, item.item());
116        });
117        // Apply sound
118        if let Some(sound) = result.sound {
119            // Vanilla `BucketItem.use`: `bucketPickupBlock.getPickupSound()...player.playSound(...)`.
120            context.player.play_sound(sound, 1.0, 1.0);
121        }
122
123        // Give filled bucket
124        create_filled_result(context, result.filled_bucket, true);
125        context.world.game_event(
126            &vanilla_game_events::FLUID_PICKUP,
127            hit_pos,
128            &GameEventContext::new(Some(context.player.as_ref()), None),
129        );
130
131        return InteractionResult::Success;
132    }
133
134    // TODO: Remove fallback once all waterloggable blocks implement pickup_block.
135    if let Some(result) = pickup_waterlogged_block(
136        block_behavior,
137        context.world,
138        hit_pos,
139        hit_state,
140        Some(context.player),
141    ) {
142        context.inv.with_item(|item| {
143            context
144                .player
145                .award_stat(&vanilla_stat_types::ITEM_USED, item.item());
146        });
147        if let Some(sound) = result.sound {
148            // Vanilla `BucketItem.use`: `bucketPickupBlock.getPickupSound()...player.playSound(...)`.
149            context.player.play_sound(sound, 1.0, 1.0);
150        }
151
152        create_filled_result(context, result.filled_bucket, true);
153        context.world.game_event(
154            &vanilla_game_events::FLUID_PICKUP,
155            hit_pos,
156            &GameEventContext::new(Some(context.player.as_ref()), None),
157        );
158
159        return InteractionResult::Success;
160    }
161
162    // Nothing was picked up — no fluid source block and no waterlogged block found.
163    // Vanilla returns FAIL here so the client knows no item change occurred.
164    InteractionResult::Fail
165}

172fn use_filled_bucket(fluid_block: BlockRef, context: &mut UseItemContext) -> InteractionResult {
173    // Raytrace to find target block
174    let (start, end) = context.player.get_ray_endpoints();
175    let (ray_block, ray_dir) = context.world.raytrace(start, end, |pos, world| {
176        let state = world.get_block_state(pos);
177        let block = state.get_block();
178        // Filled buckets use ClipContext.Fluid.NONE: ignore fluid shapes, but
179        // still test the block shape of waterlogged/container blocks.
180        if block == &vanilla_blocks::AIR {
181            return RaytraceAction::Pass;
182        }
183        RaytraceAction::CheckShape
184    });
185
186    // Vanilla returns PASS when raytrace misses (allows other handlers to try)
187    let (Some(clicked_pos), Some(direction)) = (ray_block, ray_dir) else {
188        return InteractionResult::Pass;
189    };
190
191    // If the block is out of bounds, return fail
192    if !context.world.is_in_valid_bounds(clicked_pos) {
193        return InteractionResult::Fail;
194    }
195
196    let item = context.inv.with_item(|item| item.clone());
197    if !context
198        .player
199        .may_use_item_at(direction.relative(clicked_pos), direction, &item)
200    {
201        return InteractionResult::Fail;
202    }
203
204    let clicked_state = context.world.get_block_state(clicked_pos);
205    let is_sneaking = context.player.is_crouching();
206
207    // Define fluid placement logic as a closure to reuse for primary/secondary targets.
208    // `check_sneak`: true for primary attempt, false for secondary (vanilla parity:
209    // recursive emptyContents passes hitResult=null for fallback, bypassing sneak check).
210    let try_place_fluid = |pos: BlockPos, check_sneak: bool| -> bool {
211        if !context.world.is_in_valid_bounds(pos) {
212            return false;
213        }
214
215        let state = context.world.get_block_state(pos);
216        let fluid_state = state.get_fluid_state();
217
218        // Vanilla parity (bl4): when sneaking, only air allows placement at this position.
219        // Non-air blocks redirect to the neighbor — handled by the secondary call.
220        // The secondary call bypasses this check (hitResult == null in vanilla).
221        if check_sneak && is_sneaking && !state.get_block().config.is_air {
222            return false;
223        }
224
225        let is_water_bucket = fluid_block == &vanilla_blocks::WATER;
226        let behavior = BLOCK_BEHAVIORS.get_behavior(state.get_block());
227        let is_liquid_container = state.is_liquid_container();
228        let can_place_liquid = is_water_bucket
229            && is_liquid_container
230            && behavior.can_place_liquid_with_player(
231                state,
232                FluidState::source(&vanilla_fluids::WATER).fluid_id,
233                Some(context.player),
234            );
235        let can_replace = state.can_be_replaced_by_fluid(fluid_block);
236
237        // Vanilla parity: block must be replaceable or liquid-container-admissible for placement.
238        if !can_replace && !can_place_liquid {
239            return false;
240        }
241
242        // Vanilla parity: in worlds where water evaporates (e.g. the Nether),
243        // water buckets fizz out without placing any fluid.
244        // TODO: Per-position environment attributes (vanilla uses EnvironmentAttributes.WATER_EVAPORATES per-pos)
245        if is_water_bucket && context.world.dimension_type.water_evaporates {
246            context
247                .world
248                .level_event(level_events::PARTICLES_WATER_EVAPORATING, pos, 0, None);
249            return true;
250        }
251
252        // 1. Try LiquidBlockContainer handling (only if Water bucket).
253        if is_water_bucket && is_liquid_container {
254            let source_water = FluidState::source(&vanilla_fluids::WATER);
255            behavior.place_liquid(context.world, pos, state, source_water);
256            play_empty_sound_and_event(context, pos, true);
257            return true;
258        }
259
260        // 2. Try Standard Placement (Replaceable block)
261        if can_replace {
262            // If same fluid already exists and is source, just consume bucket (parity)
263            let is_same_fluid = if is_water_bucket {
264                fluid_state.is_water()
265            } else {
266                fluid_state.is_lava()
267            };
268
269            if is_same_fluid && fluid_state.is_source() {
270                play_empty_sound_and_event(context, pos, is_water_bucket);
271                return true;
272            }
273
274            // Vanilla parity: destroy non-liquid replaceable blocks first so they
275            // drop their items (e.g. tall grass, flowers, snow layers).
276            if !state.get_block().config.liquid && !state.get_block().config.is_air {
277                context.player.get_world().destroy_block(pos, true);
278            }
279
280            // Place fluid block
281            let fluid_state_to_place = fluid_block.default_state();
282            if context
283                .world
284                .set_block(pos, fluid_state_to_place, UpdateFlags::UPDATE_ALL_IMMEDIATE)
285            {
286                let fluid_ref = if is_water_bucket {
287                    &vanilla_fluids::WATER
288                } else {
289                    &vanilla_fluids::LAVA
290                };
291                let tick_delay = FLUID_BEHAVIORS
292                    .get_behavior(fluid_ref)
293                    .tick_delay(context.world);
294                context
295                    .world
296                    .schedule_fluid_tick_default(pos, fluid_ref, tick_delay);
297
298                play_empty_sound_and_event(context, pos, is_water_bucket);
299
300                return true;
301            }
302        }
303        false
304    };
305
306    // Vanilla parity (BucketItem.java): position selection mirrors
307    // `instanceof LiquidBlockContainer && content == Fluids.WATER ? pos : directionOffsetPos`.
308    // If primary fails, secondary retries at the offset pos without sneak check,
309    // matching vanilla's recursive `emptyContents(hitResult=null)` fallback.
310    let is_water_bucket = fluid_block == &vanilla_blocks::WATER;
311    let primary_pos =
312        filled_bucket_primary_pos(clicked_state, clicked_pos, direction, is_water_bucket);
313
314    // Attempt Primary (with sneak check)
315    if try_place_fluid(primary_pos, true) {
316        let result_stack = filled_bucket_success_stack(context);
317        create_filled_result(context, result_stack, true);
318        return InteractionResult::Success;
319    }
320
321    // Attempt Secondary (Fallback — no sneak check, matching vanilla hitResult=null).
322    // Vanilla's emptyContents always recurses with hitResult=null at the offset position
323    // when the primary attempt fails, regardless of bucket type.
324    let secondary_pos = direction.relative(clicked_pos);
325    if try_place_fluid(secondary_pos, false) {
326        let result_stack = filled_bucket_success_stack(context);
327        create_filled_result(context, result_stack, true);
328        return InteractionResult::Success;
329    }
330
331    InteractionResult::Fail
332}

334fn play_empty_sound_and_event(context: &UseItemContext, pos: BlockPos, is_water_bucket: bool) {
335    let sound_event = if is_water_bucket {
336        &sound_events::ITEM_BUCKET_EMPTY
337    } else {
338        &sound_events::ITEM_BUCKET_EMPTY_LAVA
339    };
340    context.world.play_block_sound(
341        sound_event,
342        pos,
343        1.0,
344        1.0,
345        // Vanilla `BucketItem.playEmptySound`: `level.playSound(user, pos, ...)` excludes the placing player.
346        Some(context.player.id()),
347    );
348    context.world.game_event(
349        &vanilla_game_events::FLUID_PLACE,
350        pos,
351        &GameEventContext::new(Some(context.player.as_ref()), None),
352    );
353}
354
355fn filled_bucket_primary_pos(
356    clicked_state: BlockStateId,
357    clicked_pos: BlockPos,
358    direction: Direction,
359    is_water_bucket: bool,
360) -> BlockPos {
361    if is_water_bucket && clicked_state.is_liquid_container() {
362        clicked_pos
363    } else {
364        direction.relative(clicked_pos)
365    }
