1use crate::behavior::{InteractionResult, ItemBehavior};
2use crate::entity::LivingEntity;
3use crate::player::Player;
4use steel_macros::item_behavior;
5use steel_registry::data_components::vanilla_components::CUSTOM_NAME;
6use steel_registry::item_stack::ItemStack;
7use steel_utils::types::InteractionHand;
8
9/// Vanilla name tag behavior.
10#[item_behavior]
11pub struct NameTagItem;
12
13impl ItemBehavior for NameTagItem {
14    fn interact_living_entity(
15        &self,
16        stack: &mut ItemStack,
17        _player: &Player,
18        target: &dyn LivingEntity,
19        _hand: InteractionHand,
20    ) -> InteractionResult {
21        if target.entity_type().can_serialize
22            && let Some(component) = stack.get(CUSTOM_NAME)
23        {
24            if LivingEntity::is_alive(target) {
25                target.set_custom_name(Some(component.clone()));
26                if let Some(mob) = target.as_mob() {
27                    mob.set_persistence_required();
28                }
29                stack.shrink(1);
30            }
31            InteractionResult::Success
32        } else {
33            InteractionResult::Pass
34        }
35    }
36}
