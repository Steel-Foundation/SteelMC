1//! Shears item behavior (`ShearsItem`).
2//!
3//! Right-clicking a growing plant head that is not at max age crops it, matching
4//! vanilla `ShearsItem.useOn`.
5

7use steel_registry::{blocks::block_state_ext::BlockStateExt, sound_events, vanilla_game_events};

15/// Behavior for shears.
16#[item_behavior]
17pub struct ShearsItem;
18
19impl ShearsItem {
20    /// Crops a growing plant head to max age so `random_tick` no longer extends it.
21    fn block_age(
22        context: &UseOnContext<'_>,
23        plant: &dyn GrowingPlantHeadBehavior,
24        pos: BlockPos,
25        state: BlockStateId,
26    ) {
27        let new_state = plant.get_max_age_state(state);
28        context
29            .world
30            .set_block(pos, new_state, UpdateFlags::UPDATE_ALL);
31        context.world.game_event(
32            &vanilla_game_events::BLOCK_CHANGE,
33            pos,
34            &GameEventContext::new(Some(context.player), Some(new_state)),
35        );
36    }
37}
38
39impl ItemBehavior for ShearsItem {
40    fn use_on(&self, context: &mut UseOnContext) -> InteractionResult {
41        let pos = context.hit_result.block_pos;
42        let state = context.world.get_block_state(pos);
43        let behavior = BLOCK_BEHAVIORS.get_behavior(state.get_block());
44        let Some(plant) = behavior.as_growing_plant_head() else {
45            return InteractionResult::Pass;
46        };
47        if plant.is_max_age(state) {
48            return InteractionResult::Pass;
49        }
50
51        // TODO: Trigger ITEM_USED_ON_BLOCK once advancement criteria dispatch is implemented.
52        context.world.play_block_sound(
53            &sound_events::BLOCK_GROWING_PLANT_CROP,
54            pos,
55            1.0,
56            1.0,
57            Some(context.player.id()),
58        );
59        Self::block_age(context, plant, pos, state);
60
61        let has_infinite_materials = context.player.has_infinite_materials();
62        context
63            .inv
64            .with_item(|item| item.hurt_and_break(1, has_infinite_materials));
65
66        InteractionResult::Success
67    }
68}

188        init_globals_once();
189        let world = fresh_test_world(key);

195        assert!(world.set_block(pos, state, UpdateFlags::UPDATE_NONE));

197        let player = TestPlayerBuilder::new(Arc::clone(&world), "ShearsTester", 1).build();

232    fn sound_ids(packets: &SyncMutex<Vec<EncodedPacket>>) -> Vec<i32> {
233        packets
234            .lock()
235            .iter()
236            .filter_map(|packet| {
237                let mut cursor = Cursor::new(packet.encoded_data.as_slice());
238                VarInt::read(&mut cursor).ok()?;
239                if VarInt::read(&mut cursor).ok()?.0 != C_SOUND {
240                    return None;
241                }
242                Some(VarInt::read(&mut cursor).ok()?.0)
243            })
244            .collect()
245    }

247    fn use_shears(fixture: &ShearsFixture) -> InteractionResult {
248        let mut context = UseOnContext::new(
249            &fixture.player,

256    }

258    #[test]
259    fn use_on_crops_growing_head_and_emits_vanilla_effects() {
260        let non_max_age = MAX_AGE - 1;
261        let fixture = create_fixture("shears_crop", non_max_age);
262
263        assert_eq!(use_shears(&fixture), InteractionResult::Success);
264        assert_eq!(
265            fixture
266                .world
267                .get_block_state(fixture.pos)
268                .get_value(&BlockStateProperties::AGE_25),
269            MAX_AGE,
270        );
271        assert_eq!(
272            fixture
273                .player
274                .inventory
275                .lock()
276                .get_item_in_hand(InteractionHand::MainHand)
277                .get_damage_value(),
278            1,
279        );
280        assert_eq!(
281            sound_ids(&fixture.packets),
282            [sound_events::BLOCK_GROWING_PLANT_CROP.packet_holder_id()],
283        );
284        assert_eq!(
285            fixture.events.lock().as_slice(),
286            [&vanilla_game_events::BLOCK_CHANGE],
287        );
288    }
