1//! Block and item behavior system.
2//!
3//! This module contains the behavior traits and registries that define how
4//! blocks and items behave dynamically. This is separate from the static data
5//! in steel-registry to maintain a clean separation between constant data and
6//! functional/dynamic behavior.
7//!
8//! # Architecture
9//!
10//! After the main registry (`steel-registry`) is frozen, behavior registries
11//! are created:
12//! - `BlockBehaviorRegistry` - assigns default or custom behaviors to each block
13//! - `ItemBehaviorRegistry` - assigns default or custom behaviors to each item
14//!
15//! # Usage
16//!
17//! ```ignore
18//! use steel_core::behavior::{init_behaviors, BLOCK_BEHAVIORS, ITEM_BEHAVIORS};
19//!
20//! // After registry is frozen, call once at startup:
21//! init_behaviors();
22//!
23//! // Then access behaviors via the global registries:
24//! let behavior = BLOCK_BEHAVIORS.get_behavior(block);
25//! ```

109/// Wrapper for the global block behavior registry that implements `Deref`.
110pub struct BlockBehaviorLock(OnceLock<BlockBehaviorRegistry>);

112impl Deref for BlockBehaviorLock {
113    type Target = BlockBehaviorRegistry;
114
115    fn deref(&self) -> &Self::Target {
116        self.0.get().expect("Block behaviors not initialized")

131/// Extension trait for `BlockStateId` that provides access to behavior-dependent methods.
132///
133/// This is separate from `BlockStateExt` (in steel-registry) because these methods
134/// require access to the behavior registry which lives in steel-core.
135pub trait BlockStateBehaviorExt {
136    /// Returns whether this block state belongs to a vanilla `LiquidBlockContainer`.
137    fn is_liquid_container(&self) -> bool;
138
139    /// Returns whether this block state can be replaced by the given fluid block.
140    fn can_be_replaced_by_fluid(&self, fluid_block: BlockRef) -> bool;
141
142    /// Returns whether this block state can be replaced in this placement context.
143    fn can_be_replaced(&self, context: &BlockPlaceContext<'_>) -> bool;
144
145    /// Returns whether this block state is pathfindable for the supplied vanilla computation type.
146    fn is_pathfindable(&self, computation_type: PathComputationType) -> bool;
147
148    /// Returns whether this block state extends `BedBlock`
149    fn is_bed(&self) -> bool;
150
151    /// Returns whether this block state can be occupied by a forced respawn position
152    fn is_possible_to_respawn_in_this(&self) -> bool;
153}
154
155impl BlockStateBehaviorExt for BlockStateId {
156    fn is_liquid_container(&self) -> bool {
157        let block = self.get_block();
158        let behavior = BLOCK_BEHAVIORS.get_behavior(block);
159        behavior.is_liquid_container(*self)
160    }
161
162    fn can_be_replaced_by_fluid(&self, fluid_block: BlockRef) -> bool {
163        let block = self.get_block();
164        let behavior = BLOCK_BEHAVIORS.get_behavior(block);
165        behavior.can_be_replaced_by_fluid(*self, fluid_block)
166    }
167
168    fn can_be_replaced(&self, context: &BlockPlaceContext<'_>) -> bool {
169        let block = self.get_block();
170        let behavior = BLOCK_BEHAVIORS.get_behavior(block);
171        behavior.can_be_replaced(*self, context)
172    }
173
174    fn is_pathfindable(&self, computation_type: PathComputationType) -> bool {
175        let block = self.get_block();
176        let behavior = BLOCK_BEHAVIORS.get_behavior(block);
177        behavior.is_pathfindable(*self, computation_type)
178    }
179
180    fn is_bed(&self) -> bool {
181        let block = self.get_block();
182        BLOCK_BEHAVIORS.get_behavior(block).is_bed()
183    }
184
185    fn is_possible_to_respawn_in_this(&self) -> bool {
186        let block = self.get_block();
187        let behavior = BLOCK_BEHAVIORS.get_behavior(block);
188        behavior.is_possible_to_respawn_in_this(*self)
189    }
190}

192/// Global block behavior registry.

202/// Initializes the global behavior registries.
203///
204/// This should be called after the main registry is frozen. Repeated calls are a no-op.
205pub fn init_behaviors() {
206    BLOCK_BEHAVIORS.0.get_or_init(|| {
207        let mut block_behaviors = BlockBehaviorRegistry::new();
208        register_block_behaviors(&mut block_behaviors);
209        block_behaviors
210    });
211
212    FLUID_BEHAVIORS.0.get_or_init(|| {
213        let mut fluid_behaviors = FluidBehaviorRegistry::new();
214
215        // Water: WaterFluid implements FluidBehavior directly
216        let water_behavior: Box<dyn FluidBehavior> = Box::new(WaterFluid);
217        // Both WATER and FLOWING_WATER share the same behavior
218        fluid_behaviors.set_behavior(&vanilla_fluids::WATER, water_behavior);
219        fluid_behaviors.set_behavior(&vanilla_fluids::FLOWING_WATER, Box::new(WaterFluid));
220
221        // Lava: LavaFluid implements FluidBehavior directly
222        let lava_behavior: Box<dyn FluidBehavior> = Box::new(LavaFluid);
223        fluid_behaviors.set_behavior(&vanilla_fluids::LAVA, lava_behavior);
224        fluid_behaviors.set_behavior(&vanilla_fluids::FLOWING_LAVA, Box::new(LavaFluid));
225
226        fluid_behaviors
227    });
228
229    ITEM_BEHAVIORS.0.get_or_init(|| {
230        let mut item_behaviors = ItemBehaviorRegistry::new();
231        register_item_behaviors(&mut item_behaviors);
232        item_behaviors
233    });
234
235    CONSUME_EFFECT_BEHAVIORS.0.get_or_init(|| {
236        let mut consume_effect_behaviors = ConsumeEffectBehaviorRegistry::new();
237
238        let apply_effects: Box<dyn ConsumeEffectBehavior> = Box::new(ApplyEffectsBehavior);
239        consume_effect_behaviors
240            .set_behavior(&vanilla_consume_effect_types::APPLY_EFFECTS, apply_effects);
241
242        let remove_effects: Box<dyn ConsumeEffectBehavior> = Box::new(RemoveEffectsBehavior);
243        consume_effect_behaviors.set_behavior(
244            &vanilla_consume_effect_types::REMOVE_EFFECTS,
245            remove_effects,
246        );
247
248        let clear_all_effects: Box<dyn ConsumeEffectBehavior> = Box::new(ClearAllEffectsBehavior);
249        consume_effect_behaviors.set_behavior(
250            &vanilla_consume_effect_types::CLEAR_ALL_EFFECTS,
251            clear_all_effects,
252        );
253
254        let play_sound: Box<dyn ConsumeEffectBehavior> = Box::new(PlaySoundBehavior);
255        consume_effect_behaviors
256            .set_behavior(&vanilla_consume_effect_types::PLAY_SOUND, play_sound);
257
258        let teleport_randomly: Box<dyn ConsumeEffectBehavior> = Box::new(TeleportRandomlyBehavior);
259        consume_effect_behaviors.set_behavior(
260            &vanilla_consume_effect_types::TELEPORT_RANDOMLY,
261            teleport_randomly,
262        );
263
264        consume_effect_behaviors
265    });
266
267    MOB_EFFECT_BEHAVIORS.0.get_or_init(|| {
268        let mut mob_effect_behaviors = MobEffectBehaviorRegistry::new();
269
270        // HealOrHarmMobEffect backs both Instant Health and Instant Damage.
271        let instant_health: Box<dyn MobEffectBehavior> =
272            Box::new(HealOrHarmBehavior { is_harm: false });
273        mob_effect_behaviors.set_behavior(vanilla_mob_effects::INSTANT_HEALTH, instant_health);
274
275        let instant_damage: Box<dyn MobEffectBehavior> =
276            Box::new(HealOrHarmBehavior { is_harm: true });
277        mob_effect_behaviors.set_behavior(vanilla_mob_effects::INSTANT_DAMAGE, instant_damage);
278
279        let saturation: Box<dyn MobEffectBehavior> = Box::new(SaturationBehavior);
280        mob_effect_behaviors.set_behavior(vanilla_mob_effects::SATURATION, saturation);
281
282        let wither: Box<dyn MobEffectBehavior> = Box::new(WitherBehavior);
283        mob_effect_behaviors.set_behavior(vanilla_mob_effects::WITHER, wither);
284
285        let poison: Box<dyn MobEffectBehavior> = Box::new(PoisonBehavior);
286        mob_effect_behaviors.set_behavior(vanilla_mob_effects::POISON, poison);
287
288        let regeneration: Box<dyn MobEffectBehavior> = Box::new(RegenerationBehavior);
289        mob_effect_behaviors.set_behavior(vanilla_mob_effects::REGENERATION, regeneration);
290
291        let hunger: Box<dyn MobEffectBehavior> = Box::new(HungerBehavior);
292        mob_effect_behaviors.set_behavior(vanilla_mob_effects::HUNGER, hunger);
293
294        let absorption: Box<dyn MobEffectBehavior> = Box::new(AbsorptionBehavior);
295        mob_effect_behaviors.set_behavior(vanilla_mob_effects::ABSORPTION, absorption);
296
297        let bad_omen: Box<dyn MobEffectBehavior> = Box::new(BadOmenBehavior);
298        mob_effect_behaviors.set_behavior(vanilla_mob_effects::BAD_OMEN, bad_omen);
299
300        let raid_omen: Box<dyn MobEffectBehavior> = Box::new(RaidOmenBehavior);
301        mob_effect_behaviors.set_behavior(vanilla_mob_effects::RAID_OMEN, raid_omen);
302
303        let oozing: Box<dyn MobEffectBehavior> = Box::new(OozingBehavior);
304        mob_effect_behaviors.set_behavior(vanilla_mob_effects::OOZING, oozing);
305
306        let weaving: Box<dyn MobEffectBehavior> = Box::new(WeavingBehavior);
307        mob_effect_behaviors.set_behavior(vanilla_mob_effects::WEAVING, weaving);
308
309        let wind_charged: Box<dyn MobEffectBehavior> = Box::new(WindChargedBehavior);
310        mob_effect_behaviors.set_behavior(vanilla_mob_effects::WIND_CHARGED, wind_charged);
311
312        let infested: Box<dyn MobEffectBehavior> = Box::new(InfestedBehavior);
313        mob_effect_behaviors.set_behavior(vanilla_mob_effects::INFESTED, infested);
314
315        mob_effect_behaviors
316    });
