23/// Result of an interaction (item use, block use, etc.)
24#[derive(Debug, Clone, Copy, PartialEq, Eq)]
25pub enum InteractionResult {
26    /// The interaction succeeded and consumed the action.
27    Success,
28    /// The interaction succeeded and the server should broadcast the swing.
29    SuccessServer,
30    /// The interaction consumed the action without swinging.
31    Consume,
32    /// The interaction failed and consumed the action.
33    Fail,
34    /// The interaction did not apply; try the next handler.
35    Pass,
36    /// Try the empty-hand interaction on the block.
37    TryEmptyHandInteraction,
38}
39
40impl InteractionResult {
41    /// Returns true if this result consumes the action (Success or Fail).
42    /// Pass and `TryEmptyHandInteraction` do not consume the action.
43    #[must_use]
44    pub const fn consumes_action(self) -> bool {
45        matches!(
46            self,
47            InteractionResult::Success
48                | InteractionResult::SuccessServer
49                | InteractionResult::Consume
50                | InteractionResult::Fail
51        )
52    }
53
54    /// Returns true when vanilla requests the server to broadcast the swing.
55    #[must_use]
56    pub const fn should_swing_server(self) -> bool {
57        matches!(self, InteractionResult::SuccessServer)
58    }
59
60    /// Returns true for vanilla `InteractionResult.Success` variants that run
61    /// item-use side effects such as `minecraft:use_cooldown`.
62    #[must_use]
63    pub const fn should_apply_item_use_side_effects(self) -> bool {
64        matches!(
65            self,
66            InteractionResult::Success
67                | InteractionResult::SuccessServer
68                | InteractionResult::Consume
69        )
70    }
71}

73/// Context for placing a block.
74///
75/// Vanilla porting map:
76/// - `UseOnContext.getClickedPos()` is [`Self::hit_pos`].
77/// - `BlockPlaceContext.getClickedPos()` is [`Self::place_pos`].
78/// - `BlockPlaceContext.replacingClickedOnBlock()` is
79///   [`Self::replaces_clicked_block`].
80///
81/// When translating vanilla block placement code, do not map
82/// `BlockPlaceContext.getClickedPos()` to [`Self::hit_pos`].
83pub struct BlockPlaceContext<'a> {
84    /// Raw block position from the hit result.
85    ///
86    /// Vanilla equivalent: `UseOnContext.getClickedPos()`.
87    hit_pos: BlockPos,
88    /// The face of the block that was clicked.
89    clicked_face: Direction,
90    /// The exact location where the click occurred.
91    click_location: DVec3,
92    /// Whether the click was inside the block.
93    inside: bool,
94    /// Position where the block will be placed.
95    ///
96    /// Vanilla equivalent: `BlockPlaceContext.getClickedPos()`. Vanilla returns
97    /// the raw hit position only when replacing the clicked block; otherwise it
98    /// returns the adjacent block position in the clicked-face direction.
99    place_pos: BlockPos,
100    /// Whether placement replaces the hit block itself.
101    ///
102    /// Vanilla equivalent: `BlockPlaceContext.replacingClickedOnBlock()`.
103    replaces_clicked_block: bool,
104    /// The world where the block is being placed.
105    pub world: &'a Arc<World>,
106    source: PlacementSource<'a>,
107    mode: PlacementMode,
108}
109
110impl<'a> BlockPlaceContext<'a> {
111    /// Creates a placement context from a source and effective hit result.
112    #[must_use]
113    pub fn new(
114        world: &'a Arc<World>,
115        source: PlacementSource<'a>,
116        hit_result: &BlockHitResult,
117    ) -> Self {
118        Self::with_mode(world, source, hit_result, PlacementMode::Standard)
119    }
120
121    fn with_mode(
122        world: &'a Arc<World>,
123        source: PlacementSource<'a>,
124        hit_result: &BlockHitResult,
125        mode: PlacementMode,
126    ) -> Self {
127        let hit_pos = hit_result.block_pos;
128        let mut context = Self {
129            hit_pos,
130            clicked_face: hit_result.direction,
131            click_location: hit_result.location,
132            inside: hit_result.inside,
133            place_pos: hit_pos,
134            replaces_clicked_block: true,
135            world,
136            source,
137            mode,
138        };
139        context.resolve_placement_geometry();
140        context
141    }
142
143    /// Creates vanilla's playerless `DirectionalPlaceContext` equivalent.
144    #[must_use]
145    pub fn directional(
146        world: &'a Arc<World>,
147        pos: BlockPos,
148        direction: Direction,
149        stack: &'a mut ItemStack,
150        clicked_face: Direction,
151    ) -> Self {
152        let hit_result = BlockHitResult {
153            location: DVec3::new(
154                f64::from(pos.x()) + 0.5,
155                f64::from(pos.y()),
156                f64::from(pos.z()) + 0.5,
157            ),
158            direction: clicked_face,
159            block_pos: pos,
160            miss: false,
161            inside: false,
162            world_border_hit: false,
163        };
164        let source = PlacementSource::direct(
165            None,
166            InteractionHand::MainHand,
167            stack,
168            PlacementOrientation::Directional { direction },
169            false,
170        );
171        Self::with_mode(world, source, &hit_result, PlacementMode::Directional)
172    }
173
174    fn resolve_placement_geometry(&mut self) {
175        self.place_pos = self.hit_pos;
176        self.replaces_clicked_block = true;
177        self.replaces_clicked_block = self
178            .world
179            .get_block_state(self.hit_pos)
180            .can_be_replaced(self);
181        if self.mode == PlacementMode::Standard && !self.replaces_clicked_block {
182            self.place_pos = self.clicked_face.relative(self.hit_pos);
183        }
184    }
185
186    /// Returns whether the effective placement position can be replaced.
187    #[must_use]
188    pub fn can_place(&self) -> bool {
189        if self.mode == PlacementMode::Directional {
190            return self
191                .world
192                .get_block_state(self.hit_pos)
193                .can_be_replaced(self);
194        }
195
196        self.replaces_clicked_block
197            || self
198                .world
199                .get_block_state(self.place_pos)
200                .can_be_replaced(self)
201    }
202
203    /// Creates the shifted context used by vanilla `BlockPlaceContext.at`.
204    #[must_use]
205    pub fn at(mut self, pos: BlockPos, direction: Direction) -> Self {
206        let (step_x, step_y, step_z) = direction.offset();
207        self.hit_pos = pos;
208        self.clicked_face = direction;
209        self.click_location = DVec3::new(
210            f64::from(pos.x()) + 0.5 + f64::from(step_x) * 0.5,
211            f64::from(pos.y()) + 0.5 + f64::from(step_y) * 0.5,
212            f64::from(pos.z()) + 0.5 + f64::from(step_z) * 0.5,
213        );
214        self.inside = false;
215        self.mode = PlacementMode::Standard;
216        self.resolve_placement_geometry();
217        self
218    }

306    /// Returns the direction the player is looking at most directly.
307    ///
308    /// This considers both yaw and pitch to determine the nearest direction
309    /// among all 6 directions (UP, DOWN, NORTH, SOUTH, EAST, WEST).
310    ///
311    /// Based on Java's `Direction.orderedByNearest(Entity)[0]`. Unlike
312    /// [`Self::get_nearest_looking_directions`], this does not reorder around
313    /// the clicked face when placing beside a non-replaceable block.
314    #[must_use]
315    pub fn get_nearest_looking_direction(&self) -> Direction {
316        self.source.orientation.nearest_looking_direction()
317    }
318
319    /// Returns the vertical direction the player is looking toward.
320    ///
321    /// Based on Java's `BlockPlaceContext.getNearestLookingVerticalDirection()`.
322    #[must_use]
323    pub const fn get_nearest_looking_vertical_direction(&self) -> Direction {
324        self.source.orientation.nearest_vertical_direction()
325    }
326    /// Returns all 6 directions ordered by how closely the player is looking at them.
327    ///
328    /// Based on Java's `BlockPlaceContext.getNearestLookingDirections()`.
329    /// When not replacing the clicked block, the opposite of the clicked face
330    /// is moved to the front of the array.
331    #[must_use]
332    pub fn get_nearest_looking_directions(&self) -> [Direction; 6] {
333        let (mut directions, adjust_for_replacement) = self.source.orientation.directions();
334
335        // If not replacing the clicked block, prioritize the opposite of clicked face
336        if adjust_for_replacement && !self.replaces_clicked_block {
337            let clicked_opposite = self.clicked_face.opposite();
338            if let Some(index) = directions.iter().position(|&d| d == clicked_opposite)
339                && index > 0
340            {
341                directions.copy_within(0..index, 1);
342                directions[0] = clicked_opposite;
343            }
344        }
345
346        directions
347    }

353        let fluid_state = get_fluid_state(self.world, self.place_pos);
354        fluid_state.is_source() && fluid_state.is_water()

361        let fluid_state = get_fluid_state(self.world, self.place_pos);
362        fluid_state.is_full() && fluid_state.is_water()
363    }

372/// Placement direction behavior derived from either a player or a synthetic caller.
373#[derive(Clone, Copy, Debug)]
374pub enum PlacementOrientation {
375    /// Vanilla player-derived yaw and pitch.
376    Player {
377        /// Player yaw in degrees.
378        rotation: f32,
379        /// Player pitch in degrees.
380        pitch: f32,
381    },
382    /// Vanilla `DirectionalPlaceContext` orientation.
383    Directional {
384        /// Synthetic placement direction.
385        direction: Direction,
386    },
387}
388
389impl PlacementOrientation {
390    fn nearest_looking_direction(self) -> Direction {
391        match self {
392            Self::Player { rotation, pitch } => Direction::ordered_by_nearest(rotation, pitch)[0],
393            Self::Directional { .. } => Direction::Down,
394        }
395    }
396
397    fn horizontal_direction(self) -> Direction {
398        match self {
399            Self::Player { rotation, .. } => Direction::from_yaw(rotation),
400            Self::Directional { direction } if direction.is_horizontal() => direction,
401            Self::Directional { .. } => Direction::North,
402        }
403    }
404
405    const fn rotation(self) -> f32 {
406        match self {
407            Self::Player { rotation, .. } => rotation,
408            Self::Directional { direction } => match direction {
409                Direction::Down | Direction::Up => -DEGREE_90,
410                Direction::South => 0.0,
411                Direction::West => DEGREE_90,
412                Direction::North => DEGREE_180,
413                Direction::East => DEGREE_270,
414            },
415        }
416    }
417
418    const fn nearest_vertical_direction(self) -> Direction {
419        match self {
420            Self::Player { pitch, .. } if pitch < 0.0 => Direction::Up,
421            Self::Player { .. } | Self::Directional { .. } => Direction::Down,
422        }
423    }
424
425    fn directions(self) -> ([Direction; 6], bool) {
426        match self {
427            Self::Player { rotation, pitch } => {
428                (Direction::ordered_by_nearest(rotation, pitch), true)
429            }
430            Self::Directional { direction } => (directional_placement_directions(direction), false),
431        }
432    }
433}
434
435const fn directional_placement_directions(direction: Direction) -> [Direction; 6] {
436    match direction {
437        Direction::Down => [
438            Direction::Down,
439            Direction::North,
440            Direction::East,
441            Direction::South,
442            Direction::West,
443            Direction::Up,
444        ],
445        Direction::Up => [
446            Direction::Down,
447            Direction::Up,
448            Direction::North,
449            Direction::East,
450            Direction::South,
451            Direction::West,
452        ],
453        Direction::North => [
454            Direction::Down,
455            Direction::North,
456            Direction::East,
457            Direction::West,
458            Direction::Up,
459            Direction::South,
460        ],
461        Direction::South => [
462            Direction::Down,
463            Direction::South,
464            Direction::East,
465            Direction::West,
466            Direction::Up,
467            Direction::North,
468        ],
469        Direction::West => [
470            Direction::Down,
471            Direction::West,
472            Direction::South,
473            Direction::Up,
474            Direction::North,
475            Direction::East,
476        ],
477        Direction::East => [
478            Direction::Down,
479            Direction::East,
480            Direction::South,
481            Direction::Up,
482            Direction::North,
483            Direction::West,
484        ],
485    }
486}

493/// Player and item data retained across one block placement operation.
494///
495/// Player-backed access deliberately re-resolves the selected hand under short
496/// inventory locks. The stack may therefore change between accesses; retaining
497/// one inventory guard across behavior dispatch could deadlock callbacks which
498/// open a menu. Direct access retains the exact borrowed stack for the whole
499/// operation.
500pub struct PlacementSource<'a> {
501    player: Option<&'a Player>,
502    hand: InteractionHand,
503    item: PlacementItemSource<'a>,
504    orientation: PlacementOrientation,
505    is_secondary_use_active: bool,
506}
507
508impl<'a> PlacementSource<'a> {
509    /// Creates a placement source backed by a player's live hand.
510    #[must_use]
511    pub fn player_hand(player: &'a Player, inv: &InventoryAccess) -> Self {
512        let (rotation, pitch) = player.rotation();
513        let hand = inv.hand;
514        Self {
515            player: Some(player),
516            hand,
517            item: PlacementItemSource::PlayerHand(InventoryAccess::new(
518                Arc::clone(&inv.inventory),
519                hand,
520            )),
521            orientation: PlacementOrientation::Player { rotation, pitch },
522            is_secondary_use_active: player.is_secondary_use_active(),
523        }
524    }
525
526    /// Creates a source backed by a directly borrowed stack.
527    #[must_use]
528    pub const fn direct(
529        player: Option<&'a Player>,
530        hand: InteractionHand,
531        stack: &'a mut ItemStack,
532        orientation: PlacementOrientation,
533        is_secondary_use_active: bool,
534    ) -> Self {
535        Self {
536            player,
537            hand,
538            item: PlacementItemSource::Direct(stack),
539            orientation,
540            is_secondary_use_active,
541        }
542    }
543
544    /// Returns the player associated with this source, if any.
545    #[must_use]
546    pub const fn player(&self) -> Option<&Player> {
547        self.player
548    }
549
550    /// Returns the interaction hand associated with this source.
551    #[must_use]
552    pub const fn hand(&self) -> InteractionHand {
553        self.hand
554    }
555
556    /// Runs `f` with read access to this source's current stack.
557    pub fn with_item<R>(&self, f: impl FnOnce(&ItemStack) -> R) -> R {
558        match &self.item {
559            PlacementItemSource::PlayerHand(inv) => inv.with_item(|item| f(item)),
560            PlacementItemSource::Direct(item) => f(item),
561        }
562    }
563
564    /// Runs `f` with mutable access to this source's current stack.
565    pub fn with_item_mut<R>(&mut self, f: impl FnOnce(&mut ItemStack) -> R) -> R {
566        match &mut self.item {
567            PlacementItemSource::PlayerHand(inv) => inv.with_item(f),
568            PlacementItemSource::Direct(item) => f(item),
569        }
570    }
571}

573/// Access to the player's inventory.
574///
575/// This handle does not hold the inventory lock by itself. Use the closure
576/// methods to keep lock scopes short and avoid carrying an inventory guard
577/// through block behavior, world mutation, or menu opening.
578#[derive(Clone)]
579pub struct InventoryAccess {
580    inventory: Shared<PlayerInventory>,
581    hand: InteractionHand,
582}
583
584impl InventoryAccess {
585    /// Creates a new `InventoryAccess` instance.
586    pub const fn new(inventory: Shared<PlayerInventory>, hand: InteractionHand) -> Self {
587        Self { inventory, hand }
588    }
589
590    /// Runs `f` with mutable access to the item in the player's hand.
591    pub fn with_item<R>(&self, f: impl FnOnce(&mut ItemStack) -> R) -> R {
592        let mut inventory = self.inventory.lock();
593        inventory.mutate_item_in_hand(self.hand, f)
594    }
595
596    /// Runs `f` with mutable access to the player's inventory.
597    pub fn with_inventory<R>(&self, f: impl FnOnce(&mut PlayerInventory) -> R) -> R {
598        let mut inventory = self.inventory.lock();
599        f(&mut inventory)
600    }
601
602    /// Runs `f` with a container guard containing the player's inventory.
603    ///
604    /// Prefer [`Self::with_item`] or [`Self::with_inventory`] unless an operation
605    /// must interoperate with APIs that require `ContainerLockGuard`.
606    pub fn with_guard<R>(&self, f: impl FnOnce(&mut ContainerLockGuard) -> R) -> R {
607        let inv_ref = ContainerRef::from(Arc::clone(&self.inventory));
608        let mut guard = ContainerLockGuard::lock_all(&[&inv_ref]);
609        f(&mut guard)
610    }
611}

696    /// The current world
697    pub world: &'a Arc<World>,
698    /// The entity carrying the item
699    pub owner: &'a dyn Entity,

706enum InventoryTickSource<'a> {
707    PlayerInventory {
708        inventory: &'a Shared<PlayerInventory>,
709        index: usize,
710    },
711    Equipment {
712        owner: &'a dyn LivingEntity,
713        slot: EquipmentSlot,
714    },

716
717impl<'a> InventoryTickContext<'a> {
718    pub(crate) fn tick_player_inventory(world: &'a Arc<World>, player: &'a Player) {
719        for index in 0..PlayerInventory::INVENTORY_SIZE {
720            let (item, selected) = {
721                let inventory = player.inventory.lock();
722                let stack = &inventory.get_items()[index];
723                if stack.is_empty() {
724                    continue;
725                }
726                (
727                    stack.item(),
728                    index == usize::from(inventory.get_selected_slot()),
729                )
730            };
731
732            let mut context = Self {
733                world,
734                owner: player,
735                slot: selected.then_some(EquipmentSlot::MainHand),
736                item,
737                source: InventoryTickSource::PlayerInventory {
738                    inventory: &player.inventory,
739                    index,
740                },
741            };
742            ITEM_BEHAVIORS
743                .get_behavior(item)
744                .inventory_tick(&mut context);
745        }
746    }

748    pub(crate) fn tick_equipment(
749        world: &'a Arc<World>,
750        owner: &'a dyn LivingEntity,
751        slots: impl IntoIterator<Item = EquipmentSlot>,
752    ) {
753        for slot in slots {
754            let mut item = None;
755            owner.with_equipment_slot(slot, &mut |stack| {
756                if !stack.is_empty() {
757                    item = Some(stack.item());
758                }
759            });
760            let Some(item) = item else {
761                continue;
762            };
763
764            let mut context = Self {
765                world,
766                owner,
767                slot: Some(slot),
768                item,
769                source: InventoryTickSource::Equipment { owner, slot },
770            };
771            ITEM_BEHAVIORS
772                .get_behavior(item)
773                .inventory_tick(&mut context);
774        }
775    }
776
777    /// Runs `f` on the stack that is being ticked, or returns `None` if the slot no longer contains it.
778    pub fn with_item<R>(&self, f: impl FnOnce(&mut ItemStack) -> R) -> Option<R> {
779        match self.source {
780            InventoryTickSource::PlayerInventory { inventory, index } => inventory
781                .lock()
782                .with_item_mut(index, |stack| stack.is(self.item).then(|| f(stack))),
783            InventoryTickSource::Equipment { owner, slot } => {
784                let mut f = Some(f);
785                let mut result = None;
786                owner.with_equipment_slot_mut(slot, &mut |stack| {
787                    if stack.is(self.item)
788                        && let Some(f) = f.take()
789                    {
790                        result = Some(f(stack));
791                    }
792                });
793                result
794            }
795        }
796    }
