40/// The kind of melee hit, which selects its attack sound.
41///
42/// Vanilla passes these as separate flags to `attackVisualEffects`, but they are
43/// mutually exclusive: sweeps require a non-critical hit, and stabs are their own path.
44#[derive(Clone, Copy)]
45enum AttackVisualKind {
46    Critical,
47    Sweep,
48    Stab,
49    Normal { full_strength: bool },
50}

52impl AttackVisualKind {
53    const fn melee(critical: bool, sweep: bool, full_strength: bool) -> Self {
54        if critical {
55            Self::Critical
56        } else if sweep {
57            Self::Sweep
58        } else {
59            Self::Normal { full_strength }
60        }
61    }

116    /// Ticks vanilla attack-strength recovery and resets it on main-hand item changes.
117    pub(in crate::player) fn tick_attack_strength(&self) {
118        self.tick_state.lock().advance_attack_strength_ticker();
119
120        let main_hand_item = self.weapon_item();
121
122        let mut last_item = self.last_item_in_main_hand.lock();
123        if ItemStack::matches(&last_item, &main_hand_item) {
124            return;
125        }
126
127        if !ItemStack::is_same_item(&last_item, &main_hand_item) {
128            self.reset_attack_strength_ticker();
129        }
130
131        *last_item = main_hand_item;
132    }

158    fn attack_strength_scale_for_delay(
159        &self,
160        partial_tick: f32,
161        attack_strength_delay: f32,
162    ) -> f32 {
163        let ticker = self.tick_state.lock().attack_strength_ticker() as f32;

184    fn cause_extra_knockback(
185        &self,
186        entity: &dyn Entity,
187        knockback_amount: f64,
188        old_movement: DVec3,
189    ) {
190        if knockback_amount > 0.0 {
191            let yaw_radians = self.rotation().0.to_radians();
192            let yaw_sin = f64::from(trig::sin(yaw_radians));
193            let yaw_cos = f64::from(trig::cos(yaw_radians));
194            if let Some(living_target) = entity.as_living_entity() {
195                living_target.knockback(knockback_amount, yaw_sin, -yaw_cos);
196            } else {
197                entity.push_impulse(DVec3::new(
198                    -yaw_sin * knockback_amount,
199                    0.1,
200                    yaw_cos * knockback_amount,
201                ));
202            }
203
204            let velocity = self.velocity();
205            self.set_velocity(DVec3::new(velocity.x * 0.6, velocity.y, velocity.z * 0.6));
206            self.set_sprinting(false);
207        }
208
209        if entity.entity_type() == &vanilla_entities::PLAYER
210            && entity.hurt_marked()
211            && let Some(player) = self.get_world().players.get_by_entity_id(entity.id())
212        {
213            let velocity = entity.velocity();
214            player.send_packet(CSetEntityMotion::new(entity.id(), velocity));
215            entity.clear_hurt_mark();
216            entity.set_velocity(old_movement);
217        }
218    }

226    /// Returns true if the target box is within the player's attack range for `item_stack`.
227    #[must_use]
228    pub fn is_within_attack_range_with_buffer(
229        &self,
230        item_stack: &ItemStack,
231        aabb: WorldAabb,
232        buffer: f64,
233    ) -> bool {
234        let distance = aabb.distance_to_sqr(self.eye_position()).sqrt();
235        let (min_reach, max_reach, hitbox_margin) =
236            if let Some(attack_range) = item_stack.get_attack_range() {
237                if self.game_mode() == GameType::Creative {
238                    (
239                        attack_range.min_creative_reach,
240                        attack_range.max_creative_reach,
241                        attack_range.hitbox_margin,
242                    )
243                } else {
244                    (
245                        attack_range.min_reach,
246                        attack_range.max_reach,
247                        attack_range.hitbox_margin,
248                    )
249                }
250            } else {
251                (0.0, self.entity_interaction_range() as f32, 0.0)
252            };
253        let min_reach = f64::from(min_reach) - f64::from(hitbox_margin) - buffer;
254        let max_reach = f64::from(max_reach) + f64::from(hitbox_margin) + buffer;
255        distance >= min_reach && distance <= max_reach
256    }

259    #[must_use]

263        buffer: f64,
264    ) -> bool {

269    fn attack_range_for_item(&self, item_stack: &ItemStack) -> (f64, f64, f64) {
270        let Some(attack_range) = item_stack.get_attack_range() else {
271            return (0.0, self.entity_interaction_range(), 0.0);
272        };
273
274        let (min_reach, max_reach) = if self.game_mode() == GameType::Creative {
275            (
276                attack_range.min_creative_reach,
277                attack_range.max_creative_reach,
278            )
279        } else {
280            (attack_range.min_reach, attack_range.max_reach)
281        };
282        (
283            f64::from(min_reach),
284            f64::from(max_reach),
285            f64::from(attack_range.hitbox_margin),
286        )
287    }

289    fn piercing_hit_entities(&self, item_stack: &ItemStack, world: &World) -> Vec<SharedEntity> {
290        let look = self.look_angle();
291        if look.length_squared() <= f64::EPSILON {
292            return Vec::new();
293        }
294
295        let (min_reach, max_reach, hitbox_margin) = self.attack_range_for_item(item_stack);
296        let eye_position = self.eye_position();
297        let from = eye_position + look * min_reach;
298        let movement_extension = self.known_movement().dot(look).max(0.0);
299        let mut to = eye_position + look * (max_reach + movement_extension);
300
301        let block_hit = world.clip(eye_position, to, ClipBlockShape::Collider, ClipFluid::None);
302        if !block_hit.is_miss() {
303            to = block_hit.location;
304            if eye_position.distance_squared(to) < eye_position.distance_squared(from) {
305                return Vec::new();
306            }
307        }
308
309        let search_area = WorldAabb::new(from.x, from.y, from.z, from.x, from.y, from.z)
310            .inflate_xyz(hitbox_margin, hitbox_margin, hitbox_margin)
311            .expand_towards(to - from)
312            .inflate(1.0);
313        let mut hits = world
314            .get_entities_in_aabb_matching(&search_area, |entity| {
315                self.can_piercing_hit_entity(entity)
316            })
317            .into_iter()
318            .filter_map(|entity| {
319                piercing_ray_hit_t(world, entity.bounding_box(), from, to, hitbox_margin)
320                    .map(|hit_t| (hit_t, entity))
321            })
322            .collect::<Vec<_>>();
323        hits.sort_by(|(left, _), (right, _)| left.total_cmp(right));
324        hits.into_iter().map(|(_, entity)| entity).collect()
325    }

327    fn can_piercing_hit_entity(&self, target: &dyn Entity) -> bool {
328        target.id() != self.id()
329            && !target.is_invulnerable()
330            && target
331                .as_living_entity()
332                .is_some_and(|living| living.invulnerable_time() > 0)
333            && target.is_alive()
334            && target.can_be_hit_by_projectile()
335            && target
336                .as_player()
337                .is_none_or(|target_player| self.can_harm_player(target_player))
338            && !self.is_passenger_of_same_vehicle(target)
339    }

341    pub(super) fn piercing_attack(
342        self: &Arc<Self>,
343        item_stack: &ItemStack,
344        piercing_weapon: &PiercingWeapon,
345    ) {
346        let world = self.get_world();
347        let base_damage = self
348            .attributes()
349            .lock()
350            .required_value(vanilla_attributes::ATTACK_DAMAGE) as f32;
351        let mut hit_something = false;
352        for target in self.piercing_hit_entities(item_stack, &world) {
353            hit_something |= self.stab_attack(
354                &target,
355                base_damage,
356                true,
357                piercing_weapon.deals_knockback,
358                piercing_weapon.dismounts,
359            );
360        }
361
362        self.reset_attack_strength_ticker();
363        enchantment_helper::do_post_piercing_attack_effects(&world, self.as_ref());
364        if hit_something {
365            self.play_sound_holder(piercing_weapon.hit_sound.as_ref());
366        }
367        self.play_sound_holder(piercing_weapon.sound.as_ref());
368        self.swing(
369            InteractionHand::MainHand,
370            self.attack_animation(InteractionHand::MainHand),
371            false,
372        );
373    }

375    fn stab_attack(
376        self: &Arc<Self>,
377        target: &SharedEntity,
378        base_damage: f32,
379        deals_damage: bool,
380        deals_knockback: bool,
381        dismounts: bool,
382    ) -> bool {
383        let entity = target.as_ref();
384        if self.cannot_attack(entity) {
385            return false;
386        }
387
388        let attacking_item = self.weapon_item();
389        let damage_source = self.attack_damage_source(&attacking_item);
390        let enchantment_context = EnchantmentDamageContext::new(
391            entity.entity_type(),
392            Some(self.entity_type()),
393            Some(self.entity_type()),
394            &damage_source,
395        );
396        let enchanted_damage =
397            enchantment_helper::modify_damage(&attacking_item, &enchantment_context, base_damage);
398        let attack_strength_scale = self.attack_strength_scale(0.5);
399        let magic_boost = attack_strength_scale * (enchanted_damage - base_damage);
400        let base_damage = base_damage * Self::base_damage_scale_factor(attack_strength_scale);
401        if deals_knockback && self.deflect_projectile(entity) {
402            return true;
403        }
404        let damage = base_damage + magic_boost;
405        let old_movement = entity.velocity();
406        let mut affected = deals_knockback;
407        let damage_dealt = deals_damage
408            && entity
409                .level()
410                .is_some_and(|world| entity.hurt(&world, &damage_source, damage));
411        affected |= damage_dealt;
412        if deals_knockback {
413            self.cause_extra_knockback(
414                entity,
415                0.4 + Self::get_knockback(0.0, &attacking_item, &enchantment_context),
416                old_movement,
417            );
418        }
419        if dismounts && entity.is_passenger() {
420            affected = true;
421            entity.stop_riding();
422        }
423
424        if !affected {
425            return false;
426        }
427
428        self.attack_visual_effects(entity, AttackVisualKind::Stab, magic_boost);
429        self.item_attack_interaction(target, attacking_item, &damage_source, damage_dealt);
430        self.set_last_hurt_mob(Some(target));
431        self.cause_food_exhaustion(food_constants::EXHAUSTION_ATTACK);
432        true
433    }

448    /// Redirects a redirectable projectile along this player's look direction
449    /// instead of attacking it.
450    fn deflect_projectile(self: &Arc<Self>, entity: &dyn Entity) -> bool {
451        if !REGISTRY.entity_types.is_in_tag(
452            entity.entity_type(),
453            &EntityTypeTag::REDIRECTABLE_PROJECTILE,
454        ) {
455            return false;
456        }
457        let Some(projectile) = entity.as_projectile() else {
458            return false;
459        };
460        let owner: SharedEntity = Arc::<Player>::clone(self);
461        if !projectile.deflect(
462            ProjectileDeflection::AimDeflect,
463            Some(self.as_ref()),
464            Some(EntityReference::from_entity(&owner)),
465            true,
466        ) {
467            return false;
468        }
469        self.play_attack_sound(&sound_events::ENTITY_PLAYER_ATTACK_NODAMAGE);
470        true
471    }

478    /// Attacks an entity with the player's main-hand base damage.
479    ///
480    /// Returns `true` if the target accepted damage.
481    #[must_use]
482    pub fn attack(self: &Arc<Self>, target: &SharedEntity) -> bool {
483        let entity = target.as_ref();
484        if self.cannot_attack(entity) {
485            return false;
486        }
487
488        let attacking_item = self.weapon_item();
489        let (attack_damage, attack_speed, attack_knockback) = {
490            let attributes = self.attributes().lock();
491            (
492                attributes.required_value(vanilla_attributes::ATTACK_DAMAGE) as f32,
493                attributes.required_value(vanilla_attributes::ATTACK_SPEED),
494                attributes.required_value(vanilla_attributes::ATTACK_KNOCKBACK),
495            )
496        };
497        let attack_strength_delay = Self::attack_strength_delay_from_speed(attack_speed);
498        let attack_strength_scale =
499            self.attack_strength_scale_for_delay(0.5, attack_strength_delay);
500        let damage_source = self.attack_damage_source(&attacking_item);
501        let enchantment_context = EnchantmentDamageContext::new(
502            entity.entity_type(),
503            Some(self.entity_type()),
504            Some(self.entity_type()),
505            &damage_source,
506        );
507        let enchanted_damage =
508            enchantment_helper::modify_damage(&attacking_item, &enchantment_context, attack_damage);
509        let magic_boost = attack_strength_scale * (enchanted_damage - attack_damage);
510        let mut base_damage = attack_damage * Self::base_damage_scale_factor(attack_strength_scale);
511        self.reset_attack_strength_ticker();
512        if self.deflect_projectile(entity) {
513            return false;
514        }
515
516        if base_damage <= 0.0 && magic_boost <= 0.0 {
517            // Vanilla still runs post-piercing attack effects when no damage is dealt.
518            let world = self.get_world();
519            enchantment_helper::do_post_piercing_attack_effects(&world, self.as_ref());
520            return false;
521        }
522
523        let full_strength_attack = attack_strength_scale > FULL_ATTACK_STRENGTH_SCALE_THRESHOLD;
524        let knockback_attack = self.is_sprinting() && full_strength_attack;
525        if knockback_attack {
526            self.play_attack_sound(&sound_events::ENTITY_PLAYER_ATTACK_KNOCKBACK);
527        }
528        base_damage += ITEM_BEHAVIORS
529            .get_behavior(attacking_item.item())
530            .get_attack_damage_bonus(self.as_ref(), entity, base_damage, &damage_source);
531        let critical_attack = full_strength_attack && self.can_critical_attack(entity);
532        if critical_attack {
533            base_damage *= CRITICAL_ATTACK_DAMAGE_MULTIPLIER;
534        }
535        let total_damage = base_damage + magic_boost;
536        let sweep_attack = self.is_sweep_attack(
537            &attacking_item,
538            full_strength_attack,
539            critical_attack,
540            knockback_attack,
541        );
542
543        let old_entity_living_health = entity
544            .as_living_entity()
545            .map_or(0.0, LivingEntity::get_health);
546
547        let old_movement = entity.velocity();
548        let Some(target_world) = entity.level() else {
549            return false;
550        };
551        let was_hurt = entity.hurt(&target_world, &damage_source, total_damage);
552        if was_hurt {
553            let sprint_knockback = if knockback_attack { 0.5 } else { 0.0 };
554            self.cause_extra_knockback(
555                entity,
556                Self::get_knockback(attack_knockback, &attacking_item, &enchantment_context)
557                    + sprint_knockback,
558                old_movement,
559            );
560            if sweep_attack {
561                self.do_sweep_attack(entity, base_damage, &damage_source, attack_strength_scale);
562            }
563            let visual_kind =
564                AttackVisualKind::melee(critical_attack, sweep_attack, full_strength_attack);
565            self.attack_visual_effects(entity, visual_kind, magic_boost);
566            self.set_last_hurt_mob(Some(target));
567            self.item_attack_interaction(target, attacking_item, &damage_source, true);
568            self.damage_stats_and_hearts(entity, old_entity_living_health);
569            self.cause_food_exhaustion(food_constants::EXHAUSTION_ATTACK);
570        } else {
571            self.play_attack_sound(&sound_events::ENTITY_PLAYER_ATTACK_NODAMAGE);
572        }
573
574        let world = self.get_world();
575        enchantment_helper::do_post_piercing_attack_effects(&world, self.as_ref());
576        was_hurt
577    }

579    fn can_critical_attack(&self, target: &dyn Entity) -> bool {
580        self.fall_distance() > 0.0
581            && !self.on_ground()
582            && !self.on_climbable()
583            && !self.is_in_water()
584            && !self.is_mobility_restricted()
585            && !self.is_passenger()
586            && target.as_living_entity().is_some()
587            && !self.is_sprinting()
588    }

594    fn is_sweep_attack(
595        &self,
596        main_hand_item: &ItemStack,
597        full_strength_attack: bool,
598        critical_attack: bool,
599        knockback_attack: bool,
600    ) -> bool {
601        if !full_strength_attack || critical_attack || knockback_attack || !self.on_ground() {
602            return false;
603        }
604
605        let known_movement = self.known_movement();
606        let approximate_speed_sq =
607            known_movement.x * known_movement.x + known_movement.z * known_movement.z;
608        let movement_speed = self
609            .attributes()
610            .lock()
611            .required_value(vanilla_attributes::MOVEMENT_SPEED);
612        let max_speed_for_sweep_attack =
613            f64::from(movement_speed as f32) * MAX_SWEEP_SPEED_MOVEMENT_SPEED_FACTOR;
614        approximate_speed_sq < max_speed_for_sweep_attack * max_speed_for_sweep_attack
615            && REGISTRY
616                .items
617                .is_in_tag(main_hand_item.item(), &ItemTag::SWORDS)
618    }

620    fn do_sweep_attack(
621        &self,
622        target: &dyn Entity,
623        base_damage: f32,
624        damage_source: &DamageSource,
625        attack_strength_scale: f32,
626    ) {
627        self.play_attack_sound(&sound_events::ENTITY_PLAYER_ATTACK_SWEEP);
628        let world = self.get_world();
629        let sweeping_damage_ratio =
630            self.attributes()
631                .lock()
632                .required_value(vanilla_attributes::SWEEPING_DAMAGE_RATIO) as f32;
633        let sweep_damage = 1.0 + sweeping_damage_ratio * base_damage;
634        let yaw_radians = self.rotation().0.to_radians();
635        let yaw_sin = f64::from(trig::sin(yaw_radians));
636        let yaw_cos = f64::from(trig::cos(yaw_radians));
637
638        let sweep_area = target.bounding_box().inflate_xyz(
639            SWEEP_AREA_INFLATE_XZ,
640            SWEEP_AREA_INFLATE_Y,
641            SWEEP_AREA_INFLATE_XZ,
642        );
643        let nearby_entities = world.get_entities_in_aabb_matching(&sweep_area, |nearby| {
644            nearby.as_living_entity().is_some() && !nearby.is_spectator()
645        });
646        let position = self.position();
647        for nearby in nearby_entities {
648            if nearby.id() == self.id()
649                || nearby.id() == target.id()
650                || self.is_allied_to(nearby.as_ref())
651                || nearby.is_marker_armor_stand()
652                || nearby.distance_to_sqr(position) >= SWEEP_MAX_DISTANCE_SQ
653            {
654                continue;
655            }
656            let Some(living_nearby) = nearby.as_living_entity() else {
657                continue;
658            };
659
660            // Re-read per target: an earlier sweep hit can change the hand (e.g. thorns death).
661            let weapon = self.weapon_item();
662            let enchantment_context = EnchantmentDamageContext::new(
663                nearby.entity_type(),
664                Some(self.entity_type()),
665                Some(self.entity_type()),
666                damage_source,
667            );
668            let enchanted_damage =
669                enchantment_helper::modify_damage(&weapon, &enchantment_context, sweep_damage)
670                    * attack_strength_scale;
671            if !nearby.hurt(&world, damage_source, enchanted_damage) {
672                continue;
673            }
674            living_nearby.knockback(SWEEP_KNOCKBACK, yaw_sin, -yaw_cos);
675            let post_attack_context = EnchantmentPostAttackContext::new(&nearby, damage_source);
676            enchantment_helper::do_post_attack_effects_with_item_source(
677                &world,
678                nearby.as_ref(),
679                weapon,
680                &post_attack_context,
681            );
682        }
683
684        let dx = -yaw_sin;
685        let dz = yaw_cos;
686        let height = f64::from(self.base().dimensions().height);
687        world.send_particles(
688            ParticleData::simple(&vanilla_particle_types::SWEEP_ATTACK),
689            DVec3::new(position.x + dx, position.y + height * 0.5, position.z + dz),
690            0,
691            DVec3::new(dx, 0.0, dz),
692            0.0,
693        );
694    }

696    fn attack_visual_effects(&self, target: &dyn Entity, kind: AttackVisualKind, magic_boost: f32) {
697        match kind {
698            AttackVisualKind::Critical => {
699                self.play_attack_sound(&sound_events::ENTITY_PLAYER_ATTACK_CRIT);
700                self.send_to_tracking_players_and_self(CAnimate::critical_hit(target.id()));
701            }
702            AttackVisualKind::Normal { full_strength } => {
703                self.play_attack_sound(if full_strength {
704                    &sound_events::ENTITY_PLAYER_ATTACK_STRONG
705                } else {
706                    &sound_events::ENTITY_PLAYER_ATTACK_WEAK
707                });
708            }
709            AttackVisualKind::Sweep | AttackVisualKind::Stab => {}
710        }
711
712        if magic_boost > 0.0 {
713            self.send_to_tracking_players_and_self(CAnimate::magic_critical_hit(target.id()));
714        }
715    }

723    fn item_attack_interaction(
724        &self,
725        target: &SharedEntity,
726        attacking_item: ItemStack,
727        damage_source: &DamageSource,
728        apply_to_target: bool,
729    ) {
730        let entity = target.as_ref();
731        // Vanilla hands a part's `parentMob` to the weapon hooks; enchantments keep the part.
732        let parent = entity.as_part_entity().and_then(PartEntity::parent);
733        let hurt_target = parent.as_deref().unwrap_or(entity);
734        let post_attack_context = EnchantmentPostAttackContext::new(target, damage_source);
735        let item_hurt_enemy = {
736            let mut inventory = self.inventory.lock();
737            inventory.mutate_item_in_hand(InteractionHand::MainHand, |stack| {
738                if stack.is_empty() {
739                    return false;
740                }
741                let behavior = ITEM_BEHAVIORS.get_behavior(stack.item());
742                if let Some(living_target) = hurt_target.as_living_entity() {
743                    behavior.hurt_enemy(stack, living_target, self);
744                }
745                stack.get_weapon().is_some()
746            })
747        };
748
749        if apply_to_target {
750            let world = self.get_world();
751            enchantment_helper::do_post_attack_effects_with_item_source(
752                &world,
753                entity,
754                attacking_item,
755                &post_attack_context,
756            );
757        }
758
759        if !item_hurt_enemy {
760            return;
761        }
762
763        let Some(living_target) = hurt_target.as_living_entity() else {
764            return;
765        };
766        let has_infinite_materials = self.has_infinite_materials();
767        let mut inventory = self.inventory.lock();
768        inventory.mutate_item_in_hand(InteractionHand::MainHand, |stack| {
769            if stack.is_empty() {
770                return;
771            }
772            let behavior = ITEM_BEHAVIORS.get_behavior(stack.item());
773            behavior.post_hurt_enemy(stack, living_target, self);
774            if let Some(damage) = behavior.item_damage_per_attack(stack) {
775                stack.hurt_and_break(damage, has_infinite_materials);
776            }
777        });
778    }

780    /// Interacts with an entity using the held item.
781    pub fn interact_on(
782        &self,
783        entity: &dyn Entity,
784        hand: InteractionHand,
785        location: DVec3,
786    ) -> InteractionResult {
787        if self.is_spectator() {
788            // TODO: Open entity menu providers in spectator once that foundation exists.
789            return InteractionResult::Pass;
790        }
791
792        let inventory_access = InventoryAccess::new(Arc::clone(&self.inventory), hand);
793        let original_count = inventory_access.with_item(|item| item.count);
794        let result = entity.interact(self, hand, location);
795
796        if self.has_infinite_materials() {
797            inventory_access.with_item(|item| {
798                if item.count < original_count {
799                    item.count = original_count;
800                }
801            });
802        }
803
804        if result.consumes_action() {
805            return result;
806        }
807
808        if inventory_access.with_item(|item| item.is_empty()) {
809            return InteractionResult::Pass;
810        }
811        let Some(living_entity) = entity.as_living_entity() else {
812            return InteractionResult::Pass;
813        };
814        let result = living_entity.interact_living_entity_with_equippable(self, hand);
815        if self.has_infinite_materials() {
816            inventory_access.with_item(|item| {
817                if item.count < original_count {
818                    item.count = original_count;
819                }
820            });
821        }
822        if result.consumes_action() {
823            return result;
824        }
825
826        let item_ref = inventory_access.with_item(|item| item.item());
827        let item_behavior = ITEM_BEHAVIORS.get_behavior(item_ref);
828        let result = inventory_access.with_item(|item| {
829            item_behavior.interact_living_entity(item, self, living_entity, hand)
830        });
831        if self.has_infinite_materials() {
832            inventory_access.with_item(|item| {
833                if item.count < original_count {
834                    item.count = original_count;
835                }
836            });
837        }
838        result
839    }

841    /// Handles a client request to attack an entity.
842    pub fn handle_attack(self: &Arc<Self>, packet: SAttack) {
843        if !self.has_client_loaded() || self.is_spectator() {
844            return;
845        }
846
847        let world = self.get_world();
848        let Some(target) = world.get_accessible_entity_or_part_by_id(packet.entity_id) else {
849            return;
850        };
851
852        self.reset_last_action_time();
853
854        let target_pos = target.block_position();
855        if !world.world_border_snapshot().is_within_bounds_with_margin(
856            f64::from(target_pos.x()),
857            f64::from(target_pos.z()),
858            0.0,
859        ) {
860            return;
861        }
862
863        let main_hand_item = self.weapon_item();
864
865        if !self.is_within_attack_range_with_buffer(
866            &main_hand_item,
867            target.bounding_box(),
868            ATTACK_RANGE_BUFFER,
869        ) {
870            return;
871        }
872
873        if main_hand_item.get_piercing_weapon().is_some() {
874            return;
875        }
876
877        if Self::is_invalid_attack_target(self.id(), target.id(), target.entity_type()) {
878            self.disconnect(Self::invalid_entity_attacked_message());
879            log::warn!(
880                "Player {} tried to attack an invalid entity",
881                self.gameprofile.name()
882            );
883            return;
884        }
885
886        if self.cannot_attack_with_item(&main_hand_item, 5) {
887            return;
888        }
889
890        let _ = self.attack(&target);
891    }
892
893    pub(super) fn cannot_attack_with_item(&self, item_stack: &ItemStack, tolerance: i32) -> bool {
894        let required_strength = item_stack.minimum_attack_charge();
895        if required_strength <= 0.0 {
896            return false;
897        }
898
899        let optimistic_strength = {
900            let ticker = self.tick_state.lock().attack_strength_ticker() + tolerance;
901            ticker as f32 / self.current_item_attack_strength_delay()
902        };
903        optimistic_strength < required_strength
904    }

906    pub(super) fn is_invalid_attack_target(
907        player_id: i32,
908        target_id: i32,
909        target_type: EntityTypeRef,
910    ) -> bool {
911        target_id == player_id
912            || target_type == &vanilla_entities::ITEM
913            || target_type == &vanilla_entities::EXPERIENCE_ORB
914    }

916    /// Handles a client request to interact with an entity.
917    pub fn handle_interact(&self, packet: SInteract) {
918        if !self.has_client_loaded() {
919            return;
920        }
921
922        let world = self.get_world();
923        self.reset_last_action_time();
924        let target = world.get_accessible_entity_or_part_by_id(packet.entity_id);
925        self.set_crouching(packet.using_secondary_action);
926        let Some(target) = target else {
927            return;
928        };
929
930        let target_pos = target.block_position();
931        if !world.world_border_snapshot().is_within_bounds_with_margin(
932            f64::from(target_pos.x()),
933            f64::from(target_pos.z()),
934            0.0,
935        ) {
936            return;
937        }
938
939        if !self.is_within_entity_interaction_range_with_buffer(
940            target.bounding_box(),
941            ENTITY_INTERACTION_RANGE_BUFFER,
942        ) {
943            return;
944        }
945
946        let animation = self.interact_animation(packet.hand);
947        let result = self.interact_on(target.as_ref(), packet.hand, packet.location);
948        self.swing_after_interaction(packet.hand, animation, result);
949        self.broadcast_inventory_changes();
950    }

952    /// Awards stats and sends particles when an entity gets attacked by this player.
953    pub fn damage_stats_and_hearts(&self, entity: &dyn Entity, old_entity_living_health: f32) {
954        const PARTICLES_PER_HEALTH: f32 = 0.5;
955        const PARTICLE_SPREAD_XZ: f64 = 0.1;
956        const PARTICLE_SPEED: f64 = 0.2;
957
958        if let Some(entity) = entity.as_living_entity() {
959            let actual_damage = old_entity_living_health - entity.get_health();
960            self.award_custom_stat_with_count(
961                &vanilla_custom_stats::DAMAGE_DEALT,
962                round_f32(actual_damage * 10.0),
963            );
964
965            let count = (actual_damage * 0.5).round() as i32;
966            let offset = DVec3::new(
967                0.0,
968                f64::from(entity.base().dimensions().height * PARTICLES_PER_HEALTH),
969                0.0,
970            );
971            self.get_world().send_particles(
972                ParticleData::simple(&vanilla_particle_types::DAMAGE_INDICATOR),
973                entity.position().add(offset),
974                count,
975                DVec3::new(PARTICLE_SPREAD_XZ, 0.0, PARTICLE_SPREAD_XZ),
976                PARTICLE_SPEED,
977            );
978        }
979    }
