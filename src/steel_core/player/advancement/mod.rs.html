23/// Manages a player's collection of advancements.
24///
25/// This handles saving, loading, and tracking the state of granted / revoked advancements.
26#[derive(Debug, Clone)]
27pub struct PlayerAdvancement {
28    /// the progress of the player for each advancement
29    pub(crate) progress: AdvancementProgressMap,
30    is_first_packet: bool,
31    roots_to_update: FxHashSet<AdvancementNodeRef>,
32    visible: FxHashSet<AdvancementRef>,
33    progress_changed: FxHashSet<AdvancementRef>,
34    /// represent the las selected advancement tab
35    pub last_selected_tab: Option<AdvancementRef>,
36}
37
38impl Default for PlayerAdvancement {
39    fn default() -> Self {
40        Self {
41            is_first_packet: true,
42            progress: AdvancementProgressMap::default(),
43            roots_to_update: FxHashSet::default(),
44            visible: FxHashSet::default(),
45            progress_changed: FxHashSet::default(),
46            last_selected_tab: Option::default(),
47        }
48    }
49}
50
51impl PlayerAdvancement {
52    /// Award a criterion to a player. If this completes the advancement,
53    /// grant the advancement along with its rewards and display the corresponding chat message.
54    pub(crate) fn award(
55        &mut self,
56        player: &Player,
57        advancement: AdvancementRef,
58        criterion: &str,
59    ) -> bool {
60        let mut result = false;
61        let progress = self.progress.get_mut_or_start_progress(advancement);
62        let was_done = progress.is_done();
63        if progress.grant_progress(criterion) {
64            //TODO: register advancement listener
65            self.progress_changed.insert(advancement);
66            result = true;
67            if !was_done && progress.is_done() {
68                grant_reward(player, advancement.rewards());
69                if let Some(display) = advancement.display()
70                    && display.announce_chat
71                    && player
72                        .level()
73                        .is_some_and(|level| level.get_game_rule(&SHOW_ADVANCEMENT_MESSAGES))
74                {
75                    player.server().broadcast_system_chat(
76                        &display
77                            .frame_type
78                            .create_announcement(advancement, player.display_name()),
79                        None,
80                    );
81                }
82            }
83        }
84
85        if !was_done && progress.is_done() {
86            self.mark_for_visibility_update(advancement);
87        }
88        result
89    }
90
91    /// revoke the specified criterion for a player.
92    pub fn revoke(&mut self, advancement: AdvancementRef, criterion: &str) -> bool {
93        let mut result = false;
94        let progress = self.progress.get_mut_or_start_progress(advancement);
95        let was_done = progress.is_done();
96        if progress.revoke_progress(criterion) {
97            //TODO: unregister advancement listener
98            self.progress_changed.insert(advancement);
99            result = true;
100        }
101
102        if was_done && !progress.is_done() {
103            self.mark_for_visibility_update(advancement);
104        }
105
106        result
107    }
108
109    /// mark the advancement to be sent to the client next tick
110    fn mark_for_visibility_update(&mut self, advancement: AdvancementRef) {
111        let node = REGISTRY.advancements.by_key(advancement.key());
112        if let Some(node) = node {
113            self.roots_to_update
114                .insert(REGISTRY.advancements.root(node));
115        }
116    }
117
118    /// modify the added and the removed from the available tree status
119    fn update_tree_visibility(
120        &mut self,
121        root: AdvancementNodeRef,
122        added: &mut Vec<AdvancementNodeRef>,
123        removed: &mut Vec<Identifier>,
124    ) {
125        visibility_evaluator::evaluate_visibility(
126            root,
127            self,
128            &mut |player_advancement, node| {
129                player_advancement
130                    .progress
131                    .get_mut_or_start_progress(node.value())
132                    .is_done()
133            },
134            &mut move |player_advancement, node, should_be_visible| {
135                let advancement = node.value();
136                if should_be_visible {
137                    if player_advancement.visible.insert(advancement) {
138                        added.push(node);
139                        if player_advancement.progress.has_progress(advancement) {
140                            player_advancement.progress_changed.insert(advancement);
141                        }
142                    }
143                } else if player_advancement.visible.remove(advancement) {
144                    removed.push(advancement.key().clone());
145                }
146            },
147        );
148    }
149
150    /// send the advancement update packet to a player with the updated tree
151    pub fn flush_dirty(&mut self, player: &Player, show_advancement: bool) {
152        if self.is_first_packet
153            || !self.roots_to_update.is_empty()
154            || !self.progress_changed.is_empty()
155        {
156            let mut progress: FxHashMap<Identifier, &AdvancementProgress> = FxHashMap::default();
157            let mut added: Vec<AdvancementNodeRef> = Vec::new();
158            let mut removed: Vec<Identifier> = Vec::new();
159            for root in self.roots_to_update.clone() {
160                self.update_tree_visibility(root, &mut added, &mut removed);
161            }
162            self.roots_to_update.clear();
163            for advancement in &self.progress_changed {
164                if self.visible.contains(advancement) {
165                    progress.insert(
166                        advancement.key().clone(),
167                        self.progress.get_progress(advancement).unwrap_or_else(|| {
168                            unreachable!("progress changed should only contains valid advancement")
169                        }),
170                    );
171                }
172            }
173            self.progress_changed.clear();
174            if !progress.is_empty() || !added.is_empty() || !removed.is_empty() {
175                let parsed_progress: Vec<AdvancementProgressData> = progress
176                    .into_iter()
177                    .map(|(key, val)| AdvancementProgressData {
178                        id: key,
179                        progress: val
180                            .criteria
181                            .iter()
182                            .map(|(key, val)| Criteria {
183                                criterion_id: key.clone(),
184                                achieve_date: val.0.map(|time| {
185                                    time.duration_since(UNIX_EPOCH)
186                                        .map_or(0, |d| d.as_millis() as i64)
187                                }),
188                            })
189                            .collect(),
190                    })
191                    .collect();
192                player.send_packet(CUpdateAdvancements::new(
193                    self.is_first_packet,
194                    added,
195                    parsed_progress,
196                    removed,
197                    show_advancement,
198                ));
199            }
200        }
201        self.is_first_packet = false;
202    }
203
204    /// change the selected tab and update the client with it
205    pub fn set_selected_tab(&mut self, player: &Player, advancement: Option<AdvancementRef>) {
206        let old = self.last_selected_tab;
207        if let Some(value) = advancement
208            && value.is_root()
209            && value.display().is_some()
210        {
211            self.last_selected_tab = advancement;
212        } else {
213            self.last_selected_tab = None;
214        }
215
216        if old != self.last_selected_tab {
217            player.send_packet(CSelectAdvancementsTab::new(
218                self.last_selected_tab.map(|e| e.key().clone()),
219            ));
220        }
221    }
