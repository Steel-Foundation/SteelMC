1//! Server-side validation of client message acknowledgements.
2
3use std::collections::VecDeque;
4
5use steel_protocol::packets::game::LastSeenMessagesUpdate;
6
7const CHECKSUM_HASH_MULTIPLIER: u8 = 31;
8const NO_CHECKSUM: u8 = 0;
9
10/// Entry tracking a sent message signature
11#[derive(Debug, Clone)]
12struct TrackedEntry {
13    signature: Option<Box<[u8]>>,
14    pending: bool, // true if not yet acknowledged by client
15}
16
17/// Validates that the client is properly acknowledging messages sent by the server
18#[derive(Debug)]
19pub struct LastSeenMessagesValidator {
20    tracked_messages: VecDeque<Option<TrackedEntry>>,
21    last_pending_signature: Option<Box<[u8]>>,
22}
23
24impl Default for LastSeenMessagesValidator {
25    fn default() -> Self {
26        Self::new()
27    }
28}
29
30impl LastSeenMessagesValidator {
31    /// Creates a new validator
32    #[must_use]
33    pub fn new() -> Self {
34        let mut tracked_messages = VecDeque::with_capacity(LastSeenMessagesUpdate::MAX_ENTRIES);
35        for _ in 0..LastSeenMessagesUpdate::MAX_ENTRIES {
36            tracked_messages.push_back(None);
37        }
38        Self {
39            tracked_messages,
40            last_pending_signature: None,
41        }
42    }
43
44    /// Adds a pending message signature that the client should acknowledge
45    /// Matches vanilla's deduplication: only adds if different from last pending message
46    pub fn add_pending(&mut self, signature: Option<Box<[u8]>>) {
47        // Only add if this signature is different from the last one we added
48        // This prevents duplicates when the same message is processed multiple times
49        if signature.as_ref() != self.last_pending_signature.as_ref() {
50            let entry = TrackedEntry {
51                signature: signature.clone(),
52                pending: true,
53            };
54            self.tracked_messages.push_back(Some(entry));
55            self.last_pending_signature = signature;
56        }
57    }
58
59    /// Gets the number of tracked messages
60    #[must_use]
61    pub const fn tracked_count(&self) -> usize {
62        self.tracked_messages.len()
63    }
64
65    /// Applies an offset (removes old acknowledged messages from tracking)
66    pub fn apply_offset(&mut self, offset: i32) -> Result<(), String> {
67        let removable = self
68            .tracked_messages
69            .len()
70            .saturating_sub(LastSeenMessagesUpdate::MAX_ENTRIES);
71        if offset < 0 || offset as usize > removable {
72            return Err(format!(
73                "Advanced last seen window by {offset} messages, but expected at most {removable}"
74            ));
75        }
76        for _ in 0..offset {
77            self.tracked_messages.pop_front();
78        }
79        Ok(())
80    }
81
82    /// Applies an acknowledgement update from the client
83    /// acknowledged: `BitSet` of 20 bits indicating which messages in the window are acknowledged
84    /// offset: How many old messages to remove from the window
85    /// checksum: Optional checksum for validation (0 = skip checksum)
86    pub fn apply_update(
87        &mut self,
88        acknowledged: [u8; LastSeenMessagesUpdate::ACKNOWLEDGED_BYTES],
89        offset: i32,
90        checksum: u8,
91    ) -> Result<Vec<Box<[u8]>>, String> {
92        let max_entries = LastSeenMessagesUpdate::MAX_ENTRIES;
93        let bits_per_byte = u8::BITS as usize;
94        let used_bits_in_last_byte = max_entries % bits_per_byte;
95        let last_byte = acknowledged[LastSeenMessagesUpdate::ACKNOWLEDGED_BYTES - 1];
96        if used_bits_in_last_byte != 0 && last_byte >> used_bits_in_last_byte != 0 {
97            return Err(format!(
98                "Last seen update contained acknowledgements outside the {max_entries}-message window"
99            ));
100        }
101
102        self.apply_offset(offset)?;
103
104        let count = acknowledged
105            .iter()
106            .map(|byte| byte.count_ones())
107            .sum::<u32>() as usize;
108        let mut acknowledged_signatures = Vec::with_capacity(count);
109
110        for (index, tracked) in self
111            .tracked_messages
112            .iter_mut()
113            .take(max_entries)
114            .enumerate()
115        {
116            let is_acknowledged =
117                (acknowledged[index / bits_per_byte] & (1 << (index % bits_per_byte))) != 0;
118
119            if is_acknowledged {
120                let Some(entry) = tracked else {
121                    return Err(format!(
122                        "Last seen update acknowledged unknown or previously ignored message at index {index}"
123                    ));
124                };
125
126                entry.pending = false;
127                if let Some(signature) = &entry.signature {
128                    acknowledged_signatures.push(signature.clone());
129                }
130            } else {
131                if tracked.as_ref().is_some_and(|entry| !entry.pending) {
132                    return Err(format!(
133                        "Last seen update ignored previously acknowledged message at index {index}"
134                    ));
135                }
136
137                *tracked = None;
138            }
139        }
140
141        if checksum != NO_CHECKSUM {
142            let computed = acknowledged_signatures.iter().fold(1_u8, |sum, signature| {
143                let hash = signature.iter().fold(1_u8, |hash, byte| {
144                    hash.wrapping_mul(CHECKSUM_HASH_MULTIPLIER)
145                        .wrapping_add(*byte)
146                });
147
148                sum.wrapping_mul(CHECKSUM_HASH_MULTIPLIER)
149                    .wrapping_add(hash)
150            });
151
152            let expected = if computed == NO_CHECKSUM { 1 } else { computed };
153            if checksum != expected {
154                return Err("Checksum mismatch on last seen update".to_owned());
155            }
156        }
157
158        Ok(acknowledged_signatures)
159    }
160}
