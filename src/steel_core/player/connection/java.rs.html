60/// Cross-player concurrency permitted for a scheduled packet handler.
61#[derive(Clone, Copy, Debug, Eq, PartialEq)]
62pub(crate) enum ScheduledPacketExecution {
63    /// The handler may overlap handlers for other players, but never its own player lane.
64    ///
65    /// Shared mutations must be fully linearized by their resource locks, and the handler must
66    /// tolerate cross-player execution order differing from packet submission order.
67    PlayerLocal,
68    /// The handler may overlap player-local work, but not another serialized handler. Serialized
69    /// handlers start in global packet submission order.
70    Serialized,
71    /// The handler is a global submission-order barrier and must not overlap scheduled work.
72    Exclusive,
73}

150    /// Returns the handler's audited cross-player concurrency class.
151    ///
152    /// This match is intentionally exhaustive so every newly implemented packet requires an
153    /// explicit concurrency decision.
154    pub(crate) const fn execution(&self) -> ScheduledPacketExecution {
155        match &self.0 {
156            // These handlers touch player-owned state or use an individually linearizable shared
157            // operation. The player lane preserves same-player order.
158            ScheduledPlayPacketKind::ChatSessionUpdate(_)
159            | ScheduledPlayPacketKind::ClientInformation(_)
160            | ScheduledPlayPacketKind::ClientTickEnd
161            | ScheduledPlayPacketKind::PlayerLoaded
162            | ScheduledPlayPacketKind::ChatCommand(_)
163            | ScheduledPlayPacketKind::CommandSuggestion(_)
164            | ScheduledPlayPacketKind::ContainerClose(_)
165            | ScheduledPlayPacketKind::SetCreativeModeSlot(_)
166            | ScheduledPlayPacketKind::PlayerInput(_)
167            | ScheduledPlayPacketKind::PlayerAbilities(_)
168            | ScheduledPlayPacketKind::SetCarriedItem(_)
169            | ScheduledPlayPacketKind::Swing(_)
170            | ScheduledPlayPacketKind::PickItemFromBlock(_)
171            | ScheduledPlayPacketKind::ClientCommand(_) => ScheduledPacketExecution::PlayerLocal,
172            ScheduledPlayPacketKind::PlayerCommand(packet) => match packet.action {
173                PlayerCommandAction::StartSprinting
174                | PlayerCommandAction::StopSprinting
175                | PlayerCommandAction::StartFallFlying => ScheduledPacketExecution::PlayerLocal,
176                PlayerCommandAction::LeaveBed => ScheduledPacketExecution::Serialized,
177                // These handlers are not implemented, so their eventual vehicle transaction
178                // cannot yet be audited against concurrently player-local work.
179                PlayerCommandAction::StartRidingJump
180                | PlayerCommandAction::StopRidingJump
181                | PlayerCommandAction::OpenVehicleInventory => ScheduledPacketExecution::Exclusive,
182            },
183            ScheduledPlayPacketKind::PlayerAction(packet) => match packet.action {
184                PlayerAction::AbortDestroyBlock | PlayerAction::SwapItemWithOffhand => {
185                    ScheduledPacketExecution::PlayerLocal
186                }
187                PlayerAction::StartDestroyBlock
188                | PlayerAction::StopDestroyBlock
189                | PlayerAction::DropAllItems
190                | PlayerAction::DropItem => ScheduledPacketExecution::Serialized,
191                // Active-use release may invoke item behavior and mutate inventory, while stab
192                // spans independently locked targets; neither can overlap player-local work.
193                PlayerAction::ReleaseUseItem | PlayerAction::Stab => {
194                    ScheduledPacketExecution::Exclusive
195                }
196            },
197            // Position, world, menu, chat, and domain mutations may overlap player-local work but
198            // retain one global mutation order matching the packet submission order.
199            ScheduledPlayPacketKind::AcceptTeleportation(_)
200            | ScheduledPlayPacketKind::Chat(_)
201            | ScheduledPlayPacketKind::ChatAck(_)
202            | ScheduledPlayPacketKind::MovePlayer(_)
203            | ScheduledPlayPacketKind::MoveVehicle(_)
204            | ScheduledPlayPacketKind::ContainerClick(_)
205            | ScheduledPlayPacketKind::RenameItem(_)
206            | ScheduledPlayPacketKind::UseItemOn(_)
207            | ScheduledPlayPacketKind::UseItem(_)
208            | ScheduledPlayPacketKind::SignUpdate(_)
209            | ScheduledPlayPacketKind::SpectatorAction(_)
210            | ScheduledPlayPacketKind::ChangeGameMode(_)
211            | ScheduledPlayPacketKind::ChangeDifficulty(_) => ScheduledPacketExecution::Serialized,
212            // Combat spans source and target state, custom payloads have no constrained resource
213            // contract, and the unimplemented menu handlers have no auditable transaction yet.
214            ScheduledPlayPacketKind::Attack(_)
215            | ScheduledPlayPacketKind::Interact(_)
216            | ScheduledPlayPacketKind::CustomPayload(_)
217            | ScheduledPlayPacketKind::ContainerButtonClick(_)
218            | ScheduledPlayPacketKind::ContainerSlotStateChanged(_) => {
219                ScheduledPacketExecution::Exclusive
220            }
221        }
222    }

237    pub(crate) fn handle(self, player: Arc<Player>, server: &Arc<Server>) {
238        if !player.has_joined_world() && !self.can_process_before_join() {
239            return;
240        }
241
242        match self.0 {
243            ScheduledPlayPacketKind::AcceptTeleportation(packet) => {
244                player.handle_accept_teleportation(packet);
245            }
246            ScheduledPlayPacketKind::Attack(packet) => player.handle_attack(packet),
247            ScheduledPlayPacketKind::Interact(packet) => player.handle_interact(packet),
248            ScheduledPlayPacketKind::CustomPayload(packet) => {
249                player.handle_custom_payload(packet);
250            }
251            ScheduledPlayPacketKind::Chat(packet) => {
252                player.handle_chat(*packet, Arc::clone(&player));
253            }
254            ScheduledPlayPacketKind::ChatAck(packet) => player.handle_chat_ack(packet),
255            ScheduledPlayPacketKind::ChatSessionUpdate(packet) => {
256                player.handle_chat_session_update(packet);
257            }
258            ScheduledPlayPacketKind::ClientInformation(packet) => {
259                player.handle_client_information(packet);
260            }
261            ScheduledPlayPacketKind::ClientTickEnd => player.handle_client_tick_end(),
262            ScheduledPlayPacketKind::MovePlayer(packet) => player.handle_move_player(packet),
263            ScheduledPlayPacketKind::MoveVehicle(packet) => player.handle_move_vehicle(packet),
264            ScheduledPlayPacketKind::PlayerLoaded => {
265                if player.mark_client_loaded_from_network() {
266                    player.send_inventory_to_remote();
267                }
268            }
269            ScheduledPlayPacketKind::ChatCommand(packet) => {
270                player.reset_last_action_time();
271                if server
272                    .submit_command(CommandSender::Player(Arc::clone(&player)), packet.command)
273                    .is_err()
274                {
275                    player.send_message(
276                        &TextComponent::const_plain("Command queue is full").color(Color::Red),
277                    );
278                }
279                player.detect_command_rate_spam();
280            }
281            ScheduledPlayPacketKind::CommandSuggestion(packet) => {
282                if server
283                    .submit_command_suggestions(Arc::clone(&player), packet.id, packet.command)
284                    .is_err()
285                {
286                    player.send_packet(CCommandSuggestions::new(packet.id, 0, 0, Vec::new()));
287                }
288            }
289            ScheduledPlayPacketKind::ContainerButtonClick(packet) => {
290                player.handle_container_button_click(packet);
291            }
292            ScheduledPlayPacketKind::ContainerClick(packet) => {
293                player.handle_container_click(packet);
294            }
295            ScheduledPlayPacketKind::ContainerClose(packet) => {
296                player.handle_container_close(packet);
297            }
298            ScheduledPlayPacketKind::ContainerSlotStateChanged(packet) => {
299                player.handle_container_slot_state_changed(packet);
300            }
301            ScheduledPlayPacketKind::SetCreativeModeSlot(packet) => {
302                player.handle_set_creative_mode_slot(packet);
303            }
304            ScheduledPlayPacketKind::PlayerInput(packet) => player.handle_player_input(packet),
305            ScheduledPlayPacketKind::PlayerCommand(packet) => {
306                player.handle_player_command(packet);
307            }
308            ScheduledPlayPacketKind::PlayerAbilities(packet) => {
309                player.handle_player_abilities(packet);
310            }
311            ScheduledPlayPacketKind::RenameItem(packet) => player.handle_rename_item(packet),
312            ScheduledPlayPacketKind::UseItemOn(packet) => player.handle_use_item_on(packet),
313            ScheduledPlayPacketKind::UseItem(packet) => player.handle_use_item(packet),
314            ScheduledPlayPacketKind::SetCarriedItem(packet) => {
315                player.handle_set_carried_item(packet);
316            }
317            ScheduledPlayPacketKind::Swing(packet) => player.handle_animate(packet),
318            ScheduledPlayPacketKind::PlayerAction(packet) => {
319                player.handle_player_action(packet);
320            }
321            ScheduledPlayPacketKind::PickItemFromBlock(packet) => {
322                player.handle_pick_item_from_block(packet);
323            }
324            ScheduledPlayPacketKind::SignUpdate(packet) => player.handle_sign_update(packet),
325            ScheduledPlayPacketKind::SpectatorAction(packet) => {
326                player.handle_spectator_action(packet);
327            }
328            ScheduledPlayPacketKind::ClientCommand(packet) => {
329                player.handle_client_command(packet.action);
330            }
331            ScheduledPlayPacketKind::ChangeGameMode(packet) => {
332                handle_client_request(&player, server, packet.gamemode);
333            }
334            ScheduledPlayPacketKind::ChangeDifficulty(packet) => {
335                player.handle_change_difficulty(packet.difficulty);
336            }
337        }
338    }

425    async fn write_packet_now(&self, packet: &EncodedPacket) -> Result<(), PacketError> {
426        let mut network_writer = self.network_writer.lock().await;
427        let Some(network_writer) = network_writer.as_mut() else {
428            return Err(PacketError::ConnectionClosed);
429        };
430        network_writer.write_packet(packet).await
431    }
432
433    async fn finish_disconnect(&self, disconnect_packet: Option<EncodedPacket>) {
434        let finish = async {
435            let Some(mut network_writer) = self.network_writer.lock().await.take() else {
436                return Ok(());
437            };
438            let Some(packet) = disconnect_packet else {
439                return Ok(());
440            };
441            network_writer.write_packet(&packet).await
442        };
443
444        match timeout(DISCONNECT_FLUSH_TIMEOUT, finish).await {
445            Ok(Ok(())) => {}
446            Ok(Err(error)) => log::debug!(
447                "Best-effort disconnect write for client {} failed: {error}",
448                self.id
449            ),
450            Err(_) => log::debug!(
451                "Best-effort disconnect write for client {} timed out",
452                self.id
453            ),
454        }
455    }

457    /// Ticks the connection.

462    fn keep_connection_alive(&self) {
463        let mut tracker = self.keep_alive_tracker.lock();
464        let now = SystemTime::now()
465            .duration_since(UNIX_EPOCH)
466            .expect("System time before UNIX EPOCH")
467            .as_millis() as u64;
468
469        if now - tracker.alive_time >= 15000 {
470            if tracker.alive_pending {
471                self.disconnect(translations::DISCONNECT_TIMEOUT.msg());
472            } else {
473                tracker.alive_pending = true;
474                tracker.alive_id = now;
475                tracker.alive_time = now;
476                self.send_packet(CKeepAlive::new(tracker.alive_id as i64));
477            }
478        }
479    }

481    /// Handles a keep alive packet.
482    #[expect(
483        clippy::cast_possible_truncation,
484        reason = "latency saturates at u32::MAX ms (~49 days), which is unreachable in practice"
485    )]
486    fn handle_keep_alive(&self, packet: SKeepAlive) {
487        let mut tracker = self.keep_alive_tracker.lock();
488        if tracker.alive_pending && packet.id as u64 == tracker.alive_id {
489            let now = SystemTime::now()
490                .duration_since(UNIX_EPOCH)
491                .expect("System time before UNIX EPOCH")
492                .as_millis() as u64;
493
494            let time = now.saturating_sub(tracker.alive_time) as u32;
495            tracker.alive_pending = false;
496            drop(tracker);
497            let mut latency = self.latency.lock();
498            *latency = (*latency * 3 + time) / 4;
499        } else {
500            self.disconnect(translations::DISCONNECT_TIMEOUT.msg());
501        }
502    }

504    /// Returns the current latency in milliseconds.

506    #[must_use]
507    pub fn latency(&self) -> i32 {
508        *self.latency.lock() as i32
509    }

511    /// Disconnects the client.
512    pub fn disconnect(&self, reason: impl Into<TextComponent>) {
513        let packet = match EncodedPacket::from_bare(
514            CDisconnect::new(&reason.into(), self),
515            self.compression,
516            ConnectionProtocol::Play,
517        ) {
518            Ok(packet) => packet,
519            Err(err) => {
520                log::warn!(
521                    "Failed to encode disconnect packet for client {}: {err}",
522                    self.id
523                );
524                self.close();
525                return;
526            }
527        };
528        if self
529            .outgoing_packets
530            .send(OutboundPacket::Disconnect(packet))
531            .is_err()
532        {
533            self.close();
534            return;
535        }
536        self.close();
537    }

609    /// Decodes and dispatches one packet received from the client.
610    fn process_packet(
611        &self,
612        packet: RawPacket,
613        player: Arc<Player>,
614        server: &Server,
615    ) -> Result<(), PacketError> {
616        if !player.has_joined_world() && !Self::can_process_before_join(packet.id) {
617            return Ok(());
618        }
619
620        let payload_bytes = packet.payload().len();
621        let Some(packet) = Self::decode_domain_gated_packet(packet, &player)? else {
622            return Ok(());
623        };
624
625        match packet {
626            DecodedPlayPacket::Scheduled(packet) => {
627                server.schedule_play_packet(player, packet, payload_bytes);
628            }
629            DecodedPlayPacket::Immediate(packet) => {
630                self.handle_immediate_packet(packet, &player);
631            }
632        }
633        Ok(())
634    }
635
636    fn decode_domain_gated_packet(
637        packet: RawPacket,
638        player: &Player,
639    ) -> Result<Option<DecodedPlayPacket>, PacketError> {
640        let maintenance_packet = matches!(packet.id, play::S_KEEP_ALIVE | play::S_PING_REQUEST);
641        let handshake_packet = Self::can_process_during_domain_handshake(packet.id);
642        if packet.id == play::S_CLIENT_COMMAND {
643            let decoded = Self::decode_play_packet(packet)?;
644            let perform_respawn = matches!(
645                &decoded,
646                DecodedPlayPacket::Scheduled(packet) if packet.is_perform_respawn()
647            );
648            if player.gate_domain_switch_packet(false, perform_respawn) {
649                return Ok(Some(decoded));
650            }
651            return Ok(None);
652        }
653
654        if !maintenance_packet && !player.gate_domain_switch_packet(handshake_packet, false) {
655            return Ok(None);
656        }
657
658        Self::decode_play_packet(packet).map(Some)
659    }

802    fn handle_immediate_packet(&self, packet: ImmediatePlayPacket, player: &Player) {
803        match packet {
804            ImmediatePlayPacket::KeepAlive(packet) => self.handle_keep_alive(packet),
805            ImmediatePlayPacket::PingRequest(packet) => {
806                player.send_packet(CPongResponse::new(packet.time));
807            }
808            ImmediatePlayPacket::ChunkBatchReceived(packet) => {
809                player
810                    .chunk_sender
811                    .lock()
812                    .on_chunk_batch_received_by_client(packet.desired_chunks_per_tick);
813            }
814            ImmediatePlayPacket::Unknown(id) => log::info!("play packet id {id} is not known"),
815        }
816    }

818    /// Listens for packets from the client.
819    pub async fn listener(
820        &self,
821        mut reader: TCPNetworkDecoder<BufReader<OwnedReadHalf>>,
822        server: Arc<Server>,
823    ) {
824        loop {
825            select! {
826                () = self.wait_for_close() => {
827                    break;
828                }
829                packet = reader.get_raw_packet() => {
830                    match packet {
831                        Ok(packet) => {
832                            if let Some(player) = self.player.upgrade()
833                                && let Err(err) = self.process_packet(packet, player, &server) {
834                                log::warn!(
835                                    "Failed to get packet from client {}: {err}",
836                                    self.id
837                                );
838                            }
839                        }
840                        Err(err) => {
841                            log::debug!("Failed to get raw packet from client {}: {err}", self.id);
842                            self.close();
843                        }
844                    }
845                }
846            }
847        }
848    }

850    /// Sends packets to the client.
851    ///
852    pub async fn sender(&self, mut sender_recv: UnboundedReceiver<OutboundPacket>) {
853        let disconnect_packet = loop {
854            select! {
855                biased;
856                () = self.wait_for_close() => {
857                    break Self::take_queued_disconnect(&mut sender_recv);
858                }
859                outbound = sender_recv.recv() => {
860                    if let Some(outbound) = outbound {
861                        let (packet, close_after_write) = match outbound {
862                            OutboundPacket::Packet(packet) => (packet, false),
863                            OutboundPacket::Disconnect(packet) => (packet, true),
864                        };
865
866                        if close_after_write {
867                            self.close();
868                            break Some(packet);
869                        }
870
871                        let write_result = self.write_packet_now(&packet);
872                        select! {
873                            biased;
874                            () = self.wait_for_close() => {
875                                break Self::take_queued_disconnect(&mut sender_recv);
876                            },
877                            result = write_result => {
878                                if let Err(err) = result {
879                                    log::warn!("Failed to send packet to client {}: {err}", self.id);
880                                    self.close();
881                                    break None;
882                                }
883                            }
884                        }
885                    } else {
886                        //log::warn!(
887                        //    "Internal packet_sender_recv channel closed for client {}",
888                        //    self.id
889                        //);
890                        self.close();
891                        break None;
892                    }
893                }
894            }
895        };
896
897        self.finish_disconnect(disconnect_packet).await;
898    }
899
900    fn take_queued_disconnect(
901        sender_recv: &mut UnboundedReceiver<OutboundPacket>,
902    ) -> Option<EncodedPacket> {
903        let mut disconnect_packet = None;
904        loop {
905            match sender_recv.try_recv() {
906                Ok(OutboundPacket::Packet(_)) => {}
907                Ok(OutboundPacket::Disconnect(packet)) => disconnect_packet = Some(packet),
908                Err(TryRecvError::Empty | TryRecvError::Disconnected) => break,
909            }
910        }
911        disconnect_packet
912    }
