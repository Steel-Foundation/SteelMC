1//! MC `Vec3` operations for `DVec3`.
2//!
3//! Axis rotations use Steel trig lookup table X and Z rotate in the opposite direction from `glam`'s built-in methods.
4
5use glam::DVec3;
6
7use crate::trig;
8
9/// length below which mc normalization returns zero
10pub const NORMALIZE_EPSILON: f64 = 1.0e-5_f32 as f64;
11
12/// normalizes a vec with minecraft cutoff and componentwise division
13#[inline]
14#[must_use]
15pub fn normalize(vector: DVec3) -> DVec3 {
16    let length = vector.length();
17    if length < NORMALIZE_EPSILON {
18        DVec3::ZERO
19    } else {
20        vector / length
21    }
22}
23
24/// Returns the table-backed cosine and sine vanilla's rotations index.
25fn cos_sin(radians: f32) -> (f64, f64) {
26    let angle = f64::from(radians);
27    (f64::from(trig::cos(angle)), f64::from(trig::sin(angle)))
28}
29
30/// Rotates the vector around the X axis and positive angles rotate +Y toward -Z
31#[inline]
32#[must_use]
33pub fn x_rot(vector: DVec3, radians: f32) -> DVec3 {
34    let (cos, sin) = cos_sin(radians);
35    DVec3::new(
36        vector.x,
37        vector.y * cos + vector.z * sin,
38        vector.z * cos - vector.y * sin,
39    )
40}
41
42/// Rotates the vector around the Y axis and positive angles rotate +X toward -Z
43#[inline]
44#[must_use]
45pub fn y_rot(vector: DVec3, radians: f32) -> DVec3 {
46    let (cos, sin) = cos_sin(radians);
47    DVec3::new(
48        vector.x * cos + vector.z * sin,
49        vector.y,
50        vector.z * cos - vector.x * sin,
51    )
52}
53
54/// Rotates the vector around the Z axis and positive angles rotate +X toward -Y
55#[inline]
56#[must_use]
57pub fn z_rot(vector: DVec3, radians: f32) -> DVec3 {
58    let (cos, sin) = cos_sin(radians);
59    DVec3::new(
60        vector.x * cos + vector.y * sin,
61        vector.y * cos - vector.x * sin,
62        vector.z,
63    )
64}

66#[cfg(test)]
67mod tests {
68    use super::*;
69    use std::f32::consts::FRAC_PI_2;
70
71    /// The sine table steps by 2π / 65536, bounding the error on a unit vector.
72    const TABLE_EPSILON: f64 = 1e-4;
73
74    fn assert_close(actual: DVec3, expected: DVec3) {
75        assert!(
76            (actual - expected).length() < TABLE_EPSILON,
77            "expected {expected:?}, got {actual:?}"
78        );
79    }
80
81    #[test]
82    fn rotating_by_zero_returns_the_input() {
83        let vector = DVec3::new(1.0, 2.0, 3.0);
84
85        assert_eq!(x_rot(vector, 0.0).to_array(), vector.to_array());
86        assert_eq!(y_rot(vector, 0.0).to_array(), vector.to_array());
87        assert_eq!(z_rot(vector, 0.0).to_array(), vector.to_array());
88    }
89
90    #[test]
91    fn quarter_turns_match_vanilla_axis_mapping() {
92        assert_close(x_rot(DVec3::Y, FRAC_PI_2), DVec3::new(0.0, 0.0, -1.0));
93        assert_close(y_rot(DVec3::X, FRAC_PI_2), DVec3::new(0.0, 0.0, -1.0));
94        assert_close(z_rot(DVec3::X, FRAC_PI_2), DVec3::new(0.0, -1.0, 0.0));
95    }
96
97    /// Pins the sign convention that reaching for `DVec3::rotate_x` breaks.
98    #[test]
99    fn x_rot_opposes_glam_rotate_x_while_y_rot_agrees() {
100        let down = DVec3::new(0.0, -1.0, 0.0);
101        let radians = 30.0_f32.to_radians();
102
103        assert_close(x_rot(down, radians), DVec3::new(0.0, -0.866_025, 0.5));
104        assert_close(
105            down.rotate_x(f64::from(radians)),
106            DVec3::new(0.0, -0.866_025, -0.5),
107        );
108
109        let vector = DVec3::new(1.0, 0.0, 0.5);
110        assert_close(y_rot(vector, radians), vector.rotate_y(f64::from(radians)));
111    }
112}
