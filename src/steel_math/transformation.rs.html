1//! 3D affine transforms
2
3use glam::{Mat3, Mat4, Quat, Vec3, Vec4Swizzles};
4use std::f32::consts;
5use std::mem;
6
7const DECOMPOSITION_EPSILON: f32 = 1.0e-6;
8const DECOMPOSITION_EPSILON_F64: f64 = 1.0e-6;
9const JACOBI_ITERATIONS: usize = 5;
10
11/// Affine transform `T * L * S * R`, right rotation acts first
12#[derive(Debug, Clone, Copy, PartialEq)]
13pub struct Transformation {
14    /// Translation
15    pub translation: Vec3,
16    /// Rotation after scaling
17    pub left_rotation: Quat,
18    /// Scale
19    pub scale: Vec3,
20    /// Rotation before scaling
21    pub right_rotation: Quat,
22}
23
24impl Transformation {
25    /// Identity transform
26    pub const IDENTITY: Self = Transformation {
27        translation: Vec3::ZERO,
28        left_rotation: Quat::IDENTITY,
29        scale: Vec3::ONE,
30        right_rotation: Quat::IDENTITY,
31    };
32
33    /// Compose the matrix
34    #[must_use]
35    pub fn compose(self) -> Mat4 {
36        Mat4::from_translation(self.translation)
37            * Mat4::from_quat(self.left_rotation)
38            * Mat4::from_scale(self.scale)
39            * Mat4::from_quat(self.right_rotation)
40    }
41
42    fn approx_givens_quat(a11: f32, a12: f32, a22: f32) -> Givens {
43        let cos_half = 2.0 * (a11 - a22);
44        let sin_half = a12;
45
46        if Givens::G * sin_half * sin_half < cos_half * cos_half {
47            Givens::from_unnormalized(sin_half, cos_half)
48        } else {
49            Givens::PI_4
50        }
51    }
52
53    fn qr_givens_quat(a: f32, b: f32) -> Givens {
54        let p = f64::from(a).hypot(f64::from(b)) as f32;
55
56        let mut sin_half = if p > DECOMPOSITION_EPSILON { b } else { 0.0 };
57        let mut cos_half = a.abs() + p.max(DECOMPOSITION_EPSILON);
58        if a < 0.0 {
59            mem::swap(&mut sin_half, &mut cos_half);
60        }
61
62        Givens::from_unnormalized(sin_half, cos_half)
63    }
64
65    fn joml_mul(left: Mat3, right: Mat3) -> Mat3 {
66        let column = |v: Vec3| left.x_axis * v.x + (left.y_axis * v.y + left.z_axis * v.z);
67        Mat3::from_cols(
68            column(right.x_axis),
69            column(right.y_axis),
70            column(right.z_axis),
71        )
72    }
73
74    fn similarity_transform(a: &mut Mat3, q: Mat3) {
75        *a = Self::joml_mul(q.transpose(), Self::joml_mul(*a, q));
76    }
77
78    fn step_jacobi(m: &mut Mat3, result: &mut Quat) {
79        if m.col(0)[1] * m.col(0)[1] + m.col(1)[0] * m.col(1)[0] > DECOMPOSITION_EPSILON {
80            let g = Self::approx_givens_quat(
81                m.col(0)[0],
82                f32::midpoint(m.col(0)[1], m.col(1)[0]),
83                m.col(1)[1],
84            );
85
86            *result *= g.around_z_quat();
87            Self::similarity_transform(m, g.around_z_mat());
88        }
89
90        if m.col(0)[2] * m.col(0)[2] + m.col(2)[0] * m.col(2)[0] > DECOMPOSITION_EPSILON {
91            let g = Self::approx_givens_quat(
92                m.col(0)[0],
93                f32::midpoint(m.col(0)[2], m.col(2)[0]),
94                m.col(2)[2],
95            )
96            .inverse();
97
98            *result *= g.around_y_quat();
99            Self::similarity_transform(m, g.around_y_mat());
100        }
101
102        if m.col(1)[2] * m.col(1)[2] + m.col(2)[1] * m.col(2)[1] > DECOMPOSITION_EPSILON {
103            let g = Self::approx_givens_quat(
104                m.col(1)[1],
105                f32::midpoint(m.col(1)[2], m.col(2)[1]),
106                m.col(2)[2],
107            );
108
109            *result *= g.around_x_quat();
110            Self::similarity_transform(m, g.around_x_mat());
111        }
112    }
113
114    /// Split the matrix into translation, rotations and scale using SVD
115    #[expect(clippy::similar_names, reason = "matches the stages of decomposition")]
116    #[must_use]
117    pub fn decompose(matrix: Mat4) -> Self {
118        let scale_factor = 1.0 / matrix.col(3)[3];
119        let input = matrix * scale_factor;
120
121        let translation = input.w_axis.xyz();
122
123        let mat = Mat3::from_mat4(input);
124
125        let mut ata = Self::joml_mul(mat.transpose(), mat);
126
127        let mut right_rotation = Quat::IDENTITY;
128        for _ in 0..JACOBI_ITERATIONS {
129            Self::step_jacobi(&mut ata, &mut right_rotation);
130        }
131        let [x, y, z, w] = right_rotation.to_array();
132        let length_squared = x * x + (y * y + (z * z + w * w));
133        let inv_sqrt = 1.0 / (f64::from(length_squared).sqrt() as f32);
134        let [x, y, z, w] = [x, y, z, w].map(|component| component * inv_sqrt);
135
136        right_rotation = Quat::from_xyzw(x, y, z, w);
137
138        let zero_column_0 = f64::from(ata.col(0)[0]) < DECOMPOSITION_EPSILON_F64;
139        let zero_column_1 = f64::from(ata.col(1)[1]) < DECOMPOSITION_EPSILON_F64;
140
141        let (ww, xx, yy, zz) = (w * w, x * x, y * y, z * z);
142        let (zw, xy, xz, yw, yz, xw) = (
143            2.0 * (z * w),
144            2.0 * (x * y),
145            2.0 * (x * z),
146            2.0 * (y * w),
147            2.0 * (y * z),
148            2.0 * (x * w),
149        );
150
151        let rotation = Mat3::from_cols(
152            Vec3::new(ww + xx - zz - yy, xy + zw, xz - yw),
153            Vec3::new(xy - zw, yy - zz + ww - xx, yz + xw),
154            Vec3::new(yw + xz, yz - xw, zz - yy - xx + ww),
155        );
156
157        let column = |v: Vec3| (mat.x_axis * v.x + mat.y_axis * v.y) + mat.z_axis * v.z;
158        let u012s = Mat3::from_cols(
159            column(rotation.x_axis),
160            column(rotation.y_axis),
161            column(rotation.z_axis),
162        );
163
164        let mut left_rotation = Quat::IDENTITY;
165
166        // QR rotations in Z, Y, X order
167        let givens = if zero_column_0 {
168            Self::qr_givens_quat(u012s.col(1)[1], -u012s.col(1)[0])
169        } else {
170            Self::qr_givens_quat(u012s.col(0)[0], u012s.col(0)[1])
171        };
172
173        let mut u12s = givens.around_z_mat();
174        left_rotation *= givens.around_z_quat();
175        u12s = Self::joml_mul(u12s.transpose(), u012s);
176
177        let givens = if zero_column_0 {
178            Self::qr_givens_quat(u12s.col(2)[2], -u12s.col(2)[0])
179        } else {
180            Self::qr_givens_quat(u12s.col(0)[0], u12s.col(0)[2])
181        }
182        .inverse();
183        let mut u2s = givens.around_y_mat();
184        left_rotation *= givens.around_y_quat();
185        u2s = Self::joml_mul(u2s.transpose(), u12s);
186
187        let givens = if zero_column_1 {
188            Self::qr_givens_quat(u2s.col(2)[2], -u2s.col(2)[1])
189        } else {
190            Self::qr_givens_quat(u2s.col(1)[1], u2s.col(1)[2])
191        };
192        let mut sm = givens.around_x_mat();
193        left_rotation *= givens.around_x_quat();
194        sm = Self::joml_mul(sm.transpose(), u2s);
195
196        let scale = Vec3::new(sm.col(0)[0], sm.col(1)[1], sm.col(2)[2]);
197
198        right_rotation = right_rotation.conjugate();
199
200        Transformation {
201            translation,
202            left_rotation,
203            scale,
204            right_rotation,
205        }
206    }
207}

209#[derive(Clone, Copy)]
210struct Givens {
211    sin_half: f32,
212    cos_half: f32,
213}
214
215impl Givens {
216    pub const G: f32 = 3.0 + 2.0 * consts::SQRT_2;
217    const SIN_PI_8: f32 = 0.382_683_46;
218    const COS_PI_8: f32 = 0.923_879_5;
219
220    pub const PI_4: Self = Self::new(Self::SIN_PI_8, Self::COS_PI_8);
221
222    pub const fn new(sin_half: f32, cos_half: f32) -> Self {
223        Self { sin_half, cos_half }
224    }
225
226    fn from_unnormalized(sin_half: f32, cos_half: f32) -> Self {
227        let length_squared = sin_half * sin_half + cos_half * cos_half;
228        let inv_sqrt = 1.0 / (f64::from(length_squared).sqrt() as f32);
229        Self::new(inv_sqrt * sin_half, inv_sqrt * cos_half)
230    }
231
232    fn inverse(self) -> Self {
233        Self::new(-self.sin_half, self.cos_half)
234    }
235
236    fn cos(self) -> f32 {
237        self.cos_half * self.cos_half - self.sin_half * self.sin_half
238    }
239
240    fn sin(self) -> f32 {
241        2.0 * self.sin_half * self.cos_half
242    }
243
244    const fn around_x_quat(self) -> Quat {
245        Quat::from_xyzw(self.sin_half, 0.0, 0.0, self.cos_half)
246    }
247    const fn around_y_quat(self) -> Quat {
248        Quat::from_xyzw(0.0, self.sin_half, 0.0, self.cos_half)
249    }
250    const fn around_z_quat(self) -> Quat {
251        Quat::from_xyzw(0.0, 0.0, self.sin_half, self.cos_half)
252    }
253
254    fn around_x_mat(self) -> Mat3 {
255        let cos = self.cos();
256        let sin = self.sin();
257        Mat3::from_cols(
258            Vec3::new(1.0, 0.0, 0.0),
259            Vec3::new(0.0, cos, sin),
260            Vec3::new(0.0, -sin, cos),
261        )
262    }
263
264    fn around_y_mat(self) -> Mat3 {
265        let cos = self.cos();
266        let sin = self.sin();
267        Mat3::from_cols(
268            Vec3::new(cos, 0.0, -sin),
269            Vec3::new(0.0, 1.0, 0.0),
270            Vec3::new(sin, 0.0, cos),
271        )
272    }
273
274    fn around_z_mat(self) -> Mat3 {
275        let cos = self.cos();
276        let sin = self.sin();
277        Mat3::from_cols(
278            Vec3::new(cos, sin, 0.0),
279            Vec3::new(-sin, cos, 0.0),
280            Vec3::new(0.0, 0.0, 1.0),
281        )
282    }
283}

285impl From<Mat4> for Transformation {
286    fn from(matrix: Mat4) -> Self {
287        Transformation::decompose(matrix)
288    }
289}
290
291impl From<Transformation> for Mat4 {
292    fn from(t: Transformation) -> Self {
293        Transformation::compose(t)
294    }
295}

302    fn check_transformation(transformation: Transformation, given_mat: Mat4) {
303        let new = Transformation::decompose(transformation.compose());
304
305        assert!(
306            transformation.compose().abs_diff_eq(new.compose(), 1.0e-5),
307            "Transformation matrices are not approximately equal: {transformation:?} {new:?}"
308        );
309        assert!(
310            transformation.compose().abs_diff_eq(given_mat, 1.0e-5),
311            "The new matrix and the given matrix are not approximately equal: {:?} {given_mat:?}",
312            transformation.compose()
313        );
314    }

337                Vec4::new(2.494_755_3, 1.422_634_6, -0.867_355_7, 0.0),

343
344        let transformation = Transformation {
