1//! `NoiseChunk`: cell-based terrain density evaluation with trilinear interpolation.
2//!
3//! Matches vanilla's `NoiseChunk` + `NoiseBasedChunkGenerator.doFill()` flow.
4//!
5//! Vanilla wraps density functions with `Interpolated` markers. Only the inner
6//! functions (arguments to `Interpolated`) are evaluated at cell corners; the
7//! outer operations (squeeze, min, etc.) are applied per-block after trilinear
8//! interpolation. Markers with identical evaluations share a channel.
9//!
10//! Cell dimensions depend on the dimension's noise settings.
11
12use std::marker::PhantomData;
13use std::simd::f32x4;
14use steel_utils::SIMD_BATCH;
15use steel_worldgen::density::{ColumnCache, DimensionNoises, NoiseSettings};
16
17use crate::noise::Beardifier;
18
19/// Maximum number of interpolation channels supported.
20/// Overworld uses 8 (1 terrain + 4 noodle caves + 3 vein channels), nether/end use 1.
21const MAX_INTERP: usize = 16;
22
23/// Maximum slice length (`z_corners` * `corners_y`) across all dimensions.
24/// Overworld: (16/4+1) * (384/8+1) = 5 * 49 = 245. Rounded up for headroom.
25const MAX_SLICE_LEN: usize = 256;
26/// The largest Vanilla noise cell height (overworld's is 8).
27const MAX_CELL_HEIGHT: usize = 16;
28
29/// Stores density values at cell corners for a single chunk and provides
30/// trilinear interpolation between corners for block-level resolution.
31///
32/// Supports multiple interpolation channels matching vanilla's multi-interpolator
33/// system. Distinct marker evaluations get channels filled at cell corners
34/// and interpolated independently.
35///
36/// Storage is per-corner `SoA` — `slice[corner_idx * MAX_INTERP + ch]` — so 4
37/// adjacent channels' values at a given corner sit in contiguous memory,
38/// allowing the terrain sampler to keep Vanilla's f32 density values without
39/// widening an intermediate result.
40pub struct NoiseChunk<N: DimensionNoises> {
41    /// One slice per cell-X boundary, holding density values at the cell
42    /// corners on that X-plane. Length is `cell_count_xz + 1`. Indexed as
43    /// `slices[cx][corner_idx * MAX_INTERP + ch]` where
44    /// `corner_idx = z_corner * corners_y + y_corner` (range `[0, slice_len)`)
45    /// and `ch` is the interpolation channel (range `[0, interp_count)`).
46    ///
47    /// We keep all slices materialized rather than alternating two buffers so
48    /// the slice-fill phase can run in parallel: each `cx` boundary's noise
49    /// tree evaluation is independent. The per-block trilerp loop then
50    /// indexes `slices[cx]` and `slices[cx + 1]` sequentially.
51    slices: Vec<Box<[f32; MAX_INTERP * MAX_SLICE_LEN]>>,
52    /// Number of active interpolation channels.
53    interp_count: usize,
54    /// Number of Y corners per Z column (`cell_count_y` + 1).
55    corners_y: usize,
56
57    /// Per-corner block-Y values, precomputed once at construction.
58    /// Same for every slice fill (depends only on `cell_min_y`,
59    /// `cell_height`, and `corners_y`).
60    block_ys: Vec<i32>,
61
62    /// First cell X/Z in world coordinates (cell index, not block).
63    first_cell_x: i32,
64    first_cell_z: i32,
65    /// Minimum cell Y index.
66    cell_min_y: i32,
67    /// Number of cells in Y direction.
68    cell_count_y: usize,
69    /// Number of cells per chunk in XZ.
70    cell_count_xz: usize,
71
72    _phantom: PhantomData<N>,
73}
74
75impl<N: DimensionNoises> NoiseChunk<N> {
76    /// Create a new `NoiseChunk` for the given chunk position.
77    ///
78    /// `chunk_min_block_x` and `chunk_min_block_z` are the world-space block
79    /// coordinates of the chunk's northwest corner.
80    #[must_use]
81    #[expect(
82        clippy::missing_panics_doc,
83        reason = "panic is a compile-time constant check"
84    )]
85    pub fn new(chunk_min_block_x: i32, chunk_min_block_z: i32) -> Self {
86        let cell_width = N::Settings::CELL_WIDTH;
87        let cell_height = N::Settings::CELL_HEIGHT;
88        let min_y = N::Settings::MIN_Y;
89        let height = N::Settings::HEIGHT;
90
91        let first_cell_x = chunk_min_block_x.div_euclid(cell_width);
92        let first_cell_z = chunk_min_block_z.div_euclid(cell_width);
93        let cell_min_y = min_y.div_euclid(cell_height);
94
95        let cell_count_xz = (16 / cell_width) as usize;
96        let cell_count_y = (height / cell_height) as usize;
97        let corners_y = cell_count_y + 1;
98        let z_corners = cell_count_xz + 1;
99        let slice_len = z_corners * corners_y;
100
101        let interp_count = N::interpolated_count();
102        assert!(
103            slice_len <= MAX_SLICE_LEN,
104            "slice_len {slice_len} exceeds MAX_SLICE_LEN {MAX_SLICE_LEN}"
105        );
106        assert!(
107            interp_count <= MAX_INTERP,
108            "interp_count {interp_count} exceeds MAX_INTERP {MAX_INTERP}"
109        );
110
111        let block_ys: Vec<i32> = (0..corners_y)
112            .map(|cy| (cy as i32 + cell_min_y) * cell_height)
113            .collect();
114
115        let n_slices = cell_count_xz + 1;
116        let mut slices = Vec::with_capacity(n_slices);
117        for _ in 0..n_slices {
118            // This is a per-chunk constructor, so the stack temporary is fine.
119            slices.push(Box::new([0.0_f32; MAX_INTERP * MAX_SLICE_LEN]));
120        }
121
122        Self {
123            slices,
124            interp_count,
125            corners_y,
126            block_ys,
127            first_cell_x,
128            first_cell_z,
129            cell_min_y,
130            cell_count_y,
131            cell_count_xz,
132            _phantom: PhantomData,
133        }
134    }
135
136    /// Fill the slice buffer for the given cell X. Free-standing function so
137    /// each parallel slice-fill can run on its own thread with its own
138    /// `ColumnCache` clone.
139    #[expect(
140        clippy::too_many_arguments,
141        reason = "slice filling needs the precomputed geometry and per-thread cache"
142    )]
143    fn fill_slice_into(
144        slice: &mut [f32; MAX_INTERP * MAX_SLICE_LEN],
145        cell_x: i32,
146        block_ys: &[i32],
147        blended_column: &mut [f32],
148        interp_count: usize,
149        corners_y: usize,
150        cell_count_xz: usize,
151        first_cell_z: i32,
152        noises: &N,
153        cache: &mut N::ColumnCache,
154    ) {
155        let cell_width = N::Settings::CELL_WIDTH;
156
157        let block_x = cell_x * cell_width;
158
159        let mut values = [0.0_f32; MAX_INTERP * SIMD_BATCH];
160
161        for cz in 0..=cell_count_xz {
162            let cell_z = first_cell_z + cz as i32;
163            let block_z = cell_z * cell_width;
164
165            // Ensure column cache for this (x, z)
166            cache.ensure(block_x, block_z, noises);
167
168            // SIMD-batch blended noise for the entire Y column.
169            noises.compute_noise_column(block_x, block_ys, block_z, blended_column);
170
171            let (y_batches, y_tail) = block_ys.as_chunks::<SIMD_BATCH>();
172            let (blended_batches, blended_tail) = blended_column.as_chunks::<SIMD_BATCH>();
173            debug_assert_eq!(y_batches.len(), blended_batches.len());
174            debug_assert_eq!(y_tail.len(), blended_tail.len());
175
176            for (batch, (ys, blended_noise_values)) in
177                y_batches.iter().zip(blended_batches.iter()).enumerate()
178            {
179                let cy = batch * SIMD_BATCH;
180                noises.fill_cell_corner_densities_y_simd::<SIMD_BATCH>(
181                    cache,
182                    block_x,
183                    *ys,
184                    block_z,
185                    *blended_noise_values,
186                    &mut values[..interp_count * SIMD_BATCH],
187                );
188                for lane in 0..SIMD_BATCH {
189                    let corner_idx = cz * corners_y + cy + lane;
190                    let base = corner_idx * MAX_INTERP;
191                    let value_base = lane * interp_count;
192                    slice[base..base + interp_count]
193                        .copy_from_slice(&values[value_base..value_base + interp_count]);
194                }
195            }
196
197            let tail_start = y_batches.len() * SIMD_BATCH;
198            for (tail_offset, (&block_y, &blended_noise_value)) in
199                y_tail.iter().zip(blended_tail.iter()).enumerate()
200            {
201                let cy = tail_start + tail_offset;
202
203                noises.fill_cell_corner_densities(
204                    cache,
205                    block_x,
206                    block_y,
207                    block_z,
208                    blended_noise_value,
209                    &mut values[..interp_count],
210                );
211
212                let corner_idx = cz * corners_y + cy;
213                let base = corner_idx * MAX_INTERP;
214                slice[base..base + interp_count].copy_from_slice(&values[..interp_count]);
215            }
216        }
217    }

219    /// Fill the chunk with terrain blocks using multi-channel trilinear interpolation.
220    ///
221    /// For each block position:
222    /// 1. Trilinearly interpolate each channel independently from cell corners
223    /// 2. Apply outer operations (squeeze, min, etc.) via `combine_interpolated`
224    /// 3. Call `place_block` with the final density
225    #[expect(
226        clippy::too_many_lines,
227        reason = "keeps the cell interpolation and block traversal together"
228    )]
229    pub fn fill<F>(
230        &mut self,
231        noises: &N,
232        cache: &mut N::ColumnCache,
233        beardifier: Option<&Beardifier>,
234        mut place_block: F,
235    ) where
236        F: FnMut(usize, i32, usize, f32, &[f32], &mut N::ColumnCache),
237    {
238        let cell_width = N::Settings::CELL_WIDTH;
239        let cell_height = N::Settings::CELL_HEIGHT;
240        let cell_count_xz = self.cell_count_xz;
241        let cell_count_y = self.cell_count_y;
242        let interp_count = self.interp_count;
243        let corners_y = self.corners_y;
244        let first_cell_x = self.first_cell_x;
245        let first_cell_z = self.first_cell_z;
246        let block_ys: &[i32] = &self.block_ys;
247
248        // Pre-fill ALL slices sequentially. Each `(cell_x boundary)` slice is an
249        // independent noise-tree evaluation; the grid in `cache` is set up by the
250        // caller via `init_grid` and is read-only here, while each slice only
251        // overwrites the cache's per-column active fields — so one cache is reused
252        // across slices without cloning. The chunk pipeline already parallelises
253        // across chunks, so parallelising the 5 slices here would nest rayon work
254        // and add coordination + cache-clone overhead with no spare cores to use.
255        let n_slices = cell_count_xz + 1;
256        let mut local_blended = vec![0.0_f32; corners_y];
257        for (cx_off, slice) in self.slices[..n_slices].iter_mut().enumerate() {
258            let cell_x = first_cell_x + cx_off as i32;
259            Self::fill_slice_into(
260                slice,
261                cell_x,
262                block_ys,
263                &mut local_blended,
264                interp_count,
265                corners_y,
266                cell_count_xz,
267                first_cell_z,
268                noises,
269                cache,
270            );
271        }
272
273        let y_count = cell_height as usize;
274        debug_assert!(y_count <= MAX_CELL_HEIGHT);
275        let inverse_height = f32x4::splat(1.0 / cell_height as f32);
276        let channel_batches = interp_count.div_ceil(4);
277        // Reuse Z partials across X without changing vanilla's Z/X/Y arithmetic.
278        let mut z_partials =
279            vec![[f32x4::splat(0.0); 2]; cell_width as usize * corners_y * channel_batches];
280
281        for cell_x_idx in 0..cell_count_xz {
282            let s0: &[f32] = &self.slices[cell_x_idx][..];
283            let s1: &[f32] = &self.slices[cell_x_idx + 1][..];
284
285            for cell_z_idx in 0..cell_count_xz {
286                let z0_base = cell_z_idx * corners_y;
287                let z1_base = (cell_z_idx + 1) * corners_y;
288
289                for z_in_cell in 0..cell_width as usize {
290                    let factor_z = f32x4::splat(z_in_cell as f32 / cell_width as f32);
291                    for corner_y in 0..corners_y {
292                        let i0 = (z0_base + corner_y) * MAX_INTERP;
293                        let i1 = (z1_base + corner_y) * MAX_INTERP;
294                        let row = (z_in_cell * corners_y + corner_y) * channel_batches;
295                        for (batch, channel) in (0..interp_count).step_by(4).enumerate() {
296                            let corner = |slice: &[f32], base| {
297                                // SAFETY: construction bounds the corner count by
298                                // MAX_SLICE_LEN and interp_count by MAX_INTERP.
299                                // Both Z planes and corner_y are in that grid;
300                                // MAX_INTERP is a multiple of four, so the padded
301                                // final batch also stays inside its corner.
302                                f32x4::from_slice(unsafe {
303                                    slice.get_unchecked(base + channel..base + channel + 4)
304                                })
305                            };
306                            let left0 = corner(s0, i0);
307                            let left1 = corner(s0, i1);
308                            let right0 = corner(s1, i0);
309                            let right1 = corner(s1, i1);
310                            z_partials[row + batch] = [
311                                left0 + factor_z * (left1 - left0),
312                                right0 + factor_z * (right1 - right0),
313                            ];
314                        }
315                    }
316                }
317
318                for x_in_cell in 0..cell_width {
319                    let factor_x = x_in_cell as f32 / cell_width as f32;
320                    let factor_x_v = f32x4::splat(factor_x);
321                    let local_x = (cell_x_idx as i32 * cell_width + x_in_cell) as usize;
322
323                    for z_in_cell in 0..cell_width {
324                        let local_z = (cell_z_idx as i32 * cell_width + z_in_cell) as usize;
325
326                        for cell_y_idx in (0..cell_count_y).rev() {
327                            let cell_world_y = (self.cell_min_y + cell_y_idx as i32) * cell_height;
328                            let row =
329                                (z_in_cell as usize * corners_y + cell_y_idx) * channel_batches;
330                            let mut values_by_y = [[0.0_f32; MAX_INTERP]; MAX_CELL_HEIGHT];
331
332                            // Advance Y by repeated additions, matching fillCell.
333                            for (batch, channel) in (0..interp_count).step_by(4).enumerate() {
334                                let [v00, v10] = z_partials[row + batch];
335                                let [v01, v11] = z_partials[row + channel_batches + batch];
336                                let v0 = v00 + factor_x_v * (v10 - v00);
337                                let v1 = v01 + factor_x_v * (v11 - v01);
338                                let step = (v1 - v0) * inverse_height;
339                                let mut value = v0 + step * f32x4::splat(0.0);
340                                for values in values_by_y.iter_mut().take(y_count) {
341                                    values[channel..channel + 4].copy_from_slice(&value.to_array());
342                                    value += step;
343                                }
344                            }
345
346                            for y_in_cell in (0..cell_height).rev() {
347                                let world_y = cell_world_y + y_in_cell;
348                                let interpolated = &values_by_y[y_in_cell as usize][..interp_count];
349
350                                // Apply outer operations per-block.
351                                // x/z are 0 because vanilla's outer operations (squeeze, add, mul,
352                                // quarter_negative, blend_alpha, blend_offset) are x/z-independent;
353                                // only Y matters for YClampedGradient.
354                                let mut density = noises.combine_interpolated(
355                                    cache,
356                                    &interpolated[..interp_count],
357                                    0,
358                                    world_y,
359                                    0,
360                                );
361
362                                // Vanilla integrates beardifier as `add(final_density, beardifier)`
363                                // wrapped in `cacheAllInCell` — i.e. evaluated per-block, after the
364                                // outer ops on `final_density` have run. Adding it at cell corners
365                                // would put it inside the squeeze and trilerp it linearly across
366                                // the cell, both of which diverge from vanilla for large beardifier
367                                // values inside a structure's pieces.
368                                let world_x = cell_x_idx as i32 * cell_width
369                                    + x_in_cell
370                                    + self.first_cell_x * cell_width;
371                                let world_z = cell_z_idx as i32 * cell_width
372                                    + z_in_cell
373                                    + self.first_cell_z * cell_width;
374                                if let Some(beard) = beardifier {
375                                    density += beard.compute(world_x, world_y, world_z) as f32;
376                                }
377
378                                place_block(
379                                    local_x,
380                                    world_y,
381                                    local_z,
382                                    density,
383                                    &interpolated[..interp_count],
384                                    cache,
385                                );
386                            }
387                        }
388                    }
389                }
390            }
391
392            // No swap needed: all slices are pre-filled and indexed directly
393            // via `self.slices[cell_x_idx]` / `[cell_x_idx + 1]`.
394        }
395    }

397    /// Pre-fills the density/richness buffers used by material ore rules.
398    ///
399    /// `MaterialRuleContext.getDensitiesInChunk` samples these functions over
400    /// the complete chunk volume before the surface scan begins. Keeping the
401    /// compact `f32` results instead of the full interpolation channels avoids
402    /// retaining the noise chunk across generation stages.
403    #[must_use]
404    pub fn prefill_material_ore_vein_values(
405        &self,
406        noises: &N,
407        cache: &mut N::ColumnCache,
408    ) -> Box<[f32]> {
409        let value_count = N::material_ore_vein_value_count();
410        if value_count == 0 {
411            return Box::default();
412        }
413
414        let cell_width = N::Settings::CELL_WIDTH;
415        let cell_height = N::Settings::CELL_HEIGHT;
416        let min_y = N::Settings::MIN_Y;
417        let mut values = vec![0.0; 16 * 16 * N::Settings::HEIGHT as usize * value_count];
418        let mut interpolated = [0.0_f32; MAX_INTERP];
419
420        for cell_x_idx in 0..self.cell_count_xz {
421            for cell_z_idx in 0..self.cell_count_xz {
422                let z0_base = cell_z_idx * self.corners_y;
423                let z1_base = (cell_z_idx + 1) * self.corners_y;
424                let s0 = &*self.slices[cell_x_idx];
425                let s1 = &*self.slices[cell_x_idx + 1];
426
427                for x_in_cell in 0..cell_width {
428                    let factor_x = x_in_cell as f32 / cell_width as f32;
429                    let local_x = (cell_x_idx as i32 * cell_width + x_in_cell) as usize;
430                    let world_x =
431                        self.first_cell_x * cell_width + cell_x_idx as i32 * cell_width + x_in_cell;
432
433                    for z_in_cell in 0..cell_width {
434                        let factor_z = z_in_cell as f32 / cell_width as f32;
435                        let local_z = (cell_z_idx as i32 * cell_width + z_in_cell) as usize;
436                        let world_z = self.first_cell_z * cell_width
437                            + cell_z_idx as i32 * cell_width
438                            + z_in_cell;
439
440                        for cell_y_idx in (0..self.cell_count_y).rev() {
441                            let i0_base = (z0_base + cell_y_idx) * MAX_INTERP;
442                            let i1_base = (z1_base + cell_y_idx) * MAX_INTERP;
443                            let i0_next = i0_base + MAX_INTERP;
444                            let i1_next = i1_base + MAX_INTERP;
445                            let mut values_by_y = [[0.0_f32; MAX_INTERP]; MAX_CELL_HEIGHT];
446                            let y_count = cell_height as usize;
447                            debug_assert!(y_count <= MAX_CELL_HEIGHT);
448
449                            for channel in 0..self.interp_count {
450                                let n000 = s0[i0_base + channel];
451                                let n001 = s0[i1_base + channel];
452                                let n100 = s1[i0_base + channel];
453                                let n101 = s1[i1_base + channel];
454                                let n010 = s0[i0_next + channel];
455                                let n011 = s0[i1_next + channel];
456                                let n110 = s1[i0_next + channel];
457                                let n111 = s1[i1_next + channel];
458                                let v00 = n000 + factor_z * (n001 - n000);
459                                let v01 = n010 + factor_z * (n011 - n010);
460                                let v10 = n100 + factor_z * (n101 - n100);
461                                let v11 = n110 + factor_z * (n111 - n110);
462                                let v0 = v00 + factor_x * (v10 - v00);
463                                let v1 = v01 + factor_x * (v11 - v01);
464                                let step = (v1 - v0) * (1.0_f32 / cell_height as f32);
465                                let mut value = v0 + step * 0.0;
466                                for values in values_by_y.iter_mut().take(y_count) {
467                                    values[channel] = value;
468                                    value += step;
469                                }
470                            }
471
472                            for y_in_cell in (0..cell_height).rev() {
473                                let world_y =
474                                    (self.cell_min_y + cell_y_idx as i32) * cell_height + y_in_cell;
475                                interpolated[..self.interp_count].copy_from_slice(
476                                    &values_by_y[y_in_cell as usize][..self.interp_count],
477                                );
478
479                                let relative_y = (world_y - min_y) as usize;
480                                let offset =
481                                    (relative_y * 16 * 16 + local_z * 16 + local_x) * value_count;
482                                noises.fill_material_ore_vein_values(
483                                    cache,
484                                    &interpolated[..self.interp_count],
485                                    world_x,
486                                    world_y,
487                                    world_z,
488                                    &mut values[offset..offset + value_count],
489                                );
490                            }
491                        }
492                    }
493                }
494            }
495        }
496
497        values.into_boxed_slice()
498    }
